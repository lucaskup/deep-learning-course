\begin{enumerate}[I)]

% ------------------------------------------------------------------
% Q1 — Fácil (NER: esquema BIO)
% ------------------------------------------------------------------
\Needspace{19\baselineskip}
\item Na tarefa de \textit{Named Entity Recognition} (NER) com esquema
      \textbf{BIO2}, as categorias são: \textbf{B}egin (início de entidade),
      \textbf{I}nside (continuação) e \textbf{O}utside. Para a frase

\begin{center}
  \texttt{Lucas trabalha na PUCRS em Porto Alegre}
\end{center}

considerando as entidades \textit{Lucas} (pessoa), \textit{PUCRS} (organização)
e \textit{Porto Alegre} (local, dois tokens), tratando cada palavra como um token, qual sequência de
\textit{tags} é correta?

\begin{enumerate}[a)]
    \item \texttt{B-PER, O, O, B-ORG, O, B-LOC, I-LOC}
    \item \texttt{B-PER, O, O, B-ORG, O, I-LOC, I-LOC}
    \item \texttt{B-PER, O, O, I-ORG, O, B-LOC, I-LOC}
    \item \texttt{I-PER, O, O, B-ORG, O, B-LOC, I-LOC}
\end{enumerate}
% R: a

% ------------------------------------------------------------------
% Q2 — Médio (BERT: contagem de parâmetros de atenção)
% ------------------------------------------------------------------
\Needspace{18\baselineskip}
\item O BERT-Base tem $L = 12$ camadas, dimensão oculta $d = 768$ e
      $H = 12$ cabeças de atenção com $d_k = d/H = 64$ por cabeça.
      Em uma única camada de \textit{self-attention}, as projeções são:
      \[
        \mathbf{W}^Q,\; \mathbf{W}^K,\; \mathbf{W}^V \in \mathbb{R}^{d \times (H \cdot d_k)},
        \qquad
        \mathbf{W}^O \in \mathbb{R}^{(H \cdot d_k) \times d}.
      \]
      Ignorando vieses, quantos parâmetros treináveis existem nessas
      \textbf{quatro} matrizes em uma única camada?

\begin{enumerate}[a)]
    \item $589\,824$
    \item $1\,769\,472$
    \item $2\,359\,296$
    \item $4\,718\,592$
\end{enumerate}
% R: c

% ------------------------------------------------------------------
% Q3 — Médio (MLM: regra 80/10/10 e desalinhamento treino-inferência)
% ------------------------------------------------------------------
\Needspace{18\baselineskip}
\item No pré-treino do BERT, MLM seleciona $15\%$ das posições.
      Entre as selecionadas, $80\%$ recebem \texttt{[MASK]}, $10\%$
      recebem um token aleatório e $10\%$ preservam o token original.
      Considere substituir todos os tokens selecionados por \texttt{[MASK]}.
      Qual efeito dessa mudança motiva a mistura original?

\begin{enumerate}[a)]
    \item Reduzir o vocabulário de saída disponível para as posições supervisionadas no pré-treino.
    \item Aumentar a diferença entre as entradas supervisionadas no pré-treino e nas tarefas discriminativas.
    \item Eliminar os gradientes dos estados ocultos nas posições selecionadas para a perda de reconstrução.
    \item Remover a informação posicional dos tokens substituídos durante a construção da entrada do encoder.
\end{enumerate}
% R: b

% ------------------------------------------------------------------
% Q4 — Médio (BERT: remoção dos segment embeddings)
% ------------------------------------------------------------------
\Needspace{19\baselineskip}
\item Um BERT recebe \texttt{[CLS] pergunta [SEP] contexto [SEP]},
      com representação de entrada
      \[
        \mathbf{x}_i=\mathbf e_{\text{token}}(w_i)
          +\mathbf e_{\text{segmento}}(s_i)+\mathbf e_{\text{posição}}(i).
      \]
      Um pesquisador remove apenas $\mathbf e_{\text{segmento}}$, mantendo
      os tokens, suas posições e a máscara de atenção.
      Qual descrição corresponde à alteração realizada?

\begin{enumerate}[a)]
    \item A indicação posicional é removida, preservando os separadores e os embeddings de segmento.
    \item A marcação por separadores é removida, preservando as posições e os embeddings de tokens.
    \item A indicação aditiva do segmento é removida, preservando os separadores e os embeddings posicionais.
    \item A interação entre segmentos é removida, preservando os tokens e as posições de cada segmento.
\end{enumerate}
% R: c

% ------------------------------------------------------------------
% Q5 — Médio (ELECTRA vs. BERT: posições supervisionadas)
% ------------------------------------------------------------------
\Needspace{15\baselineskip}
\item Considere uma entrada com $T$ posições válidas, sem padding.
      No BERT, o conjunto $M$ reúne as posições selecionadas para MLM.
      No ELECTRA, um gerador propõe substituições e o discriminador
      usa o objetivo \textit{Replaced Token Detection} (RTD).
      Em quais posições são calculados os termos de perda do MLM
      e do RTD do discriminador, respectivamente?

\begin{enumerate}[a)]
    \item MLM: nas $T$ posições; RTD: nas posições de $M$.
    \item MLM: nas posições de $M$; RTD: nas $T$ posições.
    \item MLM: nas posições de $M$; RTD: nas posições de $M$.
    \item MLM: nas $T$ posições; RTD: nas $T$ posições.
\end{enumerate}
% R: b

% ------------------------------------------------------------------
% Q6 — Médio (CLS em STS: objetivo de adaptação)
% ------------------------------------------------------------------
\Needspace{17\baselineskip}
\item Um engenheiro compara frases pelo cosseno entre estados finais
      de \texttt{[CLS]} de um BERT sem adaptação.
      Na avaliação de Reimers e Gurevych (2019), essa abordagem obteve
      $100\rho=16{,}50$ no STS-B, enquanto a média de embeddings GloVe
      obteve $100\rho=58{,}02$, com $\rho$ indicando Spearman.
      Qual objetivo de adaptação atua diretamente sobre a organização
      dos vetores para essa comparação?

\begin{enumerate}[a)]
    \item Aproximar pares semanticamente similares e afastar pares diferentes no espaço de embeddings.
    \item Prever tokens selecionados usando o contexto das posições de cada sentença corrompida.
    \item Prever se um segmento de texto é a continuação de outro no documento.
    \item Distinguir tokens originais de substitutos amostrados por um gerador auxiliar de tokens.
\end{enumerate}
% R: a

% ------------------------------------------------------------------
% Q7 — Médio (mean pooling mascarado: cálculo com token [PAD])
% ------------------------------------------------------------------
\Needspace{23\baselineskip}
\item No SBERT, o \textit{mean pooling} mascarado calcula o vetor de
      sentença excluindo posições \texttt{[PAD]}:
      \[
        \mathbf{u} = \frac{\sum_{i=1}^{L} m_i\,\mathbf{h}_i}{\sum_{i=1}^{L} m_i},
        \qquad m_i \in \{0,1\}.
      \]
      Uma entrada preenchida tem $L = 3$ posições com os seguintes estados ocultos
      $\mathbf{h}_i \in \mathbb{R}^2$ e máscara de atenção:

\begin{center}
\begin{tabular}{lcc}
  \toprule
  Token & $\mathbf{h}_i$ & $m_i$ \\
  \midrule
  $w_1$ & $[2,\;0]$ & $1$ \\
  $w_2$ & $[0,\;2]$ & $1$ \\
  $w_3$ (\texttt{[PAD]}) & $[1,\;3]$ & $0$ \\
  \bottomrule
\end{tabular}
\end{center}

Qual é o vetor de sentença $\mathbf{u}$ produzido pelo \textit{mean pooling} mascarado?

\begin{enumerate}[a)]
    \item $\bigl[1,\;\tfrac{5}{3}\bigr] \approx [1.00,\;1.67]$
    \item $[2,\;2]$
    \item $\bigl[\tfrac{2}{3},\;\tfrac{2}{3}\bigr] \approx [0.67,\;0.67]$
    \item $[1,\;1]$
\end{enumerate}
% R: d

% ------------------------------------------------------------------
% Q8 — Médio (bi-encoder vs. cross-encoder: latência de consulta)
% ------------------------------------------------------------------
\Needspace{16\baselineskip}
\item Um cross-encoder avalia uma consulta contra $100\,000$ documentos,
      executando as inferências sequencialmente, sem batching. Cada par
      custa $10\text{ ms}$.
      Um bi-encoder já possui os documentos indexados: codificar a consulta
      custa $10\text{ ms}$ e a busca vetorial custa mais $10\text{ ms}$.
      Desconsidere os demais custos.
      Qual é a razão entre a latência do cross-encoder e a do bi-encoder?

\begin{enumerate}[a)]
    \item $10\,000\times$
    \item $25\,000\times$
    \item $50\,000\times$
    \item $100\,000\times$
\end{enumerate}
% R: c

% ------------------------------------------------------------------
% Q9 — Médio (InfoNCE: efeito do tamanho de batch)
% ------------------------------------------------------------------
\Needspace{14\baselineskip}
\item Um treinamento usa InfoNCE direcional com pares $(x_i,x_i^+)$:
      cada âncora $x_i$ é comparada aos positivos de todos os pares do lote.
      Há um único positivo rotulado por âncora, e os demais candidatos
      são tratados como negativos.
      Ao aumentar o lote de $4$ para $1\,024$ pares, quantos candidatos
      negativos passam a participar da perda de cada âncora?

\begin{enumerate}[a)]
    \item $1\,023$
    \item $1\,024$
    \item $2\,046$
    \item $2\,047$
\end{enumerate}
% R: a

% ------------------------------------------------------------------
% Q10 — Médio (triplet loss: cálculo numérico)
% ------------------------------------------------------------------
\Needspace{18\baselineskip}
\item Considere três \textit{embeddings}
      (já normalizados) e a \textit{triplet loss} com margem $\alpha = 0.5$:
      \[
        \mathbf{f}(x) = [1.0,\;0.0], \quad
        \mathbf{f}(x^+) = [0.6,\;0.8], \quad
        \mathbf{f}(x^-) = [-0.6,\;0.8],
      \]
      \[
        \mathcal{L}_{\text{triplet}} = \max\!\bigl(0,\;
          \|\mathbf{f}(x)-\mathbf{f}(x^+)\|_2^2 - \|\mathbf{f}(x)-\mathbf{f}(x^-)\|_2^2 + \alpha\bigr).
      \]
      Qual é o valor de $\mathcal{L}_{\text{triplet}}$?

\begin{enumerate}[a)]
    \item $0.50$
    \item $1.90$
    \item $2.90$
    \item $0$
\end{enumerate}
% R: d

% ------------------------------------------------------------------
% Q11 — Médio (MLM loss: cálculo numérico)
% ------------------------------------------------------------------
\Needspace{17\baselineskip}
\item O BERT pré-treina com a perda MLM:
      \[
        \mathcal{L}_{\text{MLM}}
          = -\frac{1}{|M|}\sum_{i \in M} \ln p_\theta(w_i \mid \tilde{\mathbf w}).
      \]
      A sequência corrompida é $\tilde{\mathbf w}$, e $M$ reúne as posições
      selecionadas para predição. Em um passo de treino, $|M| = 3$. O modelo
      atribui as seguintes probabilidades ao \textit{token} correto em cada
      posição: $p_1 = 0.50$,\ $p_2 = 0.25$,\ $p_3 = 0.125$.
      Qual é o valor de $\mathcal{L}_{\text{MLM}}$?

\begin{enumerate}[a)]
    \item $\approx 0.693$
    \item $\approx 1.386$
    \item $\approx 2.079$
    \item $\approx 4.159$
\end{enumerate}
% R: b

\end{enumerate}

\section*{Soluções}

% --- Solução Q1 ---
\subsection*{I: Resposta: (a)}
O esquema BIO2 exige que o \textbf{primeiro token} de qualquer entidade receba
a \textit{tag} \textbf{B-} (Begin), e os tokens subsequentes da mesma entidade
recebam \textbf{I-} (Inside). Tokens fora de entidades recebem \textbf{O}.
Na frase:
\begin{itemize}
  \item ``Lucas'' $\to$ \texttt{B-PER} (inicia a entidade pessoa).
  \item ``PUCRS'' $\to$ \texttt{B-ORG} (inicia a entidade organização).
  \item ``Porto'' $\to$ \texttt{B-LOC} (inicia a entidade local de dois tokens).
  \item ``Alegre'' $\to$ \texttt{I-LOC} (continua a mesma entidade local).
\end{itemize}
A opção~(b) usa \texttt{I-LOC} para ``Porto'' (erro: nunca se inicia uma
entidade com \texttt{I-}).
A opção~(c) usa \texttt{I-ORG} para ``PUCRS'' (mesmo erro para organização).
A opção~(d) usa \texttt{I-PER} para ``Lucas'' (mesmo erro para pessoa).

% --- Solução Q2 ---
\subsection*{II: Resposta: (c)}
Como $H \cdot d_k = 12 \times 64 = 768 = d$, todas as quatro matrizes pertencem a
$\mathbb{R}^{768 \times 768}$. O número de parâmetros em cada uma é
$768 \times 768 = 589\,824$, e no total:
\[
  4 \times 589\,824 = 2\,359\,296.
\]
A opção~(a) conta apenas $\mathbf{W}^Q$.
A opção~(b) omite a projeção de saída $\mathbf{W}^O$, que transforma a concatenação das
cabeças de volta à dimensão $d$ e é tão grande quanto as demais.
A opção~(d) duplica o resultado correto (erro de dupla contagem).

% --- Solução Q3 ---
\subsection*{III: Resposta: (b)}
A regra $80/10/10$ permite que parte das posições supervisionadas receba
um token visível, original ou aleatório. Se todas recebessem \texttt{[MASK]},
a supervisão direta de reconstrução ficaria restrita a entradas com esse símbolo,
enquanto tarefas discriminativas usualmente recebem texto sem \texttt{[MASK]}.
As posições não selecionadas continuam contendo tokens reais, portanto o modelo
continua aprendendo representações desses tokens.

A opção~(a) é incorreta: o vocabulário de saída permanece inalterado.
A opção~(c) é incorreta: o símbolo de entrada compartilhado não anula os
gradientes, pois os estados dependem também das posições e do contexto.
A opção~(d) é incorreta: a corrupção substitui tokens antes da construção
dos embeddings; os embeddings posicionais continuam sendo somados à entrada.

% --- Solução Q4 ---
\subsection*{IV: Resposta: (c)}
A entrada passa a ser
\[
  \mathbf x_i=\mathbf e_{\text{token}}(w_i)+\mathbf e_{\text{posição}}(i).
\]
A indicação aditiva por $\mathbf e_A$ e $\mathbf e_B$ é removida.
Os separadores e as posições continuam fornecendo informação sobre a
organização da sequência. O efeito no desempenho exige avaliação empírica,
inclusive em tarefas com apenas um segmento.

A opção~(a) remove o componente posicional, que foi mantido.
A opção~(b) remove os tokens separadores, que foram mantidos.
A opção~(d) confunde a remoção de um embedding com uma restrição de atenção:
a máscara continua permitindo interação entre pergunta e contexto.

% --- Solução Q5 ---
\subsection*{V: Resposta: (b)}
MLM calcula termos de reconstrução nas posições de $M$. RTD calcula
termos de classificação em todas as $T$ posições válidas da entrada.
Essa supervisão mais densa é uma fonte de eficiência do ELECTRA.
A classificação do discriminador verifica se o token coincide com o original;
uma amostra do gerador que reproduza o original recebe o rótulo de token original.

Posições fora de $M$ também podem receber gradientes do MLM indiretamente,
pois seus estados participam da contextualização das posições supervisionadas.
A comparação diz respeito aos termos de perda explícitos por posição.

A opção~(a) troca os conjuntos de posições dos dois objetivos.
A opção~(c) restringe indevidamente RTD às posições de $M$.
A opção~(d) estende indevidamente a perda MLM a todas as posições.

% --- Solução Q6 ---
\subsection*{VI: Resposta: (a)}
Um objetivo que compara pares de frases atua sobre a geometria usada na
avaliação por cosseno. A definição dos pares e a função de perda orientam
a aproximação de exemplos similares e a separação de exemplos diferentes.
MLM e NSP supervisionam outras tarefas, portanto o bom desempenho nesses
objetivos não garante embeddings de sentença adequados a STS.

A anisotropia pode prejudicar a discriminação por cosseno, mas os resultados
citados não demonstram que NSP seja sua causa exclusiva, nem que GloVe seja
sempre superior a representações de BERT.

A opção~(b) descreve reconstrução MLM, sem otimização direta da comparação de frases.
A opção~(c) descreve NSP, que classifica a relação entre segmentos apresentados juntos.
A opção~(d) descreve RTD, que supervisiona decisões sobre tokens.

% --- Solução Q7 ---
\subsection*{VII: Resposta: (d)}
A máscara zera a contribuição de $w_3$; o numerador é
$m_1\mathbf{h}_1 + m_2\mathbf{h}_2 + m_3\mathbf{h}_3
 = [2,\,0] + [0,\,2] + \mathbf{0} = [2,\,2]$
e o denominador é $\sum m_i = 1 + 1 + 0 = 2$:
\[
  \mathbf{u} = \frac{[2,\,2]}{2} = [1,\,1].
\]
A opção~(a) inclui $\mathbf{h}_3 = [1,\,3]$ do \texttt{[PAD]} no numerador e usa $L = 3$:
$([2,0]+[0,2]+[1,3])/3 = [3,5]/3 = [1,\,\tfrac{5}{3}]$: resultado distorcido pela posição de \textit{padding}.
A opção~(b) aplica corretamente a máscara ao numerador, mas omite a divisão por $\sum m_i$,
produzindo a soma bruta $[2,\,2]$.
A opção~(c) exclui o \texttt{[PAD]} do numerador (correto), mas divide por $L = 3$ em vez
de $\sum m_i = 2$, gerando $[2,\,2]/3 = [\tfrac{2}{3},\,\tfrac{2}{3}]$: erro de normalização.

% --- Solução Q8 ---
\subsection*{VIII: Resposta: (c)}
Sob a execução sequencial especificada no enunciado:
\[
  t_{\text{cross}}=100\,000\cdot10\text{ ms}=1\,000\,000\text{ ms}.
\]
O bi-encoder codifica a consulta e depois executa a busca vetorial:
\[
  t_{\text{bi}}=10\text{ ms}+10\text{ ms}=20\text{ ms},
  \qquad
  \frac{t_{\text{cross}}}{t_{\text{bi}}}=50\,000.
\]
Os valores são hipóteses deste exemplo. Em uma aplicação, batching, hardware,
índice e tamanho dos vetores afetam a comparação.

A opção~(a) equivaleria a atribuir $100\text{ ms}$ ao bi-encoder.
A opção~(b) equivaleria a atribuir $40\text{ ms}$ ao bi-encoder.
A opção~(d) desconsidera os $10\text{ ms}$ da busca vetorial.

% --- Solução Q9 ---
\subsection*{IX: Resposta: (a)}
Para cada âncora, o objetivo direcional compara $N$ candidatos:
\[
  \ell_i=-\ln\frac{\exp(\operatorname{sim}(f(x_i),f(x_i^+))/\tau)}
    {\sum_{j=1}^{N}\exp(\operatorname{sim}(f(x_i),f(x_j^+))/\tau)}.
\]
Um candidato é o positivo rotulado. Os demais são tratados como negativos,
portanto há $N-1=1\,023$ candidatos negativos por âncora.
O aumento do lote amplia as comparações, mas o efeito na qualidade depende
dos dados, da temperatura e da presença de falsos negativos. Lotes maiores
também oferecem mais oportunidades de colisões semânticas.

A opção~(b) inclui o positivo na contagem de negativos.
A opção~(c) duplica a contagem, como se as duas direções fossem reunidas na perda de cada âncora.
A opção~(d) conta como negativos todos os outros $2N-1$ elementos do conjunto de pares,
embora o objetivo especificado use apenas os $N$ candidatos $x_j^+$.

% --- Solução Q10 ---
\subsection*{X: Resposta: (d)}
\textbf{Cálculo das distâncias quadráticas:}
\begin{align*}
  \|\mathbf{f}(x) - \mathbf{f}(x^+)\|_2^2
    &= (1.0 - 0.6)^2 + (0.0 - 0.8)^2
     = 0.16 + 0.64 = 0.80, \\
  \|\mathbf{f}(x) - \mathbf{f}(x^-)\|_2^2
    &= (1.0 - (-0.6))^2 + (0.0 - 0.8)^2
     = 2.56 + 0.64 = 3.20.
\end{align*}
\textbf{Aplicando a fórmula:}
\[
  \mathcal{L}_{\text{triplet}}
    = \max(0,\; 0.80 - 3.20 + 0.50)
    = \max(0,\; -1.90)
    = 0.
\]
O positivo ($d^2 = 0.80$) já está $2.40$ unidades mais próximo que o negativo
($d^2 = 3.20$), excedendo a margem $\alpha = 0.50$: o par não viola a
restrição e não gera gradiente.
A opção~(a) ocorreria se as duas distâncias fossem iguais ($d^+ = d^-$); nesse
caso $\mathcal{L} = \alpha = 0.50$.
A opção~(b), $1.90$, troca o sinal do argumento negativo antes da aplicação do máximo.
A opção~(c), $2.90$, usa $d_-^2-d_+^2+\alpha=3.20-0.80+0.50$,
invertendo o papel das distâncias na fórmula.

% --- Solução Q11 ---
\subsection*{XI: Resposta: (b)}
As probabilidades dos tokens corretos são $p_1 = \tfrac{1}{2}$,
$p_2 = \tfrac{1}{4}$, $p_3 = \tfrac{1}{8}$. Aplicando a fórmula com $|M| = 3$:
\begin{align*}
  \mathcal{L}_{\text{MLM}}
    &= -\frac{1}{3}\Bigl(\ln\tfrac{1}{2} + \ln\tfrac{1}{4} + \ln\tfrac{1}{8}\Bigr) \\
    &= -\frac{1}{3}\Bigl(-\ln 2 - 2\ln 2 - 3\ln 2\Bigr) \\
    &= \frac{6\ln 2}{3} = 2\ln 2 \approx 1.386.
\end{align*}
\textbf{Análise das alternativas incorretas:}
\begin{itemize}
  \item \emph{(a) $\approx 0.693$}: considera apenas $-\ln(0.50) = \ln 2$,
        ou seja, somente a primeira posição, sem a média.
  \item \emph{(c) $\approx 2.079$}: usa $|M| = 2$ no denominador ($6\ln 2 / 2 = 3\ln 2$),
        como se apenas duas posições fossem selecionadas.
  \item \emph{(d) $\approx 4.159$}: omite a normalização $1/|M|$ por completo
        ($-(\ln\tfrac{1}{2}+\ln\tfrac{1}{4}+\ln\tfrac{1}{8}) = 6\ln 2$).
\end{itemize}
