  \section{Modelos Geradores -- Revisão}

  %--- Slide: definição ---
  \begin{frame}{Modelos Geradores}
    \begin{itemize}
      \item O objetivo de um \textbf{modelo gerador} é aprender uma
      \textcolor{EncRed}{distribuição de probabilidade} sobre uma
      variável $\mathbf{X}$ usando dados observados
      $\{\mathbf{x}^{(i)}\}_{i=1}^{N}$.
      \item A densidade $p_{X}(\mathbf{x})$ é parametrizada por $\thetav$.
      \item Já vimos:
      \begin{itemize}
        \item \textbf{GANs} (jogo min-max),
        \item \textbf{Normalizing Flows} (transformações invertíveis),
        \item \textbf{VAEs} (encoder/decoder + ELBO).
      \end{itemize}
    \end{itemize}
  \end{frame}

  %--- Slide: ideia diffusion ---
  \begin{frame}{A próxima ideia: \textit{Diffusion Models}\footfullcite{ho2020denoising}}
    \begin{itemize}
      \item Definir um \textcolor{EncRed}{\textit{encoder}
      \textbf{sem parâmetros treináveis}} que degrada uma imagem
      adicionando ruído gaussiano de maneira sequencial.
      \item Treinar um \textcolor{DecBlue}{\textit{decoder}} para
      \textbf{retirar o ruído} adicionado.
    \end{itemize}
    \vspace{0.5em}
    \begin{block}{Intuição}
      Aprender a remover ruído um pequeno passo de cada vez é mais
      fácil que aprender a gerar uma imagem inteira do zero.
      Empilhando muitos passos pequenos, conseguimos transformar ruído
      puro em uma amostra realista.
    \end{block}
  \end{frame}

  %======================================================================
  % SECTION 2: ENCODER / PROCESSO DE DIFUSÃO
  %======================================================================
  \section{Encoder: Processo de Difusão}

  %--- Slide: visão geral ---
  \begin{frame}{Visão Geral -- Difusão}
    \begin{itemize}
      \item No processo de difusão (\textit{forward}) adicionamos ruído
      a cada passo até a imagem ficar indistinguível do ruído.
      \item No processo de \textit{denoising} (decoder) treinamos uma
      rede neural para \textbf{remover um passo} de ruído.
      \item Após o treinamento podemos gerar novas instâncias partindo
      de $\mathbf{z}_T \sim \mathcal{N}(0,\mathbf{I})$.
    \end{itemize}
  \end{frame}


  %--------------------------------------------------------------------
  % Definição da difusão (forward)
  %--------------------------------------------------------------------

  %--- Slide: Difusão definição ---
  \begin{frame}{Difusão -- Definição}
    O processo de difusão (\textit{forward}) mapeia um exemplo $\mathbf{x}$ por
    uma série de etapas intermediárias $\mathbf{z}_1, \mathbf{z}_2, \ldots, \mathbf{z}_T$ via
    adição de ruído:
    \vspace{0.3em}
    \[
      \mathbf{z}_1 = \sqrt{1 - \beta_1}\cdot \mathbf{x} + \sqrt{\beta_1}\cdot \boldsymbol{\epsilon}_1
    \]
    \[
      \mathbf{z}_t = \sqrt{1 - \beta_t}\cdot \mathbf{z}_{t-1} + \sqrt{\beta_t}\cdot \boldsymbol{\epsilon}_t
      \qquad \forall t \in \{2, 3, \ldots, T\}
    \]
    \vspace{0.2em}
    \begin{itemize}
      \item $\boldsymbol{\epsilon}_t \sim \mathcal{N}(0,\mathbf{I})$: ruído amostrado
      da normal padrão.
      \item Hiperparâmetros $\beta_t \in [0,1]$ são o
      \textbf{\textit{noise schedule}} (controlam a velocidade da
      difusão).
    \end{itemize}
  \end{frame}

  %--- Slide: Forma probabilística ---
  \begin{frame}{Difusão -- Forma probabilística (Markov)}
    Equivalentemente, podemos escrever:
    \vspace{0.3em}
    \[
      q(\mathbf{z}_1 \mid \mathbf{x}) = \mathcal{N}_{\mathbf{z}_1}\!\left[\sqrt{1 - \beta_1}\cdot \mathbf{x},\, \beta_1 \cdot \mathbf{I}\right]
    \]
    \[
      q(\mathbf{z}_t \mid \mathbf{z}_{t-1}) = \mathcal{N}_{\mathbf{z}_t}\!\left[\sqrt{1 - \beta_t}\cdot \mathbf{z}_{t-1},\, \beta_t \cdot \mathbf{I}\right]
      \qquad \forall t \geq 2
    \]
    \vspace{0.3em}
    \begin{block}{Cadeia de Markov}
      $\mathbf{z}_t$ depende apenas de $\mathbf{z}_{t-1}$. Por isso, a distribuição
      conjunta fatoriza:
      \[
        q(\mathbf{z}_{1\ldots T} \mid \mathbf{x}) = q(\mathbf{z}_1 \mid \mathbf{x})\, \prod_{t=2}^{T} q(\mathbf{z}_t \mid \mathbf{z}_{t-1}).
      \]
    \end{block}
  \end{frame}

  %--- Slide: trajetórias 1D e convergência para ruído ---
  \begin{frame}{Difusão -- Trajetórias 1D}
    \begin{columns}[T]
      \begin{column}{0.46\textwidth}

          \item Partimos de \textbf{três instâncias} $\mathbf{x}^{(1)}, \mathbf{x}^{(2)},

          \item Conforme os passos de difusão são aplicados, as
          instâncias são combinadas com ruído e vão aos poucos perdendo
          identidade.
          \item Após muitos passos, a distribuição converge para a
          distribuição do ruído $\mathcal{N}(0,\mathbf{I})$.
        \end{itemize}
      \end{column}
      \begin{column}{0.52\textwidth}

  %--------------------------------------------------------------------
  % Kernel de difusão: forma fechada q(z_t | x)
  %--------------------------------------------------------------------

  %--- Slide: motivação kernel ---
  \begin{frame}{Kernel de Difusão -- Motivação}
    \begin{itemize}
      \item Como o processo de difusão é probabilístico, para uma mesma
      instância $\mathbf{x}$ obtemos \textbf{várias amostras} $\mathbf{z}_t$ em cada
      instante.
      \item O processo definido até aqui é \textbf{sequencial}: para
      chegar em $\mathbf{z}_t$ precisamos passar por $\mathbf{z}_1, \ldots, \mathbf{z}_{t-1}$.
      \item Para acelerar, vamos derivar uma \textbf{forma fechada}
      para amostrar diretamente $\mathbf{z}_t$ a partir de $\mathbf{x}$.
    \end{itemize}
  \end{frame}

  %--- Slide: kernel - derivação z2 ---
  \begin{frame}{Kernel de Difusão -- Substituindo $\mathbf{z}_1$ em $\mathbf{z}_2$}
    Partindo das definições:
    \vspace{0.2em}
    \[
      \mathbf{z}_1 = \sqrt{1 - \beta_1}\cdot \mathbf{x} + \sqrt{\beta_1}\cdot \boldsymbol{\epsilon}_1,
      \qquad
      \mathbf{z}_2 = \sqrt{1 - \beta_2}\cdot \mathbf{z}_1 + \sqrt{\beta_2}\cdot \boldsymbol{\epsilon}_2.
    \]
    \vspace{0.2em}
    Substituindo $\mathbf{z}_1$:
    \[
      \mathbf{z}_2 = \sqrt{1 - \beta_2}\!\left(\sqrt{1 - \beta_1}\cdot \mathbf{x} + \sqrt{\beta_1}\cdot \boldsymbol{\epsilon}_1\right) + \sqrt{\beta_2}\cdot \boldsymbol{\epsilon}_2.
    \]
  \end{frame}

  %--- Slide: kernel - manipulação algébrica ---
  \begin{frame}{Kernel de Difusão -- Manipulação algébrica}
    Reescrevendo $\sqrt{\beta_1} = \sqrt{1 - (1 - \beta_1)}$:
    \vspace{0.2em}
    \[
      \mathbf{z}_2 = \sqrt{1-\beta_2}\!\left(\sqrt{1-\beta_1}\,\mathbf{x} + \sqrt{1-(1-\beta_1)}\,\boldsymbol{\epsilon}_1\right) + \sqrt{\beta_2}\,\boldsymbol{\epsilon}_2.
    \]
    \vspace{0.2em}
    Distribuindo:
    \[
      \mathbf{z}_2 = \sqrt{(1-\beta_2)(1-\beta_1)}\,\mathbf{x}
      + \sqrt{(1-\beta_2) - (1-\beta_2)(1-\beta_1)}\,\boldsymbol{\epsilon}_1
      + \sqrt{\beta_2}\,\boldsymbol{\epsilon}_2.
    \]
  \end{frame}

  %--- Slide: soma de gaussianas ---
  \begin{frame}{Kernel de Difusão -- Soma de gaussianas}
    \begin{alertblock}{Lembrete}
      A soma de duas gaussianas independentes com médias nulas e
      desvios $\sigma_a, \sigma_b$ é uma gaussiana com média zero e
      variância $\sigma_a^2 + \sigma_b^2$.
    \end{alertblock}
    \vspace{0.3em}
    Os dois últimos termos viram um único ruído gaussiano:
    \[
      \mathbf{z}_2 = \sqrt{(1-\beta_2)(1-\beta_1)}\cdot \mathbf{x}
      + \sqrt{1 - (1-\beta_2)(1-\beta_1)}\cdot \boldsymbol{\epsilon}.
    \]
    \begin{block}{Observação}
      Podemos repetir esse processo para todos os $\mathbf{z}_t$, sempre
      acumulando os fatores $(1 - \beta_s)$.
    \end{block}
  \end{frame}

  %--- Slide: kernel forma geral ---
  \begin{frame}{Kernel de Difusão -- Forma geral}
    Definindo $\alpha_t = \prod_{s=1}^{t}(1 - \beta_s)$, temos:
    \vspace{0.3em}
    \[
      \boxed{\mathbf{z}_t = \sqrt{\alpha_t}\cdot \mathbf{x} + \sqrt{1 - \alpha_t}\cdot \boldsymbol{\epsilon}}
      \qquad \boldsymbol{\epsilon} \sim \mathcal{N}(0,\mathbf{I})
    \]
    \vspace{0.3em}
    Em forma probabilística:
    \[
      q(\mathbf{z}_t \mid \mathbf{x}) = \mathcal{N}_{\mathbf{z}_t}\!\left[\sqrt{\alpha_t}\cdot \mathbf{x},\; (1 - \alpha_t)\cdot \mathbf{I}\right].
    \]
    \begin{block}{Vantagem prática}
      Podemos amostrar $\mathbf{z}_t$ \textbf{diretamente} a partir de $\mathbf{x}$ sem
      simular toda a cadeia. Isso permite treinamento por
      \textit{minibatch} eficiente.
    \end{block}
  \end{frame}

  %--- Slide: Exemplo numérico schedule ---
  \begin{frame}{Exemplo Numérico: \textit{Schedule} $\beta$ e Sinal vs Ruído}
    \footnotesize Pense em $\mathbf{x}$ como uma \emph{imagem limpa} (e.g., um
    pixel ou um vetor latente). $\beta_t$ é a \emph{variância de ruído}
    adicionada no passo $t$ do \textit{forward process}; $\alpha_t$ é o
    quanto sobrou de $\mathbf{x}$ após $t$ passos. Aqui usamos $T = 4$ para
    visualização, DDPM real tem $T = 1000$ e $\beta$ na faixa
    $10^{-4}$ a $0.02$, mas a curva é a mesma.

    \normalsize
    \textit{Schedule} linear com $T = 4$:
    $\boldsymbol{\beta} = (0.1,\, 0.2,\, 0.3,\, 0.4)$.
    Acumulamos $\alpha_t = \prod_{s=1}^{t}(1 - \beta_s)$:
    \vspace{0.2em}
    \begin{columns}[T]
      \begin{column}{0.55\textwidth}
        \centering
        \begin{tabular}{@{}c|c|c|c@{}}
          \toprule
          $t$ & $\alpha_t$  & $\sqrt{\alpha_t}$ \,(sinal)& $\sqrt{1-\alpha_t}$ \,(ruído) \\
          \midrule
          1 & 0.900 & 0.949 & 0.316 \\
          2 & 0.720 & 0.849 & 0.529 \\
          3 & 0.504 & 0.710 & 0.704 \\
          4 & 0.302 & 0.550 & 0.836 \\
          \bottomrule
        \end{tabular}
        \vspace{0.4em}\\
        \scriptsize Ex.: $\alpha_3 = 0.9 \cdot 0.8 \cdot 0.7 = 0.504$.
      \end{column}
      \begin{column}{0.42\textwidth}
        Para $\mathbf{x} = 1$:
        \[
          \mathbf{z}_t = \sqrt{\alpha_t}\cdot 1 + \sqrt{1 - \alpha_t}\cdot \boldsymbol{\epsilon}
        \]
        \vspace{-0.4em}
        Em $t = 3$ o sinal e o ruído se cruzam; em $t = T$ grande,
        $\alpha_T \to 0$ e $\mathbf{z}_T \approx \mathcal{N}(0, \mathbf{I})$
        \textbf{por construção}, é o que permite o \textit{decoder}
        começar a amostragem de ruído puro.
      \end{column}
    \end{columns}
    \note{
      Em DDPM real, $T = 1000$ e $\beta$ é uma rampa linear de $10^{-4}$ a
      $0.02$, o que mantém $\alpha_T$ próximo de zero sem destruir
      sinal cedo demais. A intuição é a mesma da tabela acima.
      Schedules de \emph{cosseno} (Nichol \& Dhariwal, 2021) destroem o
      sinal mais devagar no meio da cadeia e costumam superar o linear em
      qualidade, sobretudo quando $T$ é grande.
    }
  \end{frame}

  % SECTION 3: DECODER
  %======================================================================
  \section{Decoder}

  %--------------------------------------------------------------------
  % Distribuição condicional da difusão
  %--------------------------------------------------------------------

  %--- Slide: motivação ---
  \begin{frame}{Distribuição condicional $q(\mathbf{z}_{t-1} \mid \mathbf{z}_t, \mathbf{x})$}
    Para o decoder vamos precisar de uma expressão para a difusão
    \textbf{revertida no tempo}, condicionada em $\mathbf{x}$:
    \vspace{0.3em}
    \[
      q(\mathbf{z}_{t-1} \mid \mathbf{z}_t, \mathbf{x}) = \frac{q(\mathbf{z}_t \mid \mathbf{z}_{t-1}, \mathbf{x})\, q(\mathbf{z}_{t-1} \mid \mathbf{x})}{q(\mathbf{z}_t \mid \mathbf{x})}
    \]
    \vspace{0.3em}
    Por regra de Bayes. Vamos manipular essa expressão.
  \end{frame}

  %--- Slide: simplificação Markov ---
  \begin{frame}{Distribuição condicional -- Simplificação Markov}
    \begin{itemize}
      \item Como difusão é Markov:
      $q(\mathbf{z}_t \mid \mathbf{z}_{t-1}, \mathbf{x}) = q(\mathbf{z}_t \mid \mathbf{z}_{t-1})$.
      \item Tirando o denominador, a igualdade vira proporcionalidade:
    \end{itemize}
    \vspace{0.3em}
    \[
      q(\mathbf{z}_{t-1} \mid \mathbf{z}_t, \mathbf{x}) \;\propto\; q(\mathbf{z}_t \mid \mathbf{z}_{t-1})\, q(\mathbf{z}_{t-1} \mid \mathbf{x}).
    \]
  \end{frame}

  %--- Slide: produto de gaussianas ---
  \begin{frame}{Distribuição condicional -- Produto de gaussianas}
    Substituindo cada termo pela sua respectiva normal:
    \[
      \propto
      \mathcal{N}_{\mathbf{z}_t}\!\left[\sqrt{1 - \beta_t}\cdot \mathbf{z}_{t-1},\, \beta_t\,\mathbf{I}\right]
      \cdot
      \mathcal{N}_{\mathbf{z}_{t-1}}\!\left[\sqrt{\alpha_{t-1}}\cdot \mathbf{x},\, (1 - \alpha_{t-1})\,\mathbf{I}\right].
    \]
    \begin{alertblock}{Mudança de variável}
      O primeiro fator é uma gaussiana em $\mathbf{z}_t$, mas pode ser reescrito
      como uma gaussiana em $\mathbf{z}_{t-1}$:
      \[
        \mathcal{N}_{\mathbf{z}_t}\!\left[\sqrt{1 - \beta_t}\cdot \mathbf{z}_{t-1},\, \beta_t\,\mathbf{I}\right]
        \;\propto\;
        \mathcal{N}_{\mathbf{z}_{t-1}}\!\left[\tfrac{1}{\sqrt{1 - \beta_t}}\, \mathbf{z}_t,\;
        \tfrac{\beta_t}{1 - \beta_t}\,\mathbf{I}\right].
      \]
    \end{alertblock}
    O produto de duas gaussianas em $\mathbf{z}_{t-1}$ é proporcional a uma
    gaussiana em $\mathbf{z}_{t-1}$; como o lado esquerdo é uma distribuição, a
    normalização é automática:
    \[
      q(\mathbf{z}_{t-1} \mid \mathbf{z}_t, \mathbf{x}) = \mathcal{N}_{\mathbf{z}_{t-1}}\!\left[
        \tfrac{(1 - \alpha_{t-1})}{1 - \alpha_t}\sqrt{1 - \beta_t}\, \mathbf{z}_t
        + \tfrac{\sqrt{\alpha_{t-1}}\,\beta_t}{1 - \alpha_t}\, \mathbf{x},\;
        \tfrac{\beta_t (1 - \alpha_{t-1})}{1 - \alpha_t}\,\mathbf{I}
      \right]
    \]
    \note{
      A identidade segue de completar o quadrado: o expoente
      $-(\mathbf{z}_t - \sqrt{1-\beta_t}\,\mathbf{z}_{t-1})^2 / (2\beta_t)$ é o mesmo de
      uma gaussiana em $\mathbf{z}_{t-1}$ com média $\mathbf{z}_t/\sqrt{1-\beta_t}$ e
      variância $\beta_t/(1-\beta_t)$, a menos de fator constante em
      $\mathbf{z}_{t-1}$. No produto final, usar
      $\alpha_t = \alpha_{t-1}(1-\beta_t)$ faz os denominadores
      colapsarem em $1 - \alpha_t$.
    }
  \end{frame}

  %--------------------------------------------------------------------
  % Aproximação do reverse process
  %--------------------------------------------------------------------

  %--- Slide: decoder ---
  \begin{frame}{Decoder (\textit{Reverse Process})}
    O aprendizado está \textbf{todo no decoder}:
    \vspace{0.3em}
    \[
      \mathbf{z}_T \;\to\; \mathbf{z}_{T-1} \;\to\; \cdots \;\to\; \mathbf{z}_1 \;\to\; \mathbf{x}.
    \]
    \vspace{0.3em}
    \begin{itemize}
      \item A distribuição reversa real $q(\mathbf{z}_{t-1} \mid \mathbf{z}_t)$ é
      complexa e potencialmente multimodal, depende dos dados.
      \item Na prática, \textbf{aproximamos} essa reversa por uma
      \emph{normal}:
    \end{itemize}
    \[
      p(\mathbf{z}_T) = \mathcal{N}_{\mathbf{z}_T}[0,\mathbf{I}],
      \qquad
      p(\mathbf{z}_{t-1} \mid \mathbf{z}_t,\thetav_t) = \mathcal{N}_{\mathbf{z}_{t-1}}\!\left[f_t[\mathbf{z}_t,\thetav_t],\; \sigma_t^{2}\,\mathbf{I}\right],
    \]
    \[
      p(\mathbf{x} \mid \mathbf{z}_1,\thetav_1) = \mathcal{N}_{\mathbf{x}}\!\left[f_1[\mathbf{z}_1,\thetav_1],\; \sigma_1^{2}\,\mathbf{I}\right].
    \]
    \begin{block}{Quando essa aproximação é razoável}
      Se o número de passos $T$ for grande e $\beta_t$ for próximo de
      0, cada passo da reversa é quase determinístico e a aproximação
      gaussiana é boa.
    \end{block}
  \end{frame}

  %--------------------------------------------------------------------
  % Treinamento e ELBO
  %--------------------------------------------------------------------

  %--- Slide: motivação treinamento ---
  \begin{frame}{Treinamento -- O problema}
    Maximizar a log-verossimilhança dos dados é o objetivo:
    \vspace{0.3em}
    \[
      \hat{\thetav}_{1\ldots T} = \argmax_{\thetav_{1\ldots T}}
      \left[\sum_{i=1}^{N} \log p(\mathbf{x}^{(i)} \mid \thetav_{1\ldots T})\right]
    \]
    \[
      p(\mathbf{x} \mid \thetav_{1\ldots T}) = \int p(\mathbf{x}, \mathbf{z}_{1\ldots T} \mid \thetav_{1\ldots T})\, d\mathbf{z}_{1\ldots T}
    \]
    \begin{alertblock}{Inviável}
      Essa integral é \textbf{intratável}. Vamos usar o
      \textbf{ELBO} como substituto.
    \end{alertblock}
  \end{frame}

  %--- Slide: distribuições conjuntas ---
  \begin{frame}{Treinamento -- Distribuições conjuntas}
    \begin{block}{Conjunta do modelo (decoder)}
      \[
        p(\mathbf{x}, \mathbf{z}_{1\ldots T} \mid \thetav_{1\ldots T})
        = p(\mathbf{x} \mid \mathbf{z}_1,\thetav_1)\, \prod_{t=2}^{T} p(\mathbf{z}_{t-1} \mid \mathbf{z}_t,\thetav_t)\cdot p(\mathbf{z}_T)
      \]
    \end{block}
    \begin{block}{Conjunta da difusão (encoder)}
      \[
        q(\mathbf{z}_{1\ldots T} \mid \mathbf{x}) = q(\mathbf{z}_1 \mid \mathbf{x})\, \prod_{t=2}^{T} q(\mathbf{z}_t \mid \mathbf{z}_{t-1})
      \]
    \end{block}
    \vspace{0.2em}
    Tudo gaussiano $\Rightarrow$ tudo tratável passo a passo.
  \end{frame}

  %--- Slide: derivação ELBO ---
  \begin{frame}{Evidence Lower Bound (ELBO)}
    Multiplicamos e dividimos por $q(\mathbf{z}_{1\ldots T}\mid \mathbf{x})$, e aplicamos
    a desigualdade de Jensen:
    \vspace{0.2em}
    \begin{align*}
      \log p(\mathbf{x} \mid \thetav_{1\ldots T})
      &= \log\!\left[\int p(\mathbf{x}, \mathbf{z}_{1\ldots T} \mid \thetav_{1\ldots T})\, d\mathbf{z}\right] \\
      &= \log\!\left[\int q(\mathbf{z}_{1\ldots T} \mid \mathbf{x})\frac{p(\mathbf{x}, \mathbf{z}_{1\ldots T} \mid \thetav_{1\ldots T})}{q(\mathbf{z}_{1\ldots T} \mid \mathbf{x})}\, d\mathbf{z}\right] \\
      &\geq \int q(\mathbf{z}_{1\ldots T} \mid \mathbf{x}) \log\!\left[\frac{p(\mathbf{x}, \mathbf{z}_{1\ldots T} \mid \thetav_{1\ldots T})}{q(\mathbf{z}_{1\ldots T} \mid \mathbf{x})}\right] d\mathbf{z}
      \quad \text{(Jensen)}
    \end{align*}
    \vspace{0.1em}
    Vamos trabalhar nesse termo (o ELBO).
  \end{frame}

  %--- Slide: ELBO substituindo ---
  \begin{frame}{ELBO -- Substituindo as conjuntas}
    Substituindo as definições do encoder e decoder:
    \vspace{0.2em}
    \[
      \log\!\left[\frac{p(\mathbf{x}|\mathbf{z}_1,\thetav_1) \prod_{t=2}^{T} p(\mathbf{z}_{t-1}|\mathbf{z}_t,\thetav_t)\cdot p(\mathbf{z}_T)}{q(\mathbf{z}_1|\mathbf{x})\prod_{t=2}^{T} q(\mathbf{z}_t|\mathbf{z}_{t-1})}\right]
    \]
    \vspace{0.1em}
    \[
      = \log\!\left[\frac{p(\mathbf{x}|\mathbf{z}_1,\thetav_1)}{q(\mathbf{z}_1|\mathbf{x})}\right]
      + \log\!\left[\frac{\prod_{t=2}^{T} p(\mathbf{z}_{t-1}|\mathbf{z}_t,\thetav_t)}{\prod_{t=2}^{T} q(\mathbf{z}_t|\mathbf{z}_{t-1})}\right]
      + \log p(\mathbf{z}_T).
    \]
  \end{frame}

  %--- Slide: ELBO Bayes no q ---
  \begin{frame}{ELBO -- Aplicando Bayes em $q(\mathbf{z}_t|\mathbf{z}_{t-1})$}
    Pela regra de Bayes (e Markov):
    \[
      q(\mathbf{z}_t \mid \mathbf{z}_{t-1}) = \frac{q(\mathbf{z}_{t-1} \mid \mathbf{z}_t, \mathbf{x})\, q(\mathbf{z}_t \mid \mathbf{x})}{q(\mathbf{z}_{t-1} \mid \mathbf{x})}.
    \]
    Aplicando isso na razão dentro do produtório, vários termos se
    cancelam, sobrando apenas $q(\mathbf{z}_1 \mid \mathbf{x})$ no numerador e
    $q(\mathbf{z}_T \mid \mathbf{x})$ no denominador:
    \[
      \prod_{t=2}^{T}\frac{1}{q(\mathbf{z}_t \mid \mathbf{z}_{t-1})} = \prod_{t=2}^{T}\frac{q(\mathbf{z}_{t-1} \mid \mathbf{x})}{q(\mathbf{z}_{t-1} \mid \mathbf{z}_t, \mathbf{x})\, q(\mathbf{z}_t \mid \mathbf{x})}.
    \]
  \end{frame}

  %--- Slide: ELBO cancelamento explícito ---
  \begin{frame}{ELBO -- O cancelamento em detalhe}
    Separamos o produtório em uma parte que permanece e outra cujos
    termos se cancelam:
    \[
      \prod_{t=2}^{T}\frac{q(\mathbf{z}_{t-1} \mid \mathbf{x})}{q(\mathbf{z}_{t-1} \mid \mathbf{z}_t, \mathbf{x})\, q(\mathbf{z}_t \mid \mathbf{x})}
      = \underbrace{\prod_{t=2}^{T}\frac{1}{q(\mathbf{z}_{t-1} \mid \mathbf{z}_t, \mathbf{x})}}_{\text{permanece}}
        \cdot \underbrace{\prod_{t=2}^{T}\frac{q(\mathbf{z}_{t-1} \mid \mathbf{x})}{q(\mathbf{z}_t \mid \mathbf{x})}}_{\text{cancela}}
    \]
    \vspace{0.4em}
    Ao expandir essa segunda parte, cada denominador cancela o
    numerador seguinte, sobrando apenas os extremos:
    \[
      \prod_{t=2}^{T}\frac{q(\mathbf{z}_{t-1} \mid \mathbf{x})}{q(\mathbf{z}_t \mid \mathbf{x})}
      = \frac{q(\mathbf{z}_1 \mid \mathbf{x})}{\cancel{q(\mathbf{z}_2 \mid \mathbf{x})}}\cdot
        \frac{\cancel{q(\mathbf{z}_2 \mid \mathbf{x})}}{\cancel{q(\mathbf{z}_3 \mid \mathbf{x})}}\cdots
        \frac{\cancel{q(\mathbf{z}_{T-1} \mid \mathbf{x})}}{q(\mathbf{z}_T \mid \mathbf{x})}
      = \frac{q(\mathbf{z}_1 \mid \mathbf{x})}{q(\mathbf{z}_T \mid \mathbf{x})}.
    \]
  \end{frame}

  %--- Slide: ELBO simplificação ---
  \begin{frame}{ELBO -- Simplificação}
    Após o cancelamento, a expressão fica:
    \vspace{0.2em}
    \[
      = \log\!\left[\frac{p(\mathbf{x}|\mathbf{z}_1,\thetav_1)\,q(\mathbf{z}_1|\mathbf{x})}{q(\mathbf{z}_1|\mathbf{x})}\right]
      + \log\!\left[\frac{\prod_{t=2}^{T} p(\mathbf{z}_{t-1}|\mathbf{z}_t,\thetav_t)}{\prod_{t=2}^{T} q(\mathbf{z}_{t-1}|\mathbf{z}_t,\mathbf{x})}\right]
      + \log\!\left[\frac{p(\mathbf{z}_T)}{q(\mathbf{z}_T|\mathbf{x})}\right]
    \]
    \vspace{0.2em}
    \[
      = \log p(\mathbf{x}|\mathbf{z}_1,\thetav_1)
      + \log\!\left[\frac{\prod_{t=2}^{T} p(\mathbf{z}_{t-1}|\mathbf{z}_t,\thetav_t)}{\prod_{t=2}^{T} q(\mathbf{z}_{t-1}|\mathbf{z}_t,\mathbf{x})}\right]
      + \cancel{\log\!\left[\tfrac{p(\mathbf{z}_T)}{q(\mathbf{z}_T|\mathbf{x})}\right]}^{\approx 0}
    \]
    \vspace{0.2em}
    O último termo é desprezível: $q(\mathbf{z}_T|\mathbf{x})\approx p(\mathbf{z}_T)$ para $T$
    grande.
  \end{frame}

  %--- Slide: ELBO formulação final ---
  \begin{frame}{ELBO -- Formulação final}
    Tomando esperança em $q(\mathbf{z}_{1\ldots T} \mid \mathbf{x})$:
    \vspace{0.2em}
    \[
      \boxed{
        \mathrm{ELBO}[\thetav_{1\ldots T}]
        = \mathbb{E}_{q(\mathbf{z}_1|\mathbf{x})}\!\big[\log p(\mathbf{x}|\mathbf{z}_1,\thetav_1)\big]
        - \sum_{t=2}^{T} \mathbb{E}_{q(\mathbf{z}_t|\mathbf{x})}\!\big[D_{KL}\!\big[q(\mathbf{z}_{t-1}|\mathbf{z}_t,\mathbf{x})\,\|\,p(\mathbf{z}_{t-1}|\mathbf{z}_t,\thetav_t)\big]\big]
      }
    \]
    \vspace{0.3em}
    \begin{itemize}
      \item \textcolor{ReconGreen}{\textbf{Termo de reconstrução}}:
      $\log p(\mathbf{x}|\mathbf{z}_1,\thetav_1)$ é uma normal $\Rightarrow$ MSE.
      \item \textcolor{DecBlue}{\textbf{Termos KL}}: ambas
      distribuições são normais $\Rightarrow$ \textbf{forma fechada}!
    \end{itemize}
  \end{frame}

  %--- Slide: KL fechada ---
  \begin{frame}{ELBO -- KL entre normais}
    A KL entre duas gaussianas tem expressão fechada. Aplicando-a:
    \vspace{0.4em}
    \begin{block}{KL desenvolvida}
      \[
        \mathbb{E}_{q(\mathbf{z}_t|\mathbf{x})}\!\left[D_{KL}\!\left(q\,\|\,p\right)\right]
        =
        \tfrac{1}{2\sigma_t^{2}}\!\left|
          \tfrac{1-\alpha_{t-1}}{1-\alpha_t}\sqrt{1-\beta_t}\, \mathbf{z}_t
          + \tfrac{\sqrt{\alpha_{t-1}}\,\beta_t}{1-\alpha_t}\, \mathbf{x}
          - f_t[\mathbf{z}_t,\thetav_t]
        \right|^{2}
      \]
    \end{block}
    \vspace{0.3em}
    Substituindo na expressão do ELBO obtemos a \textbf{Diffusion Loss
    Function}.
  \end{frame}

  %--- Slide: Diffusion Loss ---
  \begin{frame}{\textit{Diffusion Loss Function}}
    \resizebox{\linewidth}{!}{$\displaystyle
      \mathcal{L}[\thetav_{1\ldots T}]
      = \underbrace{\sum_{i=1}^{N}\!\big(-\log\mathcal{N}_{\mathbf{x}}[f_1[\mathbf{z}_1,\thetav_1],\sigma_1^2\mathbf{I}]\big)}_{\textcolor{ReconGreen}{\text{reconstrução}}}
      + \underbrace{\sum_{t=2}^{T}\tfrac{1}{2\sigma_t^{2}}\left|
        \tfrac{1-\alpha_{t-1}}{1-\alpha_t}\sqrt{1-\beta_t}\, \mathbf{z}_t
        + \tfrac{\sqrt{\alpha_{t-1}}\,\beta_t}{1-\alpha_t}\, \mathbf{x}
        - f_t[\mathbf{z}_t,\thetav_t]
      \right|^{2}}_{\textcolor{DecBlue}{\text{erro quadrático no \textit{timestep} } t}}
    $}
    \vspace{0.5em}
    \begin{itemize}
      \item Forma de \textbf{MSE} em vez de log-verossimilhança
      intratável.
      \item Cada termo $t$ pede a rede $f_t$ a prever a média da
      reversa em $\mathbf{z}_{t-1}$.
    \end{itemize}
  \end{frame}

  %--------------------------------------------------------------------
  % Reparametrização da loss e da rede
  %--------------------------------------------------------------------

  %--- Slide: porém ---
  \begin{frame}{Porém\ldots\ um detalhe empírico}
    \begin{alertblock}{Observação experimental}
      Após treinar diversos modelos difusores que tentavam descobrir
      \emph{diretamente} a imagem, percebeu-se empiricamente que os
      resultados eram \textbf{melhores} ao usar a rede para
      \textbf{estimar o ruído} $\boldsymbol{\epsilon}$ adicionado, e não a imagem.
    \end{alertblock}
    \vspace{0.5em}
    Vamos reparametrizar a função de custo para que a rede prediga
    $\boldsymbol{\epsilon}$ em vez de $\mathbf{z}_{t-1}$.
  \end{frame}

  %--- Slide: isolar x ---
  \begin{frame}{Reparametrizando a loss -- Isolar $\mathbf{x}$}
    Do kernel de difusão $\mathbf{z}_t = \sqrt{\alpha_t}\,\mathbf{x} + \sqrt{1-\alpha_t}\,\boldsymbol{\epsilon}$:
    \[
      \mathbf{x} = \frac{\mathbf{z}_t}{\sqrt{\alpha_t}} - \frac{\sqrt{1-\alpha_t}}{\sqrt{\alpha_t}}\,\boldsymbol{\epsilon}.
    \]
    \vspace{0.3em}
    Substituímos essa expressão de $\mathbf{x}$ dentro da loss e simplificamos.
  \end{frame}

  %--- Slide: simplificação, coeficiente de z_t ---
  \begin{frame}{Reparametrizando a loss -- Agrupando $\mathbf{z}_t$}
    O argumento dentro do quadrado (termo KL) era:
    \[
      \frac{1-\alpha_{t-1}}{1-\alpha_t}\sqrt{1-\beta_t}\, \mathbf{z}_t
      + \frac{\sqrt{\alpha_{t-1}}\,\beta_t}{1-\alpha_t}\, \mathbf{x}
      - f_t[\mathbf{z}_t,\thetav_t].
    \]
    Substituímos $\mathbf{x} = \tfrac{\mathbf{z}_t}{\sqrt{\alpha_t}} - \tfrac{\sqrt{1-\alpha_t}}{\sqrt{\alpha_t}}\,\boldsymbol{\epsilon}$
    e usamos $\sqrt{\alpha_t} = \sqrt{1-\beta_t}\,\sqrt{\alpha_{t-1}}$.
    \begin{block}{Coeficiente de $\mathbf{z}_t$}
      \[
        \frac{(1-\alpha_{t-1})\sqrt{1-\beta_t}}{1-\alpha_t}
        + \frac{\beta_t}{(1-\alpha_t)\sqrt{1-\beta_t}}
        = \frac{\overbrace{(1-\alpha_{t-1})(1-\beta_t)+\beta_t}^{=\;1-\alpha_t}}{(1-\alpha_t)\sqrt{1-\beta_t}}
        = \frac{1}{\sqrt{1-\beta_t}},
      \]
      \vspace{0.1em}
      pois $\alpha_{t-1}(1-\beta_t) = \alpha_t$.
    \end{block}
  \end{frame}

  %--- Slide: simplificação, coeficiente de epsilon e resultado ---
  \begin{frame}{Reparametrizando a loss -- Resultado}
    \begin{block}{Coeficiente de $\boldsymbol{\epsilon}$}
      \[
        -\,\frac{\sqrt{\alpha_{t-1}}\,\beta_t}{1-\alpha_t}\cdot\frac{\sqrt{1-\alpha_t}}{\sqrt{\alpha_t}}
        = -\,\frac{\beta_t}{\sqrt{1-\alpha_t}\,\sqrt{1-\beta_t}},
        \qquad \text{pois } \tfrac{\sqrt{\alpha_{t-1}}}{\sqrt{\alpha_t}} = \tfrac{1}{\sqrt{1-\beta_t}}.
      \]
    \end{block}
    Logo, o argumento se reduz a:
    \[
      \frac{1}{\sqrt{1-\beta_t}}\, \mathbf{z}_t
      - \frac{\beta_t}{\sqrt{1-\alpha_t}\sqrt{1-\beta_t}}\,\boldsymbol{\epsilon}
      - f_t[\mathbf{z}_t,\thetav_t].
    \]
    \begin{block}{Reparametrização da rede}
      Definimos $f_t$ na mesma forma, trocando $\boldsymbol{\epsilon}$ pela
      \textbf{nova rede $g_t$ que prediz o ruído}:
      \[
        f_t[\mathbf{z}_t,\thetav_t] = \frac{1}{\sqrt{1-\beta_t}}\, \mathbf{z}_t
        - \frac{\beta_t}{\sqrt{1-\alpha_t}\sqrt{1-\beta_t}}\, g_t[\mathbf{z}_t,\thetav_t].
      \]
    \end{block}
  \end{frame}

  %--- Slide: cancelamento z_t ---
  \begin{frame}{Reparametrizando a rede -- Cancelamento}
    Como $f_t$ tem a \textbf{mesma forma} do argumento, ao substituí-la
    os termos em $\mathbf{z}_t$ (coeficiente $\tfrac{1}{\sqrt{1-\beta_t}}$ idêntico)
    \textbf{se cancelam}:
    \vspace{0.3em}
    \[
      \cancel{\tfrac{1}{\sqrt{1-\beta_t}}\, \mathbf{z}_t}
      - \tfrac{\beta_t}{\sqrt{1-\alpha_t}\sqrt{1-\beta_t}}\,\boldsymbol{\epsilon}
      - \cancel{\tfrac{1}{\sqrt{1-\beta_t}}\, \mathbf{z}_t}
      + \tfrac{\beta_t}{\sqrt{1-\alpha_t}\sqrt{1-\beta_t}}\, g_t[\mathbf{z}_t,\thetav_t].
    \]
    \vspace{0.2em}
    Sobra apenas a diferença entre ruído predito e real, escalada por um
    fator:
    \[
      \frac{\beta_t}{\sqrt{1-\alpha_t}\sqrt{1-\beta_t}}\,
      \big(g_t[\mathbf{z}_t,\thetav_t] - \boldsymbol{\epsilon}\big).
    \]
  \end{frame}

  %--- Slide: loss simplificada ---
  \begin{frame}{Reparametrizando a rede -- Loss final}
    Substituindo, o termo do quadrado vira simplesmente
    $|g_t[\mathbf{z}_t,\thetav_t] - \boldsymbol{\epsilon}_t|^{2}$ a menos de uma constante
    multiplicativa, que podemos absorver durante a otimização:
    \vspace{0.3em}
    \[
      \boxed{
        \mathcal{L}[\thetav_{1\ldots T}] = \sum_{i=1}^{N}\sum_{t=2}^{T} \big|g_t[\mathbf{z}_t,\thetav_t] - \boldsymbol{\epsilon}_t\big|^{2}
      }
    \]
    \vspace{0.3em}
    \begin{block}{Função de custo do modelo difusor}
      É um \emph{erro quadrático médio} entre o ruído real $\boldsymbol{\epsilon}_t$
      adicionado em cada passo e a estimativa $g_t$ da rede.
      \textbf{Surpreendentemente simples}.
    \end{block}
    \vspace{0.2em}
    \footnotesize O termo de reconstrução ($t=1$) tem a mesma forma de MSE
    em $\boldsymbol{\epsilon}$; na prática o somatório roda de $t=1$ a $T$.
  \end{frame}

  %--- Slide: parametrizações do alvo ---
  \begin{frame}{Reparametrizando a rede -- O que a rede prevê?}
    A rede pode ser treinada para prever alvos \textbf{equivalentes}; a
    escolha afeta a estabilidade e a qualidade:
    \vspace{0.3em}
    \begin{itemize}
      \item \textbf{$\boldsymbol{\epsilon}$-prediction} (DDPM, usado aqui): estima o
      ruído $\boldsymbol{\epsilon}$. Loss MSE simples e estável; é o \emph{default}.
      \item \textbf{$\mathbf{x}_0$-prediction}: estima diretamente a imagem limpa
      $\mathbf{x}$. Equivalente pela álgebra do kernel, mas empiricamente pior em
      \emph{timesteps} de ruído alto.
      \item \textbf{$\mathbf{v}$-prediction} (Salimans \& Ho, 2022): estima
      $\mathbf{v} = \sqrt{\alpha_t}\,\boldsymbol{\epsilon} - \sqrt{1-\alpha_t}\,\mathbf{x}$, combinação de
      ruído e sinal. Estável em todo o \emph{schedule}; comum em geração de
      vídeo e em \emph{distillation}.
    \end{itemize}
    \begin{block}{Resumo}
      As três são equivalentes via o kernel
      $\mathbf{z}_t = \sqrt{\alpha_t}\,\mathbf{x} + \sqrt{1-\alpha_t}\,\boldsymbol{\epsilon}$; diferem na
      escala efetiva da loss por \emph{timestep}.
    \end{block}
  \end{frame}

  %======================================================================
  % SECTION 4: TREINAMENTO E AMOSTRAGEM
  %======================================================================
  \section{Treinamento e Amostragem}

  %--- Slide: algoritmo treinamento ---
  \begin{frame}{Algoritmo de Treinamento}
    \begin{algorithmic}
      \State \textbf{Entrada:} dados de treino $\mathbf{x}$
      \State \textbf{Saída:} parâmetros do modelo $\thetav_t$
      \Repeat
        \For{$i \in \mathcal{B}$} \Comment{para cada exemplo do \textit{minibatch}}
          \State $t \sim \mathrm{Uniform}[1,\ldots,T]$ \Comment{sorteia o passo}
          \State $\boldsymbol{\epsilon} \sim \mathcal{N}(0,\mathbf{I})$ \Comment{sorteia o ruído}
          \State $\ell_i = \big\| g_t\big[\sqrt{\alpha_t}\,\mathbf{x}_i + \sqrt{1-\alpha_t}\,\boldsymbol{\epsilon},\ \thetav_t\big] - \boldsymbol{\epsilon} \big\|^{2}$
        \EndFor
        \State Acumule as \textit{losses} do \textit{batch} e dê um passo de gradiente
      \Until{convergir}
    \end{algorithmic}
    \vspace{0.8em}
    {\footnotesize Monta-se $\mathbf{z}_t$ pelo kernel ($\sqrt{\alpha_t}\,\mathbf{x}_i +
    \sqrt{1-\alpha_t}\,\boldsymbol{\epsilon}$) e treina-se $g_t$ para prever o ruído
    $\boldsymbol{\epsilon}$.}
  \end{frame}

  %--- Slide: algoritmo amostragem ---
  \begin{frame}{Algoritmo de Amostragem}
    \begin{algorithmic}
      \State \textbf{Entrada:} modelo $g_t[\,\cdot\,,\thetav_t]$
      \State \textbf{Saída:} amostra $\mathbf{x}$
      \State $\mathbf{z}_T \sim \mathcal{N}(0,\mathbf{I})$ \Comment{última variável latente}
      \For{$t = T,\ldots,2$}
        \State $\hat{\mathbf{z}}_{t-1} = \dfrac{1}{\sqrt{1-\beta_t}}\,\mathbf{z}_t -
          \dfrac{\beta_t}{\sqrt{1-\alpha_t}\,\sqrt{1-\beta_t}}\,g_t[\mathbf{z}_t,\thetav_t]$
          \Comment{prediz o latente anterior}
        \State $\boldsymbol{\epsilon} \sim \mathcal{N}(0,\mathbf{I})$ \Comment{novo ruído}
        \State $\mathbf{z}_{t-1} = \hat{\mathbf{z}}_{t-1} + \sigma_t\,\boldsymbol{\epsilon}$ \Comment{adiciona ruído}
      \EndFor
      \State $\mathbf{x} = \dfrac{1}{\sqrt{1-\beta_1}}\,\mathbf{z}_1 -
        \dfrac{\beta_1}{\sqrt{1-\alpha_1}\,\sqrt{1-\beta_1}}\,g_1[\mathbf{z}_1,\thetav_1]$
        \Comment{gera $\mathbf{x}$ sem ruído}
    \end{algorithmic}
  \end{frame}

  \section{Implementação e Extensões}

  %--- Slide: U-Net ---
  \begin{frame}{Modelo para estimar $\boldsymbol{\epsilon}_t$ -- \textit{U-Net}}
    \centering
    \includegraphics[width=0.85\linewidth]{slides/img/05-geradores/dm-unet-architecture.png}\\
    \vspace{0.3em}
    \begin{itemize}
      \item Arquitetura padrão: \textbf{U-Net} com blocos residuais e
      auto-atenção.
      \item Recebe $\mathbf{z}_t$ + um \emph{time embedding} de $t$, devolve
      uma estimativa de $\boldsymbol{\epsilon}$ com a mesma forma de $\mathbf{z}_t$.
      \item \textit{Skip connections} preservam alta-frequência entre
      escalas.
    \end{itemize}
  \end{frame}

  %--- Slide: classifier-free guidance ---
  \begin{frame}{\textit{Classifier-Free Diffusion Guidance}\footfullcite{ho2022classifier}}
    \begin{itemize}
      \item É comum adicionar uma \textbf{informação condicionante} no
      processo de geração.
      \item Pode parecer contraintuitivo, contudo informações sobre a
      classe que queremos gerar \textbf{auxiliam o processo} e geram
      imagens com maior qualidade.
      \item O treinamento condicionado é feito adicionando algum tipo
      de \textbf{embedding} da classe à U-Net.
      \begin{itemize}
        \item Por exemplo, embeddings do BERT sobre a descrição da
        imagem (texto-para-imagem).
      \end{itemize}
      \item \emph{Classifier-Free}: durante o treino, esporadicamente
      ``descondicionamos'' a rede; na geração, interpolamos entre
      versões condicionada e incondicional para controlar a força da
      condição.
    \end{itemize}
  \end{frame}

  %--- Slide: latent diffusion ---
  \begin{frame}{\textit{Latent Diffusion Models}\footfullcite{rombach2022high}}
    \centering
    \includegraphics[width=0.62\linewidth]{slides/img/05-geradores/dm-latent-diffusion.jpeg}\\
    \vspace{0.2em}
    \begin{itemize}
      \item Aplique a difusão no \textbf{espaço latente} de um VAE, não
      em pixels.
      \item Reduz drasticamente o custo computacional (latente é
      $\sim 64\!\times\!64$ em vez de $1024\!\times\!1024$).
      \item Base de \textit{Stable Diffusion}.
    \end{itemize}
  \end{frame}

  %======================================================================
  \section*{Resumo}

  \begin{frame}{Resumo}
    \begin{itemize}
      \item \textbf{Difusão} (encoder, sem parâmetros): cadeia de
      Markov gaussiana que destrói gradualmente a estrutura de $\mathbf{x}$.
      \item \textbf{Kernel}: forma fechada para amostrar $\mathbf{z}_t$
      diretamente de $\mathbf{x}$.
      \item \textbf{Reverse process} (decoder): rede neural aprende a
      reverter um passo da difusão.
      \item \textbf{ELBO}: termo de reconstrução + soma de KLs entre
      normais (todas com forma fechada).
      \item \textbf{Reparametrização}: prever \emph{ruído} $\boldsymbol{\epsilon}$
      em vez da imagem leva a uma loss MSE simples.
      \item \textbf{Arquitetura}: U-Net com \textit{time embedding}.
      \item \textbf{Extensões}: \textit{classifier-free guidance},
      \textit{latent diffusion} (Stable Diffusion).
    \end{itemize}
  \end{frame}
