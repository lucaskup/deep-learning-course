  %--- Slide: Sonho do zero-shot ---
  \begin{frame}{O Sonho do \textit{Zero-Shot} em Visão}
    \begin{itemize}
      \item Modelos supervisionados só reconhecem o conjunto fixo de classes em que foram treinados
      \item Adicionar uma classe nova exige coletar exemplos rotulados e re-treinar
      \item \textit{Zero-shot}: classificar uma classe nunca vista no treino, descrita apenas por linguagem natural
    \end{itemize}

    \vspace{0.5em}
    \begin{block}{Definição operacional}
      Dado um modelo treinado e uma nova classe descrita por uma string
      (``\texttt{a photo of a tapir}''), prever se uma imagem pertence a essa classe
      \textbf{sem nenhum exemplo de tapir no treino}.
    \end{block}

    \vspace{0.4em}
    Para isso, precisamos de um espaço onde \textbf{imagem} e \textbf{texto} convivam.
  \end{frame}

  %--- Slide: Similaridade do cosseno ---
  \begin{frame}{Similaridade do Cosseno}
    \begin{columns}[T]
      \begin{column}{0.55\textwidth}
        \begin{block}{Definição}
          \[
            \mathrm{sim}(\mathbf{u}, \mathbf{v}) =
            \frac{\mathbf{u}^{\top}\mathbf{v}}{\|\mathbf{u}\|\,\|\mathbf{v}\|}
            = \cos\theta
          \]
        \end{block}

        \vspace{0.3em}
        Em CLIP, \textbf{normalizamos $\mathbf{u}$ e $\mathbf{v}$ para norma~1}
        antes de computar a similaridade:
        \[
          \tilde{\mathbf{u}} = \mathbf{u}/\|\mathbf{u}\|,\quad
          \tilde{\mathbf{v}} = \mathbf{v}/\|\mathbf{v}\|
        \]
        Assim $\mathrm{sim}(\tilde{\mathbf{u}},\tilde{\mathbf{v}}) =
        \tilde{\mathbf{u}}^{\top}\tilde{\mathbf{v}} \in [-1, 1]$.
      \end{column}
      \begin{column}{0.42\textwidth}
        \centering
        \begin{tikzpicture}[scale=1.2]
          \draw[->, thick] (-0.3,0) -- (2.5,0);
          \draw[->, thick] (0,-0.3) -- (0,2.0);
          \draw[->, ultra thick, ClipBlue]   (0,0) -- (2.0, 0.5)
            node[right, font=\small] {$\mathbf{u}$};
          \draw[->, ultra thick, ClipOrange] (0,0) -- (1.0, 1.7)
            node[above, font=\small] {$\mathbf{v}$};
          \draw (0.4,0.10) arc (14:60:0.42);
          \node[font=\scriptsize] at (0.65, 0.45) {$\theta$};
        \end{tikzpicture}
      \end{column}
    \end{columns}
  \end{frame}

  %--- Slide: InfoNCE formal ---
  \begin{frame}{A Perda \textit{InfoNCE}\footfullcite{oord2018infonce}}
    Em um \textit{batch} de $N$ pares $\{(\mathbf{u}_i, \mathbf{v}_i)\}_{i=1}^{N}$ (já normalizados),
    formamos $S \in \mathbb{R}^{N\times N}$ com $S_{ij}=\mathbf{u}_i^{\top}\mathbf{v}_j$. Aqui $\tau>0$ é a \textit{temperatura} (próximo slide): controle de ``afiamento'' da \textit{softmax}.

    \begin{block}{Loss simétrica de CLIP}
      \small
      Cada direção é uma \textit{softmax} com alvo na diagonal ($i\to t$: \textbf{linha} $i$; \quad $t\to i$: \textbf{coluna} $j$):
      \[
        \mathcal{L}_{i\to t} = -\frac{1}{N}\sum_{i=1}^{N}
          \log\frac{\exp(S_{ii}/\tau)}{\sum_{j=1}^{N}\exp(S_{ij}/\tau)}
      \]
      \[
        \mathcal{L}_{t\to i} = -\frac{1}{N}\sum_{j=1}^{N}
          \log\frac{\exp(S_{jj}/\tau)}{\sum_{i=1}^{N}\exp(S_{ij}/\tau)},\qquad
        \mathcal{L}_{\text{CLIP}} = \tfrac{1}{2}\left(\mathcal{L}_{i\to t} + \mathcal{L}_{t\to i}\right)
      \]
    \end{block}

    \vspace{0.1em}
    {\small InfoNCE $=$ \textbf{cross-entropy} (\textit{softmax} $+$ $-\log$) com a diagonal como rótulo: classificação de $N$ vias no \textit{batch}.}
  \end{frame}

  %--- Slide: Temperatura ---
  \begin{frame}{Por Que a Temperatura $\tau$ Importa}
    \begin{columns}[T]
      \begin{column}{0.50\textwidth}
        \begin{itemize}
          \item $\tau$ pequeno $\Rightarrow$ \textit{softmax} mais ``afiada'' (concentra probabilidade no maior \textit{logit})
          \item $\tau$ grande $\Rightarrow$ distribuição mais uniforme, gradientes menos focados
          \item Em CLIP, $\tau$ é \textbf{aprendido} (parâmetro $\log(1/\tau)$, inicializado em $\log 14$, recortado em $\log 100$)
        \end{itemize}

        \vspace{0.3em}
        \footnotesize
        Valor convergido típico: $1/\tau \approx 100$, ou seja $\tau \approx 0.01$.
      \end{column}
      \begin{column}{0.46\textwidth}
        \centering
        \resizebox{\linewidth}{!}{%
        \begin{tikzpicture}[font=\scriptsize]
          \begin{axis}[
            ybar,

            \legend{$\tau{=}1$, $\tau{=}0.3$, $\tau{=}0.05$}
          \end{axis}
        \end{tikzpicture}}
      \end{column}
    \end{columns}
  \end{frame}

  %--- Slide: Worked example I — similaridade ---
  \begin{frame}{Exemplo Trabalhado I: Matriz de Similaridade}
    Batch com $N=2$. \textit{Embeddings} já normalizados em $\mathbb{R}^3$:
    \[
      \mathbf{u}_1 = \begin{pmatrix}0.8 \\ 0.6 \\ 0\end{pmatrix},\;
      \mathbf{u}_2 = \begin{pmatrix}0 \\ 0.8 \\ 0.6\end{pmatrix},\;
      \mathbf{v}_1 = \begin{pmatrix}1 \\ 0 \\ 0\end{pmatrix},\;
      \mathbf{v}_2 = \begin{pmatrix}0 \\ 0.6 \\ 0.8\end{pmatrix}
    \]

    \vspace{0.2em}
    Cada $S_{ij} = \mathbf{u}_i^{\top}\mathbf{v}_j$:
    \begin{itemize}\setlength{\itemsep}{0pt}
      \item $S_{11} = 0.8\!\cdot\!1 + 0.6\!\cdot\!0 + 0 = \mathbf{0.80}$ (positivo)
      \item $S_{12} = 0 + 0.6\!\cdot\!0.6 + 0 = \mathbf{0.36}$ (negativo)
      \item $S_{21} = 0 + 0.8\!\cdot\!0 + 0.6\!\cdot\!0 = \mathbf{0.00}$ (negativo)
      \item $S_{22} = 0 + 0.8\!\cdot\!0.6 + 0.6\!\cdot\!0.8 = \mathbf{0.96}$ (positivo)
    \end{itemize}

    \vspace{0.3em}
    \[
      S = \begin{pmatrix} \mathbf{0.80} & 0.36 \\ 0.00 & \mathbf{0.96} \end{pmatrix}
    \]
    Diagonal $=$ pares positivos; fora da diagonal $=$ negativos. Note que $S$ \textbf{não é simétrica}.
    \note{
      Aponte: a diagonal está alta. Como $S$ não é simétrica, as duas direções
      (softmax por linha vs por coluna) darão perdas diferentes, motivando a loss
      simétrica do próximo slide.
    }
  \end{frame}

  %--- Slide: Worked example II — InfoNCE ---
  \begin{frame}{Exemplo Trabalhado II: \textit{InfoNCE} com $\tau$ Diferente}
    Aplicamos \textit{softmax} linha a linha em $S/\tau$. A perda usa a probabilidade do par correto (a \textbf{diagonal}):
    \vspace{0.2em}
    \centering
    \resizebox{\linewidth}{!}{%
    \begin{tabular}{@{}c|cc|cc|c@{}}
      \toprule
      $\tau$ & $S/\tau$ linha 1 & $\mathrm{softmax}$ & $S/\tau$ linha 2 & $\mathrm{softmax}$ & $\mathcal{L}_{i\to t}$ \\
      \midrule
      $1.0$  & $(0.80,\, 0.36)$ & $(0.608,\, 0.392)$ & $(0.00,\, 0.96)$ & $(0.277,\, 0.723)$ & $\mathbf{0.411}$ \\
      $0.1$  & $(8.0,\, 3.6)$   & $(0.988,\, 0.012)$ & $(0.0,\, 9.6)$  & $(\approx 0,\, \approx 1)$ & $\mathbf{0.006}$ \\
      $0.01$ & $(80,\, 36)$         & $(\approx 1,\, \approx 0)$ & $(0,\, 96)$    & $(\approx 0,\, \approx 1)$ & $\approx \mathbf{0}$ \\
      \bottomrule
    \end{tabular}}

    \vspace{0.3em}
    {\small Para $\tau{=}1$: $\mathcal{L}_{i\to t} = -\tfrac{1}{2}(\log 0.608 + \log 0.723) = 0.411$. A outra direção usa a \textit{softmax} por \textbf{coluna}: $\mathcal{L}_{t\to i} = -\tfrac{1}{2}(\log 0.690 + \log 0.646) = 0.404$ (\textbf{diferente}!), logo $\mathcal{L}_{\text{CLIP}} = \tfrac{1}{2}(0.411 + 0.404) = 0.408$.}

    \vspace{0.2em}
    \begin{itemize}\setlength{\itemsep}{0pt}
      \item Diminuir $\tau$ \textbf{recompensa fortemente} positivos já bem separados.
      \item Mas amplifica ruído: um negativo enganoso vira gradiente enorme.
      \item Daí a temperatura aprendível com \textit{clipping} em $\log 100$.
    \end{itemize}
  \end{frame}

  %--- Slide: Negativos no batch ---
  \begin{frame}{Negativos Vêm de Graça do \textit{Batch}}
    Em um batch com $N$ pares positivos, cada imagem ``vê'' $N{-}1$ textos negativos
    (e vice-versa). Total:
    \[
      \underbrace{N}_{\text{positivos}} \;+\; \underbrace{N(N{-}1)}_{\text{negativos}}
      = N^2 \text{ comparações na matriz } S
    \]

    \vspace{0.3em}
    \centering
    \begin{tabular}{@{}rrr@{}}
      \toprule
      $N$ & positivos & negativos \\
      \midrule
      32        & 32        & 992 \\
      256       & 256       & 65\,280 \\
      8\,192    & 8\,192    & 67\,100\,672 \\
      32\,768   & 32\,768   & $\approx 1.07 \times 10^9$ \\
      \bottomrule
    \end{tabular}

  \end{frame}

  %--- Slide: Cabeças de projeção ---
  \begin{frame}{Cabeças de Projeção: Mesmo Espaço para Texto e Imagem}
    Cada \textit{encoder} produz um vetor em sua dimensão nativa:
    \[
      \mathbf{h}_i^{\text{img}} \in \mathbb{R}^{d_v},\qquad
      \mathbf{h}_t^{\text{txt}} \in \mathbb{R}^{d_t}
    \]

    Projetamos linearmente para a \textbf{mesma dimensão} $d_e$ e normalizamos:
    \[
      \mathbf{u}_i = \frac{W_i\, \mathbf{h}_i^{\text{img}}}{\|W_i\, \mathbf{h}_i^{\text{img}}\|},\quad
      \mathbf{v}_t = \frac{W_t\, \mathbf{h}_t^{\text{txt}}}{\|W_t\, \mathbf{h}_t^{\text{txt}}\|}
    \]
    com $W_i \in \mathbb{R}^{d_e \times d_v}$ e $W_t \in \mathbb{R}^{d_e \times d_t}$.

    \vspace{0.4em}
    \begin{block}{Por que a normalização?}
      A perda usa cosseno; sem L2 a temperatura $\tau$ teria que absorver
      magnitudes diferentes entre os dois ramos.
    \end{block}
  \end{frame}

  %--- Slide: Pseudocódigo ---
  \begin{frame}{Pseudocódigo: \textit{Forward} + \textit{Loss}}
    \begin{columns}[T]
      \begin{column}{0.46\textwidth}
        \begin{itemize}
          \item Entrada: batch de $N$ imagens e $N$ textos correspondentes
          \item Saída: \textbf{escalar} $\mathcal{L}_{\text{CLIP}}$
          \item As duas \textit{cross-entropies} compartilham a mesma matriz $S$
        \end{itemize}
      \end{column}
      \begin{column}{0.50\textwidth}
        \begin{algorithmic}[1]
          \Require batch $\{(\mathrm{img}_i, \mathrm{txt}_i)\}_{i=1}^{N}$, $\tau$
          \State $\mathbf{H}^{\text{img}} \gets \texttt{ImageEncoder}(\mathrm{img})$
          \State $\mathbf{H}^{\text{txt}} \gets \texttt{TextEncoder}(\mathrm{txt})$
          \State $U \gets \texttt{normalize}(W_i\, \mathbf{H}^{\text{img}})$
          \State $V \gets \texttt{normalize}(W_t\, \mathbf{H}^{\text{txt}})$
          \State $S \gets U\, V^{\top} / \tau$
          \State $\mathbf{y} \gets [0, 1, \ldots, N{-}1]$
          \State $\mathcal{L}_1 \gets \texttt{CE}(S, \mathbf{y})$
          \State $\mathcal{L}_2 \gets \texttt{CE}(S^{\top}, \mathbf{y})$
          \State \Return $(\mathcal{L}_1 + \mathcal{L}_2)/2$
        \end{algorithmic}
      \end{column}
    \end{columns}
  \end{frame}

    \centering
    \resizebox{0.95\linewidth}{!}{%
    \begin{tabular}{@{}lrrrr@{}}
      \toprule
      Variante & Params (enc. imagem) & FLOPs/imagem & Memória de $S$ ($N{=}32{\rm k}$) & Hardware (paper)\\
      \midrule
      ResNet-50      & $\sim$38M & 4.1 GFLOPs & $\sim$4 GB (fp32) & --- \\
      ResNet-50x64   & $\sim$420M & 165 GFLOPs   & $\sim$4 GB        & 592 V100 / 18 dias \\
      ViT-B/32       & $\sim$88M & 4.4 GFLOPs & $\sim$4 GB        & --- \\
      ViT-L/14       & $\sim$304M & 81 GFLOPs    & $\sim$4 GB        & 256 V100 / 12 dias \\
      \bottomrule
    \end{tabular}}

    \vspace{0.5em}
    \begin{itemize}\setlength{\itemsep}{0pt}
      \item Memória de $S$ cresce \textbf{quadraticamente} com $N$
      \item Truques: \textit{gradient checkpointing}, \textit{mixed precision}, \textit{shard} de embeddings
      \item \textbf{Compute dominado pelos \textit{encoders}}: $\sim N \cdot \text{GFLOPs/imagem}$ contra $\sim N^2 d_e$ FLOPs do produto $U V^{\top}$ que forma $S$ (ordens de magnitude de diferença)
      \item \textbf{Memória dominada pelas ativações dos \textit{encoders}} quando $S$ é distribuído via \textit{sharded embeddings}; sem \textit{shard}, $S$ ($\sim 4$ GB) seria competitivo
    \end{itemize}
  \end{frame}

  %--- Slide: Truques de escala ---
  \begin{frame}{Truques para Escalar o \textit{Batch}}
    \begin{itemize}
      \item \textbf{\textit{Sharded embeddings}}: distribuir $U$ e $V$ entre GPUs; cada GPU calcula sua fatia de $S$
      \item \textbf{\textit{Gradient checkpointing}} nos \textit{encoders} pesados (ViT-L)
      \item \textbf{\textit{Mixed precision}} reduz memória dos forward passes pela metade
      \item \textbf{\textit{Local contrastive}}: cada GPU vê o batch global de $V$ via \textit{all-gather}, calcula sua linha de $S$ e devolve gradientes
    \end{itemize}

    \vspace{0.4em}
    \begin{block}{Pergunta comum}
      \emph{Precisamos minerar negativos difíceis (\textit{hard negative mining})?}\\
      \textbf{Não} no caso de CLIP. O batch de 32\,768 já fornece um espectro
      suficiente de negativos a cada passo.
    \end{block}
  \end{frame}

  %--- Slide: Linear probe ---
  \begin{frame}{\textit{Linear Probe Evaluation}}
    Para medir a qualidade dos \textit{embeddings} aprendidos, congelamos o \textit{encoder}
    e treinamos apenas um classificador linear:
    \begin{enumerate}
      \item Para cada imagem $\mathbf{x}$, computar $\mathbf{u} = f_{\text{CLIP}}(\mathbf{x})$
      \item Treinar regressão logística sobre $(\mathbf{u}, y)$ no \textit{dataset} alvo
      \item Reportar \textit{accuracy} no conjunto de teste
    \end{enumerate}

    \vspace{0.4em}
    \begin{block}{Por que esse protocolo?}
      Mede a separabilidade linear das classes no espaço de representação, isolando
      a qualidade dos \textit{features} de qualquer ajuste fino do \textit{backbone}.
    \end{block}

    \vspace{0.3em}
    Comparação direta com SimCLRv2, BYOL, MoCo-v3 e ResNet supervisionada na ImageNet.
  \end{frame}

  \begin{frame}{Exemplo Trabalhado III: \textit{Zero-Shot} com 3 Classes}
    Embedding da imagem do slide anterior: $\mathbf{u} = (0.6,\, 0.8,\, 0)^{\top}$. \textit{Embeddings} dos \textit{prompts}:
    \[
      \mathbf{v}_{\text{gato}}     = (0, 1.0, 0)^{\top},\;
      \mathbf{v}_{\text{cachorro}} = (0.8, 0.6, 0)^{\top},\;
      \mathbf{v}_{\text{carro}}    = (0, 0, 1.0)^{\top}
    \]

    Similaridades: $s_{\text{gato}} = 0.80,\; s_{\text{cachorro}} = 0.96,\; s_{\text{carro}} = 0.00$

    \vspace{0.4em}
    Com $\tau = 0.1$, logits $= (8.0,\, 9.6,\, 0)$. Subtraindo o máximo:
    \begin{itemize}\setlength{\itemsep}{0pt}
      \item $e^{-1.6} \approx 0.202$,\quad $e^{0} = 1$,\quad $e^{-9.6} \approx 6.77\!\times\!10^{-5}$
      \item soma $\approx 1.202$
      \item $p_{\text{gato}} \approx 0.168$,\; $p_{\text{cachorro}} \approx 0.832$,\; $p_{\text{carro}} \approx 0.00006$
    \end{itemize}

    \vspace{0.3em}
    \begin{block}{Predição}
      $\hat{y} = \argmax_k p_k = \text{\textbf{cachorro}}$.
      O classificador é montado \textbf{em tempo de inferência} a partir dos \textit{prompts};
      basta trocar as descrições para mudar o conjunto de classes.
    \end{block}
  \end{frame}

  %--- Slide: Prompt engineering ---
  \begin{frame}{\textit{Prompt Engineering} e Ensemble de Templates}
    A escolha do \textit{template} muda a \textit{accuracy} em vários pontos percentuais:
    \begin{itemize}\setlength{\itemsep}{0pt}
      \item \texttt{"\{class\}"} (apenas o nome): \textit{baseline}
      \item \texttt{"a photo of a \{class\}"}: $+1$ a $+2$ pp em ImageNet
      \item Templates específicos por dataset: \texttt{"a satellite photo of a \{class\}"}, \texttt{"a sketch of a \{class\}"}
    \end{itemize}

    \vspace{0.4em}
    \begin{block}{Ensemble de prompts}
      O paper original usa \textbf{80 templates} por classe e calcula a média dos
      \textit{embeddings} de texto antes da L2:
      \[
        \mathbf{v}_k = \mathrm{normalize}\!\left( \tfrac{1}{T} \sum_{t=1}^{T} f_{\text{txt}}(\text{prompt}_t(k)) \right)
      \]
      Ganho médio em ImageNet: $+3.5$ pp.
    \end{block}
  \end{frame}

  %--- Slide: Fine-tuning ---
  \begin{frame}{Adaptação Eficiente: Linear Head, Adapter, LoRA}
    \begin{columns}[T]
      \begin{column}{0.48\textwidth}
        \begin{block}{Cabeçalho linear}
          Congelar CLIP, treinar uma camada $W \in \mathbb{R}^{K \times d_e}$
          sobre os \textit{embeddings}.\\ \footnotesize
          \textit{Linear probe} clássico, baixíssimo custo.
        \end{block}
        \begin{block}{CLIP-Adapter}
          Pequeno MLP residual sobre $\mathbf{u}$, com mistura ponderada:
          $\mathbf{u}' = \alpha\, \texttt{MLP}(\mathbf{u}) + (1-\alpha)\, \mathbf{u}$.
        \end{block}
      \end{column}
      \begin{column}{0.48\textwidth}
        \begin{block}{LoRA sobre os \textit{encoders}}
          Adicionar matrizes de baixa ordem $W + \Delta = W + B A$ nos
          \textit{transformers} de imagem e texto.\\
          \footnotesize Apenas $0.1$--$1\%$ dos parâmetros treinados.
        \end{block}
        \begin{alertblock}{Quando NÃO fazer fine-tuning}
          Se a tarefa cabe em \textit{zero-shot} robusto, o \textit{fine-tuning}
          piora a generalização \textit{out-of-distribution}.
        \end{alertblock}
      \end{column}
    \end{columns}
  \end{frame}

  %--- Slide: Semantic retrieval ---
  \begin{frame}{\textit{Semantic Retrieval}: Buscar Imagens com Texto}
    \begin{enumerate}
      \item \textbf{Indexação} (offline): para cada imagem $\mathbf{x}_n$ no acervo, salvar $\mathbf{u}_n = f_{\text{img}}(\mathbf{x}_n)$
      \item Construir índice \textit{Approximate Nearest Neighbor} (FAISS, HNSW, ScaNN) sobre $\{\mathbf{u}_n\}$ usando produto interno
      \item \textbf{Consulta} (online): dado um texto $q$, computar $\mathbf{v}_q = f_{\text{txt}}(q)$
      \item Recuperar \textit{top-}$k$ pelo cosseno $\mathbf{v}_q^{\top}\mathbf{u}_n$
    \end{enumerate}

    \vspace{0.4em}
    \begin{block}{Custo amortizado}
      O \textit{encoder} de imagem roda \textbf{uma vez} por imagem do acervo;
      consultas custam apenas um forward de texto $+$ \textit{look-up} no índice.
    \end{block}

    \vspace{0.2em}
    Inverso: dada uma imagem, recuperar legendas similares no acervo de textos.
  \end{frame}

  \begin{frame}{Falhas Conhecidas do CLIP}
    \begin{columns}[T]
      \begin{column}{0.56\textwidth}
        \small
        \begin{itemize}\setlength{\itemsep}{2pt}
          \item \textbf{Texto na imagem}: CLIP ``lê'' palavras pintadas e enviesa a classificação
            (uma maçã com etiqueta \texttt{"iPod"} é classificada como \texttt{iPod})
          \item \textbf{Generalização para MNIST}: $88\%$ \textit{accuracy} (pior que regressão logística simples),
            dígitos manuscritos são raros em pares web
          \item \textbf{Contagem de objetos}: ``três cachorros'' vs.\ ``cinco cachorros'' é difícil
          \item \textbf{Granularidade fina}: distinguir espécies próximas exige supervisão dedicada
          \item \textbf{Tarefas espaciais}: localização precisa, profundidade, geometria
        \end{itemize}
      \end{column}
      \begin{column}{0.40\textwidth}
        \centering
        \includegraphics[width=\linewidth,keepaspectratio]{slides/img/06-topicos/ipod_apple.jpeg}
      \end{column}
    \end{columns}
  \end{frame}

  %--- Slide: Vieses sociais ---
  \begin{frame}{Vieses Herdados da Web}
    O paper original dedica uma seção a auditoria de vieses:
    \begin{itemize}
      \item Classificação errônea sistemática de pessoas negras como categorias não-humanas
        em \texttt{FairFace}
      \item Associações de gênero em prompts ocupacionais (``CEO'' enviesa para masculino)
      \item Sensibilidade a vocabulário ofensivo presente nos pares web
    \end{itemize}

    \vspace{0.5em}
    \begin{block}{Implicação prática}
      Antes de usar CLIP em produto: auditar para o seu domínio, considerar
      \textit{prompt} sanitizado, restringir vocabulário de classes, registrar limites
      do modelo.
    \end{block}
  \end{frame}

  %--- Slide: Sensibilidade a prompt ---
  \begin{frame}{Sensibilidade a \textit{Prompt}}
    \begin{columns}[T]
      \begin{column}{0.55\textwidth}
        \begin{itemize}
          \item Trocar \texttt{"\{class\}"} por \texttt{"a photo of a \{class\}"} pode mudar \textit{accuracy} em $5$--$10$ pp
          \item Templates errados podem inverter ranqueamentos
          \item Sem \textit{ground truth}, calibrar prompt requer um conjunto de validação
        \end{itemize}

        \vspace{0.3em}
        Em produção, recomenda-se:
        \begin{itemize}
          \item Ensemble de 5--80 templates
          \item Avaliar accuracy em conjunto de validação rotulado

  %--- Slide: Sucessores ---
  \begin{frame}{Sucessores: ALIGN, OpenCLIP, SigLIP}
    \begin{columns}[T]
      \begin{column}{0.48\textwidth}
        \begin{block}{ALIGN \citeauthoryear{jia2021align}}
          \begin{itemize}
            \item Google, contemporâneo do CLIP
            \item 1.8B pares (\texttt{alt-text} bruto, ainda mais ruído)
            \item Mostra que limpeza importa menos que escala
          \end{itemize}
        \end{block}
        \begin{block}{OpenCLIP \citeauthoryear{cherti2023openclip}}
          \begin{itemize}
            \item Reprodução aberta sobre LAION
            \item Estuda \textit{scaling laws} multimodais
            \item Pesos públicos para múltiplas variantes
          \end{itemize}
        \end{block}
      \end{column}
      \begin{column}{0.48\textwidth}
        \begin{block}{SigLIP (2023)}
          Substitui \textit{softmax} por \textbf{sigmoide} pareada:
          $\ell_{ij} = \log\!\left(1 + e^{-y_{ij}\, \mathbf{u}_i^{\top}\mathbf{v}_j / \tau}\right)$
          \begin{itemize}
            \item Não exige \textit{batch} grande
            \item Cada par é independente
            \item Ganho prático em \textit{batches} pequenos
          \end{itemize}
        \end{block}
        \begin{block}{EVA-CLIP, MetaCLIP, DFN}
          Variações em receita de dados, escala e arquitetura.
        \end{block}
      \end{column}
    \end{columns}
  \end{frame}

  %--- Slide: LAION-5B ---
  \begin{frame}{LAION-5B e a Democratização do Pré-treino Multimodal}
    \begin{itemize}
      \item \textbf{5.85B} pares (imagem, texto), abertos, multilíngues
      \item Filtragem usa CLIP: descarta pares com baixa similaridade $\Rightarrow$ recursão útil
      \item Possibilita treinar variantes de CLIP, Stable Diffusion, vários modelos abertos
    \end{itemize}

    \vspace{0.4em}
    \begin{alertblock}{Implicações éticas}
      \begin{itemize}
        \item Conteúdo problemático (NSFW, vieses, material sensível) presente
        \item Direitos autorais: imagens coletadas sem consentimento explícito
        \item Auditoria: \texttt{laion.ai} disponibiliza ferramentas de inspeção
      \end{itemize}
    \end{alertblock}
  \end{frame}
