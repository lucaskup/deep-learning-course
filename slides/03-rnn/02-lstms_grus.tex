% Desenha uma célula recorrente vanilla com as operações internas (mesma macro
% do deck 01). Os nós internos são posicionados em relação ao nó "+" e a caixa
% da célula se ajusta ao conteúdo (biblioteca fit), com margem igual em todos
% os lados. Argumentos:
%   #1 sufixo dos nomes dos nós, pode ser vazio    #2 posição x do centro
%   #3 rótulo da operação x·θ_ih      #4 rótulo da operação h·θ_hh
%   #5 rótulo da operação h·θ_ho      #6 rótulo do nó de entrada
%   #7 rótulo do nó de saída          #8 rótulo ŷ ao lado do nó de saída (ou vazio)
\newcommand{\rnncellgraph}[8]{%
    \node[draw, circle, inner sep=1pt, fill=white] (plus#1) at (#2, 0) {$+$};
    \node[opnode, minimum width=1.45cm, left=0.3cm of plus#1] (hh#1) {#4};
    \node[opnode, minimum width=1.45cm, right=0.3cm of plus#1] (tanh#1) {$\tanh$};
    \node[opnode, below=0.4cm of plus#1] (xh#1) {#3};
    \node[opnode, fill=PastelPink, above=0.4cm of plus#1] (out#1) {#5};
    \begin{scope}[on background layer]
        \node[rnncell, fit=(hh#1)(tanh#1)(xh#1)(out#1), inner sep=0.25cm] (cell#1) {};
    \end{scope}
    \node[inputnode] (xnode#1) at ($(plus#1 |- cell#1.south) + (0, -0.85)$) {#6};
    \node[outputnode, minimum size=0.85cm] (ynode#1) at ($(plus#1 |- cell#1.north) + (0, 0.95)$) {#7};
    \ifx\relax#8\relax\else
        \node[font=\footnotesize, anchor=south west, inner sep=1pt] at (ynode#1.north east) {#8};
    \fi
    \draw[->] (xnode#1) -- (xh#1);
    \draw[->] (xh#1) -- (plus#1);
    \draw[->] (hh#1) -- (plus#1);
    \draw[->] (plus#1) -- (tanh#1);
    \draw[->] (tanh#1) |- (out#1);
    \draw[->] (out#1) -- (ynode#1);
}

%========================================================================
\section{Problemas em Redes Recorrentes}
%========================================================================

\begin{frame}{Memória de Curto Prazo}
\begin{itemize}
    \item Considere a frase a seguir, em que existe uma quebra de sentimento entre as duas partes:
    \item \textit{``\textcolor{red}{O filme é péssimo e ninguém deveria perder tempo assistindo.} \textcolor{blue}{Contudo eu admiro a tentativa de Hollywood de aumentar a representatividade em termos de história e atores, certamente será possível colher frutos sobre essa tentativa em outras obras no futuro.}''}
    \item Para classificar o sentimento global, a rede precisa lembrar do início da frase quando chega ao final.
    \item Com uma RNN simples, esse tipo de \textbf{dependência de longo alcance} é difícil de manter.
\end{itemize}
\end{frame}

\begin{frame}{Estado Interno é Sobrescrito a Cada Passo}
\begin{itemize}
    \item Em uma RNN \textit{vanilla}, o estado interno $\mathbf{h}_t$ é totalmente recalculado a cada passo de tempo.
\end{itemize}
\vspace{0.4em}
\centering
\begin{tikzpicture}[every node/.style={font=\scriptsize}, scale=0.95, transform shape]
    \foreach \i in {0,1,2,3,4,5}{
        \node[rnncell, minimum width=1.0cm, minimum height=0.8cm] (c\i) at (1.6*\i, 0) {};
        \node[inputnode] (x\i) at (1.6*\i, -1.1) {$x_\i$};
        \draw[->, thick] (x\i) -- (c\i);

    \draw[->, thick] (xn) -- (cn);
    \draw[->, thick] (cn) -- (yn);
    \draw[->, thick] (c5) -- node[above, font=\scriptsize] {$h_5$} (10, 0);

    \node[font=\scriptsize, text=red] at (4.8, -1.8) {péssimo};
    \node[font=\scriptsize, text=red] at (6.4, -1.8) {e};

\vspace{0.4em}
\begin{itemize}\footnotesize
    \item Em cada célula a entrada \textcolor{red}{atualiza completamente} o estado.
    \item Manter informação por muitos passos de tempo é difícil, esse é o problema da \textbf{memória de curto prazo}.
\end{itemize}
\end{frame}

\begin{frame}{Solução: Aprender Quando Atualizar e Quando Esquecer}
\begin{itemize}
    \item A ideia é deixar a própria rede aprender, a cada passo, quanta informação anterior deve ser mantida e quanta informação nova deve entrar no estado.
    \item Essa é a motivação da arquitetura \textit{Long Short-Term Memory}\footfullcite{hochreiter1997lstm} (LSTM).
    \item É a arquitetura com mais ``partes'' que veremos até agora, mas todas as partes se reduzem a operações já conhecidas:
    \begin{itemize}
        \item multiplicação de matrizes,
        \item funções de ativação (sigmoid, $\tanh$),
        \item operações ponto a ponto (soma, multiplicação).
    \end{itemize}
\end{itemize}
\end{frame}

%========================================================================
\section{LSTMs}
%========================================================================

\begin{frame}{Revisitando RNNs}
\begin{columns}
    \begin{column}{.55\linewidth}
        \begin{itemize}
            \item Em uma RNN \textit{vanilla} o estado oculto $\mathbf{h}_{t-1}$ é combinado com a entrada $\mathbf{x}_t$ por uma camada densa com ativação $\tanh$.
            \item Repare que dentro da célula recorrente existe ``uma rede densa''.
            \item A LSTM mantém essa ideia, porém com mais camadas internas e um caminho separado para a memória de longo prazo.
        \end{itemize}
    \end{column}
    \begin{column}{.45\linewidth}
        \centering
        \resizebox{0.85\linewidth}{!}{%
        \begin{tikzpicture}
            \rnncellgraph{}{0}{$\mathbf{x}_t\,\mathbf{\Theta}_{ih}^{T} + \mathbf{b}_h$}{$\mathbf{h}_{t-1}\,\mathbf{\Theta}_{hh}^{T}$}{$\mathbf{h}_t\,\mathbf{\Theta}_{ho}^{T} + \mathbf{b}_o$}{$\mathbf{x}_t$}{$\hat{\mathbf{y}}_t$}{}
            \draw[->] (tanh.east) -- ++(0.8, 0) node[right, font=\small] {$\mathbf{h}_t$};
        \end{tikzpicture}}
    \end{column}
\end{columns}
\end{frame}

\begin{frame}{Visão Geral da Célula LSTM}
\begin{columns}
    \begin{column}{.5\linewidth}
        \begin{itemize}\small
            \item A célula carrega \textbf{dois} sinais ao longo do tempo: o estado oculto $\mathbf{h}_t$ (\textit{hidden state}) e a memória de longo prazo $\mathbf{c}_t$ (\textit{cell state}).
            \item No diagrama, a linha \textbf{superior} conduz $\mathbf{c}_t$ e a \textbf{inferior} conduz $\mathbf{h}_t$.
            \item \textit{Gates} (camadas densas com \textit{sigmoid}, saída em $[0,1]$) controlam o quanto de cada informação é mantido, atualizado ou exposto, multiplicando vetores ponto a ponto.
        \end{itemize}
    \end{column}
    \begin{column}{.5\linewidth}
        \centering
        \resizebox{\linewidth}{!}{%
        \begin{tikzpicture}[every node/.style={font=\scriptsize}]
            \lstmcell
        \end{tikzpicture}}
    \end{column}
\end{columns}

\end{frame}

\begin{frame}{Quatro Camadas Densas Dentro da Célula}
\begin{itemize}
    \item $[\mathbf{h}_{t-1}, \mathbf{x}_t]$ é a \textbf{concatenação} dos dois vetores num único vetor de dimensão $h_{\text{size}} + x_{\text{size}}$.
    \item Dentro da célula LSTM existem \textbf{quatro} camadas densas que recebem $[\mathbf{h}_{t-1}, \mathbf{x}_t]$:
    \begin{itemize}
        \item três delas funcionam como \textit{gates}, com ativação \textit{sigmoid}:
        \begin{itemize}
            \item \textit{forget gate} ($\mathbf{f}_t$): decide o quanto da memória $\mathbf{c}_{t-1}$ deve ser \textbf{esquecido}.
            \item \textit{input gate} ($\mathbf{i}_t$): decide quais partes da nova informação devem \textbf{atualizar} a memória.
            \item \textit{output gate} ($\mathbf{o}_t$): decide quais partes da memória devem \textbf{compor} o novo estado oculto.
        \end{itemize}
        \item a quarta camada usa $\tanh$ e produz uma proposta de atualização $\tilde{\mathbf{c}}_t$ para a memória.
    \end{itemize}
    \item As três \textit{gates} aprendem a regular o fluxo de informação ao longo do tempo.
\end{itemize}
\end{frame}

\begin{frame}{\textit{Forget Gate}}
\begin{columns}
    \begin{column}{.55\linewidth}
        \begin{itemize}
            \item Decide quanto da memória $\mathbf{c}_{t-1}$ deve ser \textbf{esquecida}.
            \item Recebe $[\mathbf{h}_{t-1}, \mathbf{x}_t]$, passa por uma camada densa e por uma \textit{sigmoid}:
        \end{itemize}
        \[
            \mathbf{f}_t = \sigma\bigl([\mathbf{h}_{t-1}, \mathbf{x}_t]\,\mathbf{\Theta}_f^{T} + \mathbf{b}_f\bigr)
        \]
        \begin{itemize}
            \item $\mathbf{f}_t \in [0,1]$, multiplica $\mathbf{c}_{t-1}$ ponto a ponto.
            \begin{itemize}
                \item $\mathbf{f}_t \approx 1$: mantém a memória.
                \item $\mathbf{f}_t \approx 0$: esquece.
            \end{itemize}
            \item No exemplo do filme, o \textit{forget gate} pode aprender a \textbf{manter} o sinal de ``péssimo'' enquanto a segunda parte da frase é processada.
        \end{itemize}
    \end{column}

\end{frame}

\begin{frame}{\textit{Input Gate} e Candidato $\tilde{\mathbf{c}}_t$}
\begin{columns}
    \begin{column}{.55\linewidth}
        \begin{itemize}
            \item Decide quais partes da nova entrada devem \textbf{atualizar} a memória.
            \item Duas camadas trabalham juntas:
        \end{itemize}
        \[
            \mathbf{i}_t = \sigma\bigl([\mathbf{h}_{t-1}, \mathbf{x}_t]\,\mathbf{\Theta}_i^{T} + \mathbf{b}_i\bigr)
        \]
        \[
            \tilde{\mathbf{c}}_t = \tanh\bigl([\mathbf{h}_{t-1}, \mathbf{x}_t]\,\mathbf{\Theta}_c^{T} + \mathbf{b}_c\bigr)
        \]
        \begin{itemize}
            \item $\mathbf{i}_t \in [0,1]$ filtra a contribuição.
            \item $\tilde{\mathbf{c}}_t \in [-1,1]$ é a proposta de atualização.
        \end{itemize}
    \end{column}

\begin{frame}{Atualização da Memória $\mathbf{c}_t$}
\begin{columns}
    \begin{column}{.55\linewidth}
        \begin{itemize}
            \item A nova memória combina o que foi mantido da anterior com a nova proposta filtrada:
        \end{itemize}
        \[
            \mathbf{c}_t = \mathbf{f}_t \odot \mathbf{c}_{t-1} + \mathbf{i}_t \odot \tilde{\mathbf{c}}_t
        \]
        \begin{itemize}
            \item O símbolo $\odot$ indica multiplicação ponto a ponto (\textit{Hadamard}).
            \item Repare que $\mathbf{c}_t$ tem uma rota direta no tempo, sem ativação não linear no caminho principal.
            \begin{itemize}
                \item Isso ajuda a mitigar o \textit{vanishing gradient}.
            \end{itemize}
        \end{itemize}
    \end{column}

\begin{frame}{Por que a LSTM Preserva o Gradiente}
\begin{itemize}\small
    \item Na RNN \textit{vanilla}, o gradiente entre passos distantes acumula um \textbf{produto} de fatores que saturam:
\end{itemize}
\[
    \frac{\partial \mathbf{h}_t}{\partial \mathbf{h}_{t-1}} \propto \mathbf{\Theta}_{hh}\,\bigl(1 - \tanh^2(\mathbf{z}_t)\bigr)
    \qquad\Rightarrow\qquad
    \prod_{k} \mathbf{\Theta}_{hh}\,\bigl(1 - \tanh^2(\mathbf{z}_k)\bigr)
\]
\begin{itemize}\small
    \item Como $1 - \tanh^2 \le 1$, esse produto tende a \textbf{desaparecer} em sequências longas.
    \item Na LSTM, a memória tem uma rota direta no tempo, $\mathbf{c}_t = \mathbf{f}_t \odot \mathbf{c}_{t-1} + \mathbf{i}_t \odot \tilde{\mathbf{c}}_t$. Pelo \textbf{caminho direto}:
\end{itemize}
\[
    \frac{\partial \mathbf{c}_t}{\partial \mathbf{c}_{t-1}} = \mathbf{f}_t
    \qquad\Rightarrow\qquad
    \prod_{k} \mathbf{f}_k
\]
\begin{itemize}\footnotesize
    \item Nesse caminho o gradiente passa apenas pela porta $\mathbf{f}_k$ e por uma \textbf{soma}, sem $\tanh'$ repetido nem multiplicação por matrizes.
    \item Quando o \textit{forget gate} aprende $\mathbf{f}_k \approx 1$, o gradiente é preservado por muitos passos (\textit{constant error carousel}). As demais rotas, pelas \textit{gates}, contribuem com termos menores.
\end{itemize}
\end{frame}

\begin{frame}{\textit{Output Gate} e Estado Oculto $\mathbf{h}_t$}
\begin{columns}
    \begin{column}{.55\linewidth}
        \begin{itemize}
            \item Decide quais partes da memória devem ser \textbf{expostas} no estado oculto:
        \end{itemize}
        \[
            \mathbf{o}_t = \sigma\bigl([\mathbf{h}_{t-1}, \mathbf{x}_t]\,\mathbf{\Theta}_o^{T} + \mathbf{b}_o\bigr)
        \]
        \[
            \mathbf{h}_t = \mathbf{o}_t \odot \tanh(\mathbf{c}_t)
        \]
        \begin{itemize}
            \item $\mathbf{h}_t$ é o que sai da célula a cada passo, e também alimenta a próxima célula.
            \item $\mathbf{c}_t$ continua circulando como memória de longo prazo, em paralelo a $\mathbf{h}_t$.
        \end{itemize}
    \end{column}

    \toprule

    \textit{Input gate}     & $\mathbf{i}_t = \sigma([\mathbf{h}_{t-1}, \mathbf{x}_t]\,\mathbf{\Theta}_i^{T} + \mathbf{b}_i)$ \\
    Candidato $\tilde{\mathbf{c}}_t$ & $\tilde{\mathbf{c}}_t = \tanh([\mathbf{h}_{t-1}, \mathbf{x}_t]\,\mathbf{\Theta}_c^{T} + \mathbf{b}_c)$ \\
    Memória $\mathbf{c}_t$           & $\mathbf{c}_t = \mathbf{f}_t \odot \mathbf{c}_{t-1} + \mathbf{i}_t \odot \tilde{\mathbf{c}}_t$ \\
    \textit{Output gate}    & $\mathbf{o}_t = \sigma([\mathbf{h}_{t-1}, \mathbf{x}_t]\,\mathbf{\Theta}_o^{T} + \mathbf{b}_o)$ \\
    Estado oculto $\mathbf{h}_t$     & $\mathbf{h}_t = \mathbf{o}_t \odot \tanh(\mathbf{c}_t)$ \\
    \bottomrule
\end{tabular}
\vspace{0.6em}
\begin{itemize}\footnotesize
    \item Cada peso $\mathbf{\Theta}_\bullet$ é uma matriz que multiplica o vetor concatenado $[\mathbf{h}_{t-1}, \mathbf{x}_t]$, na mesma convenção $\mathbf{x}\,\mathbf{\Theta}^{T} + \mathbf{b}$ das camadas densas.
    \item Os \textit{bias} $\mathbf{b}_\bullet$ são vetores com a mesma dimensão da saída da \textit{gate}.
    \item Sendo $h_{\text{size}}$ a dimensão do estado oculto e $x_{\text{size}}$ a dimensão da entrada, os \textit{shapes} esperados são:
    \[
        \mathbf{\Theta}_\bullet \in \mathbb{R}^{h_{\text{size}} \times (h_{\text{size}} + x_{\text{size}})}, \qquad \mathbf{b}_\bullet \in \mathbb{R}^{h_{\text{size}}}.
    \]
\end{itemize}
\end{frame}

\begin{frame}{Forward Pass Otimizado}
\begin{itemize}\small
    \item As quatro camadas densas internas operam sobre o mesmo vetor $[\mathbf{h}_{t-1}, \mathbf{x}_t]$.
    \item Em vez de quatro multiplicações de matrizes, agrupamos os pesos das quatro camadas ($\mathbf{\Theta}_i, \mathbf{\Theta}_f, \mathbf{\Theta}_c, \mathbf{\Theta}_o$) e separamos o resultado depois:
    \begin{itemize}
        \item $is$: dimensão da entrada $\mathbf{x}_t$, \quad $hs$: dimensão do estado oculto $\mathbf{h}_t$.
        \item Cada $[\mathbf{h}_{t-1}, \mathbf{x}_t]\,\mathbf{\Theta}_\bullet^{T}$ se separa em $\mathbf{x}_t\,\mathbf{W}_\bullet^{T} + \mathbf{h}_{t-1}\,\mathbf{U}_\bullet^{T}$: $\mathbf{W}$ reúne as colunas de $\mathbf{\Theta}_\bullet$ que multiplicam $\mathbf{x}_t$ e $\mathbf{U}$ as que multiplicam $\mathbf{h}_{t-1}$.
    \end{itemize}
    \[
        \mathbf{W} \in \mathbb{R}^{4hs \times is}, \quad \mathbf{U} \in \mathbb{R}^{4hs \times hs}, \quad \mathbf{b} \in \mathbb{R}^{4hs}
    \]
\end{itemize}
{\footnotesize
\begin{align*}
    \mathbf{A}_t &= \mathbf{x}_t \mathbf{W}^{T} + \mathbf{h}_{t-1} \mathbf{U}^{T} + \mathbf{b} \\
    \mathbf{i}_t &= \sigma(\mathbf{A}_t[:,\,0:hs]) \\
    \mathbf{f}_t &= \sigma(\mathbf{A}_t[:,\,hs:2hs]) \\
    \tilde{\mathbf{c}}_t &= \tanh(\mathbf{A}_t[:,\,2hs:3hs]) \\
    \mathbf{o}_t &= \sigma(\mathbf{A}_t[:,\,3hs:4hs])
\end{align*}
}
\begin{itemize}\footnotesize
    \item Essa é a convenção do PyTorch (\texttt{nn.LSTM}): \texttt{weight\_ih} ($\mathbf{W}$, \textit{shape} $4hs \times is$) e \texttt{weight\_hh} ($\mathbf{U}$, \textit{shape} $4hs \times hs$) concatenam as \textit{gates} na ordem $i, f, g, o$ (o candidato $\tilde{\mathbf{c}}_t$ é o $\mathbf{g}_t$), com dois \textit{bias} $\mathbf{b}_{ih} + \mathbf{b}_{hh}$. \textbf{Atenção}: essa ordem coloca a \textit{input gate} antes da \textit{forget gate}, invertendo a ordem das equações anteriores.
\end{itemize}
\end{frame}

\begin{frame}[fragile]{Implementação em PyTorch (1/2)}
\begin{lstlisting}[style=vscode-light, basicstyle=\ttfamily\scriptsize, language=Python]
class MyLSTM(nn.Module):
    def __init__(self, input_sz, hidden_sz):
        super().__init__()
        self.input_sz   = input_sz
        self.hidden_sz  = hidden_sz
        self.W = nn.Parameter(torch.Tensor(hidden_sz * 4, input_sz))
        self.U = nn.Parameter(torch.Tensor(hidden_sz * 4, hidden_sz))
        self.bias = nn.Parameter(torch.Tensor(hidden_sz * 4))
        self.init_weights()  # nao mostrado
\end{lstlisting}
\begin{itemize}\footnotesize
    \item As matrizes $\mathbf{W}$ e $\mathbf{U}$ contêm os pesos das quatro camadas internas concatenadas.
    \item Uma única chamada matricial substitui as quatro multiplicações originais.
\end{itemize}
\end{frame}

\begin{frame}[fragile]{Implementação em PyTorch (2/2)}
\begin{lstlisting}[style=vscode-light, basicstyle=\ttfamily\scriptsize, language=Python]
def forward(self, x):
    bs, seq_sz, _ = x.size()
    HS = self.hidden_sz
    h_t = torch.zeros(bs, HS, device=x.device)
    c_t = torch.zeros(bs, HS, device=x.device)
    hidden_seq = []
    for t in range(seq_sz):
        x_t   = x[:, t, :]
        gates = x_t @ self.W.T + h_t @ self.U.T + self.bias
        i_t = torch.sigmoid(gates[:,    :HS])
        f_t = torch.sigmoid(gates[:, HS:HS*2])
        g_t = torch.tanh(   gates[:, HS*2:HS*3])
        o_t = torch.sigmoid(gates[:, HS*3:    ])
        c_t = f_t * c_t + i_t * g_t
        h_t = o_t * torch.tanh(c_t)
        hidden_seq.append(h_t.unsqueeze(0))
    hidden_seq = torch.cat(hidden_seq, dim=0).transpose(0, 1).contiguous()
    return hidden_seq, (h_t, c_t)
\end{lstlisting}
\end{frame}

\begin{frame}{\textit{Gated Recurrent Unit}}
\begin{columns}
    \begin{column}{.55\linewidth}
        \begin{itemize}
            \item Introduzida por Cho et al. em 2014\footfullcite{cho2014gru}.
            \item É considerada uma versão \textbf{simplificada} da LSTM.
            \item Usa um único vetor de estado: $\mathbf{h}_t$ também faz o papel da memória $\mathbf{c}_t$.
            \item Em vez das três \textit{gates} da LSTM, a GRU tem apenas duas:
            \begin{itemize}
                \item \textit{reset gate} $\mathbf{r}_t$,
                \item \textit{update gate} $\mathbf{z}_t$.
            \end{itemize}
            \item Menos parâmetros e treinamento mais rápido, com desempenho competitivo.
        \end{itemize}
    \end{column}
    \begin{column}{.45\linewidth}
        \centering
        \resizebox{\linewidth}{!}{%
        \begin{tikzpicture}[every node/.style={font=\scriptsize}]
            \grucell
        \end{tikzpicture}}
    \end{column}
\end{columns}
\end{frame}

\begin{frame}{Equações da GRU}
\begin{itemize}
    \item \textit{Reset gate}: decide quanto do estado anterior é usado para computar o candidato.
\end{itemize}
\[
    \mathbf{r}_t = \sigma\bigl([\mathbf{h}_{t-1}, \mathbf{x}_t]\,\mathbf{\Theta}_r^{T} + \mathbf{b}_r\bigr)
\]
\begin{itemize}
    \item \textit{Update gate}: decide a mistura entre o estado anterior e o candidato.
\end{itemize}
\[
    \mathbf{z}_t = \sigma\bigl([\mathbf{h}_{t-1}, \mathbf{x}_t]\,\mathbf{\Theta}_z^{T} + \mathbf{b}_z\bigr)
\]
\begin{itemize}
    \item Candidato a estado oculto:
\end{itemize}
\[
    \tilde{\mathbf{h}}_t = \tanh\bigl([\mathbf{r}_t \odot \mathbf{h}_{t-1}, \mathbf{x}_t]\,\mathbf{\Theta}_h^{T} + \mathbf{b}_h\bigr)
\]
\begin{itemize}
    \item Novo estado oculto:
\end{itemize}
\[
    \mathbf{h}_t = (1 - \mathbf{z}_t) \odot \mathbf{h}_{t-1} + \mathbf{z}_t \odot \tilde{\mathbf{h}}_t
\]
\end{frame}

\begin{frame}{Exemplo de GRU: \textit{Reset} e \textit{Update Gates}}
\begin{itemize}\small
    \item Reusando $h_0 = -0.6$ e $x_0 = 0.8$, logo $[h_0, x_0] = [-0.6,\,0.8]$, com pesos:
\end{itemize}
\[
    \theta_r = [0.5,\,-0.4],\ b_r = 0.2 \qquad
    \theta_z = [-0.3,\,0.6],\ b_z = 0.1 \qquad
    \theta_h = [0.4,\,0.5],\ b_h = 0
\]
\begin{itemize}\small
    \item \textit{Reset gate}:
\end{itemize}
\[
    r_t = \sigma\bigl((-0.6)\cdot 0.5 + 0.8\cdot(-0.4) + 0.2\bigr) = \sigma(-0.42) \approx 0.3965
\]
\begin{itemize}\small
    \item \textit{Update gate}:
\end{itemize}
\[
    z_t = \sigma\bigl((-0.6)\cdot(-0.3) + 0.8\cdot 0.6 + 0.1\bigr) = \sigma(0.76) \approx 0.6814
\]
\end{frame}

\begin{frame}{Exemplo de GRU: Candidato e Novo Estado}
\begin{itemize}\small
    \item O \textit{reset gate} filtra o estado anterior \textbf{antes} de entrar no candidato:
\end{itemize}
\[
    r_t \odot h_0 = 0.3965 \cdot (-0.6) \approx -0.2379
\]
\begin{itemize}\small
    \item Candidato a estado oculto, usando $[r_t \odot h_0,\, x_0] = [-0.2379,\,0.8]$:
\end{itemize}
\[
    \tilde{h}_t = \tanh\bigl((-0.2379)\cdot 0.4 + 0.8\cdot 0.5 + 0\bigr) = \tanh(0.3048) \approx 0.2957
\]
\begin{itemize}\small
    \item Novo estado oculto, mistura controlada por $z_t$:
\end{itemize}
\[
    h_t = (1 - z_t)\,h_0 + z_t\,\tilde{h}_t = 0.3186\cdot(-0.6) + 0.6814\cdot 0.2957 \approx 0.0103
\]
\begin{itemize}\footnotesize
    \item A \textbf{mesma} porta $z_t$ decide quanto manter de $h_0$ e quanto usar do candidato. Na LSTM, essa decisão é dividida entre $f_t$ e $i_t$, com a memória $c_t$ separada.
\end{itemize}
\end{frame}

\begin{frame}{LSTM vs GRU}
\centering
\begin{tabular}{l c c}
    \toprule
                                    & LSTM            & GRU \\
    \midrule
    Estados que circulam            & $\mathbf{h}_t$ e $\mathbf{c}_t$   & apenas $\mathbf{h}_t$ \\
    Quantidade de \textit{gates}    & 3 \quad ($\mathbf{f}, \mathbf{i}, \mathbf{o}$) & 2 \quad ($\mathbf{r}, \mathbf{z}$) \\
    Camadas densas internas         & 4               & 3 \\
    Parâmetros (relativo)           & $4 \cdot d$     & $3 \cdot d$ \\
    \bottomrule
\end{tabular}
\vspace{0.6em}
\begin{itemize}\footnotesize
    \item A linha de parâmetros conta apenas as matrizes densas internas (ignora \textit{bias}). Em prática, uma GRU acaba tendo aproximadamente $75\%$ dos parâmetros de uma LSTM com a mesma dimensão oculta.
    \item Em sequências longas a LSTM costuma se comportar um pouco melhor por ter o caminho dedicado para $\mathbf{c}_t$.
    \item A GRU é mais leve e converge mais rápido, sendo uma boa escolha quando há restrição de memória ou de \textit{compute}.
\end{itemize}
\end{frame}

\begin{frame}{Variantes Comuns: \textit{Bidirectional} e \textit{Stacked}}
\begin{columns}
    \begin{column}{.5\linewidth}
        \centering
        \textbf{Bidirectional RNN}\\[0.3em]
        \resizebox{!}{3.2cm}{%
        \begin{tikzpicture}[every node/.style={font=\scriptsize}, skip/.style={->, rounded corners=3pt}]
            \foreach \i in {1,2,3}{
                \node[rnncell, minimum width=0.9cm, minimum height=0.7cm, fill=PastelGreen] (f\i) at (1.9*\i, 0) {$\overrightarrow{h}_\i$};
                \node[rnncell, minimum width=0.9cm, minimum height=0.7cm, fill=PastelPink] (b\i) at (1.9*\i, -1.2) {$\overleftarrow{h}_\i$};
                \node[inputnode] (x\i) at (1.9*\i, -2.3) {$x_\i$};
                \node[opnode] (y\i) at (1.9*\i, 1.2) {$[\overrightarrow{h}_\i; \overleftarrow{h}_\i]$};
                \draw[->] (x\i) -- (b\i);
                \draw[->] (f\i) -- (y\i);
                % x_i -> forward: contorna a célula backward pela direita
                \draw[skip] (x\i.east) -- (1.9*\i+0.62, -2.3) -- (1.9*\i+0.62, -0.6)
                    -- (1.9*\i+0.25, -0.6) -- ($(f\i.south) + (0.25, 0)$);
                % backward -> saída: contorna a célula forward pela esquerda
                \draw[skip] ($(b\i.north) + (-0.25, 0)$) -- (1.9*\i-0.25, -0.6) -- (1.9*\i-0.62, -0.6)
                    -- (1.9*\i-0.62, 0.75) -- (1.9*\i-0.25, 0.75) -- ($(y\i.south) + (-0.25, 0)$);
            }
            \foreach \i/\j in {1/2, 2/3}{
                \draw[->] (f\i) -- (f\j);
                \draw[<-] (b\i) -- (b\j);
            }
        \end{tikzpicture}}
        \begin{itemize}\scriptsize
            \item Combina \textit{forward pass} e \textit{backward pass}. Saída é a concatenação dos dois estados ocultos.
            \item Útil quando toda a sequência está disponível (e.g., \textit{encoders}). Não se aplica a geração autoregressiva.
        \end{itemize}
    \end{column}
    \begin{column}{.5\linewidth}
        \centering
        \textbf{Stacked RNN}\\[0.3em]
        \resizebox{!}{3.2cm}{%
        \begin{tikzpicture}[every node/.style={font=\scriptsize}]
            \foreach \i in {1,2,3}{
                \node[rnncell, minimum width=0.9cm, minimum height=0.7cm, fill=PastelGreen] (l1\i) at (1.3*\i, 0) {$h^{(1)}_\i$};
                \node[rnncell, minimum width=0.9cm, minimum height=0.7cm, fill=PastelYellow] (l2\i) at (1.3*\i, 1.1) {$h^{(2)}_\i$};
                \node[inputnode] (x\i) at (1.3*\i, -1.1) {$x_\i$};
                \draw[->] (x\i) -- (l1\i);
                \draw[->] (l1\i) -- (l2\i);
            }
            \foreach \i/\j in {1/2, 2/3}{
                \draw[->] (l1\i) -- (l1\j);
                \draw[->] (l2\i) -- (l2\j);
            }
        \end{tikzpicture}}
        \begin{itemize}\scriptsize
            \item Várias camadas recorrentes empilhadas. A saída da camada $\ell$ vira a entrada da camada $\ell+1$.
            \item Usado em \textit{encoders} profundos para enriquecer a representação. Cada camada pode ter sua própria LSTM ou GRU.
        \end{itemize}
    \end{column}
\end{columns}
\vspace{0.4em}
\begin{itemize}\scriptsize
    \item Ambas as variantes voltam a aparecer no contexto de \textit{encoder-decoder} no próximo deck.
\end{itemize}
\end{frame}
