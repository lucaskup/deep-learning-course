  \section{Paradigmas Arquiteturais do Transformer}

  %--- Slide: Three families ---
  \begin{frame}{As Três Famílias de Transformers}
    A arquitetura \textit{Transformer} deu origem a três famílias:
    \begin{columns}[T]
      \begin{column}{0.32\textwidth}
        \begin{block}{\textit{Encoder-Decoder}}
          \begin{itemize}
            \item T5, BART, mT5
            \item \textit{Seq2seq}: entrada $\to$ saída
            \item \textit{Cross-attention} liga os dois lados
            \item Tradução, sumarização
          \end{itemize}
        \end{block}
      \end{column}
      \begin{column}{0.32\textwidth}
        \begin{block}{\textit{Decoder-Only}}
          \begin{itemize}
            \item GPT, LLaMA, Mistral
            \item Atenção \textbf{causal} (mascarada)
            \item Geração autorregressiva
            \item \textit{Chatbots}, \textit{code completion}
          \end{itemize}
        \end{block}
      \end{column}
      \begin{column}{0.32\textwidth}
        \begin{block}{\textit{Encoder-Only}}
          \begin{itemize}
            \item BERT, RoBERTa, DeBERTa
            \item Atenção \textbf{bidirecional}
            \item Foco em \textit{compreensão}
            \item Classificação, busca, NER
          \end{itemize}
        \end{block}
      \end{column}
    \end{columns}
  \end{frame}

  \section{De Embeddings Estáticos a Contextuais}

  %--- Slide: why embeddings ---
  \begin{frame}{Por que Precisamos de \textit{Word Embeddings}?}
    \begin{columns}
      \begin{column}{0.5\textwidth}
        \textbf{Representação \textit{one-hot}}:
        \begin{itemize}
          \item Vocabulário de $|V| = 50\text{k}$ $\to$ vetores de 50k dimensões
          \item \textbf{Esparsa} (um único 1, resto zeros)
          \item Ortogonal: $\text{cos}(\text{rei}, \text{rainha}) = 0$
          \item Nenhuma semântica capturada
        \end{itemize}
      \end{column}
      \begin{column}{0.5\textwidth}
        \textbf{\textit{Word embedding}}:
        \begin{itemize}
          \item Vetor de baixa dimensão ($d = 300$, $768$, \ldots)
          \item \textbf{Denso} (sem estrutura esparsa imposta)
          \item Palavras semanticamente próximas $\to$ vetores próximos
          \item Aprendido a partir de grandes \textit{corpora}
        \end{itemize}
      \end{column}
    \end{columns}
    \vspace{0.5em}
    \begin{block}{Hipótese distribucional (Harris, 1954; Firth, 1957)}
      ``\textit{You shall know a word by the company it keeps.}'', síntese de Firth (1957). Palavras em contextos semelhantes tendem a ter significados relacionados.
    \end{block}
  \end{frame}

  %--- Slide: Word2Vec / GloVe ---
  \begin{frame}{Embeddings Estáticos: Word2Vec e GloVe}
    % \begin{columns}[T]
      % \begin{column}{0.5\textwidth}
        \textbf{Word2Vec} \footfullcite{mikolov2013word2vec}
        \begin{itemize}
          \item \textbf{Skip-gram}: prever palavras do contexto a partir da palavra central
          \item \textbf{CBOW}: prever palavra central a partir do contexto
          \item Resultado: uma matriz $\mathbf{E} \in \mathbb{R}^{|V| \times d}$
        \end{itemize}
      % \end{column}
      % \begin{column}{0.5\textwidth}
        \textbf{GloVe} \footfullcite{pennington2014glove}
        \begin{itemize}
          \item Ajuste de produtos escalares ao logaritmo de coocorrências globais
          \item Combina estatística global (como LSA) com janelas locais (como W2V)
          \item Resultado: uma matriz $\mathbf{E} \in \mathbb{R}^{|V| \times d}$
        \end{itemize}
      % \end{column}
    % \end{columns}
      Cada palavra é representada por \textit{UM} único vetor, independente do contexto.
  \end{frame}

  %--- Slide: polysemy problem ---
  \begin{frame}{O Problema da Polissemia}
    \begin{exampleblock}{Duas frases em português}
      \begin{enumerate}
        \item ``Ele sacou dinheiro no \textbf{banco} da esquina.''
        \item ``Ela sentou no \textbf{banco} do parque.''
      \end{enumerate}
    \end{exampleblock}
    \vspace{0.5em}
    \begin{columns}
      \begin{column}{0.5\textwidth}
        \textbf{Embeddings estáticos}:
        \begin{itemize}
          \item \textcolor{DarkRed}{$\mathbf{v}_{\text{banco}}^{(1)} = \mathbf{v}_{\text{banco}}^{(2)}$}
          \item O mesmo vetor reúne evidências dos diferentes usos
          \item O modelo ``confunde'' instituição financeira com assento
        \end{itemize}
      \end{column}
      \begin{column}{0.5\textwidth}
        \textbf{Embeddings contextuais}:
        \begin{itemize}
          \item \textcolor{DarkGreen}{$\mathbf{v}_{\text{banco}}^{(1)} \neq \mathbf{v}_{\text{banco}}^{(2)}$}
          \item Cada ocorrência recebe um vetor diferente
          \item O contexto \textit{molda} o vetor final
        \end{itemize}
      \end{column}
    \end{columns}
    \vspace{0.5em}
    A função que gera o \textit{embedding} deve depender do contexto $f(\text{palavra}, \text{contexto}) \to \mathbb{R}^d$.
  \end{frame}

  %--- Slide: ELMo bridge ---
  \begin{frame}{A Ponte: ELMo (2018)\footfullcite{peters2018elmo}}
    \begin{columns}[T]
      \begin{column}{0.57\textwidth}
        Duas pilhas de LSTMs modelam direções separadas:
        \begin{itemize}
          \item Forward: $p(w_t\mid w_{<t})$.
          \item Backward: $p(w_t\mid w_{>t})$.
          \item Cada direção recebe a representação de entrada, baseada em caracteres.
          \item As saídas das duas direções são concatenadas em cada camada.
        \end{itemize}
        A tarefa aprende uma combinação ponderada das camadas para
        produzir o embedding contextual.
      \end{column}
      \begin{column}{0.40\textwidth}
        \centering
        \begin{tikzpicture}[font=\scriptsize, >=stealth,
          cell/.style={draw,rounded corners,minimum width=0.9cm,minimum height=0.5cm}]
          \foreach \i/\w in {0/O,1/gato,2/sentou}{

            \node[cell,fill=CellBlue] (f\i) at (1.5*\i,1) {$\to$};
            \node[cell,fill=CellRed] (b\i) at (1.5*\i,-1) {$\leftarrow$};
            \draw[->] (w\i) -- (f\i);
            \draw[->] (w\i) -- (b\i);
          }
          \draw[->] (f0) -- (f1); \draw[->] (f1) -- (f2);
          \draw[->] (b2) -- (b1); \draw[->] (b1) -- (b0);

          \node[below=0.15cm of b1] {pilha backward};
        \end{tikzpicture}
        \par\smallskip
        {\scriptsize Esquema das direções. As camadas internas de cada pilha foram omitidas.}
      \end{column}
    \end{columns}
  \end{frame}

  %--- Slide: contextual via transformers ---
  \begin{frame}{Embeddings Contextuais via \textit{Transformers}}
    Um encoder Transformer contextualiza cada token com atenção bidirecional.
    \begin{itemize}
      \item A atenção combina vetores de valor; FFN, resíduos e normalização completam a camada.
      \item As posições são processadas em paralelo, sem recorrência temporal.
      \item A atenção encurta o caminho computacional entre posições distantes.
    \end{itemize}
    \begin{block}{Formalizando}
      Para a posição $i$, em uma sequência de comprimento $T$:
      \[
        \mathbf h_i^{(\ell)}=f_\ell\!\left(
          \mathbf h_0^{(\ell-1)},\ldots,\mathbf h_{T-1}^{(\ell-1)},i\right).
      \]
      $f_\ell$ é a camada completa do encoder.
    \end{block}
  \end{frame}

  \section{Tarefas em Nível de \textit{Token}}

  %--- Slide: general recipe ---
  \begin{frame}{A Receita Geral}
    \begin{columns}
      \begin{column}{0.5\textwidth}
        Para qualquer tarefa em nível de \textit{token}:
        \begin{enumerate}
          \item Passe a frase por um \textit{encoder} pré-treinado
          \item Extraia os \textit{embeddings} contextuais $\mathbf{h}_1, \ldots, \mathbf{h}_L$
          \item Adicione uma cabeça linear \textcolor{DarkBlue}{$\mathbf{W} \in \mathbb{R}^{d \times C}$}
          \item $\text{logits}_i = \mathbf{W}^\top \mathbf{h}_i$
          \item \textit{Softmax} sobre $C$ classes
        \end{enumerate}
      \end{column}

      \node[pl, below=0.2 of t6] (l6) {B-LOC};
      \node[pl, below=0.2 of t7] (l7) {I-LOC};
    \end{tikzpicture}}
    \vspace{0.4em}
    \begin{itemize}
      \item \textbf{Entidades multi-\textit{token}} (``Porto Alegre'') são capturadas por \textbf{B-} seguido de \textbf{I-}
      \item Cabeça linear $\mathbf{W} \in \mathbb{R}^{d \times C}$ com $C = 2k+1$ (um B- e um I- por tipo, mais O), onde $k$ é o número de categorias presentes no NER
      \item \textit{Benchmark} clássico: CoNLL-2003 \footfullcite{tjongkimsang2003conll}
    \end{itemize}
  \end{frame}

  \begin{frame}{De Palavras a Subwords: Alinhar os Rótulos}
    O tokenizer pode dividir uma palavra em vários tokens.
    \begin{block}{Exemplo de segmentação ilustrativa}
      \centering
      \begin{tabular}{lccc}
        \toprule
        Palavra & Porto & Alegre & \\
        Subwords & Porto & Ale & \texttt{\#\#gre} \\
        Rótulo de treino & B-LOC & I-LOC & ignorado \\
        \bottomrule
      \end{tabular}
    \end{block}
    \begin{itemize}
      \item Uma opção é supervisionar apenas a primeira subword de cada palavra.
      \item Tokens especiais, padding e subwords restantes são ignorados na perda.
      \item Outra opção é propagar os rótulos, respeitando o esquema BIO.
      \item A avaliação deve reconstruir as entidades no nível esperado pelo dataset.
    \end{itemize}
  \end{frame}

  \begin{frame}{Tarefa: \textit{Part-of-Speech Tagging} (POS)}
    Classificar cada \textit{token} em uma categoria gramatical (substantivo, verbo, adjetivo, \ldots).

    \vspace{0.5em}
    \centering
    \resizebox{0.9\linewidth}{!}{%

      \node[tag, below=0.2 of t4] {VERB};
      \node[tag, below=0.2 of t5] {ADV};
    \end{tikzpicture}}
    \vspace{0.5em}
    \begin{itemize}
      \item Problema clássico de NLP, resolvido com acurácia dependente do idioma, dos dados e do esquema de tags
      \item \textit{Embeddings contextuais} são ideais: a mesma palavra pode ter \textit{tag} diferente em contextos diferentes
      \begin{itemize}

          \item Eu \textbf{jogo} PlayStation todas as terças-feiras.
      \end{itemize}

    \end{itemize}
  \end{frame}

  %--- Slide: QA ---
  \begin{frame}{QA Extrativa: Localizar a Resposta\footfullcite{rajpurkar2016squad}}
    Entrada: \texttt{[CLS] pergunta [SEP] contexto [SEP]}.
    \begin{exampleblock}{Exemplo com tokenização ilustrativa}
      Pergunta: ``Quando foi fundada?''\par
      Contexto: ``A PUCRS foi fundada em 1948.''
      \smallskip
      \centering\small
      \begin{tabular}{c|rrrrrrr}
        Índice & 0&1&2&3&4&5&6\\
        \midrule
        Token & \texttt{[CLS]}&Quando&foi&fundada&?&\texttt{[SEP]}&A\\
        \midrule
        Índice & 7&8&9&10&11&12&13\\
        \midrule
        Token & PUCRS&foi&fundada&em&1948&.&\texttt{[SEP]}\\
      \end{tabular}
    \end{exampleblock}
    A resposta ocupa as posições $s=e=11$ neste exemplo.
    Os índices usados no treino vêm da tokenização e dos offsets do texto.
    \begin{itemize}
      \item O modelo prevê uma distribuição de início e outra de fim.
      \item Na inferência, selecionamos um trecho do contexto com $s\leq e$.
    \end{itemize}
  \end{frame}

  \begin{frame}{QA Extrativa: Cabeças e Perda}
    Para cada posição $i$, calculamos dois logits:
    \[
      a_i=\mathbf w_{\mathrm{start}}^\top\mathbf h_i,
      \qquad b_i=\mathbf w_{\mathrm{end}}^\top\mathbf h_i.
    \]
    O softmax normaliza \textbf{sobre as posições} da sequência, excluindo padding:
    \[
      p_{\mathrm{start}}(i)=\frac{e^{a_i}}{\sum_{j\in V}e^{a_j}},
      \qquad p_{\mathrm{end}}(i)=\frac{e^{b_i}}{\sum_{j\in V}e^{b_j}}.
    \]
    Para os índices corretos $(s,e)$:
    \[
      \mathcal L_{\mathrm{QA}}=-\log p_{\mathrm{start}}(s)-\log p_{\mathrm{end}}(e).
    \]
    $V$ reúne as posições válidas. Na seleção da resposta, restringimos
    os candidatos ao contexto. SQuAD 2.0 também exige tratar ausência de resposta.
  \end{frame}

  %--- Slide: why contextual matters for these tasks ---
  \begin{frame}{O que Muda com Embeddings Contextuais?}
    \begin{itemize}
      \item O encoder usa o contexto para representar ocorrências ambíguas.
      \item Cabeças de classificação aproveitam representações pré-treinadas.
      \item Fine-tuning adapta essas representações aos dados da tarefa.
    \end{itemize}
    \begin{block}{Resultados do BERT original\cite{devlin2019bert}}
      \begin{tabular}{lll}
        \toprule
        Tarefa & Configuração & F1 no teste\\
        \midrule
        CoNLL-2003 & BERT-Large & 92,8\\
        SQuAD 1.1 & BERT-Large, ensemble + TriviaQA & 93,2\\
        \bottomrule
      \end{tabular}
    \end{block}
    A comparação depende da métrica, da divisão dos dados e da configuração.
    O resultado de SQuAD acima usa dados adicionais e um ensemble.
  \end{frame}

  \section{BERT}

  %--- Slide: BERT intro ---
  \begin{frame}{BERT: Encoder Bidirecional Pré-Treinado\footfullcite{devlin2019bert}}
    \small \textit{Bidirectional Encoder Representations from Transformers}.\par\medskip
    
    \begin{columns}
      \begin{column}{0.55\textwidth}
        \textbf{Contribuições principais}:
        \begin{itemize}
          \item Usa apenas a \textbf{pilha encoder} do \textit{Transformer}
          \item Pré-treino \textbf{bidirecional} combinando dois objetivos:
            \begin{itemize}
              \item \textit{Masked Language Modeling} (MLM)
              \item \textit{Next Sentence Prediction} (NSP)
            \end{itemize}
          \item ``\textit{Fine-tuning}'': backbone pré-treinado + cabeça da tarefa, avaliados em 11 tarefas de NLP
          \item \textit{State-of-the-art} em quase todos os \textit{benchmarks} da época
        \end{itemize}
      \end{column}
      \begin{column}{0.42\textwidth}
        \begin{exampleblock}{Por que ``bidirecional''?}
          GPT-1 (2018) era unidirecional (causal). BERT mostrou que, para \textit{compreensão}, ver \textbf{o futuro} da frase melhora drasticamente.
          \vspace{0.3em}
          Seu objetivo de pré-treino é reconstruir tokens selecionados; geração autorregressiva exige outra formulação.
        \end{exampleblock}
      \end{column}
    \end{columns}
  \end{frame}

  %--- Slide: BERT architecture ---
  \begin{frame}{Arquitetura: Pilha de Encoders do \textit{Transformer}}
    \small
    \begin{columns}
      \begin{column}{0.67\textwidth}
        BERT = $N$ camadas de \textit{encoder}, cada uma com:
        \begin{itemize}
          \item \textit{Multi-head self-attention} \textbf{bidirecional} (sem máscara causal)
          \item \textit{Feed-forward network} (GELU)
          \item \textit{Layer Norm} + conexões residuais
        \end{itemize}
        \vspace{0.4em}
        \textbf{Duas versões oficiais}:
        \begin{center}
          \small
          \begin{tabular}{lccc}
            \toprule
             & \textbf{Base} & \textbf{Large} \\
            \midrule
            Camadas ($L$) & 12 & 24 \\
            \textit{Hidden} ($d$) & 768 & 1024 \\
            \textit{Heads} ($h$) & 12 & 16 \\
            Parâmetros & 110M & 340M \\
            \bottomrule
          \end{tabular}
        \end{center}
      \end{column}
      \begin{column}{0.30\textwidth}
        \centering
        \resizebox{0.82\linewidth}{!}{%
        \begin{tikzpicture}[
          enc/.style={draw, rounded corners, fill=CellGreen, minimum width=2.4cm, minimum height=0.50cm, font=\scriptsize},
          emb/.style={draw, fill=CellBlue, minimum width=2.4cm, minimum height=0.50cm, font=\scriptsize},
        ]
          \node[emb] (e) at (0,0) {\textit{Input Embeddings}};
          \node[enc, above=0.15 of e] (l1) {Encoder Layer 1};
          \node[enc, above=0.15 of l1] (l2) {Encoder Layer 2};
          \node[font=\scriptsize, above=0.15 of l2] (d) {$\vdots$};
          \node[enc, above=0.15 of d] (lN) {Encoder Layer $L$};
          \node[draw, fill=CellYellow, above=0.25 of lN, minimum width=2.4cm, minimum height=0.50cm, font=\scriptsize] (h) {Task Head};
        \end{tikzpicture}}
      \end{column}
    \end{columns}
  \end{frame}

  %--- Slide: training data ---
  \begin{frame}{Dados de Pré-Treino}
        BERT foi pré-treinado em:
        \begin{itemize}
          \item \textbf{BooksCorpus}: $\sim 800$M palavras de livros não publicados
          \item \textbf{Wikipedia inglesa}: $\sim 2.5$B palavras (só texto, sem tabelas/listas)
        \end{itemize}
        \vspace{0.3em}
        Total: $\sim 3.3$B palavras, $\sim 16$ GB de texto.
        \vspace{0.4em}

        \textbf{Por que esses dados?}
        \begin{itemize}
          \item Livros e artigos enciclopédicos com texto contínuo
          \item Contêm estruturas de longa distância (sentenças contíguas) $\to$ útil para NSP
          \item Linguagem formal e variada
        \end{itemize}

  \end{frame}

  %--- Slide: WordPiece reference ---
  \begin{frame}{Tokenização: WordPiece}
    BERT usa \textbf{WordPiece}.
    \begin{block}{Recapitulação rápida}
      \begin{itemize}
        \item Vocabulário de $\sim 30$k \textit{subwords}
        \item Palavras podem ser divididas em subwords; continuações são marcadas com \texttt{\#\#}
        \item Exemplo ilustrativo: \texttt{tokenização} $\to$ \texttt{token \#\#iza \#\#ção}
        \item Reduz ocorrências fora do vocabulário; \texttt{[UNK]} ainda pode ocorrer
      \end{itemize}
    \end{block}
    \vspace{0.3em}
    \textbf{Tokens especiais} adicionados pelo tokenizer do BERT:
    \begin{itemize}
      \item \texttt{[CLS]}: marca o início da sequência
      \item \texttt{[SEP]}: separador de segmentos
      \item \texttt{[PAD]}: preenchimento para lotes
      \item \texttt{[MASK]}: usado na corrupção de entrada para MLM
      \item \texttt{[UNK]}: \textit{token} desconhecido (raro com WordPiece)
    \end{itemize}
  \end{frame}

  %--- Slide: Input representation ---
  \begin{frame}{Representação de Entrada do BERT}
    O \textit{embedding} final de cada \textit{token} é a \textbf{soma} de três componentes:
    \vspace{0.4em}
    \centering
    \resizebox{0.95\linewidth}{!}{%
    \begin{tikzpicture}[
      ebox/.style={draw, minimum width=1.4cm, minimum height=0.55cm, font=\scriptsize},
      tok/.style={ebox, fill=CellBlue},
      seg/.style={ebox, fill=CellGreen},
      posemb/.style={ebox, fill=CellYellow},
      lbl/.style={ebox, fill=LightGray, font=\scriptsize\ttfamily},
    ]
      % Tokens
      \foreach \i/\t in {0/[CLS],1/o,2/gato,3/[SEP],4/dorme,5/[SEP]} {
        \node[lbl] (l\i) at (\i*1.5, 3.2) {\t};
      }
      % Token embeddings
      \foreach \i/\w in {0/CLS,1/o,2/gato,3/SEP,4/dorme,5/SEP} {
        \node[tok] (t\i) at (\i*1.5, 2.5) {$E_{\mathrm{\w}}$};
      }
      % Segment embeddings
      \foreach \i/\s in {0/A,1/A,2/A,3/A,4/B,5/B} {
        \node[seg] (s\i) at (\i*1.5, 1.7) {$E_{\s}$};
      }
      % Position embeddings
      \foreach \i in {0,...,5} {
        \node[posemb] (p\i) at (\i*1.5, 0.9) {$E_{\i}$};
      }

      % Plus signs
      \foreach \i in {0,...,5} {
        \node[font=\small] at (\i*1.5, 2.1) {$+$};
        \node[font=\small] at (\i*1.5, 1.3) {$+$};
      }

      \node[font=\scriptsize, right] at (9.3,3.2) {\textit{Tokens}};
      \node[font=\scriptsize, right] at (9.3,2.5) {\textit{Token Embeddings}};
      \node[font=\scriptsize, right] at (9.3,1.7) {\textit{Segment Embeddings}};
      \node[font=\scriptsize, right] at (9.3,0.9) {\textit{Position Embeddings}};
    \end{tikzpicture}}
    \vspace{0.3em}
    \[
      \mathbf{x}_i = E_{\text{token}}(w_i) \;+\; E_{\text{segment}}(s_i) \;+\; E_{\text{position}}(i)
    \]
    \textbf{Tudo é aprendido} (inclusive a posição, ao contrário do PE sinusoidal).
  \end{frame}

  %--- Slide: CLS token ---
  \begin{frame}{O \textit{Token} \texttt{[CLS]}}
        \texttt{[CLS]} é um \textit{token} \textbf{sempre} prefixado à sequência.

        \vspace{0.4em}
        \textbf{Papel}:
        \begin{itemize}
          \item Não é uma palavra real, começa como um \textit{embedding} aleatório aprendido
          \item Ao longo das $L$ camadas, acumula informação de \textbf{toda a frase} via \textit{self-attention}
          \item Seu estado final $\mathbf{h}_{[\text{CLS}]}^{(L)}$ é usado como \textbf{representação agregada da sequência}
          \item Cabeças de \textbf{classificação} de sentença ligam-se a ele
        \end{itemize}
        \begin{alertblock}{Observação}
          $\mathbf{h}_{[\text{CLS}]}$ do BERT não é um bom \textit{sentence embedding}.
        \end{alertblock}
  \end{frame}

  %--- Slide: SEP and segments ---
  \begin{frame}{\texttt{[SEP]} e \textit{Segment Embeddings}}
    BERT pode receber uma ou \textbf{duas} sentenças:
    \vspace{0.3em}
    \begin{block}{Formato para uma sentença}
      \texttt{[CLS] o gato dorme [SEP]}
    \end{block}
    \begin{block}{Formato para duas sentenças (par)}
      \texttt{[CLS] o gato dorme [SEP] no tapete vermelho [SEP]}
    \end{block}
    \vspace{0.3em}
    \begin{itemize}
      \item \texttt{[SEP]} marca \textbf{fronteiras} entre segmentos
      \item \textbf{Segment embeddings}: $E_A$ adicionado aos \textit{tokens} da primeira sentença, $E_B$ aos da segunda
      \item Permite tarefas de \textbf{pares de sentença}: NLI, similaridade, paráfrase, QA
    \end{itemize}
    \vspace{0.3em}
    Os segment embeddings identificam os dois segmentos no BERT.
    Separadores e posições também fornecem informação sobre as fronteiras.
  \end{frame}

  %--- Slide: fixed length + padding ---
  \begin{frame}{Comprimento Máximo e \textit{Padding}}
    \textbf{BERT tem um comprimento máximo fixo de 512 \textit{tokens}}.
    \vspace{0.4em}
    \begin{columns}
      \begin{column}{0.5\textwidth}
        \textbf{De onde vem o limite?}
        \begin{itemize}
          \item \textit{Position embeddings} aprendidos: só existem para posições $0, \ldots, 511$
          \item Atenção $O(L^2)$ $\to$ custo proibitivo para $L$ grande
          \item \textit{Batching} eficiente requer forma uniforme
        \end{itemize}
      \end{column}
      \begin{column}{0.5\textwidth}
        \textbf{Como uniformizar um \textit{batch}?}
        \begin{itemize}
          \item Frases \textbf{curtas} $\to$ \textbf{\textit{padding}} com \texttt{[PAD]} até o comprimento desejado
          \item Frases \textbf{longas} $\to$ \textbf{truncagem} (ou \textit{sliding window})
          \item \textit{Attention mask} indica quais posições são reais
        \end{itemize}
      \end{column}
    \end{columns}
    \vspace{0.4em}
    \begin{exampleblock}{Máscara de atenção}
      \centering
      \ttfamily\small
      \begin{tabular}{cccccccc}
        {[CLS]} & o & gato & {[SEP]} & {[PAD]} & {[PAD]} & {[PAD]} & {[PAD]} \\[2pt]
        1 & 1 & 1 & 1 & 0 & 0 & 0 & 0 \\
      \end{tabular}
      \rmfamily

      \vspace{0.3em}
      As colunas de chave com máscara \texttt{0} recebem $-\infty$
      nos logits de atenção. Essas posições também devem ser ignoradas no pooling e na perda.
    \end{exampleblock}
  \end{frame}

  %--- Slide: MLM objective ---
  \begin{frame}{Objetivo \#1: \textit{Masked Language Modeling} (MLM)}
    \textbf{Ideia}: selecionar $15\%$ das posições, corromper a entrada
    e prever os tokens originais nessas posições.

    \vspace{0.5em}
    \centering
    \resizebox{0.75\linewidth}{!}{%
    \begin{tikzpicture}[
      tok/.style={draw, rounded corners, fill=LightGray, minimum width=1.5cm, minimum height=0.55cm, font=\small},
      mask/.style={draw, rounded corners, fill=CellRed, minimum width=1.5cm, minimum height=0.55cm, font=\small\ttfamily},
      lbl/.style={font=\scriptsize, DarkBlue},
    ]
      \node[tok] (t1) at (0,1) {O};
      \node[tok, right=0.1 of t1] (t2) {gato};
      \node[mask, right=0.1 of t2] (t3) {[MASK]};
      \node[tok, right=0.1 of t3] (t4) {no};
      \node[tok, right=0.1 of t4] (t5) {tapete};

      \node[lbl, below=0.2 of t3] {modelo deve prever: ``sentou''};
    \end{tikzpicture}}
    \vspace{0.5em}
    \begin{itemize}
      \item Perda: \textit{cross-entropy} sobre o vocabulário para \textbf{cada posição selecionada}
      \item O modelo aprende \textbf{atenção bidirecional}: pode olhar para a esquerda E direita
      \item MLM usa contexto dos dois lados; o objetivo causal prevê o próximo token usando o contexto à esquerda
    \end{itemize}
    \[
      \mathcal{L}_{\text{MLM}} = -\frac{1}{|M|} \sum_{i \in M} \log p_\theta(w_i \mid \tilde{\mathbf w})
    \]
    onde $M$ contém as posições selecionadas e $\tilde{\mathbf w}$ é a sequência corrompida.
  \end{frame}

  %--- Slide: 80/10/10 rule ---
  \begin{frame}{A Regra $80/10/10$ do MLM}
    Dos $15\%$ \textit{tokens} selecionados para máscara, \textbf{nem todos} viram \texttt{[MASK]}:
    \vspace{0.2em}
    \begin{columns}
      \begin{column}{0.48\textwidth}
        \begin{block}{Distribuição}
          \begin{itemize}\setlength{\itemsep}{1pt}
            \item \textbf{$80\%$}: substituído por \texttt{[MASK]}
            \item \textbf{$10\%$}: substituído por \textbf{palavra aleatória}
            \item \textbf{$10\%$}: \textbf{mantido} como está
          \end{itemize}
          \vspace{0.1em}
          {\small Em todos os casos, o modelo deve prever a palavra original.}
        \end{block}
      \end{column}
      \begin{column}{0.5\textwidth}
        \textbf{Por que essa mistura?}
        \begin{itemize}\small\setlength{\itemsep}{1pt}
          \item Tarefas discriminativas usualmente recebem texto sem \texttt{[MASK]}, criando uma diferença em relação ao MLM
          \item $10\%$ aleatórios forçam o modelo a ser ``cético'' sobre \textit{tokens} visíveis
          \item $10\%$ intactos também supervisionam posições cuja entrada foi preservada
        \end{itemize}
      \end{column}
    \end{columns}
    \vspace{0.2em}
    \begin{exampleblock}{Exemplo}\small
      \begin{itemize}\setlength{\itemsep}{1pt}
        \item ``\textit{O gato} [MASK] \textit{no tapete}'' \hfill $80\%$
        \item ``\textit{O gato \textbf{banana} no tapete}'' \hfill $10\%$
        \item ``\textit{O gato sentou no tapete}'' (inalterado) \hfill $10\%$
      \end{itemize}
    \end{exampleblock}
  \end{frame}

  %--- Slide: MLM worked example ---
  \begin{frame}{MLM: Exemplo Numérico}
    Exemplo didático: cada palavra é tratada como um token.
    \begin{block}{Entrada corrompida}
      \texttt{A capital da França é [MASK]}\par
      Alvo na posição selecionada: \texttt{Paris}.
    \end{block}
    \begin{columns}[T]
      \begin{column}{0.54\textwidth}
        \begin{enumerate}
          \item O encoder produz $\mathbf h_{\texttt{[MASK]}}\in\mathbb R^{768}$.
          \item A cabeça MLM aplica projeção, GELU, normalização e saída para o vocabulário.
          \item O softmax produz probabilidades para os tokens candidatos.
        \end{enumerate}
      \end{column}
      \begin{column}{0.40\textwidth}
        \centering
        \begin{tabular}{lr}
          \toprule
          Token & Probabilidade\\
          \midrule
          Paris & 0,71\\
          Lyon & 0,08\\
          Marselha & 0,04\\
          Berlim & 0,02\\
          Demais & 0,15\\
          \bottomrule
        \end{tabular}
      \end{column}
    \end{columns}
    \medskip
    Perda na posição selecionada: $-\ln 0{,}71\approx0{,}342$.
    \par\small As probabilidades são hipotéticas; a tokenização real depende do modelo.
  \end{frame}

  %--- Slide: NSP ---
  \begin{frame}{Objetivo \#2: \textit{Next Sentence Prediction} (NSP)}
    \textbf{Ideia}: dado um par de sentenças $(A, B)$, prever se $B$ é a sentença que \textbf{segue} $A$ no texto original, ou se é uma sentença aleatória.
    \vspace{0.4em}
    \begin{columns}
      \begin{column}{0.5\textwidth}
        \begin{exampleblock}{Exemplo positivo (\texttt{IsNext})}
          $A$: ``Ele abriu a porta do carro.'' \\
          $B$: ``Em seguida, entrou e ligou o motor.''
        \end{exampleblock}
        \begin{exampleblock}{Exemplo negativo (\texttt{NotNext})}
          $A$: ``Ele abriu a porta do carro.'' \\
          $B$: ``Mercúrio é o menor planeta do sistema solar.''
        \end{exampleblock}
      \end{column}
      \begin{column}{0.48\textwidth}
        \textbf{Arquitetura}:
        \begin{itemize}
          \item Entrada: \texttt{[CLS] A [SEP] B [SEP]}
          \item Cabeça binária sobre $\mathbf{h}_{[\text{CLS}]}^{(L)}$
          \item \textit{Cross-entropy} binária
        \end{itemize}
        \vspace{0.3em}
        \textbf{Motivação}: treinar o modelo a entender \textbf{relações entre sentenças} (útil para NLI, QA, \ldots).
      \end{column}
    \end{columns}
    \vspace{0.3em}
    RoBERTa \footfullcite{liu2019roberta} dispensou NSP em sua receita
    de pré-treino. O benefício de um objetivo depende dos dados e da configuração.
  \end{frame}

  %--- Slide: Combined loss ---
  \begin{frame}{Perda Total de Pré-Treino}
    As duas perdas são combinadas em um único escalar:
    \[
      \mathcal{L}_{\text{BERT}} = \mathcal{L}_{\text{MLM}} + \mathcal{L}_{\text{NSP}}
    \]
    \vspace{0.3em}
    \begin{itemize}
      \item O \textbf{mesmo} \textit{forward pass} gera ambas as perdas
      \item $\mathcal{L}_{\text{MLM}}$ usa as posições mascaradas
      \item $\mathcal{L}_{\text{NSP}}$ usa a posição \texttt{[CLS]}
      \item \textit{Backprop} ocorre conjuntamente, com os gradientes somados
    \end{itemize}
    \vspace{0.4em}
    \begin{block}{Treinamento original}
      \begin{itemize}
        \item $1$M passos de $256$ sequências, $\sim 40$ épocas
        \item BERT-Base: 4 Cloud TPUs em configuração Pod (16 chips), cerca de 4 dias
        \item \textit{Adam} com \textit{warmup} linear
      \end{itemize}
    \end{block}
  \end{frame}

  %--- Slide: fine-tuning overview ---
  \begin{frame}{\textit{Fine-Tuning}: Uma Cabeça, Muitas Tarefas}
    Depois do pré-treino, o BERT pode ser \textbf{adaptado} a uma tarefa específica treinando por poucas épocas com uma \textbf{cabeça nova}.

    \vspace{0.4em}
    \begin{columns}[T]
      \begin{column}{0.48\textwidth}
        \textbf{Receita genérica}:
        \begin{enumerate}
          \item Carregar pesos pré-treinados
          \item Substituir a cabeça MLM / NSP por uma cabeça específica da tarefa (inicializada aleatoriamente)
          \item Treinar \textbf{tudo} de ponta a ponta por 2--5 épocas
          \item \textit{Learning rate} baixa ($\sim 2 \times 10^{-5}$)
        \end{enumerate}
      \end{column}
      \begin{column}{0.5\textwidth}
        \begin{alertblock}{Tarefas de adaptação}
          O \textbf{mesmo modelo base} resolve:
          \begin{itemize}
            \item Classificação de frase (SST-2, CoLA)
            \item Pares de frase (NLI, paráfrase)
            \item Marcação de \textit{token} (NER, POS)
          \end{itemize}
          Só muda a cabeça e o formato da entrada.
        \end{alertblock}
      \end{column}
    \end{columns}
  \end{frame}

  %--- Slide: 4 fine-tuning formats ---
  \begin{frame}{Fine-Tuning: Classificação de Sentenças}

  \begin{frame}{\textit{Feature Extraction} vs \textit{Fine-Tuning}}
    Duas estratégias para usar um modelo pré-treinado:
    \vspace{0.3em}
    \begin{columns}[T]
      \begin{column}{0.5\textwidth}
        \begin{block}{\textit{Feature Extraction} (pesos congelados)}
          \begin{itemize}
            \item Congela BERT, treina só a cabeça
            \item Reduz o custo de retropropagação
            \item Pode \textbf{cachear} os \textit{embeddings}
            \item Desempenho geralmente menor
            \item Útil quando temos poucos dados ou pouca GPU
          \end{itemize}
        \end{block}
      \end{column}
      \begin{column}{0.5\textwidth}
        \begin{block}{\textit{Fine-Tuning} (pesos descongelados)}
          \begin{itemize}
            \item Treina BERT \textbf{inteiro} + cabeça
            \item \textit{Learning rate} baixa ($2\times10^{-5}$)
            \item Pode melhorar o desempenho na tarefa
            \item Custo maior de memória e retropropagação, dependente da configuração
            \item Risco de \textit{overfitting} com poucos dados
          \end{itemize}
        \end{block}
      \end{column}
    \end{columns}
    \vspace{0.3em}
    Ainda há possibilidade de usar adaptadores \textsc{LoRA} para treinar apenas um subconjunto pequeno de parâmetros.
  \end{frame}

  %--- Slide: BERT variants ---
  \begin{frame}{O Ecossistema BERT: Principais Variantes}
    \begin{itemize}\small
      \item \textbf{RoBERTa}\cite{liu2019roberta}: mais dados, lotes maiores,
      mascaramento dinâmico e remoção de NSP.
      \item \textbf{ALBERT}\cite{lan2020albert}: fatoração dos embeddings
      e compartilhamento de parâmetros entre camadas.
      \item \textbf{DeBERTa}\cite{he2021deberta}: representações separadas
      de conteúdo e posição na atenção, com alteração da cabeça MLM.
      \item \textbf{ELECTRA}\cite{clark2020electra}: um gerador propõe
      substituições; o discriminador prevê se cada token coincide com o original.
    \end{itemize}
    \begin{block}{Arquitetura e objetivo}
      São encoders bidirecionais com mudanças no pré-treino.
      Em ELECTRA, a perda do discriminador usa todas as posições válidas,
      oferecendo supervisão mais densa que MLM.
    \end{block}
  \end{frame}

  % SECTION 5: TOKEN TO SENTENCE
  %======================================================================================
  \section{De \textit{Tokens} a Sentenças: \textit{Sentence Embeddings}}

  %--- Slide: motivation ---
  \begin{frame}{Motivação: Um Vetor por Sentença}
    \small
    Muitas aplicações precisam de \textbf{um único vetor} que represente uma sentença ou documento inteiro:
    \vspace{0.4em}
    \begin{columns}
      \begin{column}{0.5\textwidth}
        \begin{block}{Aplicações}
          \begin{itemize}
            \item \textbf{Busca semântica}: encontrar documentos similares a uma consulta
            \item \textbf{RAG}: recuperar contexto relevante para um LLM
            \item \textbf{Clustering}: agrupar documentos por tópico
            \item \textbf{Deduplicação}: detectar paráfrases
            \item \textbf{Classificação}: \textit{zero-shot} com \textit{embeddings} comparados a rótulos
            \item \textbf{Recomendação}: \textit{item-to-item}
          \end{itemize}
        \end{block}
      \end{column}
      \begin{column}{0.48\textwidth}
        \textbf{Requisitos}:
        \begin{itemize}
          \item \textbf{Rápido}: milhões de vetores pré-computados, consulta em milissegundos
          \item \textbf{Compacto}: $384$--$1024$ dimensões
          \item \textbf{Semanticamente} estruturado: similaridade cosseno deve refletir similaridade de significado
        \end{itemize}
        \vspace{0.3em}
        \textcolor{DarkBlue}{\textbf{BERT \textit{out-of-the-box} resolve isso?}}

        \vspace{0.2em}
        \textit{Não muito bem, como veremos.}
      \end{column}
    \end{columns}
  \end{frame}

  %--- Slide: naive approaches ---
  \begin{frame}{Abordagens Ingênuas}
    Como extrair \textbf{um} vetor de BERT para uma sentença?
    \vspace{0.4em}
    \begin{columns}[T]
      \begin{column}{0.33\textwidth}
        \begin{block}{\texttt{[CLS]} \textit{pooling}}
          \begin{itemize}
            \item Usar $\mathbf{h}_{[\text{CLS}]}^{(L)}$
            \item ``Parece o certo'': ele foi projetado para classificação
            \item \textcolor{DarkRed}{\textbf{Ruim}} para similaridade
          \end{itemize}
        \end{block}
      \end{column}
      \begin{column}{0.33\textwidth}
        \begin{block}{\textit{Mean pooling}}
          \begin{itemize}
            \item $\frac{\sum_i m_i\mathbf h_i}{\sum_i m_i}$
            \item Simples, robusto
            \item Usa a máscara para ignorar \texttt{[PAD]}
            \item Empiricamente \textbf{melhor} que \texttt{[CLS]}
          \end{itemize}
        \end{block}
      \end{column}
      \begin{column}{0.33\textwidth}
        \begin{block}{\textit{Max pooling}}
          \begin{itemize}
            \item $\max_i \mathbf{h}_i^{(L)}$ por dimensão
            \item Captura ``\textit{features} mais salientes''
            \item Menos usada na prática
          \end{itemize}
        \end{block}
      \end{column}
    \end{columns}
    \vspace{0.3em}
    \begin{alertblock}{Descoberta de \cite{reimers2019sbert}}
      \textbf{BERT \textit{vanilla} + \texttt{[CLS]} tem desempenho \textit{inferior} que GloVe} em \textit{Semantic Textual Similarity} (STS) usando similaridade cosseno!
    \end{alertblock}
  \end{frame}

  \begin{frame}{Mean Pooling: Por que Usar a Máscara?}
    Exemplo de estados em duas dimensões:
    \begin{center}
      \begin{tabular}{lccccc}
        \toprule
        Token & \texttt{[CLS]} & gato & \texttt{[SEP]} & \texttt{[PAD]} & \texttt{[PAD]}\\
        \midrule
        Estado $\mathbf h_i$ & $(1,0)$ & $(2,2)$ & $(0,1)$ & $(4,4)$ & $(4,4)$\\
        Máscara $m_i$ & 1 & 1 & 1 & 0 & 0\\
        \bottomrule
      \end{tabular}
    \end{center}
    \[
      \mathbf u=\frac{\sum_i m_i\mathbf h_i}{\sum_i m_i}
        =\frac{(3,3)}{3}=(1,1).
    \]
    Sem máscara, obteríamos $(11,11)/5=(2{,}2,2{,}2)$.
    \begin{itemize}
      \item O estado de uma posição de padding pode ser diferente de zero.
      \item Este exemplo inclui tokens especiais. A política deve acompanhar
      o pooling usado no treinamento do modelo.
      \item Vetores normalizados em norma $L_2$ permitem calcular cosseno por produto escalar.
    \end{itemize}
  \end{frame}

  %--- Slide: why CLS fails ---
  \begin{frame}{Por que Adaptar BERT para Similaridade?}
    \begin{itemize}
      \item No BERT original, MLM supervisiona tokens e NSP usa o estado de \texttt{[CLS]} para classificar pares.
      \item Esses objetivos não otimizam diretamente o cosseno entre vetores de frases independentes.
      \item As representações podem se concentrar em algumas direções (anisotropia), prejudicando sua discriminação por cosseno.
    \end{itemize}
    \begin{block}{Adaptar a geometria dos embeddings}
      Um objetivo de similaridade ou relevância ensina quais pares devem
      receber scores maiores. Pooling e normalização fazem parte dessa escolha.
    \end{block}
    O resultado depende do modelo, da camada e da tarefa; a avaliação
    em STS verifica essa adequação diretamente.
  \end{frame}

  \begin{frame}{Bi-Encoder e Cross-Encoder: Onde Está o Custo?}
    \begin{tabular}{p{0.27\linewidth}p{0.29\linewidth}p{0.32\linewidth}}
      \toprule
      \textbf{Cenário} & \textbf{Bi-encoder} & \textbf{Cross-encoder}\\
      \midrule

      \bottomrule
    \end{tabular}
    \begin{block}{Busca exata e aproximada}
      Comparar a consulta com todos os vetores custa $O(nd)$.
      Um índice aproximado, como HNSW, troca parte da exatidão por menor latência.
      O custo depende do índice, dos dados e dos parâmetros de busca.
    \end{block}
    Reranking limita o cross-encoder a $k$ candidatos, em vez de todo o corpus.
  \end{frame}

  % SECTION 6: CONTRASTIVE LEARNING
  %======================================================================================
  \section{Aprendizado Contrastivo}

  %--- Slide: intuition ---
  \begin{frame}{Intuição: Puxar os Similares, Empurrar os Diferentes}
    \begin{columns}
      \begin{column}{0.5\textwidth}
        O \textbf{aprendizado contrastivo} treina um \textit{encoder} $f_\theta$ tal que:
        \[
          \cos(f_\theta(x), f_\theta(x^+)) \gg \cos(f_\theta(x), f_\theta(x^-))
        \]
        \begin{itemize}
          \item $x$: \textit{anchor}
          \item $x^+$: positivo (semanticamente similar)
          \item $x^-$: negativo (diferente)
        \end{itemize}
        \vspace{0.3em}
        A supervisão pode vir de anotações ou de uma regra para construir \textbf{pares positivos}.
      \end{column}
      \begin{column}{0.48\textwidth}
        \centering
        \resizebox{0.72\linewidth}{!}{%
        \begin{tikzpicture}
          \draw[very thick, gray] (0,0) circle (2.2);
          \node[circle, fill=DarkBlue, inner sep=2pt, label={[font=\scriptsize, DarkBlue]right:$x$}] (a) at (0.3,0.4) {};
          \node[circle, fill=DarkGreen, inner sep=2pt, label={[font=\scriptsize, DarkGreen]above:$x^+$}] (p) at (0.8,0.9) {};
          \node[circle, fill=DarkRed, inner sep=2pt, label={[font=\scriptsize, DarkRed]below:$x^-_1$}] (n1) at (-1.4,-0.9) {};
          \node[circle, fill=DarkRed, inner sep=2pt, label={[font=\scriptsize, DarkRed]left:$x^-_2$}] (n2) at (-1.6,1.2) {};
          \node[circle, fill=DarkRed, inner sep=2pt, label={[font=\scriptsize, DarkRed]right:$x^-_3$}] (n3) at (1.5,-1.3) {};

          \draw[->, thick, DarkGreen] (p) -- ($(p)!0.65!(a)$);
          \draw[->, thick, DarkRed] (a) -- (n1);
          \draw[->, thick, DarkRed] (a) -- (n2);
          \draw[->, thick, DarkRed] (a) -- (n3);
        \end{tikzpicture}}

        \vspace{0.3em}
        {\scriptsize Deslocamentos esquemáticos na atualização: positivo se aproxima do anchor; negativos se afastam. O anchor é mantido fixo na ilustração.}
      \end{column}
    \end{columns}
  \end{frame}

  %--- Slide: where positive pairs come from ---
  \begin{frame}{De Onde Vêm os Pares Positivos?}
    \begin{columns}[T]
      \begin{column}{0.5\textwidth}
        \begin{block}{Pares \textit{naturais} (supervisão fraca)}
          \begin{itemize}
            \item Pergunta $\leftrightarrow$ resposta (StackExchange, Quora)
            \item Título $\leftrightarrow$ corpo (notícias, artigos)
            \item Parágrafos adjacentes
            \item Pares \textit{query}$\leftrightarrow$\textit{click} de \textit{logs}
            \item Traduções (mesma frase em línguas diferentes)
          \end{itemize}
        \end{block}
      \end{column}
      \begin{column}{0.5\textwidth}
        \begin{block}{Pares \textit{anotados} (supervisão forte)}
          \begin{itemize}
            \item NLI (SNLI, MultiNLI): \textit{entailment} $\to$ positivo, \textit{contradiction} $\to$ negativo
            \item Duplicatas do Quora
            \item MS MARCO: \textit{query}$\to$passagem relevante (relevância humana)
            \item Paráfrases geradas por humanos
          \end{itemize}
        \end{block}
      \end{column}
    \end{columns}
    \vspace{0.4em}
    \begin{alertblock}{Escala importa}
      Modelos modernos (e5, bge) treinam com \textbf{centenas de milhões} de pares positivos coletados da web com supervisão fraca, depois fazem \textit{fine-tuning} em dados anotados.
    \end{alertblock}
  \end{frame}

  %--- Slide: triplet loss ---
  \begin{frame}{\textit{Triplet Loss}}
    Usada no \textbf{FaceNet} \footfullcite{schroff2015facenet} para reconhecimento facial.
    \vspace{0.3em}
    \begin{block}{Formulação}
      \[
        \mathcal{L}_{\text{triplet}} = \max\!\left( 0,\; \| f(x) - f(x^+) \|_2^2 - \| f(x) - f(x^-) \|_2^2 + \alpha \right)
      \]
      onde $\alpha > 0$ é a \textbf{margem em distâncias quadráticas}.
    \end{block}
    \vspace{0.3em}
    \begin{itemize}
      \item A perda é zero quando a distância quadrática ao negativo supera a do positivo em pelo menos $\alpha$
      \item Representações iguais têm perda $\alpha>0$, mas isso não garante que a otimização escape desse ponto
      \item Funciona, mas precisa de \textbf{triplos} $(x, x^+, x^-)$ construídos explicitamente
      \item Limitação: \textbf{um único negativo por \textit{anchor}} $\to$ pouco sinal por \textit{step}
    \end{itemize}
    \vspace{0.3em}
    A continuação natural é usar \textbf{muitos} negativos de uma vez...
  \end{frame}

  %--- Slide: InfoNCE ---
  \begin{frame}{\textit{InfoNCE} / \textit{Multiple Negatives Ranking Loss}\footfullcite{oord2018infonce}}
    
    \begin{block}{Formulação}
      Para um \textit{batch} de $N$ pares $(x_i, x_i^+)$ com os positivos dos outros pares como candidatos negativos:
      \[
        \mathcal{L}_{\text{NCE}} = -\frac{1}{N} \sum_{i=1}^{N} \log \frac{\exp(\text{sim}(f(x_i), f(x_i^+)) / \tau)}{\sum_{j=1}^{N} \exp(\text{sim}(f(x_i), f(x_j^+)) / \tau)}
      \]
    \end{block}
    \vspace{0.3em}
    \begin{itemize}
      \item $\text{sim}$: produto escalar ou cosseno
      \item $\tau$: \textbf{temperatura} (tipicamente $\tau = 0.05$ a $0.1$)
      \item \textbf{In-batch negatives}: cada $x_j^+$ com $j \neq i$ atua como negativo para $x_i$
      \item A forma usada em sentence embeddings\cite{henderson2017mnrl} equivale a classificar entre $N$ candidatos: ``qual dos $N$ candidatos é o verdadeiro par?''
    \end{itemize}
  \end{frame}

    Exemplo ilustrativo para $q_1$, com $d_1$ como positivo e $\tau=0{,}1$:
    \begin{center}
      \begin{tabular}{lrrrr}
        \toprule
        Candidato & $d_1$ & $d_2$ & $d_3$ & $d_4$\\
        \midrule
        Cosseno & 0,8 & 0,6 & 0,2 & 0,0\\
        Logit $S_{1j}=\cos/\tau$ & 8 & 6 & 2 & 0\\
        Softmax & 0,879 & 0,119 & 0,002 & 0,0003\\
        \bottomrule
      \end{tabular}

      p(d_1\mid q_1)=\frac{1}{1+e^{-2}+e^{-6}+e^{-8}}\approx0{,}879,

      \item $d_2$ recebe quase toda a probabilidade atribuída aos negativos.
      \item A perda do lote é a média das perdas das $N$ linhas.
    \end{itemize}

  \begin{frame}{Negativos Aleatórios Saturam}
    \begin{columns}
      \begin{column}{0.55\textwidth}
        No início do treino, negativos aleatórios são informativos: quase todos dão gradiente.

        \vspace{0.3em}

          \item A maioria dos negativos aleatórios é \textbf{trivialmente diferente} do \textit{anchor}
          \item Já estão longe no espaço
          \item Gradiente próximo de zero $\to$ \textbf{sem sinal de aprendizado}
          \item O modelo ``atinge um platô''
        \end{itemize}
        \vspace{0.3em}
        \textcolor{DarkBlue}{\textbf{Precisamos de negativos que o modelo \textit{ainda confunda com positivos}.}}

  %--- Slide: hard negative mining ---
  \begin{frame}{\textit{Hard Negative Mining}}
    \textbf{Ideia}: buscar ativamente negativos que estão \textbf{próximos} do \textit{anchor} no espaço atual.

    \vspace{0.4em}
    \begin{columns}[T]
      \begin{column}{0.5\textwidth}
        \begin{block}{Estratégia 1: BM25 / léxica}
          \begin{itemize}
            \item Para cada \textit{query}, pegue documentos \textbf{não relevantes} mas com \textbf{alta sobreposição lexical}
            \item Rápido e sem modelo
            \item Usado em DPR \footfullcite{karpukhin2020dpr}
          \end{itemize}
        \end{block}
      \end{column}
      \begin{column}{0.5\textwidth}
        \begin{block}{Estratégia 2: auto-mineração}
          \begin{itemize}
            \item Usa o \textbf{\textit{checkpoint atual}} para codificar o corpus
            \item Seleciona os \textit{top-k} vizinhos mais próximos (excluindo o positivo)
            \item Atualiza os negativos \textbf{periodicamente}
            \item Usado em ANCE \footfullcite{xiong2021ance}
          \end{itemize}
        \end{block}
      \end{column}
    \end{columns}
    \vspace{0.3em}
    A dificuldade depende da consulta e do modelo. Negativos léxicos e
    densos podem ser complementares; candidatos próximos exigem verificação de relevância.
  \end{frame}

  %--- Slide: false negatives ---
  \begin{frame}{A Armadilha dos \textit{False Negatives}}
    \begin{columns}
      \begin{column}{0.55\textwidth}
        Nem todo ``negativo próximo'' é realmente negativo.
        \vspace{0.3em}
        \begin{exampleblock}{Exemplo}
          \textit{Query}: ``Qual a capital do Brasil?''\\
          \textit{Positivo rotulado}: ``Brasília é a capital.''\\
          \textit{``Negativo'' minerado}: ``A cidade de Brasília foi inaugurada em 1960 como capital.''
        \end{exampleblock}
        \vspace{0.3em}
        O segundo parágrafo é \textbf{tão relevante} quanto o positivo, mas não foi anotado. Treiná-lo como negativo ensina o modelo errado.
      \end{column}
      \begin{column}{0.42\textwidth}
        \begin{alertblock}{Mitigações}
          \begin{itemize}
            \item Filtrar negativos com \textit{score} muito alto (pode ser falso negativo)
            \item Usar um \textbf{cross-encoder} para reavaliar
            \item \textit{Consistency filtering} entre modelos
            \item Deduplicar lotes e tratar múltiplos positivos: lotes maiores também aumentam as oportunidades de falsos negativos
          \end{itemize}
        \end{alertblock}
      \end{column}
    \end{columns}
  \end{frame}

  %--- Slide: practical notes ---
  \begin{frame}{Temperatura e Tamanho de Lote}
    \begin{columns}[T]
      \begin{column}{0.48\textwidth}
        \begin{block}{Temperatura $\tau$}
          \begin{itemize}\small
            \item Menor $\tau$: softmax mais concentrado nos candidatos de maior score.
            \item Maior $\tau$: distribuição mais plana.
            \item Ajustar na validação, junto com a similaridade e os dados de treinamento.
          \end{itemize}
        \end{block}
      \end{column}
      \begin{column}{0.48\textwidth}
        \begin{block}{Número de pares $N$}
          \begin{itemize}\small
            \item Mais candidatos por anchor, com maior custo e risco de falsos negativos.
            \item O ganho depende da diversidade dos pares e da filtragem.
            \item GradCache permite ampliar o lote contrastivo sem reter todas as ativações simultaneamente.
          \end{itemize}
        \end{block}
      \end{column}
    \end{columns}
    \medskip
    A temperatura pode ser fixa ou aprendida. CLIP exemplifica o uso de
    temperatura aprendida no contraste entre imagens e textos.
  \end{frame}

  \section{SBERT: \textit{Sentence-BERT}}

  %--- Slide: SBERT intro ---
  \begin{frame}{SBERT: Adaptar BERT para Comparar Frases\footfullcite{reimers2019sbert}}
    \begin{block}{Problema}
      O cross-encoder avalia cada par conjuntamente. Codificar textos
      separadamente permite reutilizar vetores, mas o BERT sem adaptação
      tem desempenho limitado em STS por cosseno.
    \end{block}
    \begin{block}{Proposta de 2019}
      Adaptar BERT em redes siamesas ou triplet, com pooling e objetivos
      supervisionados de classificação, regressão ou comparação de distâncias.
    \end{block}
    \begin{itemize}
      \item InferSent e Universal Sentence Encoder já produziam vetores de frases.
      \item SBERT aproveita os pesos pré-treinados de BERT e RoBERTa.
      \item A biblioteca \texttt{sentence-transformers} reúne essa abordagem e extensões.
    \end{itemize}
  \end{frame}

  %--- Slide: SBERT architecture ---

  %--- Slide: SBERT objectives ---
  \begin{frame}{SBERT Original: Classificação e Regressão}
    \begin{block}{Classificação NLI}
      $\mathbf z=W^\top[\mathbf u;\mathbf v;|\mathbf u-\mathbf v|]$, com
      $W\in\mathbb R^{3d\times3}$. Softmax e entropia cruzada para
      entailment, contradiction e neutral.
    \end{block}
    \begin{block}{Regressão STS}
      O score humano $y\in[0,5]$ é normalizado: $s=y/5\in[0,1]$.
      \[
        \mathcal L_{\mathrm{STS}}=(\cos(\mathbf u,\mathbf v)-s)^2.
      \]
    \end{block}
    NLI usa cerca de um milhão de pares de SNLI e MultiNLI.
    O artigo também avalia adaptação posterior em STS.
  \end{frame}

  \begin{frame}{SBERT Original: Objetivo Triplet}
    Uma tripla contém anchor $a$, positivo $p$ e negativo $n$:
    \[
      \mathcal L=\max\left(0,
        \|\mathbf u_a-\mathbf u_p\|_2-
        \|\mathbf u_a-\mathbf u_n\|_2+\alpha\right).
    \]
    \begin{itemize}
      \item O artigo usa distância euclidiana e margem $\alpha=1$.
      \item A forma do FaceNet apresentada antes usa distâncias quadráticas.
      \item São variantes do objetivo; a margem deve acompanhar a distância escolhida.
    \end{itemize}
    \begin{block}{Treino e inferência}
      A cabeça de classificação ou o objetivo triplet supervisiona o treino.
      Na inferência, cada frase é codificada separadamente e os vetores são comparados.
    \end{block}
  \end{frame}

  \begin{frame}{Depois do SBERT: Objetivos Contrastivos}
    \begin{itemize}
      \item \textbf{SBERT (2019)}: pooling sobre BERT e adaptação
      supervisionada para produzir vetores de frases\cite{reimers2019sbert}.
      \item \textbf{SimCSE (2021)}: duas passagens com dropout constroem
      positivos; na versão supervisionada, NLI fornece positivos e
      negativos por contradição\cite{gao2021simcse}.
      \item \textbf{E5}: pré-treino contrastivo com pares coletados e
      filtrados, seguido de adaptação supervisionada\cite{wang2022e5}.
      \item \textbf{BGE}: família de modelos e recursos para embeddings
      e recuperação\cite{xiao2024bge}.
    \end{itemize}
    \begin{block}{O que varia entre modelos}
      Arquitetura, pooling, idioma, comprimento máximo, instruções,
      composição dos pares e estratégias para obter negativos.
    \end{block}
  \end{frame}

  %--- Slide: takeaway SBERT family ---
  \begin{frame}{Construir um Bi-Encoder: Decisões de Treinamento}
    \begin{enumerate}
      \item Escolher um backbone pré-treinado e seu tokenizer.
      \item Definir pooling, normalização e função de similaridade.
      \item Construir positivos conforme a tarefa: paráfrases, traduções ou consulta/documento.
      \item Escolher a perda e filtrar negativos, incluindo falsos negativos no lote.
      \item Validar no idioma, domínio e orçamento de latência da aplicação.
    \end{enumerate}
    \begin{block}{Arquitetura e objetivo}
      O backbone produz estados contextuais. O treinamento e o pooling
      determinam como o vetor agregado serve à classificação, à similaridade
      ou à recuperação.
    \end{block}
  \end{frame}

  \section{Tarefas e Avaliação de \textit{Sentence Embeddings}}

  %--- Slide: retrieval ---
  \begin{frame}{Tarefa: Retrieval (Busca Semântica)}
    Dada uma consulta, recuperar documentos relevantes do corpus.
    \begin{block}{Offline: preparar o índice}
      Codificar os documentos, respeitando o comprimento do modelo,
      e armazenar os vetores em um índice de busca.
    \end{block}
    \begin{block}{Online: atender à consulta}
      Codificar a consulta, recuperar top-$k$ candidatos e, opcionalmente,
      reordená-los com um cross-encoder.
    \end{block}
    \begin{itemize}
      \item Bibliotecas como FAISS oferecem busca exata e aproximada;
      HNSW é uma estrutura para busca aproximada.
      \item Em RAG, as passagens recuperadas compõem o contexto de um gerador.
      \item O objetivo é fornecer evidências úteis; avaliar recuperação e geração separadamente.
    \end{itemize}
  \end{frame}

  \begin{frame}{Retrieval: Avaliar os Candidatos}
    Consulta ilustrativa: ``Como trocar a senha?''
    \begin{center}\small
      \begin{tabular}{clc}
        \toprule
        Posição & Documento recuperado & Relevante?\\
        \midrule
        1 & Procedimento para alterar a senha & 1\\
        2 & Como alterar o endereço de e-mail & 0\\
        3 & Recuperação e redefinição da senha & 1\\
        4 & Configuração de notificações & 0\\
        5 & Política de privacidade & 0\\
        \bottomrule
      \end{tabular}
    \end{center}
    Suponha que existam exatamente dois documentos relevantes no corpus.
    \[
      \mathrm{Recall@2}=\frac{1}{2},\qquad \mathrm{Recall@3}=\frac{2}{2}=1.
    \]
    \begin{itemize}
      \item Recall@k mede a fração dos relevantes recuperada nos primeiros $k$.
      \item O rótulo depende de a passagem responder à consulta, além da sobreposição de palavras.
    \end{itemize}
  \end{frame}

  \begin{frame}{Retrieval: O que É DCG?}
    \textbf{DCG} significa \textit{Discounted Cumulative Gain},
    ou \textbf{ganho acumulado com desconto}.
    \begin{itemize}
      \item \textbf{Ganho}: a relevância de cada resultado para a consulta.
      \item \textbf{Desconto}: resultados em posições posteriores contribuem menos.
      \item \textbf{Acumulado}: somamos as contribuições até a posição $k$.
    \end{itemize}
    \[
      \mathrm{DCG@}k=\sum_{r=1}^{k}
      \underbrace{g(\mathrm{rel}_r)}_{\text{ganho}}
      \underbrace{\frac{1}{\log_2(r+1)}}_{\text{desconto pela posição}}.
    \]
    \small $r$ começa em 1; $\mathrm{rel}_r$ é o rótulo de relevância do
    documento na posição $r$. Usaremos relevância binária e $g(\mathrm{rel})=\mathrm{rel}$.
    \par\medskip
    Para relevância graduada, também se usa $g(\mathrm{rel})=2^{\mathrm{rel}}-1$.
    A convenção deve ser a mesma em toda a avaliação.
  \end{frame}

  \begin{frame}{Retrieval: Calcular DCG Passo a Passo}
    No exemplo da senha, os três primeiros rótulos são $(1,0,1)$:
    \begin{center}
      \begin{tabular}{ccccc}
        \toprule
        Posição $r$ & Ganho & Desconto & Contribuição & Acumulado\\
        \midrule
        1 & 1 & $1/\log_2 2=1$ & 1 & 1\\[0.3em]
        2 & 0 & $1/\log_2 3\approx0{,}631$ & 0 & 1\\[0.3em]
        3 & 1 & $1/\log_2 4=0{,}5$ & 0,5 & 1,5\\
        \bottomrule
      \end{tabular}
    \end{center}
    \[
      \mathrm{DCG@3}=1+0+0{,}5=1{,}5.
    \]
    \begin{itemize}
      \item Um resultado irrelevante contribui com zero.
      \item O mesmo ganho vale 1 na primeira posição e 0,5 na terceira.
      \item Sem desconto, tanto $(1,0,1)$ quanto $(1,1,0)$ somariam 2.
      O desconto permite distinguir onde os relevantes aparecem.
    \end{itemize}
  \end{frame}

  \begin{frame}{Retrieval: Normalizar DCG com a Ordem Ideal}
    \textbf{IDCG} é o DCG da ordem ideal, com os ganhos em ordem decrescente.
    Para a mesma consulta, o ranking ideal começa por $(1,1,0)$:
    \[
      \mathrm{IDCG@3}=1+\frac{1}{\log_2 3}+0\approx1{,}631.
    \]
    \textbf{nDCG} divide o DCG observado pelo ideal:
    \[
      \mathrm{nDCG@3}=\frac{\mathrm{DCG@3}}{\mathrm{IDCG@3}}
        =\frac{1{,}5}{1{,}631}\approx0{,}920.
    \]
    \begin{itemize}
      \item Com relevâncias não negativas e IDCG positivo, nDCG varia de 0 a 1.
      \item O valor 1 corresponde ao ganho ideal até a posição $k$.
      \item Recall@3 é 1 neste exemplo; nDCG@3 mostra que a ordem pode melhorar.
      \item Normalizar permite agregar consultas com quantidades de relevantes distintas.
    \end{itemize}
    \small Se IDCG for zero, a consulta exige uma convenção do protocolo,
    como atribuir nDCG zero ou excluí-la da média.
  \end{frame}

  %--- Slide: clustering ---
  \begin{frame}{Tarefa: \textit{Clustering}}
    \textbf{Objetivo}: agrupar documentos semanticamente similares \textbf{sem rótulos}.
    \textbf{Pipeline}:
    \begin{enumerate}
      \item Codifica os documentos
      \item Aplica $k$-means, \textit{agglomerative clustering} ou HDBSCAN
      \item Extrai tópicos / rótulos por \textit{cluster}
    \end{enumerate}
    \vspace{0.3em}
    \textbf{Aplicações}:
    \begin{itemize}
      \item Descoberta de tópicos em \textit{feedback} de usuários
      \item Organização de \textit{base} de conhecimento
      \item Detecção de spam / clones
      \item Visualização exploratória (com UMAP/t-SNE)
    \end{itemize}
  \end{frame}

  %--- Slide: classification ---
  \begin{frame}{Tarefa: Classificação via \textit{Linear Probe}}
    \textbf{Ideia}: congelar o \textit{encoder}, treinar apenas um classificador linear sobre os \textit{embeddings}.
    \vspace{0.4em}
        \textbf{Receita}:
        \begin{enumerate}
          \item Codifica $n$ exemplos rotulados em vetores
          \item Treina \textit{logistic regression} ou SVM
          \item Avalia em \textit{test set}
        \end{enumerate}
        \vspace{0.3em}
        \begin{block}{Benchmark de qualidade de embeddings}
          O \textit{linear probe accuracy} é uma medida padrão de \textbf{quão útil} um \textit{embedding} é para tarefas \textit{downstream}:
          \begin{itemize}
            \item Alta acurácia $\to$ informação acessível por uma fronteira linear
            \item Baixa acurácia $\to$ limitação da representação ou do classificador para essa tarefa
          \end{itemize}
        \end{block}
  \end{frame}

  %--- Slide: STS ---
  \begin{frame}{Tarefa: \textit{Semantic Textual Similarity} (STS)\footfullcite{cer2017sts}}
    
    Prever um \textit{score} de similaridade em $[0, 5]$ para um par de frases.
    \vspace{0.4em}
    \begin{exampleblock}{Exemplos ilustrativos de STS}
      \small
      \begin{tabular}{p{5.3cm}p{5.3cm}c}
        \toprule
        \textbf{Frase A} & \textbf{Frase B} & \textbf{Score} \\
        \midrule
        Um homem está tocando piano. & Um cara toca teclado. & $4.2$ \\
        Um gato preto dorme. & Um cão branco corre. & $0.8$ \\
        A menina está comendo. & A garota está jantando. & $3.9$ \\
        \bottomrule
      \end{tabular}
    \end{exampleblock}
    \vspace{0.4em}
    \begin{itemize}
      \item \textbf{Avaliação}: \textbf{correlação de Spearman} entre cosseno predito e \textit{score} humano
      \item Mede quão bem o espaço de \textit{embeddings} reflete similaridade semântica
      \item Não requer \textit{fine-tuning}: avalia diretamente a similaridade cosseno
    \end{itemize}
  \end{frame}

  %--- Slide: Spearman vs Pearson ---
  \begin{frame}{Pearson e Spearman: O que Cada Um Mede?}
    \begin{columns}[T]
      \begin{column}{0.48\textwidth}
        \textbf{Pearson}: associação linear.
        \[
          r=\frac{\operatorname{cov}(x,y)}{\sigma_x\sigma_y}.
        \]
        Pode ser calculado mesmo quando a relação é não linear.
        Uma relação afim crescente perfeita produz $r=1$.
      \end{column}
      \begin{column}{0.48\textwidth}
        \textbf{Spearman}: associação entre rankings.
        \[
          \rho=\operatorname{Pearson}(\operatorname{rank}(x),\operatorname{rank}(y)).
        \]
        Avalia concordância de ordem, com postos médios para empates.
        Uma relação estritamente crescente preserva os rankings.
      \end{column}
    \end{columns}
    \begin{block}{Escolha da métrica}
      Ambos são invariantes a mudanças afins positivas de escala.
      Spearman é útil quando queremos ordenar pares por similaridade.
      Reportar a métrica definida pelo benchmark e pelo protocolo de avaliação.
    \end{block}
  \end{frame}
  \begin{frame}{Spearman vs Pearson: Exemplo}

  \begin{itemize}
      \item Suponha que o modelo preveja \textbf{o quadrado} do \textit{score} humano ($\hat{y} = y^2$)
      \item \textbf{Pearson} $\approx 0.98$ (penaliza a não-linearidade).
      \item \textbf{Spearman} $= 1.00$ (ordem idêntica) 
      \item Com curvatura maior, Pearson degrada ainda mais, apesar do ranking \textbf{continuar perfeito}.
      
  \end{itemize}
      
      \centering
      \begin{tabular}{lcrrrrr}
        \toprule
        Humano & $y$ & $1$ & $2$ & $3$ & $4$ & $5$ \\
        Modelo & $\hat{y}$ & $1$ & $4$ & $9$ & $16$ & $25$ \\
        \bottomrule
      \end{tabular}
      
            
\end{frame}

  %--- Slide: NLI ---
  \begin{frame}{Tarefa: \textit{Natural Language Inference} (NLI)\footfullcite{bowman2015snli}}
    
    \textbf{Tarefa}: dada uma \textit{premissa} $P$ e uma \textit{hipótese} $H$, classificar em:
    \begin{itemize}
      \item \textbf{\textit{Entailment}}: $P$ implica $H$
      \item \textbf{\textit{Contradiction}}: $P$ e $H$ se contradizem
      \item \textbf{\textit{Neutral}}: nem uma coisa nem outra
    \end{itemize}
    \centering
    \begin{tabular}{llc}
      \toprule
      \textbf{Premise} & \textbf{Hypothesis} & \textbf{Rótulo} \\
      \midrule
      Um homem joga futebol. & Uma pessoa pratica esporte. & \textcolor{DarkGreen}{entailment} \\
      Um homem joga futebol. & Ele está dormindo. & \textcolor{DarkRed}{contradiction} \\
      Um homem joga futebol. & Ele é profissional. & neutral \\
      \bottomrule
    \end{tabular}
    \flushleft
    \vspace{0.3em}
    \textbf{Dois usos no treinamento}: SBERT original aprende a classificar
    os três rótulos. SimCSE supervisionado usa entailment como positivo
    e contradiction como negativo difícil\cite{gao2021simcse}.
    Avaliar separadamente em dados não usados no treino.
  \end{frame}

  \begin{frame}{Resumo}
    \begin{enumerate}
      \item Encoders bidirecionais produzem estados de tokens dependentes do contexto.
      \item BERT combina MLM e NSP no pré-treino; fine-tuning adapta o backbone e a cabeça à tarefa.
      \item Embeddings de sentença exigem pooling e um objetivo adequado ao uso do vetor.
      \item SBERT original usa redes siamesas ou triplet com supervisão NLI, STS ou triplas.
      \item Receitas posteriores exploram pares contrastivos, negativos no lote e mineração filtrada.
      \item Avaliar similaridade, relevância, custo e generalização exige métricas e dados específicos.
    \end{enumerate}
  \end{frame}
