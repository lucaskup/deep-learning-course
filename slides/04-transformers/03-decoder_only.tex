  \section{Do Encoder-Decoder ao Decoder-Only}

  \begin{frame}{O que Encoder e Decoder fazem?}
    \begin{columns}[T]
      \begin{column}{0.48\textwidth}
        \begin{block}{Encoder: compreensão}
          \begin{itemize}
            \item Processa \textbf{toda a entrada} de uma vez
            \item Atenção bidirecional
            \item Produz vetor de contexto rico
            \item Exemplos: BERT, RoBERTa
          \end{itemize}
        \end{block}
      \end{column}
      \begin{column}{0.48\textwidth}
        \begin{block}{Decoder: geração}
          \begin{itemize}
            \item Gera \textbf{um token por vez}
            \item Atenção causal (só vê o passado)
            \item Recebe contexto via cross-attention
            \item Exemplos: GPT, T5-decoder
          \end{itemize}
        \end{block}
      \end{column}
    \end{columns}
    \vspace{0.5em}
    \begin{center}
      \textcolor{DarkRed}{\textbf{Para LLMs, o encoder é desnecessário.}}\\
      \vspace{0.2em}
      O decoder processa o \textit{prompt} como parte da sequência.
    \end{center}
  \end{frame}

  \begin{frame}{Máscara Causal vs.\ Atenção Bidirecional}
    \begin{columns}[T]
      \begin{column}{0.50\textwidth}
        \textbf{Bidirecional (BERT/Encoder)}:\\
        cada posição vê \textit{toda} a sequência.
        \vspace{0.3em}

        \resizebox{0.85\linewidth}{!}{%
        \begin{tabular}{c|cccc}
          & $t_1$ & $t_2$ & $t_3$ & $t_4$ \\\hline
          $t_1$ & \cellcolor{CellGreen}$\checkmark$ & \cellcolor{CellGreen}$\checkmark$ & \cellcolor{CellGreen}$\checkmark$ & \cellcolor{CellGreen}$\checkmark$ \\
          $t_2$ & \cellcolor{CellGreen}$\checkmark$ & \cellcolor{CellGreen}$\checkmark$ & \cellcolor{CellGreen}$\checkmark$ & \cellcolor{CellGreen}$\checkmark$ \\
          $t_3$ & \cellcolor{CellGreen}$\checkmark$ & \cellcolor{CellGreen}$\checkmark$ & \cellcolor{CellGreen}$\checkmark$ & \cellcolor{CellGreen}$\checkmark$ \\
          $t_4$ & \cellcolor{CellGreen}$\checkmark$ & \cellcolor{CellGreen}$\checkmark$ & \cellcolor{CellGreen}$\checkmark$ & \cellcolor{CellGreen}$\checkmark$ \\
        \end{tabular}}
      \end{column}
      \begin{column}{0.46\textwidth}
        \textbf{Causal (GPT/Decoder-Only)}:\\
        posição $i$ só vê $j \le i$.
        \vspace{0.3em}

        \resizebox{0.85\linewidth}{!}{%
        \begin{tabular}{c|cccc}
          & $t_1$ & $t_2$ & $t_3$ & $t_4$ \\\hline
          $t_1$ & \cellcolor{CellGreen}$\checkmark$ & \cellcolor{CellRed}$\times$ & \cellcolor{CellRed}$\times$ & \cellcolor{CellRed}$\times$ \\
          $t_2$ & \cellcolor{CellGreen}$\checkmark$ & \cellcolor{CellGreen}$\checkmark$ & \cellcolor{CellRed}$\times$ & \cellcolor{CellRed}$\times$ \\
          $t_3$ & \cellcolor{CellGreen}$\checkmark$ & \cellcolor{CellGreen}$\checkmark$ & \cellcolor{CellGreen}$\checkmark$ & \cellcolor{CellRed}$\times$ \\
          $t_4$ & \cellcolor{CellGreen}$\checkmark$ & \cellcolor{CellGreen}$\checkmark$ & \cellcolor{CellGreen}$\checkmark$ & \cellcolor{CellGreen}$\checkmark$ \\
        \end{tabular}}
        \vspace{0.3em}
        Garante que $t_4$ não ``veja o futuro'' durante o treino.
      \end{column}
    \end{columns}
  \end{frame}

  \begin{frame}{Máscara Causal: Exemplo Numérico}
    Sequência de 3 tokens; scores $\mathbf{S} = \dfrac{\mathbf{Q}\mathbf{K}^\top}{\sqrt{d_k}}$, máscara $\mathbf{M}$ somada antes do softmax (por linha):
    \vspace{0.4em}
    \begin{center}
    \resizebox{0.94\linewidth}{!}{%
      $\underbrace{\begin{bmatrix} 2 & 1 & 3 \\ 0 & 2 & 1 \\ 1 & 0 & 2 \end{bmatrix}}_{\mathbf{S}}
      \;+\;
      \underbrace{\begin{bmatrix} 0 & \textcolor{DarkRed}{-\infty} & \textcolor{DarkRed}{-\infty} \\ 0 & 0 & \textcolor{DarkRed}{-\infty} \\ 0 & 0 & 0 \end{bmatrix}}_{\mathbf{M}}
      \;=\;
      \begin{bmatrix} 2 & \textcolor{DarkRed}{-\infty} & \textcolor{DarkRed}{-\infty} \\ 0 & 2 & \textcolor{DarkRed}{-\infty} \\ 1 & 0 & 2 \end{bmatrix}
      \;\xrightarrow{\;\text{softmax}\;}\;
      \begin{bmatrix} 1.00 & \textcolor{gray}{0} & \textcolor{gray}{0} \\ 0.12 & 0.88 & \textcolor{gray}{0} \\ 0.24 & 0.09 & 0.67 \end{bmatrix}$}
    \end{center}
    \vspace{0.5em}
    \begin{columns}[T]
      \begin{column}{0.50\textwidth}
        \begin{itemize}
          \item $e^{-\infty} = 0$: os pesos acima da diagonal \textbf{zeram} após o softmax
          \item Linha 2: $\dfrac{e^{0}}{e^{0}+e^{2}} = 0.12$, \quad $\dfrac{e^{2}}{e^{0}+e^{2}} = 0.88$
        \end{itemize}
      \end{column}
      \begin{column}{0.46\textwidth}
        \begin{exampleblock}{Leitura}
          O token 1 só atende a si mesmo; o token 3 distribui atenção pelos três tokens do passado (inclusive ele próprio).
        \end{exampleblock}
      \end{column}
    \end{columns}
  \end{frame}

  \begin{frame}{Do Encoder-Decoder para o Decoder-Only}
    \begin{columns}[T]
      \begin{column}{0.50\textwidth}
        \textbf{Ideia central}: treine apenas o decoder.\\[0.5em]
        O modelo recebe o \textbf{prompt} e continua gerando autorregressivamente.\\[0.5em]
        \begin{itemize}
          \item Sem cross-attention
          \item Sem encoder separado
          \item Arquitetura mais simples
        \end{itemize}
      \end{column}
      \begin{column}{0.46\textwidth}
        \centering
        \begin{tikzpicture}[
          block/.style={draw, rounded corners, minimum width=4.5cm, minimum height=0.50cm,
                        fill=CellYellow, align=center, font=\footnotesize},
          norm/.style={draw, rounded corners, minimum width=4.5cm, minimum height=0.50cm,
                       fill=CellBlue, align=center, font=\footnotesize},
          add/.style={draw, circle, minimum size=0.32cm, fill=white,
                      inner sep=0pt, font=\footnotesize},
          arrow/.style={->, >=Stealth, thick},
          node distance=0.28cm
        ]
          \node[norm]  (n1)   {RMSNorm};
          \node[block, above=of n1]   (attn) {Masked Self-Attention};
          \node[add,   above=of attn] (add1) {$+$};
          \node[norm,  above=of add1] (n2)   {RMSNorm};
          \node[block, above=of n2]   (ff)   {Feed-Forward (MLP)};
          \node[add,   above=of ff]   (add2) {$+$};
          \node[block, above=of add2] (lm)   {LM Head (Linear)};
          \node[below=0.4cm of n1, font=\footnotesize] (inp) {tokens de entrada};

          % Main forward flow

      \end{column}
    \end{columns}
  \end{frame}

  \begin{frame}{LM Head: do Vetor aos Logits}
    \begin{columns}[T]
      \begin{column}{0.52\textwidth}
        Após os $N$ blocos, cada posição $t$ carrega um vetor $\mathbf{h}_t \in \mathbb{R}^{d_\text{model}}$.\\[0.4em]
        \begin{block}{Projeção no vocabulário}
          $\mathbf{z} = \mathbf{h}_t\, \mathbf{W}_\text{LM}, \qquad \mathbf{W}_\text{LM} \in \mathbb{R}^{d_\text{model} \times |V|}$
        \end{block}
        \vspace{0.3em}
        Cada logit é um \textbf{produto interno}, $z_v = \mathbf{h}_t \cdot \mathbf{w}_v$, onde $\mathbf{w}_v$ é a coluna de $\mathbf{W}_\text{LM}$ associada ao token $v$: mede o alinhamento entre o estado atual e cada token do vocabulário.
        \vspace{0.3em}
        \begin{itemize}
          \item \textbf{Weight tying}: muitos modelos reutilizam a matriz de embedding ($\mathbf{W}_\text{LM} = \mathbf{W}_\text{emb}^\top$)
        \end{itemize}
      \end{column}
      \begin{column}{0.44\textwidth}
        \footnotesize\textbf{Exemplo}: $d_\text{model}=3$, $|V|=5$, uma coluna por token:\\[0.5em]
        \resizebox{\linewidth}{!}{$
          \underbrace{\begin{bmatrix}1.0 & 0.5 & -0.5\end{bmatrix}}_{\mathbf{h}_t}
          \underbrace{\begin{bmatrix}
            2.0 & 1.5 & 0.8 & 0.3 & -0.3\\
            1.6 & 1.0 & 0.2 & 0.6 & 0.0\\
            -0.4 & -0.2 & -0.2 & 0.2 & 0.4
          \end{bmatrix}}_{\mathbf{W}_\text{LM}}
        $}\\[0.6em]
        \resizebox{0.88\linewidth}{!}{$
          =\;
          \underbrace{\begin{bmatrix}
            \overset{\scriptstyle\texttt{rolha}}{\mathbf{3.0}} &
            \overset{\scriptstyle\texttt{roupa}}{2.1} &
            \overset{\scriptstyle\texttt{rua}}{1.0} &
            \overset{\scriptstyle\texttt{rainha}}{0.5} &
            \overset{\scriptstyle\texttt{rei}}{-0.5}
          \end{bmatrix}}_{\mathbf{z}}
        $}
        \vspace{0.5em}
        \begin{exampleblock}{\footnotesize Cálculo de $z_{\texttt{rolha}}$ (primeira coluna)}
          \scriptsize $1.0 \times 2.0 + 0.5 \times 1.6 + (-0.5) \times (-0.4) = 2.0 + 0.8 + 0.2 = 3.0$
        \end{exampleblock}
      \end{column}
    \end{columns}
  \end{frame}

  \begin{frame}{LM Head: dos Logits à Distribuição}
    \begin{columns}[T]
      \begin{column}{0.52\textwidth}
        O softmax converte os logits em uma distribuição de probabilidade sobre o vocabulário:
        \begin{block}{Da pontuação à probabilidade}
          $P(v \mid x_{\le t}) = \text{softmax}(\mathbf{z})_v = \dfrac{e^{z_v}}{\sum_{u \in V} e^{z_u}}$
        \end{block}
        \vspace{0.3em}
        \begin{itemize}
          \item $\mathbf{z}$: \textbf{logits}, um score por token do vocabulário
          \item A exponenciação amplia as diferenças entre scores; a soma normaliza o total para 1
        \end{itemize}
      \end{column}
      \begin{column}{0.44\textwidth}
        \footnotesize\textbf{Exemplo}: \textit{``O rato roeu a \_\_\_''}, logits do slide anterior:\\[0.4em]
        \resizebox{0.9\linewidth}{!}{%
        \begin{tabular}{lrr}
          \toprule
          \textbf{Token} & \textbf{Logit $z_v$} & $P(v \mid x_{\le t})$ \\
          \midrule
          \texttt{rolha}  & 3.0    & \cellcolor{CellGreen}\textbf{0.60} \\
          \texttt{roupa}  & 2.1    & 0.25 \\
          \texttt{rua}    & 1.0    & 0.08 \\
          \texttt{rainha} & 0.5    & 0.05 \\
          \texttt{rei}    & $-0.5$ & 0.02 \\
          \bottomrule
        \end{tabular}}
        \vspace{0.4em}
        \begin{exampleblock}{\footnotesize Cálculo}
          \scriptsize $P(\texttt{rolha}) = \dfrac{e^{3.0}}{e^{3.0}+\cdots+e^{-0.5}} = \dfrac{20.1}{33.2} \approx 0.60$
        \end{exampleblock}
        \vspace{0.2em}
        {\scriptsize A \textbf{estratégia de decodificação} (seção adiante) decide qual token emitir a partir dessa distribuição.}
      \end{column}
    \end{columns}
  \end{frame}

  \begin{frame}{Treinamento: Todas as Posições em Paralelo}
    A máscara causal permite computar a perda de \textbf{todas as posições em um único passe}:\\[0.4em]
    \begin{center}
    \resizebox{0.82\linewidth}{!}{%
    \begin{tabular}{lccccc}
      \toprule
      \textbf{Posição} & 1 & 2 & 3 & 4 & 5 \\
      \midrule
      \textbf{Entrada} & \cellcolor{CellBlue}\texttt{<s>} & \cellcolor{CellBlue}\texttt{O} & \cellcolor{CellBlue}\texttt{rato} & \cellcolor{CellBlue}\texttt{roeu} & \cellcolor{CellBlue}\texttt{a} \\
      \textbf{Alvo}    & \cellcolor{CellGreen}\texttt{O} & \cellcolor{CellGreen}\texttt{rato} & \cellcolor{CellGreen}\texttt{roeu} & \cellcolor{CellGreen}\texttt{a} & \cellcolor{CellGreen}\texttt{rolha} \\
      \textbf{Perda}   & $-\log P(\texttt{O})$ & $-\log P(\texttt{rato})$ & $-\log P(\texttt{roeu})$ & $-\log P(\texttt{a})$ & $-\log P(\texttt{rolha})$ \\
      \bottomrule
    \end{tabular}}
    \end{center}
    \vspace{0.4em}
    \begin{columns}[T]
      \begin{column}{0.50\textwidth}
        \begin{itemize}
          \item Alvo = entrada deslocada uma posição (\textit{shift-by-one})
          \item Uma sequência de $n$ tokens fornece $n$ exemplos de treino simultâneos
          \item A máscara causal garante que a posição $t$ não ``veja'' o próprio alvo
        \end{itemize}
      \end{column}
      \begin{column}{0.46\textwidth}
        \begin{alertblock}{Por que o decoder-only venceu}
          Treino massivamente paralelo (ao contrário das RNNs, que processam token a token) com um objetivo simples, aplicável a qualquer texto bruto, sem anotação.
        \end{alertblock}
      \end{column}
    \end{columns}
  \end{frame}

  \section{A Família GPT e o Paradigma Decoder-Only}

  \begin{frame}{GPT-1: Pré-Treino + Fine-Tuning\footnotemark[1]}
    \begin{columns}[T]
      \begin{column}{0.52\textwidth}
        \begin{block}{Paradigma}
          Pré-treino generativo em texto bruto; depois, \textit{fine-tuning} supervisionado para cada tarefa.
        \end{block}
        \vspace{0.3em}
        \textbf{Objetivo de pré-treino}: modelagem de linguagem causal
        \[
          \mathcal{L} = -\sum_{t} \log P(x_t \mid x_1,\ldots,x_{t-1};\,\Theta)
        \]
      \end{column}
      \begin{column}{0.44\textwidth}
        \textbf{Arquitetura e treino}:
        \begin{itemize}
          \item 117M parâmetros
          \item 12 camadas, $d=768$, 12 cabeças
          \item Contexto: 512 tokens
          \item Treino: BooksCorpus ($\approx$1B tokens)
        \end{itemize}
        \vspace{0.3em}
        \begin{exampleblock}{Contribuição}
          Mostrou que pré-treino auto-supervisionado em texto melhora fortemente o fine-tuning.
        \end{exampleblock}
      \end{column}
    \end{columns}
    \footnotetext[1]{\fullcite{radford2018gpt1}}
  \end{frame}

  \begin{frame}{GPT-2: Multitarefa Zero-Shot, sem Fine-Tuning\footnotemark[1]}
    \begin{columns}[T]
      \begin{column}{0.52\textwidth}
        \begin{block}{Paradigma}
          Zero-shot: um LM suficientemente grande realiza tarefas \textbf{sem fine-tuning} e sem exemplos.
        \end{block}
        \vspace{0.3em}
        \textit{``Language models are unsupervised multitask learners''}\\[0.4em]
        A tarefa é induzida pelo próprio prompt: por exemplo, \texttt{TL;DR:} ao final de um texto induz sumarização.
      \end{column}
      \begin{column}{0.44\textwidth}
        \textbf{Arquitetura e treino}:
        \begin{itemize}
          \item 1.5B parâmetros (XL; versões de 117M a 1.5B)
          \item 48 camadas, $d=1600$, 25 cabeças; Pre-Norm
          \item Contexto: 1024 tokens
          \item Treino: WebText ($\approx$10B tokens de texto web filtrado)
        \end{itemize}
        \vspace{0.3em}
        \begin{exampleblock}{Contribuição}
          Um único LM realiza múltiplas tarefas zero-shot.
        \end{exampleblock}
      \end{column}
    \end{columns}
    \footnotetext[1]{\fullcite{radford2019gpt2}}
  \end{frame}

  \begin{frame}{GPT-3: Few-Shot In-Context Learning\footnotemark[1]}
    \begin{columns}[T]
      \begin{column}{0.52\textwidth}
        \begin{block}{Paradigma}
          Few-shot: exemplos fornecidos \textit{no contexto}, sem atualização de pesos.
        \end{block}
        \vspace{0.3em}
        \textbf{Modos de uso em contexto}:
        \begin{itemize}
          \item \textbf{Zero-shot}: só a instrução
          \item \textbf{One-shot}: 1 exemplo no prompt
          \item \textbf{Few-shot}: K exemplos no prompt
        \end{itemize}
      \end{column}
      \begin{column}{0.44\textwidth}
        \textbf{Arquitetura e treino}:
        \begin{itemize}
          \item 175B parâmetros
          \item 96 camadas, $d=12288$, 96 cabeças
          \item Contexto: 2048 tokens
          \item Treino: 300B tokens (Common Crawl filtrado, livros, Wikipedia)
        \end{itemize}
        \vspace{0.3em}
        \begin{exampleblock}{Contribuição}
          Capacidades emergentes com escala (aritmética, tradução, raciocínio); o fine-tuning por tarefa dá lugar à engenharia de \textit{prompts}.
        \end{exampleblock}
      \end{column}
    \end{columns}
    \footnotetext[1]{\fullcite{brown2020gpt3}}
  \end{frame}

  %======================================================================================
  \section{O Bloco Decoder}

  \begin{frame}{Bloco Transformer Decoder-Only em Detalhe}
    \begin{columns}[T]
      \begin{column}{0.50\textwidth}
        O bloco básico repete-se $N$ vezes:\\[0.3em]
        \begin{enumerate}
          \item \textbf{Normalização} (Pre-Norm)
          \item \textbf{Masked Self-Attention}
          \item Conexão residual: $\mathbf{x} \mathrel{+}= \text{Attn}(\cdot)$
          \item \textbf{Normalização}
          \item \textbf{Feed-Forward} (MLP)
          \item Conexão residual: $\mathbf{x} \mathrel{+}= \text{FFN}(\cdot)$
        \end{enumerate}
        \vspace{0.3em}
        \textbf{Sem} cross-attention. Toda informação está nas $n$ posições anteriores da sequência.
      \end{column}
      \begin{column}{0.46\textwidth}
        \begin{exampleblock}{O que mudou desde 2017?}
          \begin{itemize}
            \item ReLU-FFN $\to$ \textbf{SwiGLU}
            \item LayerNorm $\to$ \textbf{RMSNorm}
            \item Post-Norm $\to$ \textbf{Pre-Norm}
            \item Codificação posicional: RoPE
          \end{itemize}
          As três primeiras mudanças serão detalhadas a seguir.
        \end{exampleblock}
      \end{column}
    \end{columns}
  \end{frame}

  \begin{frame}{FFN Padrão: Exemplo Numérico (ReLU)}
    \begin{center}
      $\text{FFN}(\mathbf{x}) = \text{ReLU}(\mathbf{x} \mathbf{W}_1)\, \mathbf{W}_2 = \max(0,\, \mathbf{x} \mathbf{W}_1)\, \mathbf{W}_2$
    \end{center}
    \begin{columns}[T]
      % ── LEFT: single-path computation flow ──────────────────────────────
      \begin{column}{0.52\textwidth}
        \centering
        \resizebox{0.68\linewidth}{!}{%
        \begin{tikzpicture}[
          vec/.style={draw, rounded corners, minimum width=3.0cm, minimum height=0.52cm,
                      fill=CellBlue, align=center, font=\small},
          lin/.style={draw, rounded corners, minimum width=2.6cm, minimum height=0.52cm,
                      fill=CellYellow, align=center, font=\small},
          act/.style={draw, rounded corners, minimum width=2.6cm, minimum height=0.52cm,
                      fill=CellRed, align=center, font=\small},
          arr/.style={->, >=Stealth, thick},
          ann/.style={font=\scriptsize, gray},
        ]
          \node[vec] (x)    at (0, 5.2) {$\mathbf{x} \in \mathbb{R}^{d_\mathrm{model}}$};
          \node[lin] (w1)   at (0, 3.9) {$\mathbf{x} \mathbf{W}_1$};
          \node[act] (relu) at (0, 2.6) {$\mathrm{ReLU}(\mathbf{x} \mathbf{W}_1)$};
          \node[lin] (w2)   at (0, 1.3) {$(\cdots)\,\mathbf{W}_2$};
          \node[vec] (y)    at (0, 0.0) {$\mathbf{y} \in \mathbb{R}^{d_\mathrm{model}}$};

          \draw[arr] (x)    -- (w1);
          \draw[arr] (w1)   -- (relu);
          \draw[arr] (relu) -- (w2);
          \draw[arr] (w2)   -- (y);

          \node[ann, right=0.12cm of w1]   {$d_m \to d_{ff}{=}4d_m$};
          \node[ann, right=0.12cm of relu] {$\max(0,\,z)$};
          \node[ann, right=0.12cm of w2]   {$d_{ff} \to d_m$};
        \end{tikzpicture}}
      \end{column}

      % ── RIGHT: numerical example ─────────────────────────────────────
      \begin{column}{0.44\textwidth}
        \vspace{1.2em}
        \footnotesize\textbf{Exemplo numérico}\\[0.1em]
        \scriptsize$d_\mathrm{model}=3,\quad d_{ff}=4\times3=12$\\[0.5em]

        {\setlength{\fboxsep}{3pt}%
        \colorbox{CellBlue}{$\mathbf{x} = [\,1.0,\;0.5,\;{-0.5}\,]$}\\[0.35em]
        \colorbox{CellYellow}{$\mathbf{x}\mathbf{W}_1 = [\,2.0,\;{-1.0},\;0.5,\;\ldots\,]$\scriptsize\textit{ (dim 12)}}\\[0.35em]
        \colorbox{CellRed}{$\mathrm{ReLU}(\mathbf{x}\mathbf{W}_1) = [\,2.0,\;\mathbf{0.0},\;0.5,\;\ldots\,]$}\\[0.3em]
        }%
        \textcolor{red!70!black}{\tiny$\Rightarrow$ ReLU \textbf{zera} o valor negativo $-1.0$!}\\[0.5em]

        {\setlength{\fboxsep}{3pt}%
        \colorbox{CellYellow}{$[\,\cdots\,]\mathbf{W}_2\;\approx\;\mathbf{y} = [\,0.9,\;{-0.1},\;0.3\,]$}
        }
      \end{column}
    \end{columns}
  \end{frame}

  \begin{frame}{SwiGLU: A FFN com \textit{Gate} das LLMs mais atuais}
    \begin{columns}[T]
      \begin{column}{0.50\textwidth}
        FFN clássica usa \textbf{ReLU}:
        \begin{block}{ReLU-FFN (Vaswani 2017)}
          $\text{FFN}(\mathbf{x}) = \max(0,\, \mathbf{x}\mathbf{W}_1)\,\mathbf{W}_2$
        \end{block}
        \vspace{0.4em}
        Llama 2/3, PaLM, Mistral usam \textbf{SwiGLU} \cite{shazeer2020swiglu}:
        \begin{block}{SwiGLU-FFN}
          $\text{FFN}(\mathbf{x}) = \bigl(\text{SiLU}(\mathbf{x}\mathbf{W}_1) \odot \mathbf{x}\mathbf{W}_3\bigr)\,\mathbf{W}_2$
        \end{block}
        onde $\text{SiLU}(z) = z \cdot \sigma(z)$ (\textit{Swish}).\\[0.3em]
        O \textbf{portão} $\mathbf{x}\mathbf{W}_3$ filtra dinamicamente a informação.
      \end{column}
      \begin{column}{0.46\textwidth}
        \begin{exampleblock}{Por que $\tfrac{8}{3}\,d_\text{model}$?}
          SwiGLU usa \textbf{3 matrizes} em vez de 2.\\
          Para manter o mesmo número de parâmetros:\\[0.2em]
          $3 \times d_\text{ff} \cdot d = 2 \times 4d \cdot d$\\[0.2em]
          $\Rightarrow d_\text{ff} = \tfrac{8}{3}\,d_\text{model}$
        \end{exampleblock}
        \vspace{0.3em}
        \begin{itemize}
          \item SwiGLU supera ReLU e GELU empiricamente
          \item Motivação teórica: portão aprende \textit{quais} dimensões ativar
          \item Alternativas: GeGLU, ReGLU
        \end{itemize}
      \end{column}
    \end{columns}
  \end{frame}

  \section{Normalização: Layer Norm e RMS Norm}

  \begin{frame}{Por que Normalizar as Ativações?}
    \begin{columns}[T]
      \begin{column}{0.50\textwidth}
        Durante o treino de redes profundas:
        \begin{itemize}
          \item Distribuição das ativações muda a cada camada (\textit{internal covariate shift})
          \item Gradientes explodem ou desaparecem
          \item Taxas de aprendizado devem ser muito pequenas
        \end{itemize}
        \vspace{0.4em}
        \textbf{Batch Norm}: normaliza sobre o batch, problemático para sequências de tamanho variável.
      \end{column}
      \begin{column}{0.46\textwidth}
        \begin{block}{Solução para Transformers}
          \textbf{Layer Normalization} \cite{ba2016layernorm}: normaliza sobre as \textit{features} de cada token, sem dependência do batch.
        \end{block}
        \vspace{0.3em}
        Permite treino estável de Transformers muito profundos com learning rates maiores.
      \end{column}
    \end{columns}
  \end{frame}

  \begin{frame}{Layer Normalization \cite{ba2016layernorm}}
    \begin{columns}[T]
      \begin{column}{0.52\textwidth}
        Dado o vetor de ativações $\mathbf{x} \in \mathbb{R}^d$ de um token:
        \[
          \mu = \frac{1}{d}\sum_{i=1}^d x_i, \qquad
          \sigma^2 = \frac{1}{d}\sum_{i=1}^d (x_i - \mu)^2
        \]
        \[
          \hat{x}_i = \frac{x_i - \mu}{\sqrt{\sigma^2 + \epsilon}}
        \]
        \[
          \text{LN}(\mathbf{x}) = \boldsymbol{\gamma} \odot \hat{\mathbf{x}} + \boldsymbol{\beta}
        \]
        $\boldsymbol{\gamma}, \boldsymbol{\beta} \in \mathbb{R}^d$: parâmetros aprendidos.
      \end{column}
      \begin{column}{0.44\textwidth}
        \begin{itemize}
          \item Independente do tamanho do batch
          \item Funciona com sequências de tamanho variável
          \item Aplicada \textbf{por token} (ao longo da dimensão $d$)
        \end{itemize}
        \vspace{0.3em}
        \begin{exampleblock}{Nota}
          A normalização ocorre ao longo da dimensão de features $d$ de cada token, não ao longo da sequência.
        \end{exampleblock}
      \end{column}
    \end{columns}
  \end{frame}

  \begin{frame}{RMS Norm: Simplificação do Layer Norm\footnotemark[1]}
    \begin{columns}[T]
      \begin{column}{0.52\textwidth}
        O re-centramento (subtração de $\mu$) é desnecessário quando a invariância de escala é suficiente.\\[0.4em]
        Normaliza apenas pela raiz do valor quadrático médio:
        \[
          \text{RMS}(\mathbf{x}) = \sqrt{\frac{1}{d}\sum_{i=1}^d x_i^2}
        \]
        \[
          \text{RMSNorm}(\mathbf{x}) = \frac{\mathbf{x}}{\text{RMS}(\mathbf{x})} \odot \boldsymbol{\gamma}
        \]
        \textbf{Sem} $\mu$, \textbf{sem} $\boldsymbol{\beta}$, mais simples e rápido.
      \end{column}
      \begin{column}{0.44\textwidth}
        \begin{itemize}
          \item Mais rápido: sem cálculo de média
          \item Mesma qualidade empírica que LayerNorm
          \item Menos parâmetros (sem $\boldsymbol{\beta}$)
        \end{itemize}
        \vspace{0.3em}
        \begin{exampleblock}{Adotado por}
          Llama 1/2/3, Mistral, Mixtral, Gemma, Falcon, PaLM, \ldots
        \end{exampleblock}
        \vspace{0.2em}
        Citado em \cite{stanfordcme295} como padrão atual para LLMs eficientes.
      \end{column}
    \end{columns}
    \footnotetext[1]{\fullcite{zhang2019rmsnorm}}
  \end{frame}

  \begin{frame}{Layer Norm, RMS Norm e Batch Norm: Exemplo Numérico}
    Lote de 2 tokens, cada um com $d=4$ \textit{features}:
    \begin{center}
      $\mathbf{X}=\begin{bmatrix}1&3&5&7\\9&7&5&3\end{bmatrix}$\qquad
      {\small\textcolor{DarkBlue}{$\rightarrow$ LN/RMS: por linha (token)}\quad\textcolor{DarkRed}{$\downarrow$ BN: por coluna (feature)}}
    \end{center}
    \begin{columns}[T]
      \begin{column}{0.52\textwidth}
        \begin{block}{LayerNorm --- token 1 (eixo $d$)}
          $\mu=4,\quad\sigma^2=5$\\[2pt]
          $\hat{\mathbf{x}}=\dfrac{\mathbf{x}-\mu}{\sqrt{\sigma^2}}=[-1.34,\,-0.45,\,0.45,\,1.34]$\\[2pt]
          {\footnotesize média $0$ e variância $1$, usando só as features do token}
        \end{block}
        \onslide<2->
        \begin{block}{RMSNorm --- token 1 (eixo $d$)}
          $\text{RMS}=\sqrt{\tfrac{1}{4}\textstyle\sum_i x_i^2}=4.58$\\[2pt]
          $\mathbf{x}/\text{RMS}=[0.22,\,0.66,\,1.09,\,1.53]$\ {\footnotesize(só reescala)}
        \end{block}
      \end{column}
      \begin{column}{0.44\textwidth}
        \onslide<3->
        \begin{alertblock}{BatchNorm --- feature 1 (lote)}
          coluna $1=[1,\,9]$:\ $\mu=5,\ \sigma^2=16$\\[2pt]
          $\hat{\mathbf{x}}=[-1,\,1]$\\[2pt]
          {\footnotesize usa estatísticas \textbf{do lote}, depende do batch}
        \end{alertblock}
        \onslide<4->
        \begin{exampleblock}{Conclusão}
          LN/RMS normalizam \textbf{por token}; BN \textbf{por feature}. Transformers usam LN/RMS por independerem do batch.
        \end{exampleblock}
      \end{column}
    \end{columns}
  \end{frame}


  \begin{frame}{Pre-Norm vs.\ Post-Norm}
    \begin{columns}[T]
      \begin{column}{0.48\textwidth}
        \begin{block}{Post-Norm (Vaswani 2017)}
          $\mathbf{x} \leftarrow \text{LN}(\mathbf{x} + \text{SubLayer}(\mathbf{x}))$\\[0.3em]
          Normalização \textbf{depois} da operação.\\
          \textcolor{DarkRed}{$\to$ gradientes instáveis com $N$ grande; requer \textit{warm-up} lento.}
        \end{block}
      \end{column}
      \begin{column}{0.48\textwidth}
        \begin{block}{Pre-Norm (GPT-2 em diante)}
          $\mathbf{x} \leftarrow \mathbf{x} + \text{SubLayer}(\text{LN}(\mathbf{x}))$\\[0.3em]
          Normalização \textbf{antes} da operação.\\
          \textcolor{DarkGreen}{$\to$ treino mais estável; gradientes fluem melhor pela conexão residual.}
        \end{block}
      \end{column}
    \end{columns}
    \vspace{0.5em}
    \begin{center}
      \textbf{Hoje}, quase todos os LLMs usam \textbf{Pre-Norm}.\\
      A conexão residual ``bypassa'' a normalização, preservando o sinal ao longo das camadas.
    \end{center}
  \end{frame}

  \section{Estratégias de Decodificação}

  \begin{frame}{Decodificação Greedy}
    \begin{columns}[T]
      \begin{column}{0.50\textwidth}
        Após o LM produzir a distribuição $P(\cdot \mid x_{<t})$:
        \begin{block}{Greedy}
          $x_t = \arg\max_v P(v \mid x_{<t})$
        \end{block}
        \vspace{0.3em}
        \begin{itemize}
          \item Rápido: nenhuma busca adicional
          \item Determinístico: mesma saída para o mesmo prompt
          \item Pode ser subótimo: maximiza cada passo isoladamente, não a sequência
        \end{itemize}
      \end{column}
      \begin{column}{0.46\textwidth}
        \begin{exampleblock}{Exemplo}
          Para \textit{``O rato roeu a \_\_\_''}:\\[0.2em]
          $P(\texttt{rolha}) = 0.60$, $P(\texttt{roupa}) = 0.25$, \ldots\\[0.2em]
          Greedy emite \texttt{rolha} \textbf{sempre}, mesmo que a continuação a partir de \texttt{roupa} fosse melhor.
        \end{exampleblock}
      \end{column}
    \end{columns}
  \end{frame}

  \begin{frame}{Beam Search}
    \begin{columns}[T]
      \begin{column}{0.50\textwidth}
        \begin{block}{Beam Search}
          Mantém $B$ hipóteses parciais; a cada passo expande todas e retém as $B$ de maior \textbf{probabilidade conjunta}.
        \end{block}
        \vspace{0.3em}
        \begin{itemize}
          \item $\mathcal{O}(B)$ mais lento que greedy
          \item Encontra sequências de maior probabilidade total que greedy perde
          \item Também determinístico
        \end{itemize}
      \end{column}
      \begin{column}{0.46\textwidth}
        \begin{exampleblock}{Exemplo com $B=2$}
          \centering
          \resizebox{\linewidth}{!}{%
          \begin{tikzpicture}[
            tok/.style={draw, rounded corners, fill=CellBlue, font=\scriptsize\ttfamily,
                        inner sep=2.5pt, minimum height=0.45cm},
            win/.style={fill=CellGreen, draw=DarkGreen, thick},
            cut/.style={fill=LightGray, draw=gray, text=gray},
            lbl/.style={font=\tiny, midway, above, sloped},
            arr/.style={->, >=Stealth, thick}
          ]
            \node[tok] (root)  at (0, 0)      {\ldots roeu a};
            \node[tok] (rolha) at (2.4, 1.0)  {rolha};
            \node[tok] (roupa) at (2.4, -0.1) {roupa};
            \node[tok, cut] (rua) at (2.4, -1.2) {rua};
            \node[tok] (toda)  at (4.6, 1.0)  {toda};
            \node[tok, win] (suja) at (4.6, -0.1) {suja};

            \draw[arr] (root) -- node[lbl] {$0.60$} (rolha);
            \draw[arr] (root) -- node[lbl] {$0.25$} (roupa);

          \raggedright O beam escolhe \texttt{roupa suja}; greedy teria travado em \texttt{rolha}.
        \end{exampleblock}
      \end{column}
    \end{columns}
  \end{frame}

  \begin{frame}{Amostragem}
    \begin{columns}[T]
      \begin{column}{0.50\textwidth}
        \begin{block}{Amostragem}
          $x_t \sim P(\cdot \mid x_{<t})$
        \end{block}
        \vspace{0.3em}
        \begin{itemize}
          \item Estocástico: saídas variam entre execuções
          \item Necessário para criatividade e diversidade
          \item Risco: tokens raros da cauda geram incoerências
        \end{itemize}
        \vspace{0.3em}
        Os parâmetros \textbf{temperature}, \textbf{top-$k$} e \textbf{top-$p$} (próximos slides) controlam essa distribuição.
      \end{column}
      \begin{column}{0.46\textwidth}
        \begin{exampleblock}{Exemplo}
          Em 100 gerações de \textit{``O rato roeu a \_\_\_''}:\\[0.2em]
          $\approx$60 emitem \texttt{rolha}, $\approx$25 \texttt{roupa}, $\approx$8 \texttt{rua}, \ldots\\[0.2em]
          Até \texttt{rei} ($P = 0.02$) aparece ocasionalmente, coerente ou não.
        \end{exampleblock}
      \end{column}
    \end{columns}
  \end{frame}

  \begin{frame}{Por que Não Usar Sempre Greedy ou Beam? \cite{holtzman2020degeneration}}
    \begin{columns}[T]
      \begin{column}{0.52\textwidth}
        \textbf{Holtzman et al.\ (2020)}: em geração aberta, maximizar a probabilidade produz \textbf{degeneração}.\\[0.4em]
        \begin{alertblock}{Saída típica de beam search (GPT-2)}
          \footnotesize\textit{``Eu adoro música. Eu adoro música. Eu adoro música. Eu adoro música. Eu adoro\ldots''}
        \end{alertblock}
        \vspace{0.3em}
        \begin{itemize}
          \item Repetição em loop: repetir aumenta a probabilidade da própria repetição
          \item Texto genérico e previsível, sem informação nova
        \end{itemize}
      \end{column}
      \begin{column}{0.44\textwidth}
        \begin{block}{Observação central}
          Texto humano \textbf{não} é o texto mais provável: humanos alternam tokens esperados e surpreendentes.
        \end{block}
        \vspace{0.3em}
        \begin{itemize}
          \item Amostragem pura resolve a repetição, mas a \textbf{cauda} da distribuição gera incoerências
          \item Solução: \textbf{amostragem truncada} (top-$k$, top-$p$), corta a cauda e amostra do resto
        \end{itemize}
      \end{column}
    \end{columns}
  \end{frame}

  \begin{frame}{Temperature, Top-$k$ e Top-$p$}
    \begin{columns}[T]
      \begin{column}{0.52\textwidth}
        \textbf{Temperature}:
        \[
          P_T(x) = \frac{\exp(z_x / T)}{\sum_v \exp(z_v / T)}
        \]
        \begin{itemize}
          \item $T \to 0$: greedy
          \item $T = 1$: distribuição original
          \item $T > 1$: distribuição mais uniforme
        \end{itemize}
        \vspace{0.3em}
        \textbf{Top-$p$}: selecione o menor conjunto de tokens cuja probabilidade acumulada $\ge p$ (ex: $p=0.9$), então amostre dentro desse conjunto.
      \end{column}
      \begin{column}{0.44\textwidth}
        \resizebox{\linewidth}{!}{%
        \begin{tikzpicture}
          \begin{axis}[
            ybar,
            bar width=0.38cm,
            xlabel={Token},
            ylabel={Probabilidade},
            symbolic x coords={$v_1$,$v_2$,$v_3$,$v_4$,$v_5$,$v_6$,$v_7$,$v_8$},
            xtick=data,
            ymin=0, ymax=0.55,
            width=7cm, height=4.5cm,
            enlarge x limits=0.08
          ]
            \addplot[fill=CellBlue] coordinates
              {($v_1$,0.45)($v_2$,0.25)($v_3$,0.15)($v_4$,0.07)($v_5$,0.04)($v_6$,0.02)($v_7$,0.01)($v_8$,0.01)};
            \draw[DarkRed, dashed, thick]
              (axis cs:{$v_4$}, 0) -- (axis cs:{$v_4$}, 0.55)
              node[above, font=\tiny, DarkRed] {top-$p{=}0.85$};
          \end{axis}
        \end{tikzpicture}}
      \end{column}
    \end{columns}
  \end{frame}

  \begin{frame}{Temperature, Top-$k$ e Top-$p$: Exemplo Numérico}
    Logits para \textit{``O rato roeu a \_\_\_''} (mesmo exemplo da LM head):
    \vspace{0.3em}
    \begin{columns}[T]
      \begin{column}{0.52\textwidth}
        \resizebox{\linewidth}{!}{%
        \begin{tabular}{lrccc}
          \toprule
          \textbf{Token} & \textbf{Logit} & $P_{T=0.5}$ & $P_{T=1}$ & $P_{T=2}$ \\
          \midrule
          \texttt{rolha} & 3.0    & \textbf{0.84} & \textbf{0.64} & \textbf{0.46} \\
          \texttt{roupa} & 2.1    & 0.14 & 0.26 & 0.29 \\
          \texttt{rua}   & 1.0    & 0.02 & 0.09 & 0.17 \\
          \texttt{rei}   & $-0.5$ & 0.00 & 0.02 & 0.08 \\
          \bottomrule
        \end{tabular}}
        \vspace{0.3em}
        \begin{itemize}
          \item $T{=}0.5$ \textbf{afia}: quase greedy
          \item $T{=}2$ \textbf{achata}: mais diversidade, mais risco
        \end{itemize}
      \end{column}
      \begin{column}{0.44\textwidth}
        Truncamento sobre $P_{T=1}$:
        \begin{block}{Top-$k$ com $k=2$}
          Mantém \texttt{rolha}, \texttt{roupa}; renormaliza:\\
          $P(\texttt{rolha}) = \dfrac{0.64}{0.64+0.26} = 0.71$
        \end{block}
        \begin{block}{Top-$p$ com $p=0.9$}
          Acumulada: $0.64 \to 0.90$; o conjunto $\{\texttt{rolha}, \texttt{roupa}\}$ já atinge $p$, os demais são cortados.
        \end{block}
        \vspace{0.2em}
        {\footnotesize Top-$k$ fixa o \textbf{tamanho} do conjunto; top-$p$ fixa a \textbf{massa} de probabilidade, adaptando-se à incerteza do modelo.}
      \end{column}
    \end{columns}
  \end{frame}

  \begin{frame}{Quando Parar de Gerar?}
    \begin{columns}[T]
      \begin{column}{0.52\textwidth}
        A geração autorregressiva não termina sozinha; três critérios de parada usuais:
        \begin{enumerate}
          \item \textbf{Token \texttt{<eos>}} (\textit{end of sequence}): o modelo emite um token especial de fim
          \item \textbf{Limite de tokens}: teto de custo e de janela de contexto
          \item \textbf{Stop sequences}: strings definidas pela aplicação (ex.: \texttt{"\textbackslash n\textbackslash n"}, fim de bloco de código)
        \end{enumerate}
      \end{column}
      \begin{column}{0.44\textwidth}
        \begin{block}{De onde vem o \texttt{<eos>}?}
          No pré-treino, documentos são concatenados separados por \texttt{<eos>}. O modelo aprende $P(\texttt{<eos>} \mid \text{contexto})$ como qualquer outro token.
        \end{block}
        \vspace{0.3em}
        \begin{exampleblock}{Na prática}
          Amostrar \texttt{<eos>} encerra a resposta; atingir o limite de tokens \textbf{trunca} a resposta no meio, situação que as APIs sinalizam (ex.: \texttt{finish\_reason}).
        \end{exampleblock}
      \end{column}
    \end{columns}
  \end{frame}

  \begin{frame}{Decodificação Guiada (\textit{Constrained Decoding}) \cite{willard2023guided}}
    \begin{columns}[T]
      \begin{column}{0.52\textwidth}
        \textbf{Problema}: LLMs podem gerar texto que não segue formato específico (JSON, SQL, código).\\[0.4em]
        \textbf{Solução}: restringir a distribuição de saída a tokens válidos segundo uma \textbf{gramática} ou \textbf{esquema}.\\[0.4em]
        \textbf{Willard \& Louf (2023)}: pré-compila um DFA para cada gramática, o mascaramento de tokens inválidos é $\mathcal{O}(1)$ por token.
      \end{column}
      \begin{column}{0.44\textwidth}
        \begin{block}{Exemplos de uso}
          \begin{itemize}
            \item Geração de JSON estruturado
            \item SQL sem erros de sintaxe
            \item Extração em formato fixo
            \item Chamadas de API com parâmetros corretos
          \end{itemize}
        \end{block}
        \vspace{0.2em}
        Implementações: \texttt{outlines}, \texttt{guidance}, \texttt{llama.cpp} grammar mode, vLLM structured outputs.
      \end{column}
    \end{columns}
  \end{frame}

  \begin{frame}{Decodificação Guiada: Máscara sobre os Logits em $q_1$}
    \small O DFA define o conjunto válido $V_{q_1} = \{\texttt{*},\ \textit{identificadores}\}$; a máscara soma $-\infty$ ao logit dos demais antes do softmax:
    \[
      P(v \mid x_{\le t}) = \text{softmax}\bigl(\mathbf{z} + \mathbf{M}_{q_1}\bigr)_v,
      \qquad
      M_{q_1}[v] = 0 \text{ se } v \in V_{q_1},\ {-\infty} \text{ caso contrário}
    \]
    \begin{columns}[T]
      \begin{column}{0.52\textwidth}
        \resizebox{0.96\linewidth}{!}{%
        \begin{tabular}{lrcrr}
          \toprule
          \textbf{Token} & \textbf{Logit $z_v$} & $v \in V_{q_1}$? & $z_v + M_{q_1}$ & $P$ \\
          \midrule
          \texttt{FROM}  & 2.5 & \cellcolor{CellRed}$\times$        & \textcolor{DarkRed}{$-\infty$} & \textcolor{gray}{0} \\
          \texttt{*}     & 2.1 & \cellcolor{CellGreen}$\checkmark$ & 2.1 & \textbf{0.49} \\
          \texttt{nome}  & 1.8 & \cellcolor{CellGreen}$\checkmark$ & 1.8 & 0.36 \\
          \texttt{WHERE} & 1.2 & \cellcolor{CellRed}$\times$        & \textcolor{DarkRed}{$-\infty$} & \textcolor{gray}{0} \\
          \texttt{idade} & 0.9 & \cellcolor{CellGreen}$\checkmark$ & 0.9 & 0.15 \\
          \texttt{;}     & 0.3 & \cellcolor{CellRed}$\times$        & \textcolor{DarkRed}{$-\infty$} & \textcolor{gray}{0} \\
          \bottomrule
        \end{tabular}}
        \vspace{0.3em}

        {\footnotesize $P(\texttt{*}) = \dfrac{e^{2.1}}{e^{2.1}+e^{1.8}+e^{0.9}} = \dfrac{8.2}{16.7} \approx 0.49$}
      \end{column}
      \begin{column}{0.44\textwidth}
        \begin{exampleblock}{Leitura}
          \small Sem a máscara, \texttt{FROM} seria o token mais provável ($P \approx 0.36$) e produziria \texttt{SELECT FROM}, inválido. Com a máscara, o softmax renormaliza a probabilidade apenas entre os tokens válidos.
        \end{exampleblock}
        \vspace{0.2em}
        {\footnotesize \textbf{Por que $\mathcal{O}(1)$ por token?} A tabela estado $\to$ máscara é pré-compilada a partir da gramática; a cada passo basta indexá-la pelo estado atual \cite{willard2023guided}.\\[0.3em]
        \scriptsize Nomes de coluna fora do vocabulário são gerados em vários subtokens; o DFA é definido sobre o vocabulário de subtokens do modelo.}
      \end{column}
    \end{columns}
  \end{frame}

  \begin{frame}{Decodificação Especulativa: Ideia Geral \cite{leviathan2023speculative}}
    \begin{columns}[T]
      \begin{column}{0.52\textwidth}
        \textbf{Gargalo}: LLMs grandes são lentos, cada token exige um passe completo.\\[0.4em]
        \textbf{Observação}: a maioria dos tokens é fácil de prever; tokens difíceis são exceção.\\[0.4em]
        \textbf{Algoritmo}:
        \begin{enumerate}
          \item \textbf{Draft model} pequeno gera $\gamma$ tokens
          \item \textbf{Target model} verifica $\gamma$ tokens \textit{em paralelo}
          \item No primeiro rejeitado: re-amostra o token e \textbf{descarta o restante} do rascunho
          \item Caso todos sejam aceitos: amostra um token extra do \textit{forward} do target ($\gamma + 1$ tokens)
        \end{enumerate}
      \end{column}
      \begin{column}{0.44\textwidth}
        \begin{block}{Propriedade}
          Saída \textbf{idêntica em distribuição} ao modelo alvo, sem perda de qualidade por construção.
        \end{block}
        \vspace{0.3em}
        \begin{exampleblock}{Speedup típico}
          $2$--$3\times$ com draft $10\times$ menor.\\
          Ex: Llama 3 70B + Llama 3 8B como draft.
        \end{exampleblock}
      \end{column}
    \end{columns}
  \end{frame}

  \begin{frame}{Decodificação Especulativa: Critério de Aceitação}
    \begin{columns}[T]
      \begin{column}{0.52\textwidth}
        Draft model $M_q$, target model $M_p$:\\[0.4em]
        Para cada token $\tilde{x}_t$ proposto pelo draft:
        \[
          \text{Aceitar com prob.} \min\!\left(1,\; \frac{p(\tilde{x}_t \mid x_{<t})}{q(\tilde{x}_t \mid x_{<t})}\right)
        \]
        Se rejeitado: re-amostre de
        \[
          p'(x) \propto \max\!\bigl(0,\, p(x)-q(x)\bigr)
        \]
        Verifique todos os $\gamma$ tokens \textbf{em um único passe} do modelo alvo.\\[0.4em]
        {\footnotesize O tamanho do rascunho $\gamma$ é um hiperparâmetro de inferência; valores típicos ficam entre 3 e 8, equilibrando taxa de aceitação e custo do draft.}
      \end{column}
      \begin{column}{0.44\textwidth}
        \begin{itemize}
          \item $p \ge q$: token \textbf{sempre aceito}
          \item Draft e target concordam $\to$ aceleração máxima
          \item Tokens difíceis: target corrige
          \item Rejeição na posição $t$: as posições $t{+}1, \ldots, \gamma$ do rascunho são \textbf{descartadas}
        \end{itemize}
        \vspace{0.3em}
        \begin{block}{Complexidade}
          $\gamma+1$ passos de $M_q$ + 1 passe de $M_p$\\
          $\to$ até $\gamma+1$ tokens gerados\\
          ($\gamma$ aceitos + 1 extra amostrado de $M_p$).\\
          Sem especulação: $\gamma+1$ passes de $M_p$.
        \end{block}
      \end{column}
    \end{columns}
  \end{frame}

  \section{Inferência Eficiente}

  \begin{frame}{KV-Cache: Como Funciona na Prática}
    \begin{columns}[T]
      \begin{column}{0.50\textwidth}
        Gerando a sequência \textit{``O rato roeu a rolha''}:
        \vspace{0.3em}
        \begin{enumerate}
          \item \textbf{Prefill}: alimenta o \textit{prompt} inteiro de uma vez\\
                $\to$ computa e \textit{armazena} $\mathbf{K}$, $\mathbf{V}$ de cada token
          \item \textbf{Decode passo 1}: prediz próximo token\\
                $\to$ computa $\mathbf{K}$, $\mathbf{V}$ \textit{só} do token novo\\
                $\to$ reutiliza $\mathbf{K}$, $\mathbf{V}$ dos tokens anteriores do cache
          \item \textbf{Decode passo 2, 3, \ldots}: idem\\
                $\to$ cache cresce a cada passo gerado
        \end{enumerate}
      \end{column}
      \begin{column}{0.46\textwidth}
        \begin{block}{Sem cache (naïve)}
          Para o $n$-ésimo token:\\
          Recalcula $\mathbf{K}$, $\mathbf{V}$ de todos os $n$ tokens\\
          Custo total: $\mathcal{O}(n^2 d)$
        \end{block}
        \vspace{0.3em}
        \begin{block}{Com cache}
          Computa $\mathbf{K}$, $\mathbf{V}$ apenas do token novo\\
          Concatena com o cache existente\\
          Custo por token: $\mathcal{O}(n d)$
        \end{block}
        \vspace{0.2em}
        {\footnotesize Memória do cache $\propto n \cdot L \cdot d_k$, cresce com a sequência e número de camadas.}
      \end{column}
    \end{columns}
  \end{frame}

  \begin{frame}{Multi-Head Attention (MHA)}{Recapitulação}
    \begin{columns}[T]
      \begin{column}{0.52\textwidth}
        Com $H$ cabeças, $d_k = d_\text{model}/H$:
        \begin{itemize}
          \item $H$ projeções de $\mathbf{Q}$, $\mathbf{K}$, $\mathbf{V}$ independentes
          \item $H$ pares de matrizes KV em cache por camada
          \item Custo de KV-cache por token: $2 \cdot H \cdot d_k \cdot L$
        \end{itemize}
        \vspace{0.3em}
        Para GPT-3 ($H=96$, $d_k=128$, $L=96$):
        \[
          \text{Cache} = 2 \times 96 \times 128 \times 96 \approx 2.4\text{M floats/token}
        \]
        Em fp16 (2 bytes/float): $\approx$4.7 MB por token; um contexto de 2048 tokens ocupa $\approx$\textbf{9.7 GB por requisição}.
      \end{column}
      \begin{column}{0.44\textwidth}
        \resizebox{\linewidth}{!}{%
        \begin{tikzpicture}[
          head/.style={draw, fill=CellBlue, minimum width=1.1cm, minimum height=1.4cm,
                       rounded corners, font=\small, align=center},
          node distance=0.4cm
        ]
          \foreach \i/\l in {1/1,2/2,3/3,4/H} {
            \node[head] (h\i) at (\i*1.3, 0) {$\mathbf{Q}_{\l}$\\$\mathbf{K}_{\l}$\\$\mathbf{V}_{\l}$};
          }
          \node[font=\small] at (4*1.3+0.9, 0) {\ldots};
          \node[font=\small] at (2.6, -1.1) {KV-cache: $H$ pares por camada};
        \end{tikzpicture}}
        \vspace{0.3em}
        Gargalo em inferência: KV-cache cresce com $n$, $H$, $L$.
      \end{column}
    \end{columns}
  \end{frame}

  \begin{frame}{Multi-Query Attention (MQA)\footnotemark[1]}
    \begin{columns}[T]
      \begin{column}{0.52\textwidth}
        Reduza $\mathbf{K}$ e $\mathbf{V}$ para \textbf{uma única cabeça} compartilhada; mantenha $H$ cabeças de $\mathbf{Q}$.
        \[
          \text{head}_h = \text{Attn}(\mathbf{Q}\mathbf{W}_h^Q,\; \mathbf{K} \mathbf{W}^K,\; \mathbf{V} \mathbf{W}^V)
        \]
        \textbf{Redução de cache:} $H\times \to 1\times$ nos tensores $\mathbf{K}$, $\mathbf{V}$.\\[0.3em]
        Cache por token: $2 \cdot d_k \cdot L$ (independente de $H$).
      \end{column}
      \begin{column}{0.44\textwidth}
        \resizebox{\linewidth}{!}{%
        \begin{tikzpicture}[
          qhead/.style={draw, fill=CellBlue, minimum width=0.8cm, minimum height=0.85cm,
                        rounded corners, font=\small, align=center},
          kvhead/.style={draw, fill=CellGreen, minimum width=1.0cm, minimum height=0.85cm,
                         rounded corners, font=\small, align=center},
          arrow/.style={->, >=Stealth, thick}
        ]
          \foreach \i in {1,2,3,4} {
            \node[qhead] (q\i) at (\i*0.95, 1.5) {$\mathbf{Q}_{\i}$};
          }
          \node[kvhead] (kv) at (2.375, 0) {$\mathbf{K}\,\mathbf{V}$};
          \foreach \i in {1,2,3,4} {
            \draw[arrow] (kv) -- (q\i);
          }
          \node[right=0.2cm of q4, font=\footnotesize] {$H$ cabeças $\mathbf{Q}$};
          \node[below=0.12cm of kv, font=\footnotesize] {1 par $\mathbf{K}$,$\mathbf{V}$};
        \end{tikzpicture}}
        \begin{itemize}
          \item KV-cache $H\times$ menor
          \item Inferência muito mais rápida
          \item Leve queda de qualidade
        \end{itemize}
      \end{column}
    \end{columns}
    \footnotetext[1]{\fullcite{shazeer2019mqa}}
  \end{frame}

  \begin{frame}{Grouped Query Attention (GQA)\footnotemark[1]}
    \begin{columns}[T]
      \begin{column}{0.52\textwidth}
        Um intermediário entre MHA e MQA.\\[0.4em]
        Divida as $H$ cabeças de $\mathbf{Q}$ em $G$ grupos. Cada grupo compartilha um par $\mathbf{K}$,$\mathbf{V}$.
        \[
          G = 1 \;\Rightarrow\; \text{MQA}
          \qquad
          G = H \;\Rightarrow\; \text{MHA}
        \]
        Llama 2 (34B e 70B), Llama 3, Mistral e Mixtral usam GQA; as variantes 7B e 13B do Llama 2 ainda usam MHA.

  \begin{frame}{Sliding Window Attention \cite{jiang2023mistral}}
    \begin{columns}[T]
      \begin{column}{0.52\textwidth}
        \textbf{Motivação}: em sequências longas, a maioria das informações úteis está \textbf{próxima} do token atual.\\[0.4em]
        Cada posição $i$ atende apenas às $w$ posições anteriores:
        \[
          \text{Attn}(i) \leftarrow \text{tokens em } [i-w,\; i]
        \]
        Custo: $\mathcal{O}(n \cdot w)$ em vez de $\mathcal{O}(n^2)$.\\[0.3em]
        Campo receptivo em $L$ camadas: $L \cdot w$.\\
        {\footnotesize Informação distante propaga-se por composição: camada $l$ agrega o que camada $l{-}1$ já resumiu da janela anterior.}
      \end{column}
      \begin{column}{0.44\textwidth}
        \begin{block}{Configuração Mistral 7B}
          \begin{itemize}
            \item Janela local: $w = 4096$
            \item 32 camadas
            \item Campo receptivo: $32 \times 4096 = 131\text{K}$
            \item Contexto declarado: 32K tokens
          \end{itemize}
        \end{block}
        \vspace{0.3em}
        \textcolor{DarkBlue}{\textbf{Resultado}}: Mistral 7B supera Llama 2 13B em benchmarks com $2\times$ menos parâmetros.
      \end{column}
    \end{columns}
  \end{frame}

  \begin{frame}{FlashAttention: Atenção Exata e Eficiente \cite{dao2022flashattention}}
    \begin{columns}[T]
      \begin{column}{0.52\textwidth}
        \small MQA/GQA/SWA \textbf{alteram} a atenção. FlashAttention ataca outro gargalo: a \textbf{movimentação de memória} na GPU.\\[0.4em]
        \begin{itemize}
          \item GPUs computam rápido, mas ler e escrever na memória principal (HBM) é lento
          \item A implementação padrão \textbf{materializa} a matriz $n \times n$ de scores na HBM
          \item FlashAttention computa a atenção \textbf{por blocos} na SRAM (cache rápido), com \textit{softmax online}, sem nunca materializar a matriz completa
        \end{itemize}
        \vspace{0.2em}
        {\scriptsize \textit{Softmax online}: acumula o softmax bloco a bloco, guardando apenas o máximo e a soma parciais dos scores, corrigidos a cada novo bloco.}
      \end{column}
      \begin{column}{0.44\textwidth}
        \begin{block}{Propriedades}
          \begin{itemize}
            \item \textbf{Exata}: mesmo resultado numérico, sem aproximação
            \item Memória extra $\mathcal{O}(n)$ em vez de $\mathcal{O}(n^2)$
            \item Speedup de $2$--$4\times$ no treino
            \item FlashAttention-2 \cite{dao2023flashattention2}: padrão 
          \end{itemize}
        \end{block}
        \vspace{0.3em}
        \begin{alertblock}{Lição}
          Parte do custo da atenção estava na \textbf{implementação}, não na matemática: otimizar antes de aproximar.
        \end{alertblock}
      \end{column}
    \end{columns}
  \end{frame}

  \section{Mistura de Especialistas (MoE)}

  \begin{frame}{Motivação: Mais Parâmetros sem Mais Compute}
    \begin{columns}[T]
      \begin{column}{0.52\textwidth}
        \textbf{Dilema}:
        \begin{itemize}
          \item Modelos maiores $\to$ melhor qualidade
          \item Modelos maiores $\to$ mais compute por token
        \end{itemize}
        \vspace{0.4em}
        \textbf{Ideia da Mistura de Especialistas (MoE)}:\\
        Substitua a FFN densa por $N_E$ redes menores (\textit{especialistas}).\\
        Cada token ativa apenas $k$ especialistas ($k \ll N_E$).
      \end{column}
      \begin{column}{0.44\textwidth}
        \begin{exampleblock}{Exemplo: Mixtral 8×7B}
          8 especialistas FFN\\
          Apenas 2 ativados por token\\
          $\to$ 46.7B parâmetros totais\\
          $\to$ compute $\approx$ 12.9B por token
        \end{exampleblock}
        \vspace{0.3em}
        {\footnotesize
        Ideia original: \textit{mixtures of experts} \cite{jacobs1991moe}.\\
        Variante esparsa: \textbf{Sparsely-Gated MoE} \cite{shazeer2017moe}.\\
        Retomada em Transformers: GShard \cite{lepikhin2021gshard} e Switch Transformer \cite{fedus2022switch}.\par}
      \end{column}
    \end{columns}
  \end{frame}

  \begin{frame}{Arquitetura MoE: Router + Especialistas}
    \begin{columns}[T]
      \begin{column}{0.46\textwidth}
        \centering
        \begin{tikzpicture}[
          router/.style={draw, fill=CellYellow, rounded corners,
                         minimum width=3cm, minimum height=0.7cm, font=\small, align=center},
          expert/.style={draw, fill=CellBlue, rounded corners,
                         minimum width=1.6cm, minimum height=0.65cm, font=\small, align=center},
          sum/.style={draw, circle, fill=white, minimum size=0.45cm, font=\small},
          arrow/.style={->, >=Stealth, thick},
          node distance=0.5cm
        ]
          \node[router] (r) {Router $G(\mathbf{x})$};
          \node[expert, above left=0.9cm and 0.2cm of r] (e1) {FFN$_1$};
          \node[expert, above=0.9cm of r]               (e2) {FFN$_2$};
          \node[expert, above right=0.9cm and 0.2cm of r] (e3) {FFN$_3$};
          \node[sum, above=2.7cm of r] (s) {$\sum$};
          \node[above=0.3cm of s, font=\small] (out) {saída};
          \node[below=0.5cm of r, font=\small] (inp) {entrada $\mathbf{x}$};

          \draw[arrow] (inp) -- (r);
          \draw[arrow] (r) -- (e1);
          \draw[arrow] (r) -- (e2);
          \draw[arrow] (r) -- (e3);
          \draw[arrow] (e1) |- (s);
          \draw[arrow] (e2) -- (s);
          \draw[arrow] (e3) |- (s);
          \draw[arrow] (s) -- (out);
        \end{tikzpicture}
      \end{column}
      \begin{column}{0.50\textwidth}
        \textbf{Em cada bloco MoE}:
        \begin{enumerate}
          \item Router calcula scores para cada especialista
          \item Seleciona top-$k$ especialistas
          \item Processa $\mathbf{x}$ nos $k$ especialistas
          \item Combina saídas ponderadas pelos scores
        \end{enumerate}
        \vspace{0.4em}
        \begin{block}{Nota}
          As camadas de \textbf{atenção} permanecem densas.\\
          Somente a FFN vira MoE.
        \end{block}
      \end{column}
    \end{columns}
  \end{frame}

  \begin{frame}{Top-$k$ Routing: Mecanismo de Gating}
    \begin{columns}[T]
      \begin{column}{0.52\textwidth}
        Dado token $\mathbf{x} \in \mathbb{R}^d$ e pesos $\mathbf{W}_g \in \mathbb{R}^{d \times N_E}$:
        \[
          \mathbf{g}(\mathbf{x}) = \text{softmax}\!\left(\text{TopK}\!\left(\mathbf{x} \mathbf{W}_g,\; k\right)\right)
        \]
        onde $\text{TopK}$ mantém os $k$ maiores scores e zera o resto.\\[0.4em]
        Saída:
        \[
          \mathbf{y} = \sum_{i \in \text{Top-}k} g_i(\mathbf{x})\; \text{FFN}_i(\mathbf{x})
        \]
      \end{column}
      \begin{column}{0.44\textwidth}
      \scriptsize
        \begin{itemize}
          \item Mixtral usa $k=2$, $N_E=8$
          \item $\sim$25\% dos parâmetros ativos por token
          \item Backward: apenas pelos $k$ especialistas ativos
        \end{itemize}
        \vspace{0.3em}
        \begin{alertblock}{Expert collapse}
          \textbf{Causa}: o router aprende a enviar tokens ao especialista que já é o melhor, que melhora ainda mais, atraindo mais tokens, em ciclo de reforço positivo.\\[0.15em]
          \textbf{Resultado}: 1--2 especialistas processam quase tudo; os demais recebem gradiente zero e nunca aprendem.\\[0.15em]
          \textbf{Solução}: \textit{Loss} auxiliar de balanceamento (próximo slide).
        \end{alertblock}
      \end{column}
    \end{columns}
  \end{frame}

  \begin{frame}{Top-$k$ Routing: Exemplo Numérico}
    Token $\mathbf{x}$, $N_E=4$ especialistas, $k=2$:
    \vspace{0.3em}
    \begin{columns}[T]
      \begin{column}{0.52\textwidth}
        {\setlength{\fboxsep}{3pt}%
        \colorbox{CellYellow}{$\mathbf{s} = \mathbf{x} \mathbf{W}_g = [\,1.2,\;\mathbf{3.1},\;0.4,\;\mathbf{2.6}\,]$}\\[0.45em]
        \colorbox{CellBlue}{$\text{TopK}(\mathbf{s}, 2) = [\,-\infty,\;3.1,\;-\infty,\;2.6\,]$}\\[0.45em]
        \colorbox{CellGreen}{$\mathbf{g} = \text{softmax} = [\,0,\;\mathbf{0.62},\;0,\;\mathbf{0.38}\,]$}\\[0.45em]
        }%
        {\footnotesize $g_2 = \dfrac{e^{3.1}}{e^{3.1}+e^{2.6}} = \dfrac{22.2}{35.7} = 0.62$}\\[0.5em]
        {\setlength{\fboxsep}{3pt}%
        \colorbox{PastelOrange}{$\mathbf{y} = 0.62\,\text{FFN}_2(\mathbf{x}) + 0.38\,\text{FFN}_4(\mathbf{x})$}
        }
      \end{column}
      \begin{column}{0.44\textwidth}
        \begin{exampleblock}{O que foi economizado}
          $\text{FFN}_1$ e $\text{FFN}_3$ \textbf{nem são computadas} para este token: com $N_E=4$ e $k=2$, metade do compute da camada é evitado.
        \end{exampleblock}
        \vspace{0.3em}
        \begin{itemize}
          \item Outro token da mesma sequência pode ativar especialistas diferentes
          \item O gradiente flui apenas pelos $k$ especialistas ativos e pelo router
        \end{itemize}
      \end{column}
    \end{columns}
  \end{frame}

  \begin{frame}{Problemas de MoE: Balanceamento de Carga}
    \begin{columns}[T]
      \begin{column}{0.50\textwidth}
        \textbf{Perda auxiliar de balanceamento}:
        \[
          \mathcal{L}_\text{aux} = \alpha \cdot N_E \cdot \sum_{i=1}^{N_E} f_i \cdot P_i
        \]
        onde: $N_E$ = número de especialistas,\\
        $\alpha$ = hiperparâmetro que pondera a perda auxiliar na perda total,\\
        $f_i$ = fração de tokens roteados ao especialista $i$,\\
        $P_i$ = score médio do router para o especialista $i$.\\[0.3em]
        Minimizar $\mathcal{L}_\text{aux}$ encoraja distribuição uniforme.
      \end{column}
      \begin{column}{0.46\textwidth}
        Outros desafios:
        \begin{itemize}
          \item \textbf{Instabilidade}: gradientes esparsas podem causar divergência
          \item \textbf{Comunicação}: especialistas distribuídos em GPUs diferentes (\textit{model parallelism})
          \item \textbf{Capacidade máxima}: tokens descartados se um especialista ficar sobrecarregado
        \end{itemize}
        \vspace{-0.1em}
        \begin{exampleblock}{Mixtral}
          \begin{itemize}
            \item Perda auxiliar com $\alpha=0.01$
            \item Roteamento token a token: top-$k$ escolhido por token e por camada
          \end{itemize}
        \end{exampleblock}
      \end{column}
    \end{columns}
  \end{frame}

  \section{Leis de Escala}

  \begin{frame}{O que Significa ``Escalar'' um LLM?}
    \begin{columns}[T]
      \begin{column}{0.50\textwidth}
        Três eixos de escala:
        \begin{enumerate}
          \item \textbf{Parâmetros} $N$: tamanho do modelo
          \item \textbf{Dados} $D$: tokens de treinamento
          \item \textbf{Compute} $C$: FLOPs totais de treino{\footnotesize\ (FLOP = \textit{Floating Point Operation}, unidade de custo aritmético)}
        \end{enumerate}
        \vspace{0.4em}
        Relação aproximada:
        \[
          C \approx 6 N D
        \]
        Cada token exige $\approx$6 FLOPs por parâmetro:\\
        2 no \textit{forward} $+$ 4 no \textit{backward} \cite{kaplan2020scaling}.
      \end{column}
      \begin{column}{0.46\textwidth}
        \begin{alertblock}{Pergunta central}
          Dado um orçamento de compute fixo, como distribuir entre $N$ (modelo maior) e $D$ (mais dados)?
        \end{alertblock}
        \vspace{0.4em}
        \begin{exampleblock}{Exemplo: Llama 2 7B (2T tokens)}
          $C \approx 6 \times (7{\times}10^{9}) \times (2{\times}10^{12}) \approx 8.4{\times}10^{22}$ FLOPs\\[2pt]
          {\footnotesize A $\sim\!1.5{\times}10^{14}$ FLOP/s úteis por A100, são $\approx$6500 GPU-dias: 3 semanas em 300 GPUs.}
        \end{exampleblock}
        \vspace{0.3em}
        Isso é exatamente o que as \textbf{leis de escala} respondem.
      \end{column}
    \end{columns}
  \end{frame}

  \begin{frame}{Leis de Escala de Kaplan et al.\ \cite{kaplan2020scaling}}
    \begin{columns}[T]
      \begin{column}{0.52\textwidth}
        \textbf{Descoberta}: a perda de teste segue \textbf{leis de potência}:
        \[
          L(N) \propto N^{-\alpha_N}, \quad
          L(D) \propto D^{-\alpha_D}
        \]
        com $\alpha_N \approx 0.076$, $\alpha_D \approx 0.095$.\\[0.5em]
        \textbf{Recomendação}: dado compute fixo, priorize \textbf{escalar $N$} e use menos dados.
      \end{column}
      \begin{column}{0.44\textwidth}
        \begin{itemize}
          \item Válido para modelos de 768 a 1.5B parâmetros (não-embedding)
          \item Curvas de perda suaves e previsíveis
          \item Arquitetura (profundidade vs.\ largura) tem impacto menor
        \end{itemize}
        \vspace{0.3em}
        \begin{alertblock}{Limitação}
          Kaplan et al.\ não otimizavam $N$ e $D$ simultaneamente; fixavam $N$ e mediam $L$ ótima em $C$.
        \end{alertblock}
      \end{column}
    \end{columns}
  \end{frame}

  \begin{frame}{Chinchilla: Treinamento Compute-Ótimo \cite{hoffmann2022chinchilla}}
    \begin{columns}[T]
      \begin{column}{0.52\textwidth}
        \textbf{Hoffmann et al.\ (2022)}, DeepMind\\[0.4em]
        Treinaram 400+ modelos variando $N$ e $D$ com mesmo $C$.\\[0.4em]
        \begin{block}{Resultado central}
          Para compute ótimo:
          \[
            N^* \propto C^{0.5}, \quad D^* \propto C^{0.5}
          \]
          \textbf{Dobrar compute $\to$ dobrar $N$ E dobrar $D$.}
        \end{block}
      \end{column}
      \begin{column}{0.44\textwidth}
        Chinchilla (70B) $\approx$ Gopher (280B) em qualidade, com $4\times$ menos compute de inferência.\\[0.5em]
        \begin{exampleblock}{Regra prática}
          Para um modelo de $N$ parâmetros, treine com pelo menos \textbf{$20N$ tokens}.
        \end{exampleblock}
        \vspace{0.3em}
        Exemplo: modelo de 7B $\to$ $\ge$140B tokens.
      \end{column}
    \end{columns}
  \end{frame}

  \section{Benchmarks e Avaliação de LLMs}

  \begin{frame}{Perplexidade: A Métrica Mais Simples}
    \textbf{Definição}: mede o quão ``surpreso'' o modelo fica ao ver o texto.
    \[
      \mathrm{PPL}(w_1,\ldots,w_N)
        = \exp\!\left(-\frac{1}{N}\sum_{i=1}^{N}\log P(w_i \mid w_1,\ldots,w_{i-1})\right)
    \]
    \begin{itemize}
      \item Menor PPL $\to$ modelo estima melhor a distribuição do corpus
      \item PPL = 1: certeza absoluta; PPL = $|V|$: uniforme sobre o vocabulário
      \item Diretamente ligada à \textit{cross-entropy loss} do treino
    \end{itemize}
    \vspace{0.4em}
    \begin{columns}[T]
      \begin{column}{0.52\textwidth}
        \begin{exampleblock}{Exemplo numérico (3 tokens)}
          $P(w_1){=}0.5,\; P(w_2){=}0.25,\; P(w_3){=}0.125$\\[2pt]
          $\mathrm{PPL} = \exp\!\left(\tfrac{0.69 + 1.39 + 2.08}{3}\right) = e^{1.39} = \textbf{4.0}$
        \end{exampleblock}
      \end{column}
      \begin{column}{0.44\textwidth}
        \begin{exampleblock}{Intuição}
          PPL = 20 $\Rightarrow$ o modelo hesita, em média, entre 20 tokens igualmente prováveis a cada passo.
        \end{exampleblock}
      \end{column}
    \end{columns}
  \end{frame}

  \begin{frame}{Perplexidade: Por que Não é Suficiente?}
    \begin{alertblock}{Limitações da perplexidade como única métrica}
      \begin{itemize}
        \item \textbf{Acurácia factual}: um modelo com PPL baixa pode ainda assim alucinar fatos
        \item \textbf{Raciocínio}: baixa surpresa no texto não implica capacidade de resolver problemas
        \item \textbf{Seguimento de instruções}: não captura se o modelo obedece comandos do usuário
        \item \textbf{Desempenho em tarefas reais}: PPL não prediz bem resultados em downstream tasks
      \end{itemize}
    \end{alertblock}
    \vspace{0.5em}
    \textcolor{DarkRed}{$\Rightarrow$ Precisamos de benchmarks específicos por tarefa para avaliar LLMs adequadamente.}
  \end{frame}

  \begin{frame}{Desafios na Avaliação de LLMs}
    \begin{columns}[T]
      \begin{column}{0.50\textwidth}
        Por que avaliar LLMs é difícil?
        \begin{itemize}
          \item Saída livre $\to$ métricas automáticas são imperfeitas
          \item \textit{Data contamination}: benchmarks no pré-treino
          \item Capacidades emergentes: falha em instâncias simples, acerta difíceis
          \item Benchmarks ficam saturados rapidamente
        \end{itemize}
      \end{column}
      \begin{column}{0.46\textwidth}
        \begin{block}{Categorias de avaliação}
          \begin{itemize}
            \item \textbf{Múltipla escolha}: ARC, MMLU
            \item \textbf{Geração + verificação}: GSM8k
            \item \textbf{Raciocínio}: HellaSwag, WinoGrande
            \item \textbf{Instrução}: MT-Bench, AlpacaEval
            \item \textbf{Arena humana}: Chatbot Arena (Elo)
          \end{itemize}
        \end{block}
      \end{column}
    \end{columns}
  \end{frame}

    \begin{columns}[T]
      \begin{column}{0.50\textwidth}
        {\footnotesize Benchmarks gerais (MMLU, ARC\ldots) medem conhecimento amplo, mas \textbf{não capturam} capacidades especializadas:}
        \begin{block}{Contexto longo}
          {\footnotesize\begin{itemize}
            \item \textbf{RULER}: tarefas sintéticas em janelas de 4K--128K tokens
            \item \textbf{NIAH}: ``agulha'' escondida em textos de 100K+ tokens
          \end{itemize}}
        \end{block}
        \begin{block}{Chamada de ferramentas}
          {\footnotesize\begin{itemize}
            \item \textbf{BFCL}: 2000 pares função/argumento em múltiplas linguagens
            \item \textbf{ToolBench}: uso de APIs reais via ReAct
          \end{itemize}}
        \end{block}
      \end{column}
      \begin{column}{0.46\textwidth}
        \begin{block}{Código}
          {\footnotesize\begin{itemize}
            \item \textbf{HumanEval}: geração de código Python; métrica \textit{pass@k}
            \item \textbf{SWE-bench}: resolve issues reais do GitHub
          \end{itemize}}
        \end{block}
        \begin{block}{Outros domínios}
          {\footnotesize\begin{itemize}
            \item \textbf{MT-Bench}: instrução multi-turno avaliada por GPT-4
            \item \textbf{MATH}: 12500 problemas de competição
            \item \textbf{MMMU}: multimodal (imagem + texto), 57 disciplinas
          \end{itemize}}
        \end{block}
        \vspace{0.2em}
        {\footnotesize \textcolor{DarkBlue}{$\Rightarrow$ Nenhum benchmark único é suficiente: avaliação requer um portfólio de tarefas.}}
