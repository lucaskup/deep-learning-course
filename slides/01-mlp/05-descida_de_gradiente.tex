  \section{Por que não força bruta}
  %======================================================================

  %--- Slide: setup do exemplo (com link da apresentação original) ---
  \begin{frame}{Um exemplo concreto: regressão logística com 1 peso e 1 bias}
    \begin{columns}
      \begin{column}{.6\linewidth}
        \begin{itemize}
          \item Uma única unidade sigmoide com 2 parâmetros (peso $w$ e bias $b$).

          \[
            \mathcal{L}(w,b)=\frac{1}{N}\sum_{i=1}^{N}\left(y^{(i)}-f(x^{(i)})\right)^{2}

          \item Pergunta natural: por que não simplesmente
            \[
              \argmin_{w,b}\,\mathcal{L}(w,b)
            \]
            via varredura por força bruta?
          \item \href{https://lucaskup.github.io/deep-learning-course/gradient-descent/\#s1}{%
              \textit{\color{blue}\underline{Vamos tentar}} (demo interativa da disciplina)}
        \end{itemize}
      \end{column}
      \begin{column}{.4\linewidth}
        \centering
        \resizebox{.9\linewidth}{!}{
          \begin{tikzpicture}[>=Latex]
            \node[font=\small] at (1.5,3.6) {$\hat{y}=f(x)$};
            \draw[->,thick] (1.5,2.6) -- (1.5,3.4);
            \node[circle,draw,fill=red!70,minimum size=1cm,text=white] (s) at (1.5,2) {$\sigma$};

  %--- Slide: por que força bruta não escala ---
  \begin{frame}{E se tentássemos varrer todos os $(w,b)$?}
    \begin{itemize}
      \item Com 2 parâmetros conseguimos visualizar a paisagem da loss em uma grade.
      \item Para 2 parâmetros, com $k$ valores em cada eixo, avaliamos $k^2$ pontos.
      \item Para uma rede com $P$ parâmetros, avaliamos $k^P$ pontos.
    \end{itemize}
    \vfill
    \begin{exampleblock}{Para fixar a ideia}
      Mesmo um MLP modesto com $P{=}1\,000$ parâmetros e apenas $k{=}10$ valores por eixo
      exigiria $10^{1000}$ avaliações. Inviável.
    \end{exampleblock}
  \end{frame}

  %--- Slide: conclusão da seção ---
  \begin{frame}{Precisamos de um método guiado}
    \begin{itemize}
      \item Força bruta ignora qualquer estrutura da função.
      \item Em problemas suaves (diferenciáveis), a própria função carrega informação
        local sobre como reduzir a loss.
      \item A descida de gradiente é justamente um método que usa essa informação
        local para escolher a direção e o tamanho do próximo passo.
    \end{itemize}
  \end{frame}

  \section{Derivando a Descida de Gradiente}
  %======================================================================

  %--- Slide: estratégia geral (vetores sem sobreposição) ---
  \begin{frame}{Estratégia: aplicar uma atualização que reduza a loss}
    \begin{columns}
      \begin{column}{.55\linewidth}
        \begin{itemize}
          \item Queremos achar uma atualização $\Delta\thetav$ tal que
            $\mathcal{L}(\thetav+\Delta\thetav) < \mathcal{L}(\thetav)$.
          \item A pergunta é: \alert{qual atualização $\Delta\thetav$ escolher?}
        \end{itemize}
        \vspace{.5em}
        {\footnotesize\color{gray!50!black}Aqui $\thetav\in\mathbb{R}^{n}$ é o vetor com todos os $n$ parâmetros do modelo empilhados: cada matriz de pesos e cada vetor de viés entram vetorizados nesse único vetor coluna.}
      \end{column}
      \begin{column}{.45\linewidth}
        \centering
        \resizebox{.7\linewidth}{!}{
          \begin{tikzpicture}[>=Latex]
            % theta
            \draw[->,thick] (0,0) -- (0.9,2.4) node[midway,above left] {$\thetav$};
            % update \Delta\thetav from origin
            \draw[->,very thick,red] (0,0) -- (1.4,0);
            \node[red,anchor=north] at (0.7,-0.05) {$\Delta\thetav$};
            % new theta = theta + \Delta\thetav
            \draw[->,thick] (0,0) -- (2.3,2.4) node[midway,below right=-1pt] {$\thetav_{\text{new}}$};
            % dashed copy of \Delta\thetav translated to tip of theta, completing the parallelogram
            \draw[dashed,thick,red] (0.9,2.4) -- (2.3,2.4);
          \end{tikzpicture}
        }
      \end{column}
    \end{columns}
  \end{frame}

  %--- Slide: refresher Taylor (com aproximações que de fato funcionam) ---
  \begin{frame}{Refresher: Série de Taylor\footnotemark}
    \begin{columns}
      \begin{column}{.55\linewidth}
        \begin{itemize}
          \item Aproxima localmente uma função suave $\mathcal{L}(\theta)$ por um polinômio em torno de $\theta_0$.
          \item Quanto mais termos, melhor a aproximação.
          \item Para deslocamentos pequenos em torno de $\theta_0$, a primeira ordem domina.
        \end{itemize}
      \end{column}
      \begin{column}{.45\linewidth}
        \centering
        \resizebox{.95\linewidth}{!}{
          \begin{tikzpicture}
            \begin{axis}[
              width=7cm, height=5cm,
              xlabel={$\theta$}, ylabel={$\mathcal{L}(\theta)$},
              axis lines=middle, xtick=\empty, ytick=\empty,
              ymin=0, ymax=2.6,
              domain=-1.5:1.5, samples=120,
              legend style={font=\tiny,at={(1.05,1)},anchor=north west}
            ]
              % function: e^{0.6 x} -- has nonzero second derivative at 0
              \addplot[thick,red] {exp(0.6*x)};
              \addlegendentry{função}
              % Taylor degree 1 at x_0=0:  1 + 0.6 x
              \addplot[thick,blue] {1 + 0.6*x};
              \addlegendentry{grau 1}
              % Taylor degree 2 at x_0=0:  1 + 0.6 x + 0.18 x^2
              \addplot[thick,Green] {1 + 0.6*x + 0.18*x*x};
              \addlegendentry{grau 2}
              \node[fill=blue,circle,inner sep=1.5pt] at (axis cs:0,1) {};
              \node[anchor=north,font=\tiny] at (axis cs:0,0.95) {$\theta_0$};
            \end{axis}
          \end{tikzpicture}
        }
      \end{column}
    \end{columns}
    Para $\theta$ escalar:
    \[
      \mathcal{L}(\theta)=\mathcal{L}(\theta_{0})
        +\frac{\mathcal{L}'(\theta_0)}{1!}(\theta-\theta_0)^{1}
        +\frac{\mathcal{L}''(\theta_0)}{2!}(\theta-\theta_0)^{2}
        +\frac{\mathcal{L}'''(\theta_0)}{3!}(\theta-\theta_0)^{3}+\cdots
    \]
    {\footnotesize\color{gray!50!black}Para $\thetav$ vetorial não há potências: o termo de grau 1 vira o produto escalar $(\thetav-\thetav_0)^{\top}\nabla\mathcal{L}(\thetav_0)$ e o de grau 2 usa a Hessiana, $\frac{1}{2!}(\thetav-\thetav_0)^{\top}\nabla^{2}\mathcal{L}(\thetav_0)\,(\thetav-\thetav_0)$. É essa forma que usaremos a seguir.}
    \footnotetext[1]{Demo interativa da disciplina:
      \href{https://lucaskup.github.io/deep-learning-course/gradient-descent/\#s2}{lucaskup.github.io/deep-learning-course/gradient-descent}.}
  \end{frame}

  %--- Slide: Taylor aplicado à loss ---
  \begin{frame}{Aplicando Taylor à Loss}
    Na série de Taylor, expandimos em torno de $\thetav_0=\thetav$ e avaliamos no ponto atualizado $\thetav+\Delta\thetav$:
    {\small
    \[
      \mathcal{L}(\thetav+\Delta\thetav) =
        \mathcal{L}(\thetav)
        + (\thetav+\Delta\thetav-\thetav)^\top\nabla_{\thetav}\mathcal{L}(\thetav)
        + \frac{1}{2!}(\thetav+\Delta\thetav-\thetav)^\top\nabla^{2}_{\thetav}\mathcal{L}(\thetav)\,(\thetav+\Delta\thetav-\thetav)
        + \cdots
    \]}
    \begin{itemize}
      \item Em todos os termos, $\thetav-\thetav$ se anula: o deslocamento é simplesmente $\Delta\thetav$.
    \end{itemize}
    \[
      \mathcal{L}(\thetav+\Delta\thetav) =
        \mathcal{L}(\thetav)
        + \Delta\thetav^\top\nabla_{\thetav}\mathcal{L}(\thetav)
        + \frac{1}{2!}\Delta\thetav^\top\nabla^{2}_{\thetav}\mathcal{L}(\thetav)\,\Delta\thetav
        + \cdots
    \]
    \begin{itemize}
      \item O termo quadrático em $\Delta\thetav$ só é desprezível se o passo for \alert{pequeno}.
    \end{itemize}
  \end{frame}

  %--- Slide: passo pequeno via taxa de aprendizado ---
  \begin{frame}{Aplicando Taylor à Loss}
    \begin{itemize}
      \item Ideia: controlar o tamanho do passo com um escalar, atualizando por $\eta \Delta\thetav$
        com $\eta>0$ pequeno, a \textit{taxa de aprendizado} (\textit{learning rate}):
        \[
          \mathcal{L}(\thetav+\eta \Delta\thetav) =
            \mathcal{L}(\thetav)
            + \eta\,\Delta\thetav^\top\nabla_{\thetav}\mathcal{L}(\thetav)
            + \frac{\eta^{2}}{2!}\Delta\thetav^\top\nabla^{2}_{\thetav}\mathcal{L}(\thetav)\,\Delta\thetav
            + \cdots
        \]
      \item Para $\eta$ suficientemente pequeno, $\eta^{2},\eta^{3},\ldots$ são
        desprezíveis e ficamos com a aproximação de primeira ordem:
        \[
          \mathcal{L}(\thetav+\eta \Delta\thetav)\approx \mathcal{L}(\thetav) + \eta\,\Delta\thetav^\top\nabla_{\thetav}\mathcal{L}(\thetav).
        \]
    \end{itemize}
  \end{frame}

  %--- Slide: critério de melhoria ---
  \begin{frame}{Critério de melhoria}
    Queremos que a atualização reduza a loss, ou seja
    \[
      \mathcal{L}(\thetav+\eta \Delta\thetav) - \mathcal{L}(\thetav) < 0.
    \]
    Substituindo a aproximação de primeira ordem:
    \[
      \eta\,\Delta\thetav^\top\nabla_{\thetav}\mathcal{L}(\thetav) < 0.
    \]
    \vfill
    \begin{itemize}
      \item Como $\eta>0$, basta exigir $\Delta\thetav^\top\nabla_{\thetav}\mathcal{L}(\thetav) < 0$.
      \item Isto é, o produto escalar entre a direção de atualização e o gradiente
        deve ser \alert{negativo}.
    \end{itemize}
  \end{frame}

  %--- Slide: produto escalar e cosseno (vetores corretos) ---
  \begin{frame}{Quando o produto escalar é negativo?}
    Seja $\alpha$ o ângulo entre $\Delta\thetav$ e $\nabla_{\thetav}\mathcal{L}(\thetav)$.
    \[
      \Delta\thetav^\top\nabla_{\thetav}\mathcal{L}(\thetav) =
        \|\Delta\thetav\|\,\|\nabla_{\thetav}\mathcal{L}(\thetav)\|\,\cos(\alpha).
    \]
    \vfill
    \begin{columns}
      \begin{column}{.55\linewidth}
        \begin{itemize}
          \item $\|\Delta\thetav\|\ge 0$ e $\|\nabla\mathcal{L}\|\ge 0$ sempre.
          \item Logo, o sinal vem inteiramente de $\cos(\alpha)$.
          \item Negativo para $\alpha\in(90^{\circ},180^{\circ})$.
          \item \alert{``Mais negativo''} para $\alpha=180^{\circ}$, ou seja
            $\Delta\thetav=-\nabla_{\thetav}\mathcal{L}$.
        \end{itemize}
      \end{column}
      \begin{column}{.45\linewidth}
        \centering
        \resizebox{.95\linewidth}{!}{
          \begin{tikzpicture}[>=Latex]
            % Gradient pointing right
            \draw[->,thick,blue] (0,0) -- (3,0) node[right] {$\nabla\mathcal{L}$};
            % -gradient pointing exactly opposite (180 degrees)
            \draw[->,thick,red] (0,0) -- (-3,0) node[left] {$\Delta\thetav^{\,*}=-\nabla\mathcal{L}$};
            % a generic \Delta\thetav making angle beta with gradient
            \draw[->,thick,Green!70!black] (0,0) -- ({3*cos(135)},{3*sin(135)}) node[above left] {$\Delta\thetav$};
            % arc for alpha
            \draw[thick] (0.9,0) arc (0:135:0.9);
            \node[font=\small] at (0.6,0.6) {$\alpha$};
          \end{tikzpicture}
        }
      \end{column}
    \end{columns}
  \end{frame}

  %--- Slide: regra final ---
  \begin{frame}{Regra de atualização da Descida de Gradiente}
    \begin{block}{Direção ótima}
      A escolha que minimiza o produto escalar é
      \(
        \Delta\thetav = -\nabla_{\thetav}\mathcal{L}(\thetav).
      \)
    \end{block}
    \vfill
    Substituindo na atualização $\thetav_{t+1}=\thetav_{t}+\eta\,\Delta\thetav$:
    \[
      \boxed{\thetav_{t+1}=\thetav_{t}-\eta\,\nabla_{\thetav}\mathcal{L}(\thetav_{t})}
    \]
    \vfill
    \begin{itemize}
      \item Cada passo move os parâmetros na direção oposta ao gradiente.
      \item $\eta$ controla o tamanho do passo.
      \item Esta é a \textbf{descida de gradiente}\footfullcite{goodfellow2016deep}.
    \end{itemize}
  \end{frame}

  %--- Slide: exemplo numérico descida de gradiente ---
  \begin{frame}{Exemplo numérico: descida de gradiente}
    \begin{block}{Cenário}
      \[
        \eta = 0.1,\qquad \thetav_t = \cvec{3}{5},\qquad
        \nabla_{\thetav}\mathcal{L}(\thetav_t) = \cvec{1}{2}.
      \]
    \end{block}
    \[
      \thetav_{t+1} = \thetav_t - \eta\,\nabla_{\thetav}\mathcal{L}(\thetav_t)
      = \cvec{3}{5} - 0.1\,\cvec{1}{2} = \cvec{2.9}{4.8}.
    \]
    \begin{itemize}
      \item O passo é proporcional ao gradiente: a segunda coordenada anda o dobro (0.2 contra 0.1).
    \end{itemize}
  \end{frame}

  \subsection{Tamanho do Mini-batch}
  %======================================================================

  %--- Slide: três tamanhos canônicos ---
  \begin{frame}{Variar quantas instâncias entram no gradiente}
    \begin{itemize}
      \item Podemos estimar o gradiente com diferentes tamanhos de \textit{batch}:
    \end{itemize}
    \vfill
    \begin{center}
      \begin{tabular}{@{}lll@{}}
        \toprule
        Tamanho do batch & Nome usual & Atualização \\
        \midrule
        $N$ (todo o dataset) & \textit{Batch Gradient Descent} & determinística \\
        $1<m<N$ & \textit{Mini-batch Gradient Descent} & estocástica \\
        $1$ & \textit{Stochastic Gradient Descent} (SGD) & altamente estocástica \\
        \bottomrule
      \end{tabular}
    \end{center}
    \vfill
    \begin{itemize}
      \item Em deep learning, ``SGD'' costuma ser usado como sinônimo de mini-batch
        com $m\ll N$.
    \end{itemize}
  \end{frame}

  %--- Slide: por que não usar sempre o batch completo ---
  \begin{frame}{Por que não usar sempre o batch completo?}
    \begin{itemize}
      \item Limitações de \textit{hardware}: o batch precisa caber na memória da GPU.
      \item Custo por passo: cada atualização vê $N$ exemplos, mesmo quando uma fração já
        seria suficiente para uma boa estimativa do gradiente.
      \item Falta de ruído: a trajetória é determinística e não consegue escapar de mínimos
        locais ou pontos de sela.
    \end{itemize}
    \vfill
    \begin{block}{Trade-off}
      Mini-batches pequenos introduzem ruído (bom para escapar) e cabem em memória,
      mas a estimativa do gradiente é mais barulhenta (passos mais erráticos).
    \end{block}
  \end{frame}

  %--- Slide: SGD vs Batch (figura limpa) ---
  \begin{frame}{Trajetória: \textit{Batch} vs \textit{Stochastic}\footfullcite{prince2023understanding}}
    \centering
    \includegraphics[height=0.66\textheight]{slides/img/01-mlp/gd-batch-vs-sgd.png}
    \begin{itemize}
      \item Esquerda: \textit{batch gradient descent} usa o gradiente exato em cada passo.
      \item Direita: \textit{stochastic gradient descent} oscila, mas explora vales que o batch ignora.
    \end{itemize}
  \end{frame}

  %--- Slide: amostragem de mini-batches (figura limpa) ---
  \begin{frame}{Mini-batches em uma paisagem real\footfullcite{prince2023understanding}}
    \begin{columns}
      \begin{column}{.55\linewidth}
        \centering
        \includegraphics[width=\linewidth,height=0.78\textheight,keepaspectratio]{slides/img/01-mlp/gd-minibatch-4panel.png}
      \end{column}
      \begin{column}{.45\linewidth}
        \begin{itemize}
          \item Em a) está a loss completa.
          \item Em b), c), d) vemos diferentes mini-batches com $m{=}3$.
          \item Cada mini-batch aproxima o gradiente verdadeiro com erro,
            o que se traduz em ruído na trajetória.
        \end{itemize}
      \end{column}
    \end{columns}
  \end{frame}

  %--- Slide: propriedades do SGD ---
  \begin{frame}{Propriedades do SGD}
    \begin{itemize}
      \item Instâncias amostradas \alert{sem reposição}; uma \textbf{época} é
        completada quando todas as instâncias foram vistas.
      \item Adiciona ruído ao treinamento, ajudando a escapar de mínimos locais e
        pontos de sela.
      \item É computacionalmente leve (uma instância por passo).
      \item Empiricamente, soluções obtidas com SGD generalizam melhor.
      \item Não converge a um ponto fixo (passos sempre flutuam ao redor do mínimo).
    \end{itemize}
  \end{frame}

  %======================================================================
  \subsection{Momentum e Nesterov}
  %======================================================================

  %--- Slide: motivação ---
  \begin{frame}{Motivação: usar a memória dos gradientes}
    \begin{itemize}
      \item Gradientes consecutivos podem oscilar fortemente (mini-batch ruidoso ou vales estreitos).
      \item Ideia: somar uma fração da atualização anterior à nova, suavizando a trajetória.
      \item Se os gradientes recentes apontam na mesma direção, aceleramos.
      \item Se mudam de direção, o efeito tende a se cancelar e o passo é amortecido.
    \end{itemize}
  \end{frame}

  %--- Slide: Momentum formalização ---
  \begin{frame}{\textit{Momentum}\footfullcite{polyak1964momentum}}
    \begin{block}{Atualização com momentum}
      \[
        \mathbf{v}_{t+1} = \beta\,\mathbf{v}_{t} + (1-\beta)\,\nabla_{\thetav}\mathcal{L}(\thetav_{t})
      \]
      \[
        \thetav_{t+1} = \thetav_{t} - \eta\,\mathbf{v}_{t+1}
      \]
    \end{block}
    \begin{itemize}
      \item $\beta\in[0,1)$ controla quanto da memória é considerado.
      \item $\mathbf{v}_t$ é uma \textbf{média móvel exponencial} dos gradientes anteriores.
      \item Para $\beta=0$, recuperamos a descida de gradiente padrão.
      \item Valores típicos: $\beta\in[0.9\,;\,0.99]$.
    \end{itemize}
    \vspace{0.4em}
    {\footnotesize\color{gray!50!black}Convenção sem $(1-\beta)$: PyTorch e o paper original de Polyak escrevem $\mathbf{v}_{t+1} = \beta\,\mathbf{v}_t + \nabla_{\thetav}\mathcal{L}$. A diferença é absorvível em $\eta$.}
  \end{frame}

  %--- Slide: comparação visual momentum (figura limpa) ---
  \begin{frame}{Efeito visual do \textit{Momentum}}
    \centering
    \includegraphics[width=.7\linewidth]{slides/img/01-mlp/gd-momentum-vs-no.png}
    \begin{itemize}
      \item Esquerda: SGD puro oscila fortemente em vales estreitos.
      \item Direita: com momentum, oscilações verticais se cancelam e a trajetória
        avança mais rápido na direção principal.
    \end{itemize}
  \end{frame}

  %--- Slide: exemplo numérico Momentum ---
  \begin{frame}{Exemplo numérico: \textit{Momentum}}
    \begin{block}{Cenário}
      \[
        \eta = 0.1,\quad \beta = 0.9,\quad \thetav_t = \cvec{3}{5},\quad
        \nabla_{\thetav}\mathcal{L}(\thetav_t) = \cvec{1}{2},\quad \mathbf{v}_t = \cvec{2}{1}.
      \]
    \end{block}
    \[
      \begin{aligned}
        \mathbf{v}_{t+1} &= 0.9\,\cvec{2}{1} + 0.1\,\cvec{1}{2} = \cvec{1.9}{1.1}\\[4pt]
        \thetav_{t+1} &= \cvec{3}{5} - 0.1\,\cvec{1.9}{1.1} = \cvec{2.81}{4.89}
      \end{aligned}
    \]
    \begin{itemize}
      \item A memória $\mathbf{v}_t$ empurra o passo além do gradiente atual: 0.19 na primeira coordenada, contra 0.1 do SGD.
    \end{itemize}
  \end{frame}

  %--- Slide: Nesterov ---
  \begin{frame}{\textit{Nesterov Accelerated Momentum}\footfullcite{nesterov1983method}}
    \begin{itemize}
      \item Momentum é, na prática, um \textit{preditor} do próximo passo.
      \item Nesterov sugere avaliar o gradiente \alert{já no ponto deslocado pelo momentum},
        e não no ponto atual.
    \end{itemize}
    \vfill
    \begin{block}{Regra de Nesterov}
      \[
        \mathbf{v}_{t+1} = \beta\,\mathbf{v}_{t} + (1-\beta)\,\nabla_{\thetav}\mathcal{L}(\thetav_{t} - \eta\,\beta\,\mathbf{v}_{t})
      \]
      \[
        \thetav_{t+1} = \thetav_{t} - \eta\,\mathbf{v}_{t+1}
      \]
    \end{block}
    \begin{itemize}
      \item Pequena correção, ganho prático em problemas com vales alongados.
    \end{itemize}
  \end{frame}

  %--- Slide: visual Nesterov (figura limpa) ---
  \begin{frame}{Nesterov: olhar à frente antes de calcular o gradiente}
    \begin{columns}
      \begin{column}{.55\linewidth}
        \centering
        \includegraphics[width=\linewidth,height=0.78\textheight,keepaspectratio]{slides/img/01-mlp/gd-nesterov.png}
      \end{column}
      \begin{column}{.45\linewidth}
        \begin{itemize}
          \item O ponto branco é onde o momentum levaria, e é nesse ponto que o gradiente é avaliado.
          \item Resultado: a trajetória do Nesterov ``corrige'' a do momentum quando há curvatura.
        \end{itemize}
      \end{column}
    \end{columns}
  \end{frame}

  %--- Slide: exemplo numérico Nesterov ---
  \begin{frame}{Exemplo numérico: Nesterov}
    \begin{block}{Cenário}
      \[
        \eta=0.1,\quad \beta=0.9,\quad \thetav_t=\cvec{3}{5},\quad \mathbf{v}_t=\cvec{2}{1}.
      \]
    \end{block}
    Ponto adiantado pelo momentum e gradiente avaliado nele:
    \[
      \thetav_t - \eta\beta\,\mathbf{v}_t = \cvec{3}{5} - 0.09\,\cvec{2}{1} = \cvec{2.82}{4.91},
      \qquad \nabla_{\thetav}\mathcal{L}\!\left(\cvec{2.82}{4.91}\right) = \cvec{0.8}{1.6}.
    \]
    \[
      \begin{aligned}
        \mathbf{v}_{t+1} &= 0.9\,\cvec{2}{1} + 0.1\,\cvec{0.8}{1.6} = \cvec{1.88}{1.06}\\[4pt]
        \thetav_{t+1} &= \cvec{3}{5} - 0.1\,\cvec{1.88}{1.06} = \cvec{2.812}{4.894}
      \end{aligned}
    \]
    \begin{itemize}
      \item O gradiente é medido já no ponto deslocado pelo momentum, não em $\thetav_t$.
    \end{itemize}
  \end{frame}

  %======================================================================
  \subsection{Métodos com Taxa Adaptativa}
  %======================================================================

  %--- Slide: motivação normalização ---
  \begin{frame}{Problema do passo fixo em paisagens anisotrópicas}
    \begin{itemize}
      \item Em uma direção a loss pode ser muito íngreme; em outra, quase plana.
      \item Com $\eta$ fixo, ou damos passos exagerados na direção íngreme, ou ficamos
        lentos na direção rasa.
      \item Solução: usar uma taxa de aprendizado \textbf{efetiva} diferente para cada
        parâmetro, baseada na magnitude histórica de seus gradientes.
    \end{itemize}
  \end{frame}

  %--- Slide: ideia da normalização ---
  \begin{frame}{Normalização do gradiente}
    Para tornar o passo independente da magnitude do gradiente, podemos definir
    \[
      \mathbf{m}_{t+1}=\nabla_{\thetav}\mathcal{L}(\thetav_t),\qquad
      \mathbf{v}_{t+1}=\big(\nabla_{\thetav}\mathcal{L}(\thetav_t)\big)^{2}
    \]
    e atualizar com
    \[
      \thetav_{t+1}=\thetav_{t}-\eta\,\frac{\mathbf{m}_{t+1}}{\sqrt{\mathbf{v}_{t+1}}+\epsilon}.
    \]
    \begin{itemize}
      \item Com a normalização, o termo que multiplica $\eta$ tem magnitude próxima de 1.
      \item O tamanho do passo passa a ser controlado essencialmente por $\eta$.
      \item $\epsilon$ é um pequeno valor positivo para estabilidade numérica.
    \end{itemize}
  \end{frame}

  %--- Slide: exemplo numérico normalização ---
  \begin{frame}{Exemplo numérico: normalização do gradiente}
    \begin{block}{Cenário}
      \[
        \eta=0.1,\quad \thetav_t=\cvec{3}{5},\quad \nabla_{\thetav}\mathcal{L}(\thetav_t)=\cvec{1}{2},\quad \epsilon\approx 0.
      \]
    \end{block}
    \[
      \mathbf{m}_{t+1} = \cvec{1}{2},\qquad \mathbf{v}_{t+1} = \cvec{1^2}{2^2} = \cvec{1}{4}.
    \]
    \[
      \thetav_{t+1} = \cvec{3}{5} - 0.1\,\cvec{1/\sqrt{1}}{2/\sqrt{4}}
      = \cvec{3}{5} - 0.1\,\cvec{1}{1} = \cvec{2.9}{4.9}.
    \]
    \begin{itemize}
      \item Divisão elemento a elemento. Apesar do gradiente assimétrico, o passo normalizado é igual nas duas coordenadas.
    \end{itemize}
  \end{frame}

  %--- Slide: AdaGrad ---
  \begin{frame}{AdaGrad: \textit{Adaptive (per-parameter) Gradient}\footfullcite{duchi2011adagrad}}
    \begin{block}{Atualização}
      \[
        \mathbf{G}_{t+1} = \mathbf{G}_{t} + \big(\nabla_{\thetav}\mathcal{L}\big)^{2}
      \]
      \[
        \thetav_{t+1} = \thetav_{t} - \frac{\eta}{\sqrt{\mathbf{G}_{t+1}}+\epsilon}\,\nabla_{\thetav}\mathcal{L}
      \]
    \end{block}
    \begin{itemize}
      \item Cada parâmetro tem sua própria taxa efetiva, baseada em $\mathbf{G}_{t}$, a soma
        acumulada dos gradientes ao quadrado.
      \item Gradientes consistentemente grandes recebem passos menores; parâmetros
        raros recebem passos maiores.
    \end{itemize}
    \begin{alertblock}{Limitação}
      $\mathbf{G}_{t}$ é monotonamente crescente, logo $\eta/(\sqrt{\mathbf{G}_{t}}+\epsilon)$ decai sempre:
      a taxa efetiva tende a zero e o aprendizado para.
    \end{alertblock}
  \end{frame}

  %--- Slide: exemplo numérico AdaGrad ---
  \begin{frame}{Exemplo numérico: AdaGrad}
    \begin{block}{Cenário}
      \[
        \eta=0.1,\quad \thetav_t=\cvec{3}{5},\quad \nabla_{\thetav}\mathcal{L}(\thetav_t)=\cvec{1}{2},\quad \mathbf{G}_t=\cvec{24}{12}.
      \]
    \end{block}
    \[
      \mathbf{G}_{t+1} = \cvec{24}{12} + \cvec{1^2}{2^2} = \cvec{25}{16}.
    \]
    \[
      \thetav_{t+1} = \cvec{3}{5} - 0.1\,\cvec{1/\sqrt{25}}{2/\sqrt{16}}
      = \cvec{3}{5} - 0.1\,\cvec{0.2}{0.5} = \cvec{2.98}{4.95}.
    \]
    \begin{itemize}
      \item $\mathbf{G}_t$ acumula os quadrados dos gradientes e só cresce. Cada coordenada ganha uma taxa efetiva própria $\eta/\sqrt{\mathbf{G}_{t+1}}$, então o passo $(0.02,\ 0.05)$ difere entre os parâmetros: fica menor onde $\mathbf{G}$ acumulou mais.
    \end{itemize}
  \end{frame}

  %--- Slide: RMSProp ---
  \begin{frame}{RMSProp: \textit{Root Mean Square Propagation}\footfullcite{tieleman2012rmsprop}}
    \begin{itemize}
      \item Apresentado por Hinton em um \textit{slide} de aula; nunca foi formalmente publicado.
      \item Disponível em praticamente todos os \textit{frameworks}.
      \item Ideia: substituir a soma do AdaGrad por uma \textbf{média móvel exponencial}.
    \end{itemize}
    \vfill
    \begin{block}{Atualização}
      \[
        \mathbb{E}[g^{2}]_{t+1} = \rho\,\mathbb{E}[g^{2}]_{t} + (1-\rho)\,(\nabla_{\thetav}\mathcal{L})^{2}
      \]
      \[
        \thetav_{t+1}=\thetav_{t} - \frac{\eta}{\sqrt{\mathbb{E}[g^{2}]_{t+1}}+\epsilon}\,\nabla_{\thetav}\mathcal{L}
      \]
    \end{block}
    \begin{itemize}
      \item $\rho$ controla a janela efetiva de memória; o termo deixa de crescer indefinidamente.
    \end{itemize}
  \end{frame}

  %--- Slide: exemplo numérico RMSProp ---
  \begin{frame}{Exemplo numérico: RMSProp}
    \begin{block}{Cenário}
      \[
        \eta=0.1,\quad \rho=0.9,\quad \thetav_t=\cvec{3}{5},\quad \nabla_{\thetav}\mathcal{L}(\thetav_t)=\cvec{1}{2},\quad \mathbb{E}[g^2]_t=\cvec{1}{4}.
      \]
    \end{block}
    \[
      \mathbb{E}[g^2]_{t+1} = 0.9\,\cvec{1}{4} + 0.1\,\cvec{1}{4} = \cvec{1}{4}.
    \]
    \[
      \thetav_{t+1} = \cvec{3}{5} - 0.1\,\cvec{1/\sqrt{1}}{2/\sqrt{4}} = \cvec{2.9}{4.9}.
    \]
    \begin{itemize}
      \item Neste cenário o passo coincide com o da normalização simples. A diferença para o AdaGrad (média móvel contra soma acumulada) aparece ao longo das iterações.
    \end{itemize}
  \end{frame}

  %--- Slide: Adam motivação ---
  \begin{frame}{Adam: \textit{Adaptive Moment Estimation}\footfullcite{kingma2015adam}}
    \begin{itemize}
      \item Junta as duas ideias vistas: \textbf{momentum} (sobre o gradiente) e
        \textbf{normalização} (pela raiz da média móvel do gradiente ao quadrado).
      \item Considerado, por muitos anos, o otimizador padrão para deep learning.
    \end{itemize}
  \end{frame}

  %--- Slide: Adam fórmulas ---
  \begin{frame}{Adam: as duas médias móveis}
    \begin{block}{Estimativas dos momentos}
      \[
        \mathbf{m}_{t+1} = \beta_1\,\mathbf{m}_{t} + (1-\beta_1)\,\nabla_{\thetav}\mathcal{L}(\thetav_{t})
      \]
      \[
        \mathbf{v}_{t+1} = \beta_2\,\mathbf{v}_{t} + (1-\beta_2)\,\big(\nabla_{\thetav}\mathcal{L}(\thetav_{t})\big)^{2}
      \]
    \end{block}
    \begin{itemize}
      \item $\beta_1\in[0,1)$, $\beta_2\in[0,1)$, com $\mathbf{m}_0=0$ e $\mathbf{v}_0=0$.
      \item $\mathbf{m}$ acompanha o primeiro momento (média) do gradiente, $\mathbf{v}$ acompanha o segundo
        momento (média do gradiente ao quadrado).
    \end{itemize}
    \vspace{0.4em}
    {\footnotesize\color{gray!50!black}Atenção à notação: aqui $\mathbf{v}$ é a média móvel do gradiente ao quadrado, não a velocidade do \textit{momentum}.}
  \end{frame}

  %--- Slide: Adam correção de viés ---
  \begin{frame}{Adam: correção de viés inicial}
    \begin{itemize}
      \item Com $\mathbf{m}_0=\mathbf{v}_0=0$, as primeiras iterações subestimam fortemente $\mathbf{m}$ e $\mathbf{v}$.
      \item Correção de viés (\textit{bias correction}):
        \[
          \widetilde{\mathbf{m}}_{t+1}=\frac{\mathbf{m}_{t+1}}{1-\beta_1^{t+1}},\qquad
          \widetilde{\mathbf{v}}_{t+1}=\frac{\mathbf{v}_{t+1}}{1-\beta_2^{t+1}}.
        \]
      \item A regra final é
        \[
          \boxed{\thetav_{t+1}=\thetav_{t}-\eta\,\frac{\widetilde{\mathbf{m}}_{t+1}}{\sqrt{\widetilde{\mathbf{v}}_{t+1}}+\epsilon}}
        \]
      \item Hiperparâmetros padrão: $\beta_1{=}0.9$, $\beta_2{=}0.999$, $\epsilon{=}10^{-8}$.
    \end{itemize}
  \end{frame}

  %--- Slide: exemplo numérico Adam ---
  \begin{frame}{Exemplo numérico: Adam (primeiro passo)}
    \begin{block}{Cenário}
      \[
        \eta=0.1,\ \beta_1=0.9,\ \beta_2=0.999,\ \thetav_0=\cvec{3}{5},\ \nabla_{\thetav}\mathcal{L}(\thetav_0)=\cvec{1}{2},\ \mathbf{m}_0=\mathbf{v}_0=0.
      \]
    \end{block}
    \[
      \mathbf{m}_1 = 0.1\,\cvec{1}{2} = \cvec{0.1}{0.2},\qquad \mathbf{v}_1 = 0.001\,\cvec{1}{4} = \cvec{0.001}{0.004}.
    \]
    Correção de viés ($t=1$):
    \[
      \widetilde{\mathbf{m}}_1 = \frac{1}{1-0.9}\cvec{0.1}{0.2} = \cvec{1}{2},\qquad
      \widetilde{\mathbf{v}}_1 = \frac{1}{1-0.999}\cvec{0.001}{0.004} = \cvec{1}{4}.
    \]
    \[
      \thetav_1 = \cvec{3}{5} - 0.1\,\cvec{1/\sqrt{1}}{2/\sqrt{4}} = \cvec{2.9}{4.9}.
    \]
    \begin{itemize}
      \item A correção de viés desfaz a subestimativa inicial: o passo por parâmetro fica $\approx \eta$.
    \end{itemize}
  \end{frame}

  %--- Slide: Adam visual em loss anisotrópica (figura limpa) ---
  \begin{frame}{Por que o Adam ajuda em loss anisotrópica}
    \begin{columns}
      \begin{column}{.55\linewidth}
        \includegraphics[width=\linewidth]{slides/img/01-mlp/gd-adam-ab.png}
      \end{column}
      \begin{column}{.45\linewidth}
        \begin{itemize}
          \item Esquerda: $\eta$ pequeno avança bem na vertical, mas é lento na horizontal.
          \item Direita: $\eta$ grande progride na horizontal, mas oscila descontroladamente
            na vertical.
          \item Adam normaliza por direção, evitando ambos os extremos.
        \end{itemize}
      \end{column}
    \end{columns}
  \end{frame}

  %--- Slide: Adam comparação completa (figura limpa) ---
  \begin{frame}{Adam comparado com outros otimizadores}
    \centering
    \includegraphics[height=0.78\textheight]{slides/img/01-mlp/gd-adam-4panel.png}
  \end{frame}

  %--- Slide: Adam recap ---
  \begin{frame}{Por que Adam virou o padrão}
    \begin{itemize}
      \item As magnitudes dos gradientes variam fortemente conforme a função de ativação,
        a camada da rede e a inicialização.
      \item Adam é mais imune a essas diferenças e menos sensível à escolha da taxa de
        aprendizado inicial.
      \item Na prática, costuma funcionar razoavelmente bem com os hiperparâmetros padrão.
    \end{itemize}
  \end{frame}

  %--- Slide: AdamW ---
  \begin{frame}{AdamW: \textit{weight decay} desacoplado\footfullcite{loshchilov2019adamw}}
    \begin{itemize}
      \item \textit{Weight decay}: encolher os pesos um pouco a cada passo, penalizando
        pesos de magnitude alta.
      \item No SGD, basta somar $\lambda\,\thetav_t$ ao gradiente: o efeito é o decaimento
        multiplicativo $(1-\eta\lambda)\,\thetav_t$.
      \item No Adam, esse termo entraria em $\mathbf{m}$ e $\mathbf{v}$ e passaria pela divisão por
        $\sqrt{\widetilde{\mathbf{v}}}$: pesos com histórico de gradientes grandes quase não decairiam.
      \item O \textbf{AdamW} desacopla o decaimento, aplicando-o fora da parte adaptativa:
    \end{itemize}
    \begin{block}{Atualização do AdamW}
      \[
        \thetav_{t+1}=\thetav_{t}-\eta\left(\frac{\widetilde{\mathbf{m}}_{t+1}}{\sqrt{\widetilde{\mathbf{v}}_{t+1}}+\epsilon}
        + \lambda\,\thetav_{t}\right)
      \]
    \end{block}
  \end{frame}

  %======================================================================
  \subsection{Muon: usando ortogonalização de matrizes}
  %======================================================================

  %--- Slide: motivação Muon ---
  \begin{frame}{Muon: \textit{MomentUm Orthogonalized by Newton-schulz}\footfullcite{jordan2024muon}}
    \begin{itemize}
      \item Por quase uma década, Adam (e variantes como AdamW) foi o padrão de fato em
        deep learning.
      \item Em 2024, o \textbf{Muon} foi proposto, apresentado pelos autores como o
        primeiro avanço fundamental desde Adam.
      \item Foco do Muon: as \alert{matrizes de pesos} das camadas ocultas (parâmetros 2D).
        Bias e \textit{embeddings} continuam usando AdamW.
    \end{itemize}
  \end{frame}

  %--- Slide: valores singulares (conceito) ---
  \begin{frame}{Valores singulares: os fatores de esticamento da matriz}
    Uma matriz $\mathbf{M}$ transforma o círculo unitário, o conjunto dos vetores de norma 1, em uma \alert{elipse}.
    \vfill
    \centering
    \begin{tikzpicture}[scale=0.72]
      \begin{scope}[shift={(-3.6,0)}]
        \draw[gray!40,->] (-1.5,0)--(1.5,0);
        \draw[gray!40,->] (0,-1.5)--(0,1.5);
        \draw[very thick,NavyBlue] (0,0) circle (1);
        \node[below] at (0,-1.8) {\small vetores de norma 1};
      \end{scope}
      \draw[->,very thick] (-1.6,0) -- node[above]{$\mathbf{M}$} (0.4,0);
      \begin{scope}[shift={(3.6,0)}]
        \draw[gray!40,->] (-2.3,0)--(2.3,0);
        \draw[gray!40,->] (0,-1.5)--(0,1.5);
        \draw[very thick,BrickRed,rotate=25] (0,0) ellipse (2 and 0.6);
        \draw[->,thick,rotate=25] (0,0) -- node[sloped,above,pos=0.6]{$\sigma_1$} (2,0);
        \draw[->,thick,rotate=25] (0,0) -- (0,0.6) node[above,inner sep=1.5pt]{$\sigma_2$};
        \node[below] at (0,-1.8) {\small a imagem é uma elipse};
      \end{scope}
    \end{tikzpicture}
    \vfill
    \raggedright
    \begin{itemize}
      \item Os \textbf{valores singulares} $\sigma_1\ge\sigma_2$ são os comprimentos dos semieixos.
      \item Vetores ao longo da direção de $\sigma$ grande saem muito amplificados; ao longo da de $\sigma$ pequeno, quase apagados.
    \end{itemize}
  \end{frame}

  %--- Slide: matrizes ortogonais (conceito) ---
  \begin{frame}{Ortogonalizar: transformar a elipse em círculo}
    Uma matriz $\mathbf{O}$ é \textbf{ortogonal} quando $\mathbf{O}^{\top}\mathbf{O}=\mathbf{I}$, ou seja, suas colunas têm norma 1 \alert{e} são ortogonais entre si.
    \vfill
    \centering
    \begin{tikzpicture}[scale=0.72]
      \begin{scope}[shift={(-3.6,0)}]
        \draw[gray!40,->] (-2.3,0)--(2.3,0);
        \draw[gray!40,->] (0,-1.5)--(0,1.5);
        \draw[very thick,BrickRed,rotate=25] (0,0) ellipse (2 and 0.6);
        \draw[->,thick,rotate=25] (0,0) -- node[sloped,above,pos=0.6]{$\sigma_1$} (2,0);
        \draw[->,thick,rotate=25] (0,0) -- (0,0.6) node[above,inner sep=1.5pt]{$\sigma_2$};
        \node[below] at (0,-1.8) {\small $\sigma_1 \gg \sigma_2$};
      \end{scope}
      \draw[->,very thick] (-1.2,0) -- node[above,align=center]{\small ortogonalizar} (1.2,0);
      \begin{scope}[shift={(3.6,0)}]
        \draw[gray!40,->] (-1.5,0)--(1.5,0);
        \draw[gray!40,->] (0,-1.5)--(0,1.5);
        \draw[very thick,NavyBlue] (0,0) circle (1);
        \draw[->,thick,rotate=25] (0,0) -- (1,0);
        \draw[->,thick,rotate=25] (0,0) -- (0,1);
        \node[below] at (0,-1.8) {\small $\sigma_1=\sigma_2=1$};
      \end{scope}
    \end{tikzpicture}
    \vfill
    \raggedright
    \begin{itemize}
      \item Uma matriz ortogonal preserva comprimentos e ângulos: o círculo unitário continua um círculo unitário.
      \item \textbf{Ortogonalizar} $\mathbf{M}$ é trocá-la pela matriz ortogonal mais próxima, mantendo as direções singulares e igualando todos os $\sigma$ a 1.
    \end{itemize}
  \end{frame}

  %--- Slide: ideia central Muon ---
  \begin{frame}{Ideia central: ortogonalizar a atualização}
    \begin{itemize}
      \item As matrizes de gradientes em redes neurais costumam ter \textit{rank efetivo}
        baixo, ou seja, são dominadas por poucas direções singulares.
      \item O passo fica mal aproveitado: direções com $\sigma$ pequeno, mesmo quando úteis
        para o aprendizado, quase não contribuem para a atualização.
      \item Muon substitui a matriz de atualização pela \alert{matriz ortogonal mais próxima},
        redistribuindo a magnitude entre todas as direções singulares.
      \item Geometricamente, o passo do Muon é a \textit{steepest descent} sob a norma
        espectral, enquanto o SGD é a \textit{steepest descent} sob a norma de Frobenius.
    \end{itemize}
  \end{frame}

  %--- Slide: steepest descent depende da norma (formulação) ---
  \begin{frame}{\textit{Steepest descent}: o passo mais íngreme depende da norma}
    Qual atualização $\Delta \mathbf{W}$ mais reduz a loss, dado um tamanho máximo de passo?
    Com $\mathbf{G}=\nabla_{\mathbf{W}}L$, a aproximação linear da loss é
    \[
      L(\mathbf{W}+\Delta \mathbf{W})\;\approx\;L(\mathbf{W})+\langle \mathbf{G},\,\Delta \mathbf{W}\rangle,
      \qquad
      \langle \mathbf{G},\,\Delta \mathbf{W}\rangle=\textstyle\sum_{ij}G_{ij}\,\Delta W_{ij}.
    \]
    O passo mais íngreme minimiza essa aproximação com o tamanho do passo limitado:
    \[
      \Delta \mathbf{W}^{*}=\operatorname*{arg\,min}_{\|\Delta \mathbf{W}\|\le \eta}\;\langle \mathbf{G},\,\Delta \mathbf{W}\rangle.
    \]
    \begin{itemize}
      \item A restrição é essencial: na aproximação linear, sem limite de tamanho o valor
        desceria sem fim.
      \item A solução depende de \alert{qual norma} mede o tamanho do passo. Vamos resolver
        o problema para duas normas: a de \textbf{Frobenius} e a \textbf{espectral}.
    \end{itemize}
  \end{frame}

  %--- Slide: solução sob Frobenius ---
  \begin{frame}{Solução sob a norma de Frobenius: o gradiente}
    A norma de Frobenius trata a matriz como um vetor longo:
    $\|\Delta \mathbf{W}\|_{F}=\sqrt{\sum_{ij}\Delta W_{ij}^{2}}$.
    Pela desigualdade de Cauchy--Schwarz,
    \[
      \langle \mathbf{G},\,\Delta \mathbf{W}\rangle \;\ge\; -\,\|\mathbf{G}\|_{F}\,\|\Delta \mathbf{W}\|_{F} \;\ge\; -\,\eta\,\|\mathbf{G}\|_{F}.
    \]
    A igualdade acontece quando $\Delta \mathbf{W}$ aponta no sentido oposto a $\mathbf{G}$ e usa todo o
    tamanho permitido:
    \[
      \Delta \mathbf{W}^{*}=-\,\eta\,\frac{\mathbf{G}}{\|\mathbf{G}\|_{F}}.
    \]
    \begin{itemize}
      \item O passo mais íngreme sob Frobenius é o \alert{gradiente} (normalizado): a descida
        de gradiente usual mede o tamanho do passo entrada por entrada, sem olhar a
        estrutura da matriz.
    \end{itemize}
  \end{frame}

  %--- Slide: solução sob a norma espectral ---
  \begin{frame}{Solução sob a norma espectral: a ortogonalização}
    A SVD escreve $\mathbf{G}$ como uma soma de matrizes de \textit{rank} 1: cada parcela pega a
    direção singular de entrada $\mathbf{v}_{i}$, leva na direção de saída $\mathbf{u}_{i}$ e estica por $\sigma_{i}$:
    \[
      \mathbf{G}=\sum_{i}\sigma_{i}\,\mathbf{u}_{i}\mathbf{v}_{i}^{\top}
      \qquad\Longrightarrow\qquad
      \langle \mathbf{G},\,\Delta \mathbf{W}\rangle=\sum_{i}\sigma_{i}\,\big(\mathbf{u}_{i}^{\top}\Delta \mathbf{W}\,\mathbf{v}_{i}\big).
    \]
    A norma espectral é o maior fator de esticamento: se $\|\Delta \mathbf{W}\|_{2}=\sigma_{1}(\Delta \mathbf{W})\le\eta$,
    nenhum vetor unitário é esticado além de $\eta$. Cada parcela então obedece
    $\mathbf{u}_{i}^{\top}\Delta \mathbf{W}\,\mathbf{v}_{i}\ge-\eta$, logo
    \[
      \langle \mathbf{G},\,\Delta \mathbf{W}\rangle\;\ge\;-\,\eta\sum_{i}\sigma_{i}.
    \]
    A igualdade exige $\Delta \mathbf{W}\,\mathbf{v}_{i}=-\eta\,\mathbf{u}_{i}$ em \alert{todas} as direções singulares
    ao mesmo tempo:
    \[
      \Delta \mathbf{W}^{*}=-\,\eta\sum_{i}\mathbf{u}_{i}\mathbf{v}_{i}^{\top}=-\,\eta\,\mathbf{U}\mathbf{V}^{\top},
    \]
    uma matriz ortogonal escalada por $\eta$, exatamente a \alert{ortogonalização} de $\mathbf{G}$.
    \vfill
    {\footnotesize\color{gray!50!black}Supomos todos os $\sigma_{i}>0$; $\mathbf{U}$ e $\mathbf{V}$ empilham os
    $\mathbf{u}_{i}$ e $\mathbf{v}_{i}$ como colunas; usamos $\langle \mathbf{u}_{i}\mathbf{v}_{i}^{\top},\,\Delta \mathbf{W}\rangle
    =\mathbf{u}_{i}^{\top}\Delta \mathbf{W}\,\mathbf{v}_{i}$ e $\|\mathbf{u}_{i}\|=\|\mathbf{v}_{i}\|=1$.}
  \end{frame}

  %--- Slide: pseudocódigo Muon ---
  \begin{frame}{Pseudocódigo do Muon}
    \begin{block}{Atualização (matriz $\mathbf{G}$ do gradiente)}
      \begin{algorithmic}[1]
        \State $\mathbf{M}_{t+1} \gets \beta \mathbf{M}_{t} + \mathbf{G}_{t}$ \Comment{momentum (acumulador de gradientes)}
        \State $\widehat{\mathbf{M}}_{t+1} \gets \mathbf{M}_{t+1}/\|\mathbf{M}_{t+1}\|_{F}$ \Comment{normaliza pela norma de Frobenius}
        \State $\mathbf{O}_{t+1} \gets \mathrm{NewtonSchulz5}(\widehat{\mathbf{M}}_{t+1})$ \Comment{aproxima a matriz ortogonal mais próxima}
        \State $\mathbf{W}_{t+1} \gets \mathbf{W}_{t} - \eta\,\mathbf{O}_{t+1}$
      \end{algorithmic}
    \end{block}
    \begin{itemize}
      \item Newton-Schulz aproxima a parte ortogonal de $\widehat{\mathbf{M}}$ com polinômios
        matriciais, sem calcular SVD; com 5 passos, roda bem em \textit{tensor cores} de GPU.
      \item A normalização da linha 2 existe só para trazer os valores singulares à região
        de convergência do Newton-Schulz: a matriz ortogonal mais próxima não muda com a escala.
      \item Quem é ortogonalizado é o momentum $\mathbf{M}_{t+1}$, não o gradiente instantâneo:
        na derivação do passo mais íngreme, $\mathbf{M}_{t+1}$ assume o papel de $\mathbf{G}$.
    \end{itemize}
    {\footnotesize\color{gray!50!black}Na implementação de referência, a normalização fica
    dentro do \texttt{NewtonSchulz5} e o momentum é Nesterov; o passo de uma matriz $m\times n$
    é reescalado por $\sqrt{\max(1,\,m/n)}$, e o Moonlight usa $0.2\sqrt{\max(m,n)}$ para
    casar com o AdamW.}
  \end{frame}

  %--- Slide: exemplo numérico Muon (parte 1) ---
  \begin{frame}{Exemplo numérico: Muon (momentum e normalização)}
    \begin{block}{Cenário: parâmetro 2D, como o Muon exige}
      \[
        \eta=0.1,\quad \beta=0.9,\quad
        \mathbf{W}_t=\begin{bmatrix}3&5\\2&4\end{bmatrix},\quad
        \mathbf{M}_t=\begin{bmatrix}2&-3\\5&2\end{bmatrix},\quad
        \mathbf{G}_t=\begin{bmatrix}1.2&1.7\\2.5&1.2\end{bmatrix}.
      \]
    \end{block}
    Passo 1, momentum (elemento a elemento):
    \[
      \mathbf{M}_{t+1}=0.9\begin{bmatrix}2&-3\\5&2\end{bmatrix}+\begin{bmatrix}1.2&1.7\\2.5&1.2\end{bmatrix}
      =\begin{bmatrix}3&-1\\7&3\end{bmatrix}.
    \]
    Passo 2, normalização pela norma de Frobenius:
    \[
      \|\mathbf{M}_{t+1}\|_{F}=\sqrt{9+1+49+9}=\sqrt{68}\approx 8.25,
      \qquad
      \widehat{\mathbf{M}}_{t+1}\approx\begin{bmatrix}0.36&-0.12\\0.85&0.36\end{bmatrix}.
    \]
    \begin{itemize}
      \item Falta o passo 3, a ortogonalização. Antes dele, vejamos quanto $\mathbf{M}_{t+1}$ estica cada direção singular.
    \end{itemize}
  \end{frame}

  %--- Slide: calculo dos valores singulares ---
  \begin{frame}{Os valores singulares de $\mathbf{M}_{t+1}$}
    Os valores singulares de $\mathbf{M}$ são as raízes quadradas dos autovalores de $\mathbf{M}^{\top}\mathbf{M}$:
    \[
      \mathbf{M}_{t+1}^{\top}\mathbf{M}_{t+1}
      =\begin{bmatrix}3&7\\-1&3\end{bmatrix}\begin{bmatrix}3&-1\\7&3\end{bmatrix}
      =\begin{bmatrix}58&18\\18&10\end{bmatrix}.
    \]
    Com traço e determinante
    \[
      \operatorname{tr}\!\big(\mathbf{M}^{\top}\mathbf{M}\big)=58+10=68,
      \qquad
      \det\!\big(\mathbf{M}^{\top}\mathbf{M}\big)=58\cdot 10-18\cdot 18=580-324=256,
    \]
    a equação característica fica
    \[
      \lambda^{2}-68\lambda+256=0,
      \qquad
      \lambda=\frac{68\pm\sqrt{4624-1024}}{2}=\frac{68\pm 60}{2}.
    \]
    \[
      \lambda_1=64,\quad \lambda_2=4
      \qquad\Longrightarrow\qquad
      \sigma_1=8,\quad \sigma_2=2.
    \]
    \begin{itemize}
      \item Vetores em uma direção singular saem amplificados 4 vezes mais que na outra: é essa assimetria que o Muon vai corrigir.
    \end{itemize}
  \end{frame}

  %--- Slide: exemplo numérico Muon (parte 2) ---
  \begin{frame}{Exemplo numérico: Muon (a tentativa ingênua)}
    As colunas de $\mathbf{M}_{t+1}=\begin{bmatrix}3&-1\\7&3\end{bmatrix}$ não são ortogonais entre si:
    \[
      (3,\,7)\cdot(-1,\,3) = -3+21 = 18 \neq 0.
    \]
    A tentativa ingênua seria normalizar cada coluna separadamente:
    \[
      \frac{(3,\,7)}{\sqrt{58}}=(0.39,\,0.92),
      \qquad
      \frac{(-1,\,3)}{\sqrt{10}}=(-0.32,\,0.95).
    \]
    \begin{alertblock}{Normalizar as colunas não basta}
      Cada coluna ficou com norma 1, mas o produto interno entre elas é $0.75$, ainda longe de zero. A matriz resultante não é ortogonal.
    \end{alertblock}
  \end{frame}

  %--- Slide: forma fechada da ortogonalização 2x2 ---
  \begin{frame}{Ortogonalização em forma fechada para matrizes $2\times 2$}
    Quando $\det \mathbf{M}>0$, a matriz ortogonal mais próxima de $\mathbf{M}$ é uma \alert{rotação}:
    \[
      \mathbf{R}(\phi)=\begin{bmatrix}\cos\phi&-\sin\phi\\ \sin\phi&\cos\phi\end{bmatrix}.
    \]
    Minimizar a distância $\|\mathbf{M}-\mathbf{R}(\phi)\|_{F}^{2}$ equivale a maximizar o produto interno
    \[
      \langle \mathbf{M},\,\mathbf{R}(\phi)\rangle
      =\underbrace{(a+d)}_{p}\cos\phi+\underbrace{(c-b)}_{q}\sin\phi,
      \qquad \mathbf{M}=\begin{bmatrix}a&b\\c&d\end{bmatrix},
    \]
    e o máximo ocorre quando $(\cos\phi,\,\sin\phi)$ aponta na direção do vetor $(p,\,q)$:
    \[
      \mathbf{O}=\frac{1}{\sqrt{p^{2}+q^{2}}}\begin{bmatrix}p&-q\\q&p\end{bmatrix}.
    \]
    {\footnotesize\color{gray!50!black}É o $\mathbf{U}\mathbf{V}^{\top}$ da derivação anterior, calculado sem SVD. Se $\det \mathbf{M}<0$, a ortogonal mais próxima é uma reflexão e a forma muda; em dimensões maiores não há fórmula tão simples, e é aí que entra o Newton-Schulz.}
  \end{frame}

  %--- Slide: exemplo numérico Muon (parte 3) ---
  \begin{frame}{Exemplo numérico: Muon (ortogonalização e atualização)}
    Passo 3, ortogonalização de $\mathbf{M}_{t+1}=\begin{bmatrix}3&-1\\7&3\end{bmatrix}$ pela forma fechada:
    \[
      \det \mathbf{M}_{t+1}=3\cdot 3-(-1)\cdot 7=16>0,\quad
      p=3+3=6,\quad q=7-(-1)=8,\quad \sqrt{p^{2}+q^{2}}=10,
    \]
    \[
      \mathbf{O}_{t+1}=\frac{1}{10}\begin{bmatrix}6&-8\\8&6\end{bmatrix}=\begin{bmatrix}0.6&-0.8\\0.8&0.6\end{bmatrix}.
    \]
    Passo 4, atualização dos pesos:
    \[
      \mathbf{W}_{t+1}=\begin{bmatrix}3&5\\2&4\end{bmatrix}-0.1\begin{bmatrix}0.6&-0.8\\0.8&0.6\end{bmatrix}
      =\begin{bmatrix}2.94&5.08\\1.92&3.94\end{bmatrix}.
    \]
    \begin{itemize}
      \item Os valores singulares passaram de $(8,\,2)$ para $(1,\,1)$: a atualização estica todas as direções singulares igualmente.
      \item Aplicamos a forma fechada a $\mathbf{M}_{t+1}$, não a $\widehat{\mathbf{M}}_{t+1}$: as duas têm a mesma
        ortogonal mais próxima. A normalização do passo 2 só importa para a convergência do Newton-Schulz.
    \end{itemize}
  \end{frame}

  %--- Slide: Adam e Muon no mesmo cenário (passo do Adam) ---
  \begin{frame}{Comparando as atualizações de Adam e Muon}
    \begin{block}{Cenário: mesmo primeiro momento para os dois otimizadores}
      \[
        \epsilon\approx 0,\quad
        \widetilde{\mathbf{m}}_{t+1}=\mathbf{M}_{t+1}=\begin{bmatrix}1&-2\\6&-7\end{bmatrix},\quad
        \widetilde{\mathbf{v}}_{t+1}=\begin{bmatrix}1&4\\36&49\end{bmatrix}.
      \]
    \end{block}
    O Adam normaliza entrada por entrada (aqui $\widetilde{\mathbf{v}}=\widetilde{\mathbf{m}}^{2}$, um histórico de gradientes estável). A direção da sua atualização é
    \[
      \mathbf{D}_{\text{Adam}}=\frac{\widetilde{\mathbf{m}}_{t+1}}{\sqrt{\widetilde{\mathbf{v}}_{t+1}}}
      =\begin{bmatrix}1/1&-2/2\\6/6&-7/7\end{bmatrix}
      =\begin{bmatrix}1&-1\\1&-1\end{bmatrix}.
    \]
    \begin{itemize}
      \item Com histórico estável, o passo do Adam vira o \alert{sinal} do momentum: toda entrada anda com a mesma magnitude, no sentido que seu próprio sinal indica.
      \item As colunas da matriz de atualização são paralelas: o passo tem \textit{rank} 1.
    \end{itemize}
  \end{frame}

  %--- Slide: diretrizes práticas ---
  \begin{frame}{Qual otimizador usar na prática?}
    \begin{itemize}
      \item \textbf{Padrão hoje}: AdamW (Adam com \textit{weight decay} desacoplado) na maioria
        dos modelos.
      \item \textbf{Visão computacional clássica}: SGD com momentum costuma generalizar tão
        bem ou melhor, especialmente com \textit{learning rate schedules} bem ajustados.
      \item \textbf{Modelos grandes (LLMs)}: AdamW continua sendo o padrão; Muon vem
        ganhando adoção para as matrizes das camadas ocultas (Moonlight, Kimi K2).
      \item \textbf{Importante}: a taxa de aprendizado $\eta$ é o hiperparâmetro de maior
        impacto em qualquer um deles.
    \end{itemize}
  \end{frame}
