  \section{Definição e exemplo simples}
  %======================================================================

  %--- Slide: motivação ---
  \begin{frame}{Por que pensar em grafo computacional?}
    \begin{itemize}
      \item \textit{Backpropagation} é a aplicação repetida da regra da cadeia
        em uma função composta.
      \item Funções complicadas (uma rede profunda inteira) podem ser
        decompostas em operações elementares.
      \item Representar essa decomposição como um \alert{grafo} torna
        explícito o caminho que cada gradiente local precisa percorrer.
      \item Mesma estrutura permite ao computador fazer a derivada
        \alert{automaticamente}, ideia central de \textit{frameworks}
        como PyTorch e JAX.
    \end{itemize}
  \end{frame}

  %--- Slide: definição de grafo computacional ---
  \begin{frame}{Definição}
    \begin{itemize}
      \item Um \alert{grafo computacional} é um DAG (\textit{directed acyclic graph})
        em que:
        \begin{itemize}
          \item nodos-folha representam \textbf{variáveis} (entradas, parâmetros);
          \item nodos internos representam \textbf{operações} elementares (soma,
            produto, $\exp$, ReLU, $\dots$);
          \item arestas indicam o fluxo de dados, das entradas em direção à
            saída final.
        \end{itemize}
      \item Avaliar a função é fazer um \alert{forward pass}: percorrer o grafo
        em ordem topológica.
      \item Calcular gradientes em relação a cada folha é fazer um
        \alert{backward pass}: percorrer o grafo na ordem inversa, aplicando
        a regra da cadeia.
    \end{itemize}
  \end{frame}

  \section{Forward e backward genéricos}
  %======================================================================

  %--- Slide: nodo genérico forward ---
  \begin{frame}{Um nodo genérico no \textit{forward}}
    \centering
    \begin{tikzpicture}[node distance=1.0cm and 1.6cm]
      \node[cgvar] (x) {$x$};
      \node[cgvar, below=of x] (y) {$y$};
      \node[cgop, right=of x, yshift=-0.6cm] (f) {$f$};
      \node[right=of f, xshift=0.2cm] (z) {$z = f(x,y)$};

      \draw[cgedge] (x) -- (f);
      \draw[cgedge] (y) -- (f);
      \draw[cgedge] (f) -- (z);
    \end{tikzpicture}
    \begin{itemize}
      \item No \textit{forward}, cada nodo recebe valores de entrada e calcula
        sua saída local.
      \item As saídas alimentam os nodos seguintes na ordem topológica.
      \item No final do grafo temos o valor da função, tipicamente uma
        \textit{loss} escalar $L$.
    \end{itemize}
  \end{frame}

  %--- Slide: nodo genérico backward ---
  \begin{frame}{O mesmo nodo no \textit{backward}}
    \centering
    \begin{tikzpicture}[node distance=1.6cm and 4.2cm]
      \node[cgvar] (x) {$x$};
      \node[cgvar, below=of x] (y) {$y$};
      \node[cgop, right=of x, yshift=-0.8cm] (f) {$f$};
      \node[right=of f, xshift=0.6cm] (z) {$z$};

      \draw[cgedge] (x) -- (f);
      \draw[cgedge] (y) -- (f);
      \draw[cgedge] (f) -- (z);

      \node[cggrad, below right=0.1cm and 0.1cm of x] {$\dfrac{\partial L}{\partial x} = \dfrac{\partial L}{\partial z}\cdot\dfrac{\partial z}{\partial x}$};
      \node[cggrad, below right=0.1cm and 0.1cm of y] {$\dfrac{\partial L}{\partial y} = \dfrac{\partial L}{\partial z}\cdot\dfrac{\partial z}{\partial y}$};
      \node[cggrad, below right=0.05cm and 0.05cm of z] {$\dfrac{\partial L}{\partial z}$};
    \end{tikzpicture}
    \begin{itemize}
      \item Cada aresta no sentido oposto carrega o \alert{gradiente
        \textit{upstream}} $\partial L / \partial z$.
      \item O nodo multiplica esse gradiente pela \alert{derivada local}
        $\partial z / \partial x$ (regra da cadeia) e envia para a aresta de
        entrada.
      \item Se uma variável aparece em múltiplos nodos, os gradientes que chegam
        nela são \textbf{somados}.
    \end{itemize}
  \end{frame}

  %--- Slide: padrões de nodos ---
  \begin{frame}{Padrões comuns de nodos}
    \begin{columns}
      \begin{column}{.6\linewidth}
        \begin{itemize}
          \item \textbf{Soma} ($+$): \textit{distribuidor}, copia o gradiente
            \textit{upstream} para todas as entradas.
          \item \textbf{Produto} ($\times$): \textit{comutador}, multiplica o
            gradiente pelo valor da \textit{outra} entrada.
          \item \textbf{Máximo} ($\max$): \textit{roteador}, envia o gradiente
            inteiro para a entrada que venceu, zero para as demais.
          \item \textbf{Cópia} (uma variável usada duas vezes): \textit{somador}
            no backward, soma os gradientes que chegam.
        \end{itemize}
      \end{column}
      \begin{column}{.4\linewidth}
        \centering
        \begin{tikzpicture}[node distance=0.7cm and 1.0cm, font=\scriptsize]
          \node[cgop] (p) {$+$};
          \node[left=of p, yshift=0.55cm] (pa) {$a$};
          \node[left=of p, yshift=-0.55cm] (pb) {$b$};
          \node[right=of p] (po) {};
          \draw[cgedge] (pa) -- (p);
          \draw[cgedge] (pb) -- (p);
          \draw[cgedge] (p) -- (po);
          \node[cggrad, below right=0.02cm and 0.02cm of p] {$g$};
          \node[cggrad, below right=0.0cm and 0.02cm of pa] {$g$};
          \node[cggrad, below right=0.0cm and 0.02cm of pb] {$g$};

          \node[cgop, below=1.6cm of p] (m) {$\times$};
          \node[left=of m, yshift=0.55cm] (ma) {$a$};
          \node[left=of m, yshift=-0.55cm] (mb) {$b$};
          \node[right=of m] (mo) {};
          \draw[cgedge] (ma) -- (m);
          \draw[cgedge] (mb) -- (m);
          \draw[cgedge] (m) -- (mo);
          \node[cggrad, below right=0.02cm and 0.02cm of m] {$g$};
          \node[cggrad, below right=0.0cm and 0.02cm of ma] {$g\,b$};
          \node[cggrad, below right=0.0cm and 0.02cm of mb] {$g\,a$};
        \end{tikzpicture}
      \end{column}
    \end{columns}
  \end{frame}

  %--- Slide: agrupando em sigmoide ---
  \begin{frame}{Agrupando nodos: a sigmoide}
    \begin{columns}
      \begin{column}{.55\linewidth}
        \begin{itemize}
          \item A sequência $\times{-}1 \to \exp \to +1 \to 1/u$
            implementa $\sigma(z) = 1/(1+e^{-z})$.
          \item Sua derivada tem forma fechada:
            $$\sigma'(z) = \sigma(z)\,(1 - \sigma(z)).$$
          \item Substituir os 4 nodos por um único nodo \textit{sigmoide}
            simplifica o grafo, evita instabilidades numéricas e cobra apenas
            uma multiplicação no \textit{backward}.
          \item Granularidade fina (flexível) versus grossa (eficiente) é
            uma decisão recorrente em \textit{frameworks} de \textit{deep learning}.
        \end{itemize}
      \end{column}
      \begin{column}{.45\linewidth}
        \centering
        \begin{tikzpicture}[node distance=1.0cm and 1.4cm, font=\small]
          \node[cgvar] (z) {$z$};
          \node[cgop, right=of z] (sig) {$\sigma$};
          \node[right=of sig, xshift=0.1cm] (out) {$\sigma(z)$};

          \draw[cgedge] (z) -- (sig);
          \draw[cgedge] (sig) -- (out);

          \node[cggrad, below right=0.1cm and 0.05cm of z] {$\sigma(1{-}\sigma)\cdot g$};
          \node[cggrad, below right=0.05cm and 0.05cm of sig] {$g$};
        \end{tikzpicture}\\
        \footnotesize \textit{Forward} entrega $\sigma(z)$;
        \textit{backward} multiplica o gradiente \textit{upstream} por
        $\sigma(1{-}\sigma)$.
      \end{column}
    \end{columns}
  \end{frame}

  \section{O MLP como grafo computacional}
  %======================================================================

  %--- Slide: nodo de produto de matrizes ---
  \begin{frame}{Nodos vetorizados: produto de matrizes}
    \begin{columns}
      \begin{column}{.58\linewidth}
        \begin{itemize}
          \item No MLP vetorizado, a pré-ativação de cada camada é um produto
            de matrizes: $\mathbf{Z} = \mathbf{A}\,\mathbf{\Theta}^{\top}$, com
            $\mathbf{A} \in \mathbb{R}^{N\times d_e}$ (entradas da camada),
            $\mathbf{\Theta} \in \mathbb{R}^{d_s\times d_e}$ (pesos) e
            $\mathbf{Z} \in \mathbb{R}^{N\times d_s}$.
          \item No \textit{backward}, o gradiente \textit{upstream} é
            $\mathbf{G} = \partial J/\partial \mathbf{Z}$, com a mesma forma de $\mathbf{Z}$:
            \[
              \frac{\partial J}{\partial \mathbf{A}} = \mathbf{G}\,\mathbf{\Theta}
              \qquad
              \frac{\partial J}{\partial \mathbf{\Theta}} = \mathbf{G}^{\top}\mathbf{A}
            \]
          \item Mesma regra do nodo $\times$ escalar: cada entrada
            recebe o gradiente multiplicado pela \textit{outra} entrada.
          \item Confira as dimensões: $\mathbf{G}\,\mathbf{\Theta}$ é $N\times d_e$ (forma de $\mathbf{A}$)
            e $\mathbf{G}^{\top}\mathbf{A}$ é $d_s\times d_e$ (forma de $\mathbf{\Theta}$).
        \end{itemize}
      \end{column}
      \begin{column}{.42\linewidth}
        \centering
        \begin{tikzpicture}[node distance=1.4cm and 1.6cm]
          \node[cgvar] (A) {$\mathbf{A}$};
          \node[cgvar, below=of A] (T) {$\mathbf{\Theta}$};
          \node[cgopw, right=of A, yshift=-1.1cm] (mm) {\textit{matmul}};
          \node[right=of mm, xshift=-0.4cm] (Z) {$\mathbf{Z}$};
          \draw[cgedge] (A) -- (mm);
          \draw[cgedge] (T) -- (mm);
          \draw[cgedge] (mm) -- (Z);
          \node[cggrad, below right=0.05cm and 0.05cm of mm] {$\mathbf{G}$};
          \node[cggrad, below right=0.0cm and 0.05cm of A] {$\mathbf{G}\,\mathbf{\Theta}$};
          \node[cggrad, below right=0.0cm and 0.05cm of T] {$\mathbf{G}^{\top}\mathbf{A}$};
        \end{tikzpicture}
      \end{column}
    \end{columns}
  \end{frame}

  %--- Slide: nodo de ativação elemento a elemento ---
  \begin{frame}{Nodos vetorizados: ativação elemento a elemento}
    \begin{columns}
      \begin{column}{.58\linewidth}
        \begin{itemize}
          \item O nodo de ativação aplica $\sigma$ em cada posição da
            matriz: $\mathbf{A} = \sigma(\mathbf{Z})$, e $\mathbf{A}$ tem a mesma forma de $\mathbf{Z}$.
          \item Como o nodo não mistura posições, o \textit{backward}
            é a regra escalar aplicada elemento a elemento:
            \[
              \frac{\partial J}{\partial \mathbf{Z}} = \mathbf{G} \odot \sigma'(\mathbf{Z})
              \qquad
              \sigma'(\mathbf{Z}) = \mathbf{A} \odot (1 - \mathbf{A})
            \]
          \item $\odot$ é o produto de Hadamard (elemento a elemento), o mesmo
            da aula de \textit{backpropagation}.
          \item Como no nodo sigmoide escalar, $\sigma'$ reaproveita a
            saída $\mathbf{A}$ já calculada no \textit{forward}.
        \end{itemize}
      \end{column}
      \begin{column}{.42\linewidth}
        \centering
        \begin{tikzpicture}[node distance=1.0cm and 1.6cm]
          \node[cgvar] (Z) {$\mathbf{Z}$};
          \node[cgop, right=of Z] (sig) {$\sigma$};
          \node[right=of sig, xshift=-0.4cm] (A) {$\mathbf{A}$};
          \draw[cgedge] (Z) -- (sig);
          \draw[cgedge] (sig) -- (A);
          \node[cggrad, below right=0.05cm and 0.05cm of sig] {$\mathbf{G}$};
          \node[cggrad, below right=0.05cm and 0.05cm of Z] {$\mathbf{G} \odot \sigma'(\mathbf{Z})$};
        \end{tikzpicture}
      \end{column}
    \end{columns}
  \end{frame}

  %--- Slide: nodo de função de custo ---
  \begin{frame}{Nodos vetorizados: função de custo}
    \begin{columns}
      \begin{column}{.58\linewidth}
        \begin{itemize}
          \item O último nodo compara a saída da rede com o alvo e
            reduz tudo a um escalar (\textit{half SSE} da aula anterior):
            \[
              J = \frac{1}{2}\sum_{i}\left(y_i - \hat{y}_i\right)^{2}
              \qquad
              \frac{\partial J}{\partial \hat{\mathbf{y}}} = \hat{\mathbf{y}} - \mathbf{y}
            \]
          \item O \textit{backward} começa nesse escalar, com
            $\partial J/\partial J = 1$; o gradiente devolvido tem a mesma
            forma de $\hat{\mathbf{y}}$ (o fator $\tfrac{1}{2}$ cancela o $2$ da
            derivada).
          \item $\mathbf{y}$ é dado do problema: não é treinável, então nenhum gradiente

  %--- Slide: fechamento ---
  \begin{frame}{Redes neurais são implementadas como Grafos}
    \begin{itemize}
      \item Podemos decompor funções em nodos com \textit{forward} e
        derivada local: nodos escalares ($+$, $\times$, $\exp$, $1/u$), o nodo
        sigmoide e os nodos vetorizados (\textit{matmul}, ativação,
        \textit{loss}).
      \item Todas as arquiteturas que veremos no curso são decompostas da
        mesma forma
        
       \item Se cada nodo conhece seu \textit{forward} e sua derivada local, o \textit{backpropagation} no grafo computacional treina qualquer arquitetura, por mais complexa que pareça.
    \end{itemize}
    
    
  \end{frame}
