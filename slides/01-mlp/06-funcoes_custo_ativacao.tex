  %--- Slide: onde aplicamos ---
  \begin{frame}{Camadas ocultas vs camada de saída}
    \begin{itemize}
      \item Em \textbf{camadas ocultas}, $\varphi$ adiciona não-linearidade: sem ela, qualquer
        rede colapsa em uma transformação afim.
      \item Em \textbf{camada de saída}, $\varphi$ é escolhida conforme a tarefa (regressão,
        classificação binária, multiclasse, multilabel).
      \item As ativações de saída interagem diretamente com a função de custo.
    \end{itemize}
  \end{frame}

  %--- Slide: requisitos ---
  \begin{frame}{O que esperamos de uma função de ativação}
    \begin{itemize}
      \item Contínua e diferenciável (quase em todo lugar): a descida de gradiente exige $\varphi'$.
      \item $\varphi'(z)$ não nula em uma faixa ampla, para que o gradiente não se anule.
      \item Computação barata: avaliada milhões de vezes por iteração.
      \item Idealmente sem saturação severa: gradientes preservam ordem de magnitude entre camadas.
    \end{itemize}
  \end{frame}

  %--- Slide: Sigmoid ---
  \begin{frame}{Sigmoid}
    \begin{columns}
      \begin{column}{.45\linewidth}
        \[
          \sigma(z) = \frac{1}{1+e^{-z}}
        \]
        \[
          \sigma'(z) = \sigma(z)\,(1-\sigma(z))
        \]
        \begin{itemize}
          \item Imagem em $(0,1)$.
          \item Derivada máxima $0.25$ em $z=0$.
          \item Histórica em redes rasas; saturação atrapalha em redes profundas.
        \end{itemize}
      \end{column}
      \begin{column}{.55\linewidth}
        \centering
        \resizebox{\linewidth}{!}{
          \begin{tikzpicture}
            \begin{axis}[
              width=8cm,height=5cm,
              xlabel={$z$},
              axis lines=middle,
              domain=-6:6, samples=120,
              ymin=-0.05, ymax=1.1,
              legend style={font=\tiny,at={(1.05,1)},anchor=north west}
            ]
              \addplot[thick,blue]   {1/(1+exp(-x))};
              \addlegendentry{$\sigma(z)$}
              \addplot[thick,Green!70!black,dashed] {exp(-x)/((1+exp(-x))^2)};
              \addlegendentry{$\sigma'(z)$}
            \end{axis}
          \end{tikzpicture}
        }
      \end{column}
    \end{columns}
  \end{frame}

  %--- Slide: tanh ---
  \begin{frame}{Tangente hiperbólica}
    \begin{columns}
      \begin{column}{.45\linewidth}
        \[
          \tanh(z) = \frac{e^{z}-e^{-z}}{e^{z}+e^{-z}}
        \]
        \[
          \tanh'(z) = 1 - \tanh^2(z)
        \]
        \begin{itemize}
          \item Imagem em $(-1,1)$, centrada em zero.
          \item Função ímpar: $\tanh(-z)=-\tanh(z)$.
          \item Sofre do mesmo problema de saturação.
        \end{itemize}
      \end{column}
      \begin{column}{.55\linewidth}
        \centering
        \resizebox{\linewidth}{!}{
          \begin{tikzpicture}
            \begin{axis}[
              width=8cm,height=5cm,
              xlabel={$z$},

              ymin=-1.1, ymax=1.1,
              legend style={font=\tiny,at={(1.05,1)},anchor=north west}

              \addplot[thick,blue]    {tanh(x)};
              \addlegendentry{$\tanh(z)$}

              \addlegendentry{$\tanh'(z)$}
            \end{axis}
          \end{tikzpicture}
        }
      \end{column}
    \end{columns}
  \end{frame}

  %--- Slide: saturação e dissipação de gradiente ---
  \begin{frame}{Saturação: dissipação de gradiente}
    \begin{columns}
      \begin{column}{.55\linewidth}
        \begin{itemize}
          \item Para $|z|$ grande, $\sigma'(z)\to 0$ e $\tanh'(z)\to 0$.
          \item A regra da cadeia em uma rede profunda multiplica essas derivadas
            camada a camada.
          \item Resultado: gradientes vão a zero, camadas iniciais quase não aprendem.
          \item Esse fenômeno é o \textit{vanishing gradient problem}.
        \end{itemize}
      \end{column}
      \begin{column}{.45\linewidth}
        \centering
        \resizebox{\linewidth}{!}{
          \begin{tikzpicture}
            \begin{axis}[
              width=7cm,height=4.5cm,
              xlabel={$z$}, ylabel={$\varphi'(z)$},
              ylabel style={at={(axis description cs:0.5,1.02)},anchor=south},
              axis lines=middle,
              domain=-6:6, samples=120,
              ymin=0, ymax=1.15,
              legend style={font=\tiny,at={(1.05,1)},anchor=north west}
            ]
              \addplot[thick,blue]    {exp(-x)/((1+exp(-x))^2)};
              \addlegendentry{$\sigma'(z)$}

              \addlegendentry{$\tanh'(z)$}
            \end{axis}
          \end{tikzpicture}
        }
      \end{column}
    \end{columns}
  \end{frame}

  %--- Slide: ReLU ---
  \begin{frame}{ReLU\footfullcite{nair2010relu}}
    \begin{columns}
      \begin{column}{.45\linewidth}
        \[
          \mathrm{ReLU}(z) = \max(0,z)
        \]
        \[
          \mathrm{ReLU}'(z) =
            \begin{cases}
              0 & z < 0 \\
              1 & z > 0
            \end{cases}
        \]
        \begin{itemize}
          \item \textit{Rectified Linear Unit}.
          \item Não satura no lado positivo: gradiente preservado.
          \item Esparsidade natural (zera o lado negativo).
          \item Padrão em redes profundas modernas.
        \end{itemize}
      \end{column}
      \begin{column}{.55\linewidth}
        \centering
        \resizebox{\linewidth}{!}{
          \begin{tikzpicture}
            \begin{axis}[
              width=8cm,height=5cm,
              xlabel={$z$},
              axis lines=middle,
              domain=-3:3, samples=120,
              ymin=-0.3, ymax=3.1,
              legend style={font=\tiny,at={(1.05,1)},anchor=north west}
            ]
              \addplot[thick,blue]              {max(0,x)};
              \addlegendentry{$\mathrm{ReLU}(z)$}
              \addplot[thick,Green!70!black,dashed,samples=2,domain=-3:0]   {0};
              \addplot[thick,Green!70!black,dashed,samples=2,domain=0.0001:3] {1};
              \addlegendentry{$\mathrm{ReLU}'(z)$}
            \end{axis}
          \end{tikzpicture}
        }
      \end{column}
    \end{columns}
  \end{frame}

  %--- Slide: dying ReLU ---
  \begin{frame}{O problema do \textit{dying ReLU}}
    \begin{itemize}
      \item Se uma unidade tem $z<0$ para todos os exemplos do dataset, $\mathrm{ReLU}'(z)=0$
        sempre.
      \item Sem gradiente passando, os pesos não atualizam: a unidade fica \alert{morta}.
      \item Em redes muito profundas, é comum uma fração significativa das unidades estar morta
        em algum momento do treinamento.
      \item Causas usuais: taxa de aprendizado muito alta, inicialização ruim.
    \end{itemize}
  \end{frame}

  %--- Slide: Leaky ReLU / PReLU ---
  \begin{frame}{Leaky ReLU\footfullcite{maas2013leaky} e PReLU\footfullcite{he2015delving}}
    \begin{columns}
      \begin{column}{.45\linewidth}
        \[
          \varphi(z)=\max(\alpha z,\, z),\quad \alpha\in(0,1)
        \]
        \[
          \varphi'(z) =
            \begin{cases}
              \alpha & z<0 \\
              1 & z>0
            \end{cases}
        \]
        \begin{itemize}
          \item Leaky ReLU: $\alpha$ fixo (tipicamente $0.01$).
          \item PReLU: $\alpha$ é treinável.
          \item Mata o problema do \textit{dying ReLU} ao manter gradiente $\alpha$ no lado negativo.
        \end{itemize}
      \end{column}
      \begin{column}{.55\linewidth}
        \centering
        \resizebox{\linewidth}{!}{
          \begin{tikzpicture}
            \begin{axis}[
              width=8cm,height=5cm,

              ymin=-0.7, ymax=3.1,
              legend style={font=\tiny,at={(1.05,1)},anchor=north west}

              \addplot[thick,blue]      {max(0.1*x, x)};

              \addplot[thick,red,dashed]    {max(0.01*x, x)};
              \addlegendentry{Leaky ReLU ($\alpha{=}0.01$)}

      \end{column}
    \end{columns}
  \end{frame}

  %--- Slide: ELU ---
  \begin{frame}{ELU\footfullcite{clevert2016elu}}
    \begin{columns}
      \begin{column}{.45\linewidth}
        \[
          \varphi(z) =
            \begin{cases}
              z & z\ge 0 \\
              \alpha(e^{z}-1) & z<0
            \end{cases}
        \]
        \[
          \varphi'(z) =
            \begin{cases}
              1 & z>0 \\
              \alpha\,e^{z} & z<0
            \end{cases}
        \]
        \begin{itemize}
          \item \textit{Exponential Linear Unit}.
          \item Suave em torno de 0; sai do regime de unidade morta exponencialmente.
          \item Empurra a média das ativações para perto de zero.
        \end{itemize}
      \end{column}
      \begin{column}{.55\linewidth}
        \centering
        \resizebox{\linewidth}{!}{
          \begin{tikzpicture}
            \begin{axis}[
              width=8cm,height=5cm,
              xlabel={$z$},
              axis lines=middle,
              domain=-4:3, samples=120,
              ymin=-1.2, ymax=3.1,
              legend style={font=\tiny,at={(1.05,1)},anchor=north west}
            ]
              \addplot[thick,blue,domain=-4:0]   {1*(exp(x)-1)};
              \addplot[thick,blue,domain=0:3]    {x};
              \addlegendentry{ELU ($\alpha{=}1$)}
            \end{axis}
          \end{tikzpicture}
        }
      \end{column}
    \end{columns}
  \end{frame}

  %--- Slide: SELU ---
  \begin{frame}{SELU\footfullcite{klambauer2017selu}}
    \begin{columns}
      \begin{column}{.5\linewidth}
        \[
          \varphi(z) = \lambda
            \begin{cases}
              z & z\ge 0 \\
              \alpha(e^{z}-1) & z<0
            \end{cases}
        \]
        \begin{itemize}
          \item \textit{Scaled ELU}: ELU multiplicada por uma constante $\lambda$.
          \item Constantes $\alpha\approx 1.6733$ e $\lambda\approx 1.0507$ são escolhidas
            para que a estatística das ativações se auto-normalize ao longo das camadas.
          \item Útil quando não usamos \textit{batch norm}.
        \end{itemize}
      \end{column}
      \begin{column}{.5\linewidth}
        \centering
        \resizebox{\linewidth}{!}{
          \begin{tikzpicture}
            \begin{axis}[
              width=8cm,height=5cm,
              xlabel={$z$},
              axis lines=middle,
              domain=-4:3, samples=120,
              ymin=-1.9, ymax=3.4,
              legend style={font=\tiny,at={(1.05,1)},anchor=north west}
            ]
              \addplot[thick,blue,domain=-4:0]   {1.0507*1.6733*(exp(x)-1)};
              \addplot[thick,blue,domain=0:3]    {1.0507*x};
              \addlegendentry{SELU}
            \end{axis}
          \end{tikzpicture}
        }
      \end{column}
    \end{columns}
  \end{frame}

  %--- Slide: Swish ---
  \begin{frame}{Swish\footfullcite{ramachandran2017swish}}
    \begin{columns}
      \begin{column}{.45\linewidth}
        \[
          \varphi(z) = z\,\sigma(\beta z)
        \]
        \begin{itemize}
          \item Não monotônica: passa por um pequeno mínimo negativo antes de subir.
          \item Diferenciável em todo o domínio.
          \item Limite $\beta\to 0$: $\varphi(z)\approx z/2$ (linear).
          \item Limite $\beta\to\infty$: $\varphi(z)\to\mathrm{ReLU}(z)$.
          \item Com $\beta=1$ é a \textbf{SiLU} (\textit{Sigmoid Linear Unit}), \texttt{nn.SiLU} no PyTorch.
        \end{itemize}
      \end{column}
      \begin{column}{.55\linewidth}
        \centering
        \resizebox{\linewidth}{!}{
          \begin{tikzpicture}
            \begin{axis}[
              width=8cm,height=5cm,
              xlabel={$z$},
              axis lines=middle,
              domain=-5:5, samples=120,
              ymin=-1.5, ymax=5.1,
              legend style={font=\tiny,at={(1.05,1)},anchor=north west}
            ]
              \addplot[thick,blue]              {x/(1+exp(-1*x))};
              \addlegendentry{$\beta=1$}
              \addplot[thick,red,dashed]    {x/(1+exp(-10*x))};
              \addlegendentry{$\beta=10$}
              \addplot[thick,Green!70!black,dotted] {x/(1+exp(-0.1*x))};
              \addlegendentry{$\beta=0.1$}
            \end{axis}
          \end{tikzpicture}
        }
      \end{column}
    \end{columns}
  \end{frame}

  %--- Slide: GELU ---
  \begin{frame}{GELU\footfullcite{hendrycks2016gelu}}
    \begin{columns}
      \begin{column}{.5\linewidth}
        \[
          \varphi(z) = z\,\Phi(z),\qquad \Phi(z)=\Pr(Z\le z)
        \]
        \begin{itemize}
          \item \textit{Gaussian Error Linear Unit}.
          \item Multiplica $z$ pela probabilidade acumulada gaussiana $\Phi(z)$,
            com $Z\sim\mathcal{N}(0,1)$.
          \item Aproximação prática: $\varphi(z)\approx z\,\sigma(1.702\,z)$ (próxima de Swish).
          \item Padrão em alguns \textit{transformers}(BERT, GPT-2/3, ViT).
        \end{itemize}
      \end{column}
      \begin{column}{.5\linewidth}
        \centering
        \resizebox{\linewidth}{!}{
          \begin{tikzpicture}
            \begin{axis}[
              width=8cm,height=5cm,
              xlabel={$z$},
              axis lines=middle,
              domain=-4:4, samples=120,
              ymin=-0.8, ymax=4.1,
              legend style={font=\tiny,at={(1.05,1)},anchor=north west}

            \end{axis}
          \end{tikzpicture}

      \end{column}
    \end{columns}
  \end{frame}

  %--- Slide: SwiGLU ---
  \begin{frame}{SwiGLU\footfullcite{shazeer2020swiglu}}
    \begin{itemize}
      \item Não é uma ativação ponto-a-ponto: combina duas projeções lineares da mesma entrada
        com \textit{gating}.
      \[
        \mathrm{SwiGLU}(\mathbf{x})
          = \mathrm{Swish}(\mathbf{x}\mathbf{W}_1)\;\odot\;(\mathbf{x}\mathbf{W}_3)
      \]
      \item $\odot$ é o produto elemento a elemento: o ramo $\mathbf{x}\mathbf{W}_3$ funciona como
        um \textit{portão} que decide quanto de cada dimensão do ramo ativado passa adiante.
      \item Um portão em $0$ desliga a dimensão; um portão negativo inverte o sinal.
        A rede aprende \textit{quais} dimensões ativar para cada entrada.
      \item Substitui o bloco \textit{feed-forward} de arquiteturas modernas como
        LLaMA, Mistral e PaLM, seguido de uma projeção de saída $\mathbf{W}_2$.
      \item Empiricamente supera ReLU e GELU com o mesmo orçamento de parâmetros.
    \end{itemize}
  \end{frame}

        }
      \end{column}
    \end{columns}
  \end{frame}

  %--- Slide: orientação prática ---
  \begin{frame}{Qual usar em uma camada oculta?}
    \begin{itemize}
      \item \textbf{Default moderno}: ReLU para CNNs e MLPs simples; GELU para
        \textit{transformers}.
      \item \textbf{Quando precisa de saída centrada em zero}: tanh em redes recorrentes
        clássicas.
      \item \textbf{Quando o \textit{dying ReLU} aparece}: trocar para Leaky ReLU ou ELU.
      \item \textbf{Em LLMs grandes}: blocos FFN passam a usar SwiGLU.
      \item \textbf{Sigmoid em camada oculta}: praticamente nunca em redes profundas.
    \end{itemize}
  \end{frame}

  % Bloco: ativações na camada de saída
  %======================================================================

  %--- Slide: visão geral ---
  \begin{frame}{A camada de saída define a interface com a tarefa}
    \begin{itemize}
      \item Regressão real: queremos $\hat y\in\mathbb{R}$ → \textbf{Linear}.
      \item Classificação binária: queremos $\hat y\in[0,1]$ (probabilidade) → \textbf{Sigmoid}.
      \item Multilabel: $K$ rótulos independentes, cada um com probabilidade própria
        → \textbf{$K$ unidades Sigmoid}.
      \item Multiclasse exclusiva: $K$ classes mutuamente exclusivas, distribuição categórica
        → \textbf{Softmax} (atua sobre a camada inteira).
    \end{itemize}
  \end{frame}

  %--- Slide: Sigmoid na saída ---
  \begin{frame}{Sigmoid na camada de saída}
    \begin{itemize}
      \item Combinada com BCE (binary cross-entropy) na função de custo.
      \item Domínio do alvo: $y\in\{0,1\}$.
      \item Saída: $\hat y = \sigma(z) = \Pr(y=1\mid \mathbf{x})$.
      \item Para multilabel basta uma sigmoid por classe; cada uma se comporta como
        um classificador binário independente.
    \end{itemize}
  \end{frame}

  %--- Slide: Softmax intro ---
  \begin{frame}{Softmax: ativação que age na camada inteira}
    \begin{columns}
      \begin{column}{.55\linewidth}
        \[
          \mathrm{softmax}(\mathbf{z})_i = \frac{e^{z_i}}{\sum_{j=1}^{K} e^{z_j}}
        \]
        \begin{itemize}
          \item Cada saída $i$ depende de todas as pré-ativações, não só de $z_i$.
          \item Saídas são positivas e somam 1: distribuição de probabilidade.
          \item Combina com a \textit{categorical cross-entropy} na função de custo.
        \end{itemize}
      \end{column}
      \begin{column}{.45\linewidth}
        \centering
        \textbf{Exemplo} ($K=3$):
        \[
          \mathrm{softmax}\!\begin{pmatrix}1.8\\0.7\\-6.0\end{pmatrix}
          \approx
          \begin{pmatrix}0.7500\\0.2497\\0.0003\end{pmatrix}
        \]
        {\footnotesize O logit mais alto domina, mas nenhuma saída é exatamente zero.}
      \end{column}
    \end{columns}
  \end{frame}

  \section{Funções de Custo}
  %======================================================================

  %--- Slide: mudança de paradigma ---
  \begin{frame}{Mudança de paradigma}
    \begin{itemize}
      \item Até aqui: a rede $f(\mathbf{x};\thetav)$ produz uma saída $\hat y$.
      \item A partir de agora: a rede produz \alert{parâmetros $\bm{\eta}$ de uma distribuição
        de probabilidade} sobre $y$.
      \item A função de custo mede o quão bem essa distribuição predita explica os dados observados.
      \item Esse é o ponto de partida da família de \textit{maximum likelihood estimators}.
    \end{itemize}
  \end{frame}

  %--- Slide: visualização da Pr(y|x) - regressão ---
  \begin{frame}{$\Pr(y\mid x)$ em uma tarefa contínua}
    \begin{columns}
      \begin{column}{.45\linewidth}
        \begin{itemize}
          \item Tarefa de regressão: $y\in\mathbb{R}$.
          \item Para cada $x$, a rede estima a média de uma gaussiana sobre $y$.
          \item As gaussianas verticais à direita ilustram $\Pr(y\mid x{=}2)$ e $\Pr(y\mid x{=}7)$.
        \end{itemize}
      \end{column}
      \begin{column}{.55\linewidth}

      \end{column}
    \end{columns}
  \end{frame}

  \begin{frame}{$\Pr(y\mid x)$ em uma tarefa discreta}

        \begin{itemize}
          \item Tarefa de classificação multiclasse: $y\in\{1,2,3,4\}$.
          \item Para cada $x$, a rede estima as probabilidades de cada classe.
          \item Distribuição categórica (Bernoulli generalizada).

  %--- Slide: MLE ---
  \begin{frame}{Estimador de máxima verossimilhança}
    \[
      \hat\thetav = \argmax_{\thetav}\;\prod_{i=1}^{N}\Pr\!\left(y^{(i)} \;\big|\; f(\mathbf{x}^{(i)};\thetav)\right)
    \]
    \vfill
    \begin{itemize}
      \item Maximiza a probabilidade conjunta dos rótulos observados, dada a entrada.
      \item Hipóteses implícitas:
        \begin{itemize}
          \item As amostras $(\mathbf{x}^{(i)}, y^{(i)})$ são \textbf{identicamente distribuídas}.
          \item São \textbf{independentes} entre si (justifica o produto).
        \end{itemize}
      \item Em uma frase: \textit{i.i.d.}
    \end{itemize}
  \end{frame}

  %--- Slide: log monotonia ---
  \begin{frame}{Por que aplicar log à verossimilhança?}
    \begin{columns}
      \begin{column}{.4\linewidth}
        \begin{itemize}
          \item $\log$ é monotônica crescente: preserva o ponto de máximo.
          \item Transforma produto em soma, que é estável numericamente.
          \item Para distribuições discretas, cada termo $\log\Pr(\cdot)\le 0$.
        \end{itemize}
      \end{column}
      \begin{column}{.6\linewidth}
        \centering
        \includegraphics[width=\linewidth]{slides/img/01-mlp/al-log-monotonic.png}
      \end{column}
    \end{columns}
  \end{frame}

  %--- Slide: log-verossimilhança ---
  \begin{frame}{Log-verossimilhança}
    \begin{align*}
      \hat\thetav
        &= \argmax_{\thetav}\;\prod_{i=1}^{N}\Pr\!\left(y^{(i)} \mid f(\mathbf{x}^{(i)};\thetav)\right) \\
        &= \argmax_{\thetav}\;\log\!\left[\prod_{i=1}^{N}\Pr\!\left(y^{(i)} \mid f(\mathbf{x}^{(i)};\thetav)\right)\right] \\
        &= \argmax_{\thetav}\;\sum_{i=1}^{N}\log\Pr\!\left(y^{(i)} \mid f(\mathbf{x}^{(i)};\thetav)\right)
    \end{align*}
  \end{frame}

  %--- Slide: NLL ---
  \begin{frame}{Convenção: minimizar a \textit{Negative Log-Likelihood}}
    \begin{align*}
      \hat\thetav
        &= \argmax_{\thetav}\;\sum_{i=1}^{N}\log\Pr\!\left(y^{(i)} \mid f(\mathbf{x}^{(i)};\thetav)\right) \\
        &= \argmin_{\thetav}\;\left[-\sum_{i=1}^{N}\log\Pr\!\left(y^{(i)} \mid f(\mathbf{x}^{(i)};\thetav)\right)\right] \\
        &= \argmin_{\thetav}\;\mathcal{L}(\thetav)
    \end{align*}
    \vfill
    \begin{itemize}
      \item $\mathcal{L}(\thetav)$ é a \textbf{Negative Log-Likelihood} (NLL).
      \item Diferentes escolhas de distribuição $\Pr(y\mid \cdot)$ produzem diferentes
        funções de custo conhecidas.
    \end{itemize}
  \end{frame}

  %--- Slide: receita geral ---
  \begin{frame}{Receita geral para construir uma função de custo}
    \begin{enumerate}
      \item Escolher uma distribuição $\Pr(y\mid \bm{\eta})$ apropriada ao domínio de $y$.
      \item Fazer a rede $f(\mathbf{x};\thetav)$ predizer os parâmetros $\bm{\eta}$ dessa distribuição.
      \item Substituir e simplificar a NLL para obter uma função de custo $\mathcal{L}(\thetav)$.
      \item Treinar minimizando $\mathcal{L}$ via descida de gradiente.
    \end{enumerate}
    \vfill
    \begin{exampleblock}{O que vem agora}
      Vamos aplicar essa receita em três casos canônicos: regressão (gaussiana),
      classificação binária (Bernoulli) e classificação multiclasse (categórica).
    \end{exampleblock}
  \end{frame}

  %======================================================================
  % Bloco: exemplo 1, regressão
  %======================================================================

  %--- Slide: distribuição gaussiana ---
  \begin{frame}{Passo 1: escolher a distribuição}
    Para $y\in\mathbb{R}$ a escolha natural é a gaussiana univariada:
    \[
      \Pr(y\mid \mu,\sigma^{2}) = \frac{1}{\sqrt{2\pi\sigma^{2}}}\,
        \exp\!\left[-\frac{(y-\mu)^{2}}{2\sigma^{2}}\right].
    \]
    \begin{itemize}
      \item $\mu$ é a média (centro da distribuição).
      \item $\sigma^{2}$ é a variância (largura).
    \end{itemize}
  \end{frame}

  %--- Slide: rede prediz a média ---
  \begin{frame}{Passo 2: rede prediz a média}
    \[
      \Pr\!\big(y \mid \alert{f(\mathbf{x};\thetav)},\sigma^{2}\big) =
        \frac{1}{\sqrt{2\pi\sigma^{2}}}\,
        \exp\!\left[-\frac{(y-\alert{f(\mathbf{x};\thetav)})^{2}}{2\sigma^{2}}\right].
    \]
    \begin{itemize}
      \item A rede passa a estimar $\mu = f(\mathbf{x};\thetav)$.
      \item A variância $\sigma^{2}$ é tratada como hiperparâmetro (constante).
      \item A camada de saída é \textbf{Linear} (média pode assumir qualquer valor real).
    \end{itemize}
  \end{frame}

  %--- Slide: simplificação ---
  \begin{frame}{Passo 3: substituir na NLL e simplificar}
    \begin{align*}
      \mathcal{L}(\thetav)
        &= -\sum_{i=1}^{N}\log\!\left[\frac{1}{\sqrt{2\pi\sigma^{2}}}\,\exp\!\left(-\frac{(y^{(i)}-f(\mathbf{x}^{(i)};\thetav))^{2}}{2\sigma^{2}}\right)\right] \\
        &= -\sum_{i=1}^{N}\log\!\left(\frac{1}{\sqrt{2\pi\sigma^{2}}}\right) + \sum_{i=1}^{N}\frac{(y^{(i)}-f(\mathbf{x}^{(i)};\thetav))^{2}}{2\sigma^{2}} \\
        &\propto \sum_{i=1}^{N}\big(y^{(i)}-f(\mathbf{x}^{(i)};\thetav)\big)^{2}
    \end{align*}
    \begin{itemize}
      \item O primeiro termo não depende de $\thetav$; a divisão por $2\sigma^{2}$ é constante.
      \item Sobra exatamente o erro quadrático.
    \end{itemize}
  \end{frame}

  %--- Slide: resultado MSE ---
  \begin{frame}{Resultado: erro quadrático médio}
    \[
      \boxed{\hat\thetav = \argmin_{\thetav}\;\sum_{i=1}^{N}\big(y^{(i)}-f(\mathbf{x}^{(i)};\thetav)\big)^{2}}
    \]
    \begin{itemize}
      \item Esta é a \textbf{função de custo $L_2$} (também chamada de MSE quando dividida por $N$).
      \item Sai naturalmente de duas hipóteses: instâncias \textit{i.i.d.} amostradas de uma
        gaussiana univariada com variância fixa.
    \end{itemize}
  \end{frame}

  %--- Slide: visual ---
  \begin{frame}{Visual: ajuste e gaussianas verticais}
    \begin{columns}
      \begin{column}{.5\linewidth}
        \centering
        \includegraphics[width=.9\linewidth,height=0.62\textheight,keepaspectratio]{slides/img/01-mlp/al-mse-residuals.png}\\
        \footnotesize Soma dos quadrados dos resíduos.
      \end{column}
      \begin{column}{.5\linewidth}
        \centering
        \includegraphics[width=.9\linewidth,height=0.62\textheight,keepaspectratio]{slides/img/01-mlp/al-mse-gaussians.png}\\
        \footnotesize $-\sum_i \log\Pr(y_i\mid f(\mathbf{x}_i;\thetav),\sigma^{2})$.
      \end{column}
    \end{columns}
  \end{frame}

  %--- Slide: Laplace -> MAE ---
  \begin{frame}{A mesma receita com outra distribuição: Laplace $\to$ MAE}
    Trocando a gaussiana pela distribuição de \textbf{Laplace}, com $\mu = f(\mathbf{x};\thetav)$
    e escala $b$ constante:
    \[
      \Pr(y\mid \mu, b) = \frac{1}{2b}\,\exp\!\left(-\frac{|y-\mu|}{b}\right).
    \]
    Substituindo na NLL:
    \begin{align*}
      \mathcal{L}(\thetav)
        &= -\sum_{i=1}^{N}\log\!\left[\frac{1}{2b}\,
           \exp\!\left(-\frac{|y^{(i)}-f(\mathbf{x}^{(i)};\thetav)|}{b}\right)\right]
         \propto \sum_{i=1}^{N}\big|y^{(i)}-f(\mathbf{x}^{(i)};\thetav)\big|
    \end{align*}
    \begin{itemize}
      \item O resultado é a \textbf{função de custo $L_1$} (MAE): penaliza desvios linearmente,
        não quadraticamente.
      \item A Laplace tem caudas mais pesadas que a gaussiana: erros grandes são menos
        surpreendentes, logo a função de custo é mais \alert{robusta a outliers}.
    \end{itemize}
  \end{frame}

  %======================================================================
  % Bloco: exemplo 2, classificação binária
  %======================================================================

  %--- Slide: Bernoulli ---
  \begin{frame}{Passo 1: distribuição Bernoulli}
    Para $y\in\{0,1\}$ usamos a Bernoulli($\lambda$):
    \[
      \Pr(y\mid \lambda) =
        \begin{cases}
          1-\lambda & y=0 \\
          \lambda & y=1
        \end{cases}
        \;=\; (1-\lambda)^{1-y}\,\lambda^{y}.
    \]
    \begin{itemize}
      \item $\lambda\in[0,1]$ é a probabilidade da classe positiva.
      \item Forma compacta usando expoentes evita o uso explícito de cases.
    \end{itemize}
  \end{frame}

  %--- Slide: rede prediz lambda ---
  \begin{frame}{Passo 2: rede prediz $\lambda$ via sigmoid}
    \begin{itemize}
      \item Saída crua da rede $f(\mathbf{x};\thetav)\in\mathbb{R}$ não respeita $\lambda\in[0,1]$.
      \item Aplicamos sigmoid na saída: $\lambda = \sigma(f(\mathbf{x};\thetav))$.
      \[
        \Pr(y\mid \mathbf{x};\thetav) =
          \big(1-\sigma(f(\mathbf{x};\thetav))\big)^{1-y}\,
          \big(\sigma(f(\mathbf{x};\thetav))\big)^{y}.
      \]
    \end{itemize}
  \end{frame}

  %--- Slide: NLL e BCE ---
  \begin{frame}{Passo 3: NLL e simplificação}
    \begin{align*}
      \mathcal{L}(\thetav)
        &= -\sum_{i=1}^{N}\log\!\left[\big(1-\sigma(f(\mathbf{x}^{(i)};\thetav))\big)^{1-y^{(i)}}\,
                                       \big(\sigma(f(\mathbf{x}^{(i)};\thetav))\big)^{y^{(i)}}\right] \\
        &= \sum_{i=1}^{N}\Big[-\big(1-y^{(i)}\big)\log\!\big(1-\sigma(f(\mathbf{x}^{(i)};\thetav))\big)
                              -y^{(i)}\log\!\big(\sigma(f(\mathbf{x}^{(i)};\thetav))\big)\Big]
    \end{align*}
    \begin{block}{Resultado: Entropia Cruzada Binária (BCE)}
      \[
        \boxed{\mathcal{L}_{\text{BCE}}(\thetav) = -\sum_{i=1}^{N}
          \Big[ y^{(i)}\log\hat\lambda^{(i)} + (1-y^{(i)})\log\!\big(1-\hat\lambda^{(i)}\big) \Big]}
      \]
      \centering com $\hat\lambda^{(i)} = \sigma\!\big(f(\mathbf{x}^{(i)};\thetav)\big)$.
    \end{block}
  \end{frame}

  %--- Slide: visualização sigmoid → Pr(y|x) ---
  \begin{frame}{Visual: \textit{piecewise} linear $\to$ sigmoid $\to$ $\Pr(y\mid x)$}
    \begin{columns}
      \begin{column}{.33\linewidth}
        \centering
        \includegraphics[width=\linewidth]{slides/img/01-mlp/al-piecewise-linear.png}\\
        \footnotesize $f(\mathbf{x};\thetav)$

      \end{column}
    \end{columns}
    \vspace{0.4em}
    \begin{itemize}
      \item A pré-ativação da camada de saída é uma função \textit{piecewise} linear.
      \item Após a sigmoid, a saída fica em $[0,1]$ e pode ser lida como uma probabilidade.
    \end{itemize}
  \end{frame}

  %--- Slide: por que não MSE em classificação ---
  \begin{frame}{Por que não usar MSE em classificação?}
    Para um exemplo $(\mathbf{x}, y)$ com saída $\hat y=\sigma(z)$:
    \vspace{-0.4em}
    \begin{align*}
      \mathcal{L}_{\text{MSE}} &= \big(\sigma(z)-y\big)^{2}
        & \frac{\partial \mathcal{L}_{\text{MSE}}}{\partial z} &= 2\big(\sigma(z)-y\big)\,\alert{\sigma'(z)} \\
      \mathcal{L}_{\text{BCE}} &= -\big[y\log\sigma(z) + (1-y)\log\big(1-\sigma(z)\big)\big]
        & \frac{\partial \mathcal{L}_{\text{BCE}}}{\partial z} &= \sigma(z)-y
    \end{align*}
    \vspace{-0.6em}
    \begin{columns}
      \begin{column}{.55\linewidth}
        \begin{itemize}
          \item Na BCE o $\sigma'$ cancela: gradiente proporcional ao erro.
          \item No MSE, $\sigma'(z)$ sobrevive: errar com confiança \alert{mata} o gradiente.
          \item MSE embute a hipótese gaussiana; $y\in\{0,1\}$ pede Bernoulli, logo BCE.
          \item $\mathcal{L}_{\text{MSE}}$ não é convexa em $z$; $\mathcal{L}_{\text{BCE}}$ é.
        \end{itemize}
      \end{column}
      \begin{column}{.45\linewidth}
        \centering
        \resizebox{.9\linewidth}{!}{
          \begin{tikzpicture}
            \begin{axis}[
              width=8cm,height=5cm,
              xlabel={$z$}, ylabel={$\left|\partial \mathcal{L}/\partial z\right|$},
              axis lines=middle,
              domain=-6:6, samples=120,
              ymin=0, ymax=1.1,
              legend style={font=\tiny,at={(0.97,0.97)},anchor=north east}
            ]
              \addplot[thick,blue] {1 - 1/(1+exp(-x))};
              \addlegendentry{BCE}
              \addplot[thick,red,dashed] {2*(1 - 1/(1+exp(-x)))*exp(-x)/((1+exp(-x))^2)};
              \addlegendentry{MSE}
            \end{axis}
          \end{tikzpicture}
        }\\
        \footnotesize Magnitude do gradiente para um exemplo com $y=1$.
      \end{column}
    \end{columns}
  \end{frame}

  %======================================================================
  % Bloco: exemplo 3, classificação multiclasse
  %======================================================================

  %--- Slide: distribuição categórica ---
  \begin{frame}{Passo 1: distribuição categórica}
    Para $y\in\{1,2,\ldots,K\}$ (exclusivo), a generalização da Bernoulli é a
    \textbf{categórica}:
    \[
      \Pr(y\mid \bm{\lambda}) = \prod_{k=1}^{K}\lambda_{k}^{[y=k]}
        \quad\text{com}\quad
        \sum_{k=1}^{K}\lambda_{k} = 1,\;\lambda_{k}\ge 0.
    \]
    \begin{itemize}
      \item $[y=k]$ vale 1 se $y=k$ e 0 caso contrário (Iverson).
      \item Cada $\lambda_k\in[0,1]$ e $\sum_k \lambda_k = 1$ formam um vetor de probabilidades.
      \item Outras notações comuns na literatura: $\pi_k$ ou $p_k$ no lugar de $\lambda_k$.
    \end{itemize}
  \end{frame}

  %--- Slide: rede prediz lambda via softmax ---
  \begin{frame}{Passo 2: rede prediz $\bm{\lambda}$ via softmax}
    \begin{itemize}
      \item Saída crua da rede: $\mathbf{z} = f(\mathbf{x};\thetav)\in\mathbb{R}^{K}$ (logits).
      \item Aplicamos softmax para obter probabilidades:
        \[
          \lambda_{k} = \mathrm{softmax}(\mathbf{z})_k = \frac{e^{z_k}}{\sum_{j=1}^{K}e^{z_j}}.
        \]
      \item Por construção, $\bm{\lambda}$ é um vetor de probabilidades válido.
    \end{itemize}
  \end{frame}

  %--- Slide: NLL multiclasse ---
  \begin{frame}{Passo 3: NLL e simplificação}
    \begin{align*}
      \mathcal{L}(\thetav)
        &= -\sum_{i=1}^{N}\log\!\left[\prod_{k=1}^{K}\lambda_{k}(\mathbf{x}^{(i)};\thetav)^{[y^{(i)}=k]}\right] \\
        &= -\sum_{i=1}^{N}\sum_{k=1}^{K} [y^{(i)}=k]\,\log\lambda_{k}(\mathbf{x}^{(i)};\thetav).
    \end{align*}
    Em forma vetorial com $\mathbf{y}^{(i)}$ \textit{one-hot}:
    \[
      \boxed{\mathcal{L}_{\text{CCE}}(\thetav) = -\sum_{i=1}^{N}\sum_{k=1}^{K} y^{(i)}_{k}\,
        \log\!\big(\mathrm{softmax}(f(\mathbf{x}^{(i)};\thetav))_k\big)}
    \]
    Esta é a \textbf{Categorical Cross-Entropy} (CCE).
  \end{frame}

  %--- Slide: substituindo softmax ---
  \begin{frame}{Forma equivalente: log-softmax}
    Como em cada amostra apenas a classe correta $k^{*} = y^{(i)}$ contribui (one-hot):
    \[
      \mathcal{L}_{\text{CCE}}(\thetav)
        = -\sum_{i=1}^{N}\log\!\left[\frac{e^{z_{k^{*}}^{(i)}}}{\sum_{j=1}^{K} e^{z_{j}^{(i)}}}\right]
        = -\sum_{i=1}^{N}\!\left[z_{k^{*}}^{(i)} - \log\!\sum_{j=1}^{K} e^{z_{j}^{(i)}}\right].
    \]
    \begin{itemize}
      \item $\log\sum e^{z_j}$ é o \textit{log-sum-exp}, calculado de forma estável em \textit{frameworks}.
      \item A softmax é invariante a deslocamentos de todos os logits por um escalar $c$:
        \[
          \mathrm{softmax}(\mathbf{z}) = \mathrm{softmax}\big(\mathbf{z}-c\,\mathbf{1}\big),
          \qquad \mathbf{1} = (1,\ldots,1)^{\top},
          \quad c\,\mathbf{1} = (c,\ldots,c)^{\top}.
        \]
        Usa-se $c=\max_j z_j$ para que nenhum $e^{z_j - c}$ estoure.
    \end{itemize}
  \end{frame}

  %--- Slide: derivada da softmax, preparação ---
  \begin{frame}{Derivada da softmax: preparação}
    Escrevendo $S = \sum_{j=1}^{K} e^{z_j}$, cada saída da softmax é um produto de dois fatores:
    \[
      \lambda_k = \frac{e^{z_k}}{S} = e^{z_k}\cdot S^{-1}.
    \]
    \begin{itemize}
      \item Como $S$ soma todos os $e^{z_j}$, $\lambda_k$ depende de \alert{todos} os $z_j$:
        a derivada completa é uma matriz (Jacobiano) $K\times K$ de termos
        $\partial\lambda_k/\partial z_j$.
      \item Ferramentas: regra do produto, $(u\,v)' = u'v + u\,v'$, e regra da cadeia
        para a potência, $\dfrac{\partial S^{-1}}{\partial z_j} = -S^{-2}\,\dfrac{\partial S}{\partial z_j}$.
      \item Há duas situações distintas: derivar em relação ao \textbf{próprio} logit
        ($j = k$) ou em relação a \textbf{outro} logit ($j \ne k$).
    \end{itemize}
  \end{frame}

  %--- Slide: derivada da softmax, caso j = k ---
  \begin{frame}{Derivada da softmax: caso $j = k$, passo a passo}
    Identificando os dois fatores: $u = e^{z_k}$, $v = S^{-1}$ com $S = \sum_{j} e^{z_j}$.
    \begin{align*}
      \frac{\partial u}{\partial z_k} &= e^{z_k}
        &&\text{(derivada da exponencial)} \\
      \frac{\partial S}{\partial z_k} &= \frac{\partial}{\partial z_k}\big(e^{z_1}+\cdots+e^{z_k}+\cdots+e^{z_K}\big) = e^{z_k}
        &&\text{(só um termo da soma depende de $z_k$)} \\
      \frac{\partial v}{\partial z_k} &= -S^{-2}\,\frac{\partial S}{\partial z_k} = -S^{-2}\,e^{z_k}
        &&\text{(regra da cadeia na potência)} \\[0.4em]
      \frac{\partial \lambda_k}{\partial z_k}
        &= u'v + u\,v'
         = e^{z_k}\,S^{-1} - e^{z_k}\,S^{-2}\,e^{z_k}
        &&\text{(regra do produto)} \\
        &= \frac{e^{z_k}}{S} - \frac{e^{z_k}}{S}\cdot\frac{e^{z_k}}{S}
        &&\text{(reescrever as potências como frações)} \\
        &= \lambda_k - \lambda_k^{2}
        &&\text{(reconhecer $\lambda_k = e^{z_k}/S$)} \\
        &= \lambda_k\,(1-\lambda_k)
        &&\text{(fatorar $\lambda_k$)}
    \end{align*}
  \end{frame}

  %--- Slide: interpretação do caso j = k ---
  \begin{frame}{Derivada da softmax: lendo o caso $j = k$}
    \[
      \frac{\partial \lambda_k}{\partial z_k} = \lambda_k\,(1-\lambda_k)
    \]
    \begin{itemize}
      \item Mesma forma da derivada da sigmoid: aumentar $z_k$ aumenta $\lambda_k$,
        com efeito máximo quando $\lambda_k = 1/2$.
      \item Satura quando $\lambda_k \to 0$ ou $\lambda_k \to 1$: logits muito confiantes
        respondem pouco a mudanças.
      \item Sempre não negativa: um logit nunca diminui a própria probabilidade.
    \end{itemize}
  \end{frame}

  %--- Slide: derivada da softmax, caso j != k ---
  \begin{frame}{Derivada da softmax: caso $j \ne k$, passo a passo}
    Mesmo produto $\lambda_k = u\,v$, mas agora derivando em relação a \textbf{outro} logit $z_j$.
    \begin{align*}
      \frac{\partial u}{\partial z_j} &= \frac{\partial}{\partial z_j}\,e^{z_k} = 0
        &&\text{($e^{z_k}$ não depende de $z_j$)} \\
      \frac{\partial S}{\partial z_j} &= \frac{\partial}{\partial z_j}\big(e^{z_1}+\cdots+e^{z_j}+\cdots+e^{z_K}\big) = e^{z_j}
        &&\text{(só o termo $e^{z_j}$ sobrevive)} \\
      \frac{\partial v}{\partial z_j} &= -S^{-2}\,\frac{\partial S}{\partial z_j} = -S^{-2}\,e^{z_j}
        &&\text{(regra da cadeia na potência)} \\[0.4em]
      \frac{\partial \lambda_k}{\partial z_j}
        &= u'v + u\,v'
         = 0\cdot S^{-1} - e^{z_k}\,S^{-2}\,e^{z_j}
        &&\text{(regra do produto)} \\
        &= -\frac{e^{z_k}}{S}\cdot\frac{e^{z_j}}{S}
        &&\text{(reescrever $S^{-2}$ como duas frações)} \\
        &= -\lambda_k\,\lambda_j
        &&\text{(reconhecer as duas softmax)}
    \end{align*}
    \vspace{-0.7em}
    \begin{itemize}
      \item Sinal negativo: aumentar outro logit \alert{rouba} massa de probabilidade de $\lambda_k$.
    \end{itemize}
  \end{frame}

  %--- Slide: unificação com Kronecker ---
  \begin{frame}{Derivada da softmax: unificando os dois casos}
    \begin{columns}
      \begin{column}{.5\linewidth}
        \[
          \frac{\partial \lambda_k}{\partial z_j} =
            \begin{cases}
              \lambda_k\,(1-\lambda_k) & j = k \\
              -\lambda_k\,\lambda_j    & j \ne k
            \end{cases}
        \]
      \end{column}
      \begin{column}{.5\linewidth}
        \[
          \boxed{\frac{\partial \lambda_k}{\partial z_j} = \lambda_k\big([k=j] - \lambda_j\big)}
        \]
      \end{column}
    \end{columns}
    \vspace{0.6em}
    \begin{itemize}
      \item O colchete de Iverson $[k=j]$ (delta de Kronecker $\delta_{kj}$) vale $1$ quando
        $j=k$ e $0$ caso contrário.
      \item Verificação: com $j=k$, $[k=j]=1$ e sobra $\lambda_k(1-\lambda_k)$;
        com $j\ne k$, $[k=j]=0$ e sobra $-\lambda_k\lambda_j$.
      \item Em forma matricial, o Jacobiano $\mathbf{J}\in\mathbb{R}^{K\times K}$ com
        $J_{kj} = \partial\lambda_k/\partial z_j$ é
        \[
          \mathbf{J} = \mathrm{diag}(\bm{\lambda}) - \bm{\lambda}\bm{\lambda}^{\top}.
        \]
    \end{itemize}
  \end{frame}

  %--- Slide: por que derivar a loss direto nos logits ---
  \begin{frame}{Por que derivar a função de custo direto nos logits?}
    Regra da cadeia atravessando o Jacobiano da softmax, com
    $\dfrac{\partial \mathcal{L}}{\partial \lambda_k} = -\dfrac{y_k}{\lambda_k}$:
    \begin{align*}
      \frac{\partial \mathcal{L}}{\partial z_j}
        &= \sum_{k=1}^{K} \frac{\partial \mathcal{L}}{\partial \lambda_k}\,
           \frac{\partial \lambda_k}{\partial z_j}
         = \sum_{k=1}^{K} \left[\left(-\frac{y_k}{\lambda_k}\right)\lambda_k\big([k=j]-\lambda_j\big)\right]
         = \sum_{k=1}^{K} \Big[-y_k\big([k=j]-\lambda_j\big)\Big] \\
        &= -y_j + \lambda_j\underbrace{\textstyle\sum_{k} y_k}_{=\,1}
         = \lambda_j - y_j.
    \end{align*}
    \begin{itemize}
      \item O gradiente nos logits é simplesmente \alert{softmax menos one-hot}:
        a divisão por $\lambda_k$, que explode quando $\lambda_k \to 0$, desaparece.
      \item O resultado é limitado, $\lambda_j - y_j \in [-1, 1]$, e se anula apenas quando a
        predição coincide com o alvo.
      \item Por isso \textit{frameworks} calculam a cross-entropy direto sobre os logits,
        sem passar pela softmax como camada separada.
    \end{itemize}
  \end{frame}

  %--- Slide: label smoothing ---
  \begin{frame}{Variante: \textit{label smoothing}}
    Substitui o alvo \textit{one-hot} por uma mistura com a distribuição
    uniforme\footfullcite{szegedy2016rethinking}, com $\varepsilon\in(0,1)$ pequeno:
    \[
      \tilde y_k = (1-\varepsilon)\,[y=k] + \frac{\varepsilon}{K},
      \qquad
      \mathcal{L}_{\text{LS}} = -\sum_{k=1}^{K}\tilde y_k\,\log\lambda_k.
    \]
    \begin{itemize}
      \item Com \textit{one-hot}, o mínimo exige $\lambda_{k^{*}}\to 1$, ou seja, logit da classe
        correta $\to\infty$: o gradiente $\lambda_j - y_j$ nunca se anula.
      \item Com $\tilde{\mathbf{y}}$, o ótimo é atingido em probabilidades finitas: penaliza
        excesso de confiança e melhora calibração\footfullcite{muller2019labelsmoothing}.
      \item Em PyTorch: \texttt{nn.CrossEntropyLoss(label\_smoothing=0.1)}.
    \end{itemize}
  \end{frame}

  \section{Métricas de Avaliação}

  %--- Slide: métricas vs loss ---
  \begin{frame}{Métricas $\ne$ função de custo}
    \begin{itemize}
      \item A função de custo é otimizada durante o treinamento; precisa ser diferenciável
        e numericamente estável.
      \item A métrica é interpretável pelo usuário final; pode ser não diferenciável
        (acurácia, F1) ou dependente de um \textit{threshold}.
      \item Quase sempre, a função de custo é uma \alert{aproximação suave} do que de fato queremos
        otimizar (a métrica).
    \end{itemize}
  \end{frame}

  %--- Slide: regressão ---
  \begin{frame}{Métricas de regressão}
    \begin{itemize}
      \item \textbf{MSE}: $\dfrac{1}{N}\sum_{i=1}^{N}(\hat y^{(i)}-y^{(i)})^{2}$.
      \item \textbf{MAE}: $\dfrac{1}{N}\sum_{i=1}^{N}|\hat y^{(i)}-y^{(i)}|$.
      \item \textbf{$R^{2}$}: $1-\dfrac{\sum_i (y^{(i)}-\hat y^{(i)})^{2}}{\sum_i (y^{(i)}-\bar y)^{2}}$,
        onde $\bar y$ é a média dos rótulos.
      \item MSE pune grandes desvios mais fortemente; MAE é mais robusto a \textit{outliers};
        $R^{2}$ é interpretável como variância explicada.
    \end{itemize}
  \end{frame}

  %--- Slide: classificação ---
  \begin{frame}{Métricas de classificação}
    \begin{columns}
      \begin{column}{.55\linewidth}
        \begin{itemize}
          \item \textbf{Acurácia}: $\dfrac{TP+TN}{TP+TN+FP+FN}$.
          \item \textbf{Recall}: $\dfrac{TP}{TP+FN}$.
          \item \textbf{Precision}: $\dfrac{TP}{TP+FP}$.
          \item \textbf{$F_\beta$-score}: $(1+\beta^{2})\dfrac{P\cdot R}{\beta^{2}P+R}$.
        \end{itemize}
      \end{column}
      \begin{column}{.45\linewidth}
        \centering
        \begin{tabular}{ll|cc}
          \toprule
            & & \multicolumn{2}{c}{\textbf{Predito}} \\
            & & Sim & Não \\
          \midrule
          \textbf{Real} & Sim & TP & FN \\
                        & Não & FP & TN \\
          \bottomrule
        \end{tabular}
      \end{column}
    \end{columns}
  \end{frame}

  %--- Slide: worked example ---
  \begin{frame}{Métricas de classificação: exemplo numérico}
    \begin{columns}
      \begin{column}{.42\linewidth}
        \centering
        \textbf{Matriz de confusão}\\[0.3em]
        \begin{tabular}{ll|cc}
          \toprule
            & & \multicolumn{2}{c}{\textbf{Predito}} \\
            & & Sim & Não \\
          \midrule
          \textbf{Real} & Sim & $8$ & $2$ \\
                        & Não & $3$ & $87$ \\
          \bottomrule
        \end{tabular}\\[0.3em]
        {\footnotesize $100$ instâncias, $10$ positivas.}
      \end{column}
      \begin{column}{.55\linewidth}
        \begin{align*}
          \text{Acurácia} &= \tfrac{8+87}{100} = 0.95 \\
          \text{Recall}   &= \tfrac{8}{8+2} = 0.80 \\
          \text{Precision}&= \tfrac{8}{8+3} \approx 0.727 \\
          F_1             &= 2\cdot\tfrac{0.727 \cdot 0.80}{0.727 + 0.80} \approx 0.762
        \end{align*}
        {\footnotesize Acurácia alta esconde recall mais baixo: classe positiva rara é mais difícil.}
      \end{column}
    \end{columns}
  \end{frame}

  %--- Slide: quando usar cada uma ---
  \begin{frame}{Escolhendo a métrica certa}
    \begin{itemize}
      \item \textbf{Datasets balanceados}: acurácia é razoável.
      \item \textbf{Datasets desbalanceados}: acurácia engana; prefira F1 ou recall específico.
      \item \textbf{Quando o custo do FN é alto} (medicina, fraude): priorize \textbf{recall}.
      \item \textbf{Quando o custo do FP é alto} (spam): priorize \textbf{precision}.
      \item \textbf{Multiclasse}: macro/micro-F1, matriz de confusão por classe.
    \end{itemize}
  \end{frame}
