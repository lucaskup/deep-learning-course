\section{Perceptron}

\begin{frame}{Perceptron}

Frank Rosenblatt\footfullcite{rosenblatt1958perceptron} fez algumas modificações no Neurônio MP
\begin{equation*}
    \mltargethat = \varphi(\mlinstance\thetav^{\top})
\end{equation*}
\begin{itemize}
    \item Entradas são números reais
    \item Cada entrada é ponderada por um parâmetro
    \item Existe um algoritmo de aprendizado para definir $\thetav$
    \item Função de ativação $\varphi(z)=\text{sinal}(z)=\begin{cases}
        +1,\quad z\geq0\\
        -1,\quad z<0
    \end{cases}$ 
\end{itemize}
\end{frame}

\begin{frame}{Perceptron}

\large ``(...) identities of this sort must be \textit{learned} (...) the number of functional units in the storage system or memory, should be less than the number of forms or memories to be retained.''\footfullcite{rosenblatt1958perceptron}
\end{frame}

\begin{frame}{Algoritmo de Aprendizado do Perceptron (PLA)}
\begin{algorithmic}[1]
\State Assuma $\X^{(i)}$ com primeira componente igual a $1$ (truque do bias)
\State Inicialize os pesos $\thetav$ aleatoriamente
\While{existirem observações mal classificadas}
  \State Selecione $\X^{(i)}$ tal que $\text{sinal}(\X^{(i)} \thetav^\top) \neq \y^{(i)}$
  \State Atualize: $\thetav \gets \thetav + \y^{(i)} \X^{(i)}$
\EndWhile
\end{algorithmic}
\end{frame}

        \begin{itemize}
          \item Descoberto empiricamente
          \item Intuição geométrica: produto escalar

          \item Sempre converge para problemas linearmente separáveis

\begin{frame}{Resolvendo XOR}
\begin{itemize}
    \item Quando organizamos vários Perceptrons em camadas chamamos de \textit{MultiLayer Perceptron} (MLP)
    \item PLA não funciona para MLP
    \item Essa limitação minou o interesse por redes neurais durante anos (Inverno da IA)
    \item Para treinar um MLP vamos precisar usar retropropagação de erros (\textit{backpropagation}) e descida de gradiente (\textit{gradient descent})
\end{itemize}
