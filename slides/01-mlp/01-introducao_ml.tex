\section{Conceitos Básicos de \textit{Machine Learning}}

\begin{frame}{Desenvolvimento de uma aplicação com AM}
\only<1>{
    O projeto de um sistema que usa \textit{Machine Learning} pode ser descrito pela figura abaixo\footnote{\scriptsize Fluxos diferentes existem na literatura de \textit{Knowledge Discovery in Databases} e em Mineração de Dados.}
    }
\only<2>{
\textbf{Experimental Design}
    \begin{itemize}
        \item Modelar o problema como uma tarefa de \textit{Machine Learning}
        \item Como você definiria o problema de treinar um reconhecedor de rostos que é capaz de reconhecer 100 pessoas diferentes?
        \begin{itemize}
            \item Como um problema de \alert{classificação multiclasse} com 100 classes será muito difícil coletar uma base dados grande o bastante
            \item Como um problema de \alert{classificação binário} onde o sistema recebe um par de imagens é mais fácil coletar dados
        \end{itemize}
    \end{itemize}
}
\only<3>{
    \textbf{Data Collection}
    \begin{itemize}
        \item É possível coletar dados que sejam representativos do problema em questão?
        \item Existe algum dataset aberto que pode nos auxiliar?
        \item Quando a coleta de dados ou a anotação é cara, existem alternativas?
    \end{itemize}
}
\only<4>{
    \textbf{Dimensionality Reduction}
    \begin{itemize}
        \item Todos os atributos coletados são úteis para resolver a tarefa?
        \item Como selecionar apenas os atributos relacionados a variável alvo? 
        \item Existem formas de criar novos atributos que sejam melhores preditores para a variável alvo?
        \item Posso resumir meus atributos em um espaço menor?
    \end{itemize}
}
\only<5>{
    \textbf{Predictor Design}
    \begin{itemize}
        \item Existe algum modelo cujo viés indutivo seja aderente ao meu problema?
        \item Qual será a métrica utilizada para avaliação do desempenho do modelo?
        \item Qual meu \textit{budget} computacional?
        \item Qual a melhor hiperparametrização dos modelos selecionados?
    \end{itemize}
}
\only<6>{
    \textbf{Error Estimation}
    \begin{itemize}
        \item Meu modelo treinado generaliza?
        \item Qual é o erro esperado do meu modelo para novos dados?
        \item Coletar mais dados pode melhorar meus resultados?
    \end{itemize}

}

\begin{frame}{Tipos de Aprendizado}
    \begin{itemize}
        \item \textbf{Supervisionado}
        \begin{itemize}
            \item Dados de treino possuem rótulos
            \item \textit{Exemplo}: classificar fotos como gato ou cachorro a partir de 10 mil imagens rotuladas
        \end{itemize}
        \item \textbf{Não Supervisionado}
        \begin{itemize}
            \item Dados de treino não possuem rótulos
            \item \textit{Exemplo}: agrupar clientes em segmentos a partir do histórico de compras, sem rótulos
        \end{itemize}
        \item \textbf{Semi Supervisionado}
        \begin{itemize}
            \item Alguns dados de treino possuem rótulo outros não
            \item \textit{Exemplo}: detecção de fraude com poucas transações rotuladas e milhões sem rótulo
        \end{itemize}
        \item \textbf{Aprendizado por Reforço}
        \begin{itemize}
            \item Agente recebe “recompensa” devido a uma série de ações
            \item \textit{Exemplo}: AlphaGo aprende a jogar Go ganhando recompensa ao final da partida
        \end{itemize}
        \item \textbf{Auto Supervisionado}
        \begin{itemize}
            \item Parte das entradas servem como “supervisão”
            \item \textit{Exemplo}: pré-treino do BERT prevendo palavras mascaradas em uma frase
        \end{itemize}
    \end{itemize}
\end{frame}

\begin{frame}{Notação}
    Considerando um dataset formado por exemplos de uma relação $\big(\mlinstancei, \mltargeti\big )$
    \begin{itemize}
        \item $\mlinstance$ é nossa matriz de atributos $N\times M$ onde $N$ é o número de instâncias e $M$ é o número de atributos indexada por $\mlinstance^{(n)}_m$
        \item $\mltarget$ é nossa variável alvo $N\times 1$
        \item Objetivo é estimar $\mltargeti$ para uma instância desconhecida $\mlinstancei$
        \begin{itemize}
            \item Quando $\mltargeti$ é uma variável discreta: \textit{Problema de Classificação} 
            \item Quando $\mltargeti$ é uma variável contínua: \textit{Problema de Regressão}
        \end{itemize}
    \end{itemize}
\end{frame}

\subsection{Regressão Linear}

\begin{frame}{Intuição}
\only<1-2>{
Queremos ajustar um \textit{hiperplano}\footnote{No $\mathbb{R}^2$ um hiperplano é equivalente a uma reta} aos nossos dados

}

Qual é a melhor \alert{hipótese}?
\[J_{\text{SSR}}=\sum_{i=1}^{N}\left(\mltargethati - \mltargeti\right)^2\]
}

\end{itemize}

            \end{itemize}
        \end{block}

    \end{column}

\end{frame}

\begin{frame}{Treinamento}
    
        Vamos derivar $J_{\text{SSR}}$ em relação a $\theta$ e igualar a 0
        \begin{align*}
            \frac{\partial J_{\text{SSR}}}{\partial\theta}&=\frac{\partial ((\mltargethat - \mltarget)^\top(\mltargethat - \mltarget))}{\partial\theta}\\
            &=\frac{\partial (\mltargethat^\top\mltargethat - \mltargethat^\top\mltarget-\mltarget^\top\mltargethat + \mltarget^\top\mltarget)}{\partial\theta}\\
            &=\frac{\partial (\mltargethat^\top\mltargethat - 2\mltargethat^\top\mltarget + \mltarget^\top\mltarget)}{\partial\theta}\\
            &=\frac{\partial (\bm{\theta}^\top\mlinstance^\top\mlinstance\bm{\theta} - 2\bm{\theta}^\top\mlinstance^\top\mltarget + \mltarget^\top\mltarget)}{\partial\theta}\\
            &=\frac{\partial (\bm{\theta}^\top\mlinstance^\top\mlinstance\bm{\theta})}{\partial\theta} - \frac{\partial(2\bm{\theta}^\top\mlinstance^\top\mltarget)}{\partial\theta}\\
            &=2\mlinstance^\top\mlinstance\bm{\theta} - 2\mlinstance^\top\mltarget\\
            &=2\mlinstance^\top(\mlinstance\bm{\theta} - \mltarget)
        \end{align*}
\end{frame}

\begin{frame}{Treinamento}
    
        Vamos derivar $J_{\text{SSR}}$ em relação a $\theta$ e igualar a 0
        \begin{align*}
            2\mlinstance^\top(\mlinstance\bm{\theta} - \mltarget) &= 0\\
            \mlinstance^\top(\mlinstance\bm{\theta} - \mltarget) &= 0\\
            \mlinstance^\top\mlinstance\bm{\theta} - \mlinstance^\top\mltarget &= 0\\

    \item $\bm{\theta}_{t+1}=\bm{\theta}_{t}-\eta\nabla_ J$
    \item $\bm{\theta}_{0}$ inicializado com valores aleatórios
    \item $\eta\in(0,1]$ taxa de aprendizado
    \item Valores típicos para $\eta$ são $10^{-3}, 10^{-4}$ ou $10^{-5}$
\end{itemize}    
\end{frame}

\begin{frame}{Descida de Gradiente}
\begin{columns}
    \begin{column}{.45\linewidth}
    \begin{itemize}
        \item Ao lado temos um algoritmo para \textit{Batch Gradient Descent} 
        \item Se selecionarmos apenas uma instância $\mlinstancei$ para estimar o valor do gradiente teremos o \textit{Stochastic Gradient Descent}
        \item A diferença entre $J_{\text{SSR}}$ de diferentes iterações pode ser usado como condição de convergência
    \end{itemize}

    \end{column}
    \begin{column}{.55\linewidth}
        \begin{algorithmic}[1]
        \Require $\mlinstance_{n\times m}$ matriz de instâncias
        \Require $\mltarget_{n\times 1}$ variável alvo
        \Require $\eta\in(0,1]$ taxa de aprendizado
        \Procedure{Gradient Descent}{}
        \State $\bm{\theta} \sim \mathcal{N}(0,1)$ 
        \Repeat 
            \State $\bm{\theta} \gets \bm{\theta}-\eta\mlinstance^\top(\mlinstance\bm{\theta}-\mltarget)$
            \State $\mltargethat\gets\mlinstance\bm{\theta}$
            \State $J_{\text{SSR}} \gets (\mltarget-\mltargethat)^\top(\mltarget-\mltargethat)$ 
        \Until{Condição de convergência}
        \State \Return $\bm{\theta}$
        \EndProcedure
        \end{algorithmic}        
    \end{column}
\end{columns}
    
\end{frame}

\subsection{Regressão Logística}
\begin{frame}{Como aplicar Regressão Linear para Classificação?}
A \textit{imagem} da Regressão Linear é incompatível com uma classificação
\begin{itemize}
    \item $p\left(\mltargethati=c|\mlinstancei\right)\in[0,1]$
    \item $\mlinstancei\bm{\theta}\in(-\infty,+\infty)$
\end{itemize}

Precisamos encontrar algum valor relacionado a probabilidade de pertencimento a uma classe $p\left(\mltargethati=c|\mlinstancei\right)$ que tenha um contradomínio similar a Regressão Linear $\mlinstancei\bm{\theta}$
    
\end{frame}

\begin{frame}{Como aplicar Regressão Linear para Classificação?}

\begin{itemize}
    \item<1-> $p\left(\mltargethati=c|\mlinstancei\right)\in[0,1]$: probabilidade de pertencimento a classe $c$ 
    \item<2-> $\frac{p\left(\mltargethati=c|\mlinstancei\right)}{1-p\left(\mltargethati=c|\mlinstancei\right)}\in[0,+\infty)$: chance de pertencimento a classe $c$
    \item<3-> $\ln{\left(\frac{p\left(\mltargethati=c|\mlinstancei\right)}{1-p\left(\mltargethati=c|\mlinstancei\right)}\right)}\in(-\infty,+\infty)$: log da chance de pertencimento a classe $c$: 
\end{itemize}

\end{frame}

\begin{frame}{Aplicando Regressão Linear para estimar o log das chances}
\small
\begin{align*}
    \ln{\left(\frac{p\left(\mltargethati=c|\mlinstancei\right)}{1-p\left(\mltargethati=c|\mlinstancei\right)}\right)} &= \mlinstancei\bm{\theta}\\
    \frac{p\left(\mltargethati=c|\mlinstancei\right)}{1-p\left(\mltargethati=c|\mlinstancei\right)} &= e^{\mlinstancei\bm{\theta}}\\
    p\left(\mltargethati=c|\mlinstancei\right) &= e^{\mlinstancei\bm{\theta}}\left(1-p\left(\mltargethati=c|\mlinstancei\right)\right)\\
    p\left(\mltargethati=c|\mlinstancei\right) +p\left(\mltargethati=c|\mlinstancei\right)e^{\mlinstancei\bm{\theta}} &= e^{\mlinstancei\bm{\theta}}\\
    p\left(\mltargethati=c|\mlinstancei\right)(1+e^{\mlinstancei\bm{\theta}}) &= e^{\mlinstancei\bm{\theta}}\\
    p\left(\mltargethati=c|\mlinstancei\right) &= \frac{e^{\mlinstancei\bm{\theta}}}{(1+e^{\mlinstancei\bm{\theta}})}= \frac{1}{e^{-\mlinstancei\bm{\theta}}(1+e^{\mlinstancei\bm{\theta}})}\\
    p\left(\mltargethati=c|\mlinstancei\right) &= \frac{1}{\left(1+e^{-\mlinstancei\bm{\theta}}\right)}\\
\end{align*}
\end{frame}

\begin{frame}{Regressão Logística}
\begin{itemize}
    \item Ao adaptar a Regressão Linear para a tarefa de classificação binária obtemos a Regressão Logística
    \item $\mltargethat = \frac{1}{\left(1+e^{-\mlinstancei\bm{\theta}}\right)}$
    \item Precisamos de uma nova função de custo
    \begin{itemize}
        \item $J_{\text{SSR}}$ não é convexo para a Regressão Logística
        \item $J_{\text{BCE}}=-\frac{1}{N}\sum_{i=1}^{N}\Big[\mltargeti\ln\mltargethati +(1-\mltargeti)\ln{(1-\mltargethati)}\Big]$
    \end{itemize}
\end{itemize}
    
\end{frame}

\begin{frame}{Treinamento}
Vamos derivar $J_{\text{BCE}}$ em relação aos $\bm{\theta}$
\begin{align*}
    \frac{\partial J_{\text{BCE}}}{\partial\bm{\theta}} &=     \frac{\partial (-\mltarget\ln\mltargethat -(1-\mltarget)\ln{(1-\mltargethat)})}{\partial\bm{\theta}}\\
    &= -\frac{\partial (\mltarget\ln\mltargethat)}{\partial\bm{\theta}} -\frac{\partial((1-\mltarget)\ln{(1-\mltargethat)})}{\partial\bm{\theta}}\\
    &= -\mltarget\frac{\partial (\ln\mltargethat)}{\partial\bm{\theta}} -(1-\mltarget)\frac{\partial\ln{(1-\mltargethat)}}{\partial\bm{\theta}}\\
    &= -\mltarget\frac{\partial (\ln\mltargethat)}{\partial\mltargethat}\frac{\partial\mltargethat}{\partial\bm{\theta}} -(1-\mltarget)\frac{\partial\ln{(1-\mltargethat)}}{\partial\mltargethat}\frac{\partial\mltargethat}{\partial\bm{\theta}}\\
    &= -\mlinstance^\top\mltarget\frac{1}{\mltargethat}\mltargethat(1-\mltargethat) -\mlinstance^\top(1-\mltarget)\left(-\frac{1}{1-\mltargethat}\right)\mltargethat(1-\mltargethat)\\
    &= \mlinstance^\top\left(-\mltarget+\mltarget\mltargethat +\mltargethat-\mltargethat\mltarget\right)\\
    &= \mlinstance^\top\left(\mltargethat-\mltarget\right)\\
\end{align*}
    
\end{frame}

\begin{frame}{Regressão Logística}
\begin{itemize}
    \item Função de Custo $J_{\text{BCE}}$ é convexa, mas não admite solução analítica
    \item O gradiente tem exatamente a mesma forma da Regressão Linear
    \item Gera uma fronteira de decisão linear (hiperplano) no espaço de atributos
    \end{itemize}
    
\end{frame}
